#!/usr/bin/env python3
import time
import sys
from processor import read_file, find_duplicates_bruteforce

def run_once(path):
    text = read_file(path)
    start = time.time()
    dups = find_duplicates_bruteforce(text)
    t = time.time() - start
    return t, dups

def print_header(kind, t):
    print("BENCHMARK:", kind, "time_seconds={:.6f}".format(t))

def main():
    args = sys.argv[1:]
    mode = "--small" if not args else args[0]
    if mode == "--small":
        path = "data/sample_input_small.txt"
        kind = "small"
    elif mode == "--large":
        path = "data/sample_input_large.txt"
        kind = "large"
    else:
        print("Usage: bench.py [--small|--large]")
        sys.exit(1)

    t, dups = run_once(path)
    print_header(kind, t)
    print("duplicate_count:", len(dups))

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Simple text processing utilities.

Intentional characteristics in baseline:
- tokenize: currently removes only a couple punctuation marks and splits on single space (split(' ')),
  which fails to split on newlines and multiple whitespace correctly.
- count_palindromes: currently counts single-letter palindromes; tests expect length >= 2 palindromes.
- find_duplicates_bruteforce: naive nested-loop duplicate detector (O(n^2)) used by bench.py.
"""

from typing import List, Dict, Set

def read_file(path: str) -> str:
    with open(path, "r", encoding="utf-8") as f:
        return f.read()

def tokenize(text: str) -> List[str]:
    # Intentionally limited punctuation removal and a split that doesn't handle newlines properly.
    t = text.lower().replace(",", "").replace(".", "").replace("!", "").replace("?", "")
    # BAD: splitting on single space misses newlines and multiple whitespace
    tokens = [w for w in t.split(" ") if w != ""]
    return tokens

def count_words(text: str) -> Dict[str, int]:
    tokens = tokenize(text)
    counts: Dict[str, int] = {}
    for w in tokens:
        counts[w] = counts.get(w, 0) + 1
    return counts

def top_n_words(text: str, n: int = 3) -> List[str]:
    counts = count_words(text)
    sorted_items = sorted(counts.items(), key=lambda kv: (-kv[1], kv[0]))
    return [k for k, _ in sorted_items[:n]]

def count_palindromes(text: str) -> int:
    tokens = tokenize(text)
    # Currently counts single-letter tokens as palindromes too
    count = 0
    for w in tokens:
        if w == w[::-1]:
            count += 1
    return count

def find_duplicates_bruteforce(text: str) -> Set[str]:
    """
    Find words that appear more than once using an O(n^2) nested loop approach.
    Returns a set of duplicate words.
    """
    tokens = tokenize(text)
    dups = set()
    n = len(tokens)
    for i in range(n):
        for j in range(i+1, n):
            if tokens[i] == tokens[j]:
                dups.add(tokens[i])
    return dups

from processor import tokenize, count_palindromes

def test_tokenize_handles_newlines():
    text = "Hello\\nWorld"
    tokens = tokenize(text)
    # Expect two tokens: 'hello' and 'world'
    assert tokens == ["hello", "world"]

def test_palindrome_count_excludes_single_letter():
    text = "Anna civic level a b"
    # 'Anna' (case-insensitive), 'civic', 'level' are palindromes of length >= 2.
    # Single letters 'a' and 'b' should not be counted.
    assert count_palindromes(text) == 3

How and why these materials were created
- Purpose: The materials form a small, self-contained Python project that requires the candidate to read an unfamiliar codebase, fix two failing tests (Task A), implement a small feature exposed via existing functions and tests (Task B), and make a focused optimization to a naive hot path used by a simple benchmark (Task C).
- Size and scope: The codebase is intentionally small (one module with ~80 lines, small utility, three test files, two input files). This keeps the work realistic but completable in 90 minutes.
- Determinism: All files are fully deterministic (concrete strings and counts). The large input is a fixed set of repeated lines (20 lines × 7 tokens = 140 tokens) to make the brute-force duplicate finder noticeably slower than an optimized method while keeping runtime reasonable in typical execution environments.

Data patterns and relationships
- Tokenization and word counts: The data files and tests rely on consistent lowercasing and punctuation removal. Tests expect tokenize to split on any whitespace (including newline), so the candidate must update the tokenize implementation to use a whitespace-aware split (str.split()) and/or improve punctuation stripping.
- Palindrome counting: The test expects palindromes of length >= 2 only. The baseline function counts single-letter palindromes as well; fixing the test requires adding a length guard.
- Task B feature: The new min_length filter must be supported by count_words and top_n_words. The test in tests/test_task_b_spec.py describes and verifies behavior: words shorter than min_length are ignored when computing top words. The candidate must integrate this parameter through the existing call chain rather than creating a separate function.
- Task C benchmark: bench.py uses find_duplicates_bruteforce which is intentionally O(n^2) by pairwise comparison across tokens. The large file repeats 7 distinct words across 20 lines (140 tokens) so the brute-force loop performs a measurable number of comparisons. The intended optimization is to replace the nested loop with a linear Counter-based approach (collections.Counter or set-based logic), reducing complexity to O(n). The candidate should record before/after timings.

Traps, ambiguities, and common errors
- tokenize splitting: The baseline splits on a literal space (" "), which fails to split on newlines and multiple spaces. Candidates might try to fix this by changing only the punctuation replacements and forget to change split(' ') to split(), which will leave tests failing.
- Palindrome counting: Candidates might normalize only case but forget to skip length-1 tokens. Another possible mistake is not normalizing before comparison (e.g., "Anna" vs "anna").
- Task B integration: A common incorrect approach is to add a new standalone function and new test that calls it, leaving existing top_n_words unchanged. The correct approach is to add an optional parameter (with a default) to count_words and top_n_words and ensure existing callers still work.
- Task C measurement: Some candidates might micro-optimize instead of changing the algorithm (e.g., minor micro-benchmarks). The intended, straightforward fix is to replace the nested loops with collections.Counter or a single pass to track seen words and duplicates. This yields a clear before/after timing difference.
- Tests: tests/test_task_b_spec.py as provided intentionally omits min_length in the call to top_n_words; the candidate must modify/add the test to include the min_length argument and adjust the assertion accordingly. The test file is the place where the candidate must add exactly one new/modified test.

- Correct tokenization: using lowercasing, removing specified punctuation, and splitting on any whitespace.
- Palindrome definition: case-insensitive palindrome with minimum length 2.
- Feature implementation: count_words(text, min_length=None) and top_n_words(text, n=3, min_length=None) must accept and propagate min_length.
- Optimization: find_duplicates_bruteforce should be replaced or supplemented by an O(n) implementation. The candidate may keep the original name but must change implementation to be faster; bench.py calls find_duplicates_bruteforce directly, so change must be in that function.
