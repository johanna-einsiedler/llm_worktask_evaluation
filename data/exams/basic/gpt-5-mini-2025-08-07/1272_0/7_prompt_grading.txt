import csv
from datetime import datetime

def read_csv(path):
    """
    Read CSV and return list of dict records.
    Expected columns: Transaction_ID,Customer_ID,Date,Amount,Region
    """
    with open(path, newline="", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        return [row for row in reader]

def parse_date(date_str):
    """
    Parse date string and return a datetime.date.
    BUG: The parsing format below is incorrect for inputs like '2023-01-05'.
    Candidate must fix this.
    """
    # BUG: wrong format used
    return datetime.strptime(date_str, "%d-%m-%Y").date()

def dedupe_records(records):
    """
    Remove duplicate transactions.
    Definition: duplicates share the same Transaction_ID (ignoring leading/trailing whitespace).
    CURRENT IMPLEMENTATION:
      - Naive O(n^2) nested loop (marked SLOW).
      - Compares Transaction_IDs without stripping whitespace (bug: fails when IDs have extra spaces).
      Candidate must both fix correctness (strip IDs) and optimize this SLOW function in Task 3.
    """
    unique = []
    for r in records:
        is_dup = False
        for u in unique:
            # SLOW: naive nested loop with extra work to simulate a hotspot.
            a = (r['Transaction_ID'] + u['Transaction_ID']).lower()
            # artificial workload to make this function measurable in benchmark
            for _ in range(2000):
                _ = a.endswith(u['Transaction_ID'])
            if u['Transaction_ID'] == r['Transaction_ID']:
                is_dup = True
                break
        if not is_dup:
            unique.append(r)
    return unique

def write_csv(records, out_file):
    if not records:
        return
    fieldnames = ["Transaction_ID","Customer_ID","Date","Amount","Region"]
    writer = csv.DictWriter(out_file, fieldnames=fieldnames)
    writer.writeheader()
    for r in records:
        writer.writerow(r)

#!/usr/bin/env python3
import time
from code.deduper import read_csv, dedupe_records

def benchmark_once():
    records = read_csv("sample_data/transactions.csv")
    t0 = time.perf_counter()
    dedupe_records(records)
    t1 = time.perf_counter()
    return (t1 - t0) * 1000.0

def main():
    runs = 5
    times = []
    for _ in range(runs):
        ms = benchmark_once()
        print(f"run_ms: {ms:.2f}")
        times.append(ms)
    avg = sum(times) / len(times)
    print(f"avg_ms: {avg:.2f}")

if __name__ == "__main__":
    main()
