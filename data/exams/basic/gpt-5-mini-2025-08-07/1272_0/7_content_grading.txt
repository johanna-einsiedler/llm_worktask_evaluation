# task_evaluation.py
import json
import os
import re
import sys
from typing import Any, Dict, List, Optional, Tuple

# Grading weights (per exam specification)
WEIGHTS = {
    "A_tests": 30,
    "A_scope": 10,
    "B_feature": 20,
    "B_test_integration": 10,
    "C_improvement": 15,
    "C_explanation": 10,
    "submission": 5,
}
MAX_TOTAL = sum(WEIGHTS.values())


def load_json(path: str) -> Any:
    """Load JSON from file with basic error handling."""
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {path}")
    except json.JSONDecodeError as e:
        raise ValueError(f"Invalid JSON in {path}: {e}")


def find_task(candidate_tasks: List[Dict], task_id: str) -> Optional[Dict]:
    """Return the task object with id == task_id or None."""
    for t in candidate_tasks:
        if str(t.get("id")).upper() == task_id:
            return t
    return None


def tests_all_passed(output: str) -> Tuple[bool, Dict[str, int]]:
    """
    Heuristic to determine whether pytest/unit-tests passed.
    Returns (all_passed, counts) where counts include passed, failed, errors.
    """
    if not isinstance(output, str):
        return False, {"passed": 0, "failed": 0, "errors": 0}

    # Look for "X passed" occurrences
    passed_match = re.search(r"(\d+)\s+passed", output, flags=re.IGNORECASE)
    passed = int(passed_match.group(1)) if passed_match else 0

    failed_match = re.search(r"(\d+)\s+failed", output, flags=re.IGNORECASE)
    failed = int(failed_match.group(1)) if failed_match else 0

    # Catch "ERROR" or "errors" reported by pytest
    error_match = re.search(r"(\d+)\s+error", output, flags=re.IGNORECASE)
    errors = int(error_match.group(1)) if error_match else 0

    # Also search for uppercase 'ERROR' or 'Traceback' as evidence of test errors
    if re.search(r"\bERROR\b", output):
        # If error was counted above, keep it, else mark at least 1
        if errors == 0:
            errors = 1

    if re.search(r"Traceback \(most recent call last\):", output):
        if errors == 0:
            errors = 1

    all_passed = (failed == 0 and errors == 0 and passed > 0)
    return all_passed, {"passed": passed, "failed": failed, "errors": errors}


def extract_benchmark_times(output: str) -> Dict[str, float]:
    """
    Parse benchmark times from bench output. Returns dict with keys 'small' and/or 'large' if found.
    Matches lines like:
      BENCHMARK: small time_seconds=0.000312
    """
    times: Dict[str, float] = {}
    if not isinstance(output, str):
        return times

    for match in re.finditer(
        r"BENCHMARK:\s*(?P<kind>\w+)\s+time_seconds=(?P<time>[0-9]*\.?[0-9]+)",
        output,
    ):
        kind = match.group("kind").lower()
        try:
            t = float(match.group("time"))
        except ValueError:
            continue
        times[kind] = t
    return times


def score_task_a(task: Dict) -> Tuple[int, List[str]]:
    """
    Task A scoring:
    - 30 pts: Task A tests pass (heuristic from test_output)
    - 10 pts: Changes minimal and well-scoped (we expect processor.py only)
    Returns (score, reasons)
    """
    reasons: List[str] = []
    score = 0
    expected_files_for_A = {"processor.py"}

    output = task.get("test_output", "")
    all_passed, counts = tests_all_passed(output)

    if all_passed:
        score += WEIGHTS["A_tests"]
        reasons.append(f"A-tests: detected {counts['passed']} passed, {counts['failed']} failed, {counts['errors']} errors -> awarding {WEIGHTS['A_tests']} pts")
    else:
        reasons.append(f"A-tests: tests did not fully pass (passed={counts['passed']}, failed={counts['failed']}, errors={counts['errors']}) -> awarding 0 pts for tests")

    modified_files = task.get("modified_files", {}) or {}
    modified_set = set(modified_files.keys())

    # Minimal changes heuristic:
    if modified_set == expected_files_for_A:
        score += WEIGHTS["A_scope"]
        reasons.append(f"A-scope: only modified {sorted(list(modified_set))} -> awarding full {WEIGHTS['A_scope']} pts")
    else:
        # If processor.py is among changed but also one extra file -> partial credit
        if "processor.py" in modified_set and len(modified_set) == 2:
            score += WEIGHTS["A_scope"] // 2
            reasons.append(f"A-scope: modified processor.py plus one other file {sorted(list(modified_set - expected_files_for_A))} -> awarding partial {WEIGHTS['A_scope']//2} pts")
        else:
            reasons.append(f"A-scope: modified files {sorted(list(modified_set))} -> expecting only processor.py for full credit -> awarding 0 pts")

    return score, reasons


def score_task_b(task: Dict) -> Tuple[int, List[str]]:
    """
    Task B scoring:
    - 20 pts: Feature implemented and tests pass (min_length implemented and test suite passes)
    - 10 pts: Added an integration unit test for the feature
    Partial credit rules:
      - If min_length implemented in count_words but not forwarded to top_n_words -> half of feature pts
      - If test missing but implementation present -> partial 10 pts
    """
    reasons: List[str] = []
    score = 0

    output = task.get("test_output", "")
    all_passed, counts = tests_all_passed(output)

    modified_files = task.get("modified_files", {}) or {}
    processor_content = modified_files.get("processor.py", "")
    test_b_content = modified_files.get("tests/test_task_b_spec.py", "")

    # Detect presence of min_length symbol in processor.py
    has_min_in_count = False
    has_min_in_top = False
    if isinstance(processor_content, str) and processor_content:
        # look for count_words(... min_length or def count_words(..., min_length
        if re.search(r"def\s+count_words\s*\(.*min_length", processor_content):
            has_min_in_count = True
        if re.search(r"def\s+top_n_words\s*\(.*min_length", processor_content):
            has_min_in_top = True

    # Check for the new unit test presence in the modified test file
    test_exists = False
    if isinstance(test_b_content, str) and "test_top_n_with_min_length" in test_b_content:
        test_exists = True

    # Evaluate feature implementation vs tests
    if all_passed and has_min_in_count and has_min_in_top:
        score += WEIGHTS["B_feature"]
        reasons.append(f"B-feature: tests passed and min_length present in count_words and top_n_words -> awarding {WEIGHTS['B_feature']} pts")
    else:
        # partial cases
        if has_min_in_count and not has_min_in_top:
            score += WEIGHTS["B_feature"] // 2
            reasons.append(f"B-feature: min_length implemented in count_words but not in top_n_words -> awarding partial {WEIGHTS['B_feature']//2} pts")
        elif has_min_in_count and not all_passed:
            # Implementation present but tests don't pass
            score += WEIGHTS["B_feature"] // 2
            reasons.append(f"B-feature: min_length implemented but tests did not pass -> awarding partial {WEIGHTS['B_feature']//2} pts")
        else:
            reasons.append("B-feature: min_length not detected in processor.py or implementation missing -> awarding 0 pts for feature")

    # Test/integration points: candidate must add exactly one unit test (or at least one)
    if test_exists:
        score += WEIGHTS["B_test_integration"]
        reasons.append(f"B-test: found test_top_n_with_min_length in tests/test_task_b_spec.py -> awarding {WEIGHTS['B_test_integration']} pts")
    else:
        # If the test file was modified but test name isn't present, give partial credit if implementation exists
        if has_min_in_count or has_min_in_top:
            score += WEIGHTS["B_test_integration"] // 2
            reasons.append(f"B-test: implementation present but specific unit test not found -> awarding partial {WEIGHTS['B_test_integration']//2} pts")
        else:
            reasons.append("B-test: no integration test found -> awarding 0 pts for test/integration")

    return score, reasons


def score_task_c(task: Dict) -> Tuple[int, List[str]]:
    """
    Task C scoring:
    - 15 pts: measurable quantitative improvement (baseline vs after)
      - full credit if improvement >= 10%
      - partial credit if improvement > 0 but < 10%
    - 10 pts: clear explanation of change and trade-offs
    Candidate should supply benchmark_before and benchmark_after numbers OR include them in test_output.
    """
    reasons: List[str] = []
    score = 0

    bench_before = task.get("benchmark_before", None)
    bench_after = task.get("benchmark_after", None)

    # If not provided, try to parse test_output for benchmark times
    test_output = task.get("test_output", "")
    parsed = extract_benchmark_times(test_output or "")

    if bench_before is None or bench_after is None:
        # attempt parsing: prefer 'large' times
        if "large" in parsed:
            # If only one large time present, we cannot compute improvement; treat as benchmark_after
            if bench_after is None:
                bench_after = parsed.get("large")
                reasons.append(f"C-benchmark: parsed bench_after from test_output large={bench_after}")
        # try small as fallback
        if bench_before is None and "small" in parsed:
            bench_before = parsed.get("small")
            reasons.append(f"C-benchmark: parsed bench_before from test_output small={bench_before}")

    # If both present, compute improvement
    if bench_before is not None and bench_after is not None:
        try:
            b_before = float(bench_before)
            b_after = float(bench_after)
            if b_before <= 0:
                reasons.append("C-benchmark: invalid non-positive benchmark_before -> awarding 0 pts for improvement")
            else:
                improvement = (b_before - b_after) / b_before
                if improvement >= 0.10:
                    score += WEIGHTS["C_improvement"]
                    reasons.append(f"C-improvement: benchmark_before={b_before:.6f}, after={b_after:.6f}, improvement={improvement*100:.1f}% -> awarding full {WEIGHTS['C_improvement']} pts")
                elif improvement > 0:
                    # small improvement -> partial
                    partial = WEIGHTS["C_improvement"] // 2
                    score += partial
                    reasons.append(f"C-improvement: improvement={improvement*100:.2f}% (<10%) -> awarding partial {partial} pts")
                else:
                    reasons.append(f"C-improvement: no positive improvement (before={b_before:.6f}, after={b_after:.6f}) -> awarding 0 pts")
        except (TypeError, ValueError):
            reasons.append("C-benchmark: could not parse benchmark_before/after as floats -> awarding 0 pts for improvement")
    else:
        # No both numbers available; check explanation for robustness claim
        explanation = (task.get("explanation") or "").lower()
        if any(kw in explanation for kw in ["prevent", "fixed", "robust", "handle", "edge", "avoid failure"]):
            # award partial for robustness fix if explanation claims it
            partial = WEIGHTS["C_improvement"] // 2
            score += partial
            reasons.append(f"C-improvement: no before/after numbers but explanation indicates robustness fix -> awarding partial {partial} pts")
        else:
            reasons.append("C-improvement: missing before/after benchmark numbers and no robustness claim -> awarding 0 pts")

    # Explanation scoring: require > ~30 chars and mention of algorithm or tradeoffs (heuristic)
    explanation_text = (task.get("explanation") or "").strip()
    explanation_lower = explanation_text.lower()
    if len(explanation_text) >= 30 and any(
        kw in explanation_lower for kw in ["counter", "set", "algorithm", "o(n)", "linear", "complexity", "cache", "trade-off", "tradeoff", "robust"]
    ):
        score += WEIGHTS["C_explanation"]
        reasons.append(f"C-explanation: explanation looks sufficient (len={len(explanation_text)}) and mentions algorithm/tradeoff -> awarding {WEIGHTS['C_explanation']} pts")
    else:
        # partial: some explanation present
        if len(explanation_text) >= 15:
            partial = WEIGHTS["C_explanation"] // 2
            score += partial
            reasons.append(f"C-explanation: short explanation present (len={len(explanation_text)}) -> awarding partial {partial} pts")
        else:
            reasons.append("C-explanation: explanation missing or too short -> awarding 0 pts for explanation")

    return score, reasons


def score_submission_quality(submission: Dict) -> Tuple[int, List[str]]:
    """
    5 points total for submission quality & reproducibility:
    - Simple checks: run_commands present and non-empty, tasks has 3 entries, top-level keys present, modified_files included where relevant.
    """
    reasons: List[str] = []
    score = 0
    # Basic schema checks
    required_top = ["name", "language", "time_taken_minutes", "run_commands", "tasks"]
    missing = [k for k in required_top if k not in submission]
    if missing:
        reasons.append(f"submission-schema: missing top-level keys: {missing} -> awarding 0 pts for submission quality")
        return score, reasons

    run_commands = submission.get("run_commands") or []
    tasks = submission.get("tasks") or []

    if not isinstance(run_commands, list) or len(run_commands) == 0:
        reasons.append("submission-schema: run_commands missing or empty -> awarding 0 pts for submission quality")
        return score, reasons

    if not isinstance(tasks, list) or len(tasks) != 3:
        reasons.append(f"submission-schema: tasks should be an array of 3 task objects, found {len(tasks)} -> awarding 0 pts")
        return score, reasons

    # If run_commands first command looks like the standard one, award full points
    first_cmd = str(run_commands[0]).strip().lower()
    if first_cmd.startswith("./run_all.sh") or first_cmd.startswith("python -m pytest"):
        score += WEIGHTS["submission"]
        reasons.append(f"submission-schema: run_commands appear valid (first: {run_commands[0]}) -> awarding full {WEIGHTS['submission']} pts")
    else:
        # partial: non-empty but not standard
        partial = WEIGHTS["submission"] // 2
        score += partial
        reasons.append(f"submission-schema: run_commands present but first command '{run_commands[0]}' is unusual -> awarding partial {partial} pts")

    return score, reasons


def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)

    submission_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    # Load files with error handling
    try:
        submission = load_json(submission_path)
    except Exception as e:
        print(f"Error loading submission JSON: {e}")
        sys.exit(2)

    try:
        answer_key = load_json(answer_key_path)
    except Exception as e:
        print(f"Error loading answer key JSON: {e}")
        sys.exit(2)

    results: Dict[str, Any] = {}
    per_task_scores: Dict[str, Dict[str, Any]] = {}
    total_score = 0
    deductions: List[str] = []

    # Validate tasks array
    candidate_tasks = submission.get("tasks", [])
    if not isinstance(candidate_tasks, list):
        print("Invalid submission: 'tasks' must be a list")
        sys.exit(2)

    # Score Task A
    taskA = find_task(candidate_tasks, "A")
    if taskA is None:
        per_task_scores["A"] = {
            "score": 0,
            "max_score": WEIGHTS["A_tests"] + WEIGHTS["A_scope"],
            "reasons": ["Task A object missing in submission"],
        }
    else:
        score_a, reasons_a = score_task_a(taskA)
        per_task_scores["A"] = {
            "score": score_a,
            "max_score": WEIGHTS["A_tests"] + WEIGHTS["A_scope"],
            "reasons": reasons_a,
        }
        total_score += score_a

    # Score Task B
    taskB = find_task(candidate_tasks, "B")
    if taskB is None:
        per_task_scores["B"] = {
            "score": 0,
            "max_score": WEIGHTS["B_feature"] + WEIGHTS["B_test_integration"],
            "reasons": ["Task B object missing in submission"],
        }
    else:
        score_b, reasons_b = score_task_b(taskB)
        per_task_scores["B"] = {
            "score": score_b,
            "max_score": WEIGHTS["B_feature"] + WEIGHTS["B_test_integration"],
            "reasons": reasons_b,
        }
        total_score += score_b

    # Score Task C
    taskC = find_task(candidate_tasks, "C")
    if taskC is None:
        per_task_scores["C"] = {
            "score": 0,
            "max_score": WEIGHTS["C_improvement"] + WEIGHTS["C_explanation"],
            "reasons": ["Task C object missing in submission"],
        }
    else:
        score_c, reasons_c = score_task_c(taskC)
        per_task_scores["C"] = {
            "score": score_c,
            "max_score": WEIGHTS["C_improvement"] + WEIGHTS["C_explanation"],
            "reasons": reasons_c,
        }
        total_score += score_c

    # Submission quality & reproducibility
    score_sub, reasons_sub = score_submission_quality(submission)
    per_task_scores["submission_quality"] = {
        "score": score_sub,
        "max_score": WEIGHTS["submission"],
        "reasons": reasons_sub,
    }
    total_score += score_sub

    # Compose final results
    results["per_task"] = per_task_scores
    results["total_points"] = total_score
    results["max_points"] = MAX_TOTAL
    pct = (float(total_score) / float(MAX_TOTAL)) * 100.0 if MAX_TOTAL > 0 else 0.0
    # Normalize to 1 decimal place
    overall_score = round(pct, 2)
    results["percentage"] = overall_score
    results["overall_score"] = overall_score  # required exact variable name in output JSON
    # Provide human-readable deductions/notes
    results["notes"] = []
    # Collect all reasons into notes
    for k, v in per_task_scores.items():
        for line in v.get("reasons", []):
            results["notes"].append(f"{k}: {line}")

    # Include the submission and answer_key top-level minimal validation
    results["validated_submission_keys"] = {
        "has_name": "name" in submission,
        "has_run_commands": bool(submission.get("run_commands")),
        "tasks_count": len(submission.get("tasks", [])),
    }

    # Save test_results.json next to script
    script_dir = os.path.dirname(os.path.abspath(__file__))
    out_path = os.path.join(script_dir, "test_results.json")
    try:
        with open(out_path, "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
    except Exception as e:
        print(f"Error writing results file: {e}")
        sys.exit(2)

    # Print a short summary to stdout
    print(f"Grading complete. Total: {total_score}/{MAX_TOTAL} ({overall_score}%).")
    print(f"Results written to {out_path}")


if __name__ == "__main__":
    main()
