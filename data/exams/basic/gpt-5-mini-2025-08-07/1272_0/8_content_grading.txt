#!/usr/bin/env python3
"""
task_evaluation.py

Automated grader for the Practical Code Maintenance Basic exam.

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

Produces:
    test_results.json (in the same directory as this script)

This grader applies the weights and rules described in the exam materials.
Only Python standard library is used.
"""

import json
import os
import re
import sys
from typing import Any, Dict, List, Tuple

# -------------------------
# Helper utilities
# -------------------------


def load_json(path: str) -> Any:
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        raise
    except json.JSONDecodeError as e:
        raise ValueError(f"Invalid JSON in {path}: {e}")


def safe_get_task(tasks_list: List[Dict], tid: str) -> Dict:
    for t in tasks_list:
        if t.get("id") == tid:
            return t
    return {}


def find_pass_fail_counts(output: str) -> Tuple[int, int]:
    """
    Attempt to extract 'X passed' and 'Y failed' counts from pytest/unittest output.
    Returns (passed, failed). If not found, returns (0,0) as default.
    """
    if not output:
        return 0, 0
    out = output
    # look for patterns like '4 passed' and '1 failed'
    m_pass = re.search(r"(\d+)\s+passed", out, flags=re.IGNORECASE)
    m_fail = re.search(r"(\d+)\s+failed", out, flags=re.IGNORECASE)
    passed = int(m_pass.group(1)) if m_pass else 0
    failed = int(m_fail.group(1)) if m_fail else 0
    # If output contains 'ERROR' or 'Traceback' consider that as failed
    if re.search(r"\b(error|traceback|exception)\b", out, flags=re.IGNORECASE):
        # mark at least one failure if nothing detected explicitly
        if failed == 0:
            failed = 1
    return passed, failed


def tests_appear_passing(output: str) -> bool:
    """
    Heuristic: return True if output indicates no failures.
    Conditions:
      - If a failed count is present and >0 => return False
      - Else if 'passed' count present and no errors reported => True
      - Else False (can't confirm)
    """
    if output is None:
        return False
    passed, failed = find_pass_fail_counts(output)
    if failed > 0:
        return False
    if passed > 0:
        # ensure no 'FAILED' or 'ERROR' markers
        if re.search(r"\b(fail|failed|error|traceback)\b", output, flags=re.IGNORECASE):
            # If failed word appears but failed count is zero, still consider fail if 'FAILED' banner present
            if re.search(r"\bFAILED\b", output, flags=re.IGNORECASE) or re.search(
                r"\bERROR\b", output, flags=re.IGNORECASE
            ):
                return False
        return True
    # else conservative: not enough evidence tests passed
    return False


def contains_min_length_in_signature(proc_content: str) -> Tuple[bool, bool]:
    """
    Check whether processor.py defines count_words and top_n_words with min_length parameter.
    Returns tuple (count_words_has_min, top_n_has_min).
    """
    if not proc_content:
        return False, False
    # naive detection: look for 'def count_words(' line and check for 'min_length' inside parentheses
    def_sig_re = re.compile(r"def\s+count_words\s*\(([^)]*)\)", re.MULTILINE)
    m = def_sig_re.search(proc_content)
    count_has = False
    if m:
        params = m.group(1)
        if "min_length" in params:
            count_has = True
    top_sig_re = re.compile(r"def\s+top_n_words\s*\(([^)]*)\)", re.MULTILINE)
    m2 = top_sig_re.search(proc_content)
    top_has = False
    if m2:
        params = m2.group(1)
        if "min_length" in params:
            top_has = True
    return count_has, top_has


def detect_counter_usage(proc_content: str) -> bool:
    """
    Detect whether collections.Counter or Counter is used in processor.py content.
    """
    if not proc_content:
        return False
    if "Counter(" in proc_content or "from collections import Counter" in proc_content:
        return True
    return False


def detect_find_duplicates_reimpl(proc_content: str) -> bool:
    """
    Detect if find_duplicates_bruteforce has been reimplemented (not the nested loop).
    Heuristic: look for nested 'for i in' 'for j in' pattern (original O(n^2) form).
    If absent and presence of Counter or linear logic found, consider reimplemented.
    """
    if not proc_content:
        return False
    nested_loop_pattern = re.search(r"for\s+\w+\s+in\s+range\([^)]*\):\s*?\n\s*for\s+\w+\s+in\s+range\(", proc_content, flags=re.MULTILINE)
    if nested_loop_pattern:
        # still nested => likely brute-force
        return False
    # if Counter used or 'counts = Counter' or 'counts =' with 'tokens' then reimplemented
    if detect_counter_usage(proc_content):
        return True
    # other heuristics: presence of 'counts = {}' + single pass 'for w in tokens' and building counts
    if re.search(r"counts\s*=\s*\{", proc_content) and re.search(r"for\s+\w+\s+in\s+tokens", proc_content):
        return True
    return False

# Grading logic
# -------------------------


def grade_task_a(candidate_task: Dict, answer_task: Dict) -> Tuple[int, int, List[str]]:
    """
    Task A: 40 points total
      - 30 pts: All Task A tests pass in candidate-submitted test_output
      - 10 pts: Changes are minimal and well-scoped (only processor.py changed) -> award full
                 If processor.py present but extra files changed => partial 5 pts
                 If processor.py missing => 0 pts
    """
    max_points = 40
    messages: List[str] = []
    score = 0

    # 30 pts for tests passing
    test_output = candidate_task.get("test_output", "")
    if tests_appear_passing(test_output):
        score += 30
        messages.append("30/30: Task A tests appear to pass (no failures detected in test output).")
    else:
        messages.append("0/30: Task A tests did not appear to pass (failing tests or errors detected).")

    # 10 pts for minimal changes: expect processor.py modified and minimal other changes
    modified = candidate_task.get("modified_files", {}) or {}
    modified_keys = set(modified.keys())
    if "processor.py" in modified_keys:
        # If only processor.py changed, full points
        others = modified_keys - {"processor.py"}
        if not others:
            score += 10
            messages.append("10/10: Only processor.py was modified for Task A (minimal, well-scoped change).")
        else:
            # partial credit if processor.py included but other files modified too
            score += 5
            messages.append(
                f"5/10: processor.py was modified but additional files were changed ({sorted(list(others))})."
            )
    else:
        messages.append("0/10: processor.py was not modified for Task A (expected).")

    return score, max_points, messages


def grade_task_b(candidate_task: Dict, answer_task: Dict) -> Tuple[int, int, List[str]]:
    """
    Task B: 30 points total
      - 20 pts: new feature works and its tests pass (min_length implemented)
          * Check processor.py has min_length in count_words and top_n_words signatures.
          * If only count_words has it but top_n_words doesn't, partial 10/20.
      - 10 pts: Integration with existing code and candidate added unit test
          * Expect tests/test_task_b_spec.py included in modified files and its content has 'min_length=4' usage
    """
    max_points = 30
    messages: List[str] = []
    score = 0

    modified = candidate_task.get("modified_files", {}) or {}
    proc_content = modified.get("processor.py", "") or ""
    # detect min_length presence in function signatures
    count_has_min, top_has_min = contains_min_length_in_signature(proc_content)

    test_output = candidate_task.get("test_output", "")

    # Feature correctness points (20)
    if count_has_min and top_has_min:
        # also require tests passed (the full test suite or at least relevant test)
        if tests_appear_passing(test_output):
            score += 20
            messages.append("20/20: min_length implemented in both count_words and top_n_words and tests pass.")
        else:
            # implemented, but tests don't appear passing
            score += 10
            messages.append("10/20: min_length implemented in both functions but tests do not appear to pass.")
    elif count_has_min and not top_has_min:
        # partial credit per rubric
        score += 10
        messages.append(
            "10/20: min_length implemented in count_words but not propagated to top_n_words (partial credit)."
        )
    else:
        messages.append("0/20: min_length not implemented in expected functions.")

    # Integration & unit test (10)
    test_file_name = "tests/test_task_b_spec.py"
    test_file_present = test_file_name in modified
    test_file_content = modified.get(test_file_name, "") or ""

    if test_file_present:
        # check the test includes min_length usage
        if "min_length" in test_file_content or "min_length=" in test_file_content:
            # ensure test output shows passing
            if tests_appear_passing(test_output):
                score += 10
                messages.append("10/10: Integration test updated in tests/test_task_b_spec.py and tests pass.")
            else:
                score += 5
                messages.append(
                    "5/10: Test for min_length present in tests/test_task_b_spec.py but tests did not appear to pass."
                )
        else:
            # test file changed but not invoking min_length properly
            score += 3
            messages.append("3/10: tests/test_task_b_spec.py was modified but does not appear to call min_length.")
    else:
        # no test added/modified in expected place
        messages.append("0/10: Expected unit test in tests/test_task_b_spec.py not provided in modified_files.")

    return score, max_points, messages


def grade_task_c(candidate_task: Dict, answer_task: Dict) -> Tuple[int, int, List[str]]:
    """
    Task C: 25 points total
      - 15 pts: Demonstrable quantitative improvement (benchmark_before and benchmark_after numbers)
          * Full points if both numbers provided and benchmark_after < benchmark_before by at least 5% (or at least a small absolute drop).
          * Partial points if improvement is small but positive.
          * If no numbers but algorithmic change detected (e.g., Counter) -> partial credit.
      - 10 pts: Explanation present and mentions algorithmic choice / trade-offs
          * Full 10 if explanation contains 'Counter', 'O(n)', 'linear', 'algorithm', or 'cache' keywords.
          * 5 if explanation non-empty but not specific.
    """
    max_points = 25
    messages: List[str] = []
    score = 0

    bench_before = candidate_task.get("benchmark_before")
    bench_after = candidate_task.get("benchmark_after")
    proc_changed_files = candidate_task.get("modified_files", {}) or {}
    proc_content = proc_changed_files.get("processor.py", "") or ""
    explanation = (candidate_task.get("explanation") or "").strip()
    test_output = candidate_task.get("test_output", "") or ""

    # 15 pts: measurable improvement
    measurable_awarded = 0
    if isinstance(bench_before, (int, float)) and isinstance(bench_after, (int, float)):
        try:
            b_before = float(bench_before)
            b_after = float(bench_after)
            if b_after < b_before:
                # compute relative improvement
                if b_before > 0:
                    rel = (b_before - b_after) / b_before
                else:
                    rel = 0.0
                # thresholds
                if rel >= 0.05 or (b_before - b_after) >= 0.001:
                    # acceptable measurable improvement
                    score += 15
                    measurable_awarded = 15
                    messages.append(
                        f"15/15: benchmark shows improvement: before={b_before:.6f}s after={b_after:.6f}s (rel improvement {rel*100:.1f}%)."
                    )
                else:
                    # small but positive improvement
                    score += 8
                    measurable_awarded = 8
                    messages.append(
                        f"8/15: small improvement detected: before={b_before:.6f}s after={b_after:.6f}s (rel {rel*100:.1f}%)."
                    )
            else:
                messages.append(
                    f"0/15: benchmark shows no improvement or slower: before={b_before:.6f}s after={b_after:.6f}s."
                )
        except Exception:
            messages.append("0/15: Invalid numeric benchmark values provided.")
    else:
        # No numeric benchmarks provided; try to detect algorithmic change
        if detect_find_duplicates_reimpl(proc_content):
            # partial credit
            score += 10
            measurable_awarded = 10
            messages.append(
                "10/15: No numeric benchmarks provided but processor.py appears to have reimplemented duplicate detection (linear approach detected); awarded partial credit."
            )
        else:
            messages.append("0/15: No numeric benchmarks and no clear algorithmic improvement detected.")

    # 10 pts: clear explanation
    explanation_award = 0
    if explanation:
        # look for keywords indicating algorithmic reasoning
        keywords = ["counter", "collections.Counter", "o(n)", "o(n^2)", "o(n2)", "linear", "algorithm", "complexity", "cache", "caching", "set(", "hash", "trade-off", "robust"]
        expl_low = explanation.lower()
        if any(k.lower() in expl_low for k in keywords):
            score += 10
            explanation_award = 10
            messages.append("10/10: Explanation provided and mentions algorithmic choice or trade-offs.")
        else:
            score += 5
            explanation_award = 5
            messages.append("5/10: Explanation provided but lacks specific algorithmic keywords; partial credit.")
    else:
        messages.append("0/10: No explanation provided for Task C.")

    return score, max_points, messages


def grade_submission_quality(candidate: Dict) -> Tuple[int, int, List[str]]:
    """
    5 points total for submission quality & reproducibility
      - run_commands present & non-empty
      - JSON contains required top-level keys and 3 tasks
      - modified_files entries included for changed files (we check that modified_files exists for each task)
    """
    max_points = 5
    messages: List[str] = []
    score = 0

    # run_commands check
    run_cmds = candidate.get("run_commands")
    if isinstance(run_cmds, list) and run_cmds:
        score += 2
        messages.append("2/2: run_commands present.")
    else:
        messages.append("0/2: run_commands missing or empty.")

    tasks = candidate.get("tasks")
    if isinstance(tasks, list) and len(tasks) == 3:
        score += 2
        messages.append("2/2: tasks array present and contains 3 task objects.")
    else:
        messages.append("0/2: tasks array missing or does not contain 3 tasks.")

    # modified_files presence: ensure each task has modified_files key (can be empty dict)
    ok = True
    if isinstance(tasks, list):
        for t in tasks:
            if "modified_files" not in t:
                ok = False
                break
    if ok:
        score += 1
        messages.append("1/1: Each task includes modified_files key (even if empty).")
    else:
        messages.append("0/1: One or more task objects missing modified_files key.")

    return score, max_points, messages


# -------------------------
# Main grading routine
# -------------------------


def grade_submission(candidate_path: str, answer_key_path: str) -> Dict:
    results: Dict[str, Any] = {
        "errors": [],
        "tasks": {},
        "total_points": 0,
        "max_points": 100,
        "overall_score": 0.0,
    }

    # Load files
    try:
        candidate = load_json(candidate_path)
    except FileNotFoundError:
        raise SystemExit(f"Candidate submission file not found: {candidate_path}")
    except Exception as e:
        raise SystemExit(f"Failed to load candidate JSON: {e}")

    try:
        answer_key = load_json(answer_key_path)
    except FileNotFoundError:
        raise SystemExit(f"Answer key file not found: {answer_key_path}")
    except Exception as e:
        raise SystemExit(f"Failed to load answer key JSON: {e}")

    # Basic validation of candidate JSON structure
    top_level_required = ["name", "language", "time_taken_minutes", "run_commands", "tasks"]
    missing = [k for k in top_level_required if k not in candidate]
    if missing:
        results["errors"].append(f"Candidate submission missing top-level keys: {missing}")

    # Locate tasks in candidate and answer_key
    cand_tasks = candidate.get("tasks", [])
    ans_tasks = answer_key.get("tasks", [])

    # Build mapping by id for convenience
    cand_map = {t.get("id"): t for t in cand_tasks if "id" in t}
    ans_map = {t.get("id"): t for t in ans_tasks if "id" in t}

    # Grade Task A
    try:
        cand_A = cand_map.get("A", {})
        ans_A = ans_map.get("A", {})
        score_a, max_a, messages_a = grade_task_a(cand_A, ans_A)
    except Exception as e:
        score_a, max_a = 0, 40
        messages_a = [f"Error grading Task A: {e}"]
    results["tasks"]["A"] = {"score": score_a, "max": max_a, "notes": messages_a}
    results["total_points"] += score_a

    # Grade Task B
    try:
        cand_B = cand_map.get("B", {})
        ans_B = ans_map.get("B", {})
        score_b, max_b, messages_b = grade_task_b(cand_B, ans_B)
    except Exception as e:
        score_b, max_b = 0, 30
        messages_b = [f"Error grading Task B: {e}"]
    results["tasks"]["B"] = {"score": score_b, "max": max_b, "notes": messages_b}
    results["total_points"] += score_b

    # Grade Task C
    try:
        cand_C = cand_map.get("C", {})
        ans_C = ans_map.get("C", {})
        score_c, max_c, messages_c = grade_task_c(cand_C, ans_C)
    except Exception as e:
        score_c, max_c = 0, 25
        messages_c = [f"Error grading Task C: {e}"]
    results["tasks"]["C"] = {"score": score_c, "max": max_c, "notes": messages_c}
    results["total_points"] += score_c

    # Submission quality
    try:
        score_sub, max_sub, messages_sub = grade_submission_quality(candidate)
    except Exception as e:
        score_sub, max_sub = 0, 5
        messages_sub = [f"Error grading submission quality: {e}"]
    results["submission_quality"] = {"score": score_sub, "max": max_sub, "notes": messages_sub}
    results["total_points"] += score_sub

    # Final totals and percentage
    results["max_points"] = 100
    results["percentage"] = round((results["total_points"] / results["max_points"]) * 100, 2)
    results["overall_score"] = results["percentage"]

    # Provide a short summary explaining deductions
    summary_lines: List[str] = []
    # Task A summary
    a_notes = results["tasks"]["A"]["notes"]
    summary_lines.append(f"Task A: {results['tasks']['A']['score']}/{results['tasks']['A']['max']}.")
    summary_lines.extend(a_notes)

    # Task B summary
    b_notes = results["tasks"]["B"]["notes"]
    summary_lines.append(f"Task B: {results['tasks']['B']['score']}/{results['tasks']['B']['max']}.")
    summary_lines.extend(b_notes)

    # Task C summary
    c_notes = results["tasks"]["C"]["notes"]
    summary_lines.append(f"Task C: {results['tasks']['C']['score']}/{results['tasks']['C']['max']}.")
    summary_lines.extend(c_notes)

    # Submission
    s_notes = results["submission_quality"]["notes"]
    summary_lines.append(f"Submission quality: {results['submission_quality']['score']}/{results['submission_quality']['max']}.")
    summary_lines.extend(s_notes)

    results["final_summary"] = "\n".join(summary_lines)

    return results


# -------------------------
# Script entrypoint
# -------------------------


def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)

    candidate_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    try:
        results = grade_submission(candidate_path, answer_key_path)
    except Exception as e:
        # write an error result file
        out = {"errors": [str(e)], "overall_score": 0.0}
        with open("test_results.json", "w", encoding="utf-8") as f:
            json.dump(out, f, indent=2)
        print(f"Grading failed: {e}")
        sys.exit(1)

    # Write results to test_results.json
    try:
        with open("test_results.json", "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
    except Exception as e:
        print(f"Failed to write test_results.json: {e}")
        sys.exit(1)

    # Print a small console summary
    print("Grading complete. Results written to test_results.json")
    print(f"Overall score: {results['overall_score']:.2f}% ({results['total_points']}/{results['max_points']})")


if __name__ == "__main__":
    main()
