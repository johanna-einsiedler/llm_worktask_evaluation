#!/usr/bin/env python3
"""
task_evaluation.py

Usage:
    python task_evaluation.py <candidate_submission.json> <answer_key.json>

This script grades a candidate submission against an answer key using the
grading weights and heuristics described in the exam evaluation guidance.
It writes a detailed grading report to test_results.json in the script's directory.

Author: Automated grader
"""

import json
import os
import re
import sys
from typing import Any, Dict, List, Tuple

# Constants: scoring weights (from evaluation guideline)
MAX_TASK_A = 40  # 30 for tests passing, 10 for minimal changes
MAX_TASK_B = 30  # 20 for feature/tests, 10 for integration + added test
MAX_TASK_C = 25  # 15 for measurable improvement, 10 for explanation
MAX_SUBMISSION = 5  # submission quality & reproducibility
TOTAL_MAX = MAX_TASK_A + MAX_TASK_B + MAX_TASK_C + MAX_SUBMISSION

RESULTS_FILENAME = "test_results.json"


# ---------- Utility helpers ----------

def load_json_file(path: str) -> Any:
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        raise SystemExit(f"Error: file not found: {path}")
    except json.JSONDecodeError as e:
        raise SystemExit(f"Error: invalid JSON in {path}: {e}")


def safe_get(d: Dict, key: str, default=None):
    return d.get(key, default)


def normalize_whitespace(s: str) -> str:
    return re.sub(r'\s+', ' ', s.strip()) if isinstance(s, str) else s


def parse_pytest_summary(output: str) -> Dict[str, Any]:
    """
    Parse pytest-style summary from output. Return dict with keys:
    - passed (int)
    - failed (int)
    - errors (int)
    - skipped (int)
    - summary_found (bool)
    - success (bool) -- True if there are zero failed/errors
    """
    result = {"passed": 0, "failed": 0, "errors": 0, "skipped": 0, "summary_found": False, "success": False}
    if not isinstance(output, str):
        return result
    # Look for lines like "4 passed", "1 failed", "1 error", "1 skipped"
    passed = re.search(r'(\d+)\s+passed', output)
    failed = re.search(r'(\d+)\s+failed', output)
    errors = re.search(r'(\d+)\s+error', output)  # 'error' singular/plural
    skipped = re.search(r'(\d+)\s+skipped', output)
    if passed:
        result["passed"] = int(passed.group(1))
    if failed:
        result["failed"] = int(failed.group(1))
    if errors:
        result["errors"] = int(errors.group(1))
    if skipped:
        result["skipped"] = int(skipped.group(1))
    # If any of these were found, consider summary_found True
    if any([passed, failed, errors, skipped]):
        result["summary_found"] = True
    # Heuristic for success: no 'failed' and no 'error' mention (case-insensitive)
    lowered = output.lower()
    if ('failed' not in lowered) and ('error' not in lowered) and result["summary_found"]:
        # If summary found and no failed/error, it's success
        result["success"] = True
    # Another safety: if output contains 'failed' but number 0, that's OK; covered above
    return result


def extract_benchmark_time_from_output(output: str, kind: str = "large") -> float:
    """
    Parse bench.py output lines like:
    BENCHMARK: large time_seconds=0.003500
    Returns float or None if not found.
    """
    if not isinstance(output, str):
        return None
    pattern = re.compile(r'BENCHMARK:\s*' + re.escape(kind) + r'\s+time_seconds=([0-9]*\.?[0-9]+)', re.IGNORECASE)
    m = pattern.search(output)
    if m:
        try:
            return float(m.group(1))
        except Exception:
            return None
    return None


def safe_is_number(x):
    return isinstance(x, (int, float))


# ---------- Grading logic per task ----------

def grade_task_a(task_obj: Dict) -> Tuple[int, List[str]]:
    """
    Task A scoring:
      - 30 pts: All Task A tests pass (pytest summary shows no failures/errors)
      - 10 pts: Changes are minimal and well-scoped (heuristic: only processor.py modified)
    Returns (points_awarded, list_of_explanations)
    """
    reasons: List[str] = []
    awarded = 0

    # 30 pts for tests passing
    test_output = safe_get(task_obj, "test_output", "")
    parse = parse_pytest_summary(test_output)
    if parse["summary_found"] and parse["success"]:
        awarded += 30
        reasons.append("Task A tests passed according to provided test_output (+30).")
    else:
        # If summary not found, attempt to detect '4 passed' (generic)
        # Provide partial guidance in explanation
        if parse["summary_found"] and not parse["success"]:
            reasons.append("Task A tests did not pass (pytest summary indicates failures or errors). (+0 for tests).")
        else:
            reasons.append("Could not detect pytest summary for Task A; assuming tests not passing. (+0 for tests).")

    # 10 pts for minimal, well-scoped changes
    modified_files = safe_get(task_obj, "modified_files", {})
    if not isinstance(modified_files, dict):
        modified_files = {}
    # Heuristic: minimal means only processor.py changed for Task A
    files_changed = set(modified_files.keys())
    if len(files_changed) == 0:
        reasons.append("No files were reported modified for Task A; cannot award points for minimal changes (+0).")
    else:
        # Acceptable minimal sets for Task A: processor.py alone
        if files_changed == {"processor.py"}:
            awarded += 10
            reasons.append("Only processor.py was changed for Task A; changes appear minimal and well-scoped (+10).")
        else:
            # If processor.py present but there are a couple of other expected changes (rare), give partial credit
            if "processor.py" in files_changed and len(files_changed) <= 2:
                awarded += 5
                reasons.append("processor.py modified along with one other file; partial credit for scoped changes (+5).")
            else:
                reasons.append(f"Modified files for Task A are {sorted(files_changed)}; not minimal (+0).")

    return awarded, reasons


def grade_task_b(task_obj: Dict) -> Tuple[int, List[str]]:
    """
    Task B scoring:
      - 20 pts: The new feature works and its test(s) pass (pytest summary shows no failures/errors)
      - 10 pts: Integration with existing code is correct and the candidate added at least one unit test (heuristic)
    Returns (points_awarded, list_of_explanations)
    """
    reasons: List[str] = []
    awarded = 0

    test_output = safe_get(task_obj, "test_output", "")
    parse = parse_pytest_summary(test_output)
    if parse["summary_found"] and parse["success"]:
        awarded += 20
        reasons.append("All tests passed in task_b test_output (+20).")
    else:
        if parse["summary_found"] and not parse["success"]:
            reasons.append("Tests did not pass after Task B changes (pytest summary indicates failures or errors) (+0).")
        else:
            reasons.append("Could not detect pytest summary for Task B; assuming tests not passing (+0).")

    # 10 pts for integration and added test
    modified_files = safe_get(task_obj, "modified_files", {}) or {}
    files_changed = set(modified_files.keys())

    processor_text = modified_files.get("processor.py", "")
    testfile_text = modified_files.get("tests/test_task_b_spec.py", None)
    # Heuristic checks:
    # - processor.py should contain 'min_length' as a parameter
    has_min_length_in_processor = isinstance(processor_text, str) and ('min_length' in processor_text)
    # - tests/test_task_b_spec.py should be included in modified_files and contain 'min_length'
    has_test_modified = isinstance(testfile_text, str) and ('min_length' in testfile_text)
    if has_min_length_in_processor and has_test_modified:
        awarded += 10
        reasons.append("processor.py implements min_length and the Task B test file was added/modified to include min_length (+10).")
    else:
        # Some partial credit possibilities:
        if has_min_length_in_processor and not has_test_modified:
            awarded += 6
            reasons.append("processor.py contains min_length but the Task B test file was not provided in modified_files; partial credit (+6).")
        elif not has_min_length_in_processor and has_test_modified:
            awarded += 4
            reasons.append("Task B test was provided but processor.py does not appear to include min_length; partial credit (+4).")
        else:
            reasons.append("No evidence of integration (min_length) or added Task B test in modified_files (+0).")

    return awarded, reasons


def grade_task_c(task_obj: Dict) -> Tuple[int, List[str]]:
    """
    Task C scoring:
      - 15 pts: Demonstrable quantitative improvement (benchmark_before, benchmark_after)
        * Full 15 if after <= 0.8 * before (>=20% improvement) OR absolute improvement is significant
        * Partial scaled credit otherwise.
      - 10 pts: Clear explanation of change, alternatives considered, and trade-offs
        * Heuristic: explanation length and keywords
    Returns (points_awarded, list_of_explanations)
    """
    reasons: List[str] = []
    awarded = 0

    before = safe_get(task_obj, "benchmark_before", None)
    after = safe_get(task_obj, "benchmark_after", None)

    # If not provided as numbers, try to parse from test_output
    if not safe_is_number(before) or not safe_is_number(after):
        # try parse test_output for 'BENCHMARK: large time_seconds=...'
        test_output = safe_get(task_obj, "test_output", "")
        parsed_large = extract_benchmark_time_from_output(test_output, kind="large")
        parsed_small = extract_benchmark_time_from_output(test_output, kind="small")
        # If both present in output, we cannot determine before/after from single run; rely on fields
        # Keep None if not found
        if parsed_large is not None and (not safe_is_number(before)):
            before = before or parsed_large  # only set if missing
        # Note: candidate should provide explicit benchmark_before/after fields; we use them preferentially.

    if safe_is_number(before) and safe_is_number(after):
        # compute improvement
        try:
            before_f = float(before)
            after_f = float(after)
            if after_f < before_f:
                improvement = (before_f - after_f) / max(before_f, 1e-12)
                improvement_pct = improvement * 100.0
                reasons.append(f"Provided benchmark_before={before_f:.6f}, benchmark_after={after_f:.6f}; improvement={improvement_pct:.1f}%")
                # Scoring thresholds:
                if improvement_pct >= 20.0:
                    awarded += 15
                    reasons.append("Improvement >=20%: full Task C performance points awarded (+15).")
                elif improvement_pct >= 5.0:
                    # partial credit scaled between 5% and 20%
                    # linear mapping: 5% -> 5 pts, 20% -> 12 pts roughly; but spec says full 15 for ~30-50%
                    # We'll use 8 pts for 5-20% improvement (reasonable partial)
                    awarded += 8
                    reasons.append("Improvement between 5% and 20%: partial Task C performance points (+8).")
                else:
                    # small improvement <5%
                    awarded += 3
                    reasons.append("Improvement <5%: small performance gain, small partial credit (+3).")
            else:
                reasons.append("No improvement observed (after >= before); no Task C performance points (+0).")
        except Exception as e:
            reasons.append(f"Error computing improvement: {e}; no Task C performance points (+0).")
    else:
        reasons.append("benchmark_before and/or benchmark_after not provided or not numeric; cannot award performance points (+0).")

    # 10 pts for explanation quality: heuristics based on length and use of expected keywords
    explanation_text = safe_get(task_obj, "explanation", "") or ""
    explanation_text_strip = normalize_whitespace(explanation_text) if explanation_text else ""
    explanation_len = len(explanation_text_strip)
    explanation_lower = explanation_text_strip.lower()

    if explanation_len >= 100:
        awarded += 10
        reasons.append("Detailed explanation provided (>=100 chars): full explanation points (+10).")
    elif explanation_len >= 40:
        awarded += 6
        reasons.append("Reasonable explanation provided (40-99 chars): partial explanation points (+6).")
    elif explanation_len >= 20:
        awarded += 3
        reasons.append("Short explanation provided (20-39 chars): small partial explanation points (+3).")
    else:
        reasons.append("No sufficient explanation (less than 20 chars): no explanation points (+0).")

    return awarded, reasons


def grade_submission_quality(submission: Dict) -> Tuple[int, List[str]]:
    """
    5 pts total for submission quality & reproducibility:
      - run_commands present and non-empty (2 pts)
      - JSON schema present: name, language, time_taken_minutes, tasks array (2 pts)
      - modified_files present for tasks that were completed (1 pt)
    Returns (points_awarded, list_of_explanations)
    """
    reasons: List[str] = []
    awarded = 0

    run_commands = safe_get(submission, "run_commands", None)
    if isinstance(run_commands, list) and len(run_commands) >= 1 and any(isinstance(c, str) and c.strip() for c in run_commands):
        awarded += 2
        reasons.append("run_commands present and non-empty (+2).")
    else:
        reasons.append("run_commands missing or empty (+0).")

    # Check top-level metadata keys
    required_keys = ["name", "language", "time_taken_minutes", "tasks"]
    missing = [k for k in required_keys if k not in submission]
    if not missing:
        awarded += 2
        reasons.append("Required top-level keys present (name, language, time_taken_minutes, tasks) (+2).")
    else:
        reasons.append(f"Missing top-level keys: {missing} (+0).")

    # Check modified_files present for tasks declared completed
    tasks = safe_get(submission, "tasks", [])
    if isinstance(tasks, list):
        completed_tasks = [t for t in tasks if safe_get(t, "status") == "completed"]
        count_ok = 0
        for t in completed_tasks:
            mf = safe_get(t, "modified_files", {})
            if isinstance(mf, dict) and len(mf) >= 0:
                count_ok += 1
        # If at least one completed task includes modified_files, award final point
        if len(completed_tasks) > 0:
            # more rigorous: ensure tasks have modified_files key present (even empty dict allowed)
            if all("modified_files" in t for t in completed_tasks):
                awarded += 1
                reasons.append("modified_files present for completed tasks (+1).")
            else:
                reasons.append("Some completed tasks missing modified_files key (+0).")
        else:
            reasons.append("No tasks reported as completed; cannot verify modified_files (+0).")
    else:
        reasons.append("tasks key is not a list (+0).")

    return awarded, reasons


# ---------- Main evaluation flow ----------

def grade_submission(candidate_path: str, answer_key_path: str) -> Dict[str, Any]:
    # Load files
    try:
        candidate = load_json_file(candidate_path)
    except SystemExit as e:
        raise
    except Exception as e:
        raise SystemExit(f"Failed to load candidate submission: {e}")

    try:
        answer_key = load_json_file(answer_key_path)
    except SystemExit as e:
        raise
    except Exception as e:
        raise SystemExit(f"Failed to load answer key: {e}")

    # Basic validation of candidate submission structure
    results: Dict[str, Any] = {}
    per_task_results: Dict[str, Any] = {}
    deductions: List[str] = []

    # Validate top-level keys minimally
    if not isinstance(candidate, dict):
        raise SystemExit("Candidate submission JSON root must be an object")

    tasks_list = safe_get(candidate, "tasks")
    if not isinstance(tasks_list, list) or len(tasks_list) == 0:
        # Still continue but mark failure
        deductions.append("Candidate submission missing tasks array or it's empty.")
        tasks_list = []

    # Build mapping id->task for easy lookup
    candidate_tasks_by_id = {}
    for t in tasks_list:
        tid = safe_get(t, "id")
        if isinstance(tid, str):
            candidate_tasks_by_id[tid.upper()] = t

    # Grade Task A
    task_a_obj = candidate_tasks_by_id.get("A", {})
    a_score, a_reasons = grade_task_a(task_a_obj)
    per_task_results["A"] = {
        "awarded": a_score,
        "max": MAX_TASK_A,
        "reasons": a_reasons
    }

    # Grade Task B
    task_b_obj = candidate_tasks_by_id.get("B", {})
    b_score, b_reasons = grade_task_b(task_b_obj)
    per_task_results["B"] = {
        "awarded": b_score,
        "max": MAX_TASK_B,
        "reasons": b_reasons
    }

    # Grade Task C
    task_c_obj = candidate_tasks_by_id.get("C", {})
    c_score, c_reasons = grade_task_c(task_c_obj)
    per_task_results["C"] = {
        "awarded": c_score,
        "max": MAX_TASK_C,
        "reasons": c_reasons
    }

    # Submission quality
    sub_score, sub_reasons = grade_submission_quality(candidate)
    per_task_results["Submission"] = {
        "awarded": sub_score,
        "max": MAX_SUBMISSION,
        "reasons": sub_reasons
    }

    total_awarded = a_score + b_score + c_score + sub_score
    percentage = (total_awarded / TOTAL_MAX) * 100.0 if TOTAL_MAX > 0 else 0.0

    # Compose final results
    results["per_task"] = per_task_results
    results["total_awarded"] = total_awarded
    results["total_max"] = TOTAL_MAX
    results["percentage"] = round(percentage, 2)
    results["overall_score"] = round(percentage, 2)  # as required variable
    results["notes"] = deductions

    # Add some metadata from candidate for convenience
    results["candidate_summary"] = {
        "name": safe_get(candidate, "name"),
        "language": safe_get(candidate, "language"),
        "time_taken_minutes": safe_get(candidate, "time_taken_minutes"),
        "run_commands": safe_get(candidate, "run_commands"),
    }

    return results


def save_results(results: Dict[str, Any], path: str = RESULTS_FILENAME):
    try:
        with open(path, "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2, ensure_ascii=False)
    except Exception as e:
        raise SystemExit(f"Failed to write results to {path}: {e}")


def main(argv: List[str]):
    if len(argv) != 3:
        print("Usage: python task_evaluation.py <candidate_submission.json> <answer_key.json>")
        sys.exit(2)
    candidate_path = argv[1]
    answer_key_path = argv[2]

    # Grade
    try:
        results = grade_submission(candidate_path, answer_key_path)
    except SystemExit as e:
        print(str(e))
        sys.exit(1)
    except Exception as e:
        print(f"Unexpected error during grading: {e}")
        sys.exit(1)

    # Save results
    save_results(results)
    # Print summary to stdout
    print("Grading complete. Results written to", RESULTS_FILENAME)
    print(f"Total: {results['total_awarded']} / {results['total_max']}  ({results['percentage']}%)")
    # Also expose overall_score in globals if required by wrappers
    # (Note: outside of function, a consumer can import this module and read overall_score variable after execution)
    return results


if __name__ == "__main__":
    # Run main with sys.argv
    final_results = main(sys.argv)

What the script checks automatically (high level):
- Task A: inspects the provided test_output to detect pytest success; enforces minimal change heuristic (processor.py only) for extra points.
- Task B: checks tests passing and verifies the presence of 'min_length' in processor.py and that the candidate added/modified the Task B test file; awards partial credit heuristically if only some evidence exists.
- Task C: requires numeric benchmark_before and benchmark_after fields (or parses bench output) and awards performance points based on measured improvement; awards explanation points based on explanation length.
- Submission quality: presence of run_commands, top-level keys, and modified_files for completed tasks.

Limitations / heuristic nature:
- Some checks (e.g., "minimal changes", "integration correctness") are heuristics; the script aims to be conservative and award partial credit where automatic verification is ambiguous. Manual review is still recommended for final grading if needed.
