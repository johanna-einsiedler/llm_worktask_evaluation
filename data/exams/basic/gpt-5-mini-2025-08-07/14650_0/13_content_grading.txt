# task_evaluation.py
import json
import os
import re
import sys

"""
Automated grading script for the "Updating Automated Tests" basic practical exam.

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

This script:
- Loads the candidate submission JSON and an answer key JSON (the answer key is used minimally;
  grading logic is based on expected fields and behaviors described in the exam spec).
- Applies a rubric to assign points across categories.
- Produces test_results.json in the same directory as this script with a detailed score breakdown,
  explanations for deductions, and overall percentage (overall_score).
- Uses only Python standard library modules.

Notes:
- The grading logic is heuristic and tolerant to formatting differences.
- If the candidate JSON is malformed or missing required fields, the script records errors and
  assigns zero or partial credit where possible.
"""

# ---------- Helper utilities ----------

def safe_load_json(path):
    """Load JSON from a file and return (data, error)."""
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f), None
    except Exception as e:
        return None, str(e)

def contains_case_insensitive(haystack, needles):
    """Return True if any of needles (string or list) appear in haystack (case-insensitive)."""
    if isinstance(needles, str):
        needles = [needles]
    hay = haystack.lower() if isinstance(haystack, str) else ""
    for n in needles:
        if n.lower() in hay:
            return True
    return False

def find_number_of_pass_fail(output_text):
    """Try to parse pytest summary counts from output. Returns (passed, failed) or (None, None)."""
    if not isinstance(output_text, str):
        return None, None
    # Look for patterns like "6 passed", "2 passed, 3 failed", "2 passed, 1 skipped, 0 failed", etc.
    passed = None
    failed = None
    # Find all 'X passed' and 'Y failed'
    m_pass = re.search(r'(\d+)\s+passed', output_text, re.IGNORECASE)
    if m_pass:
        try:
            passed = int(m_pass.group(1))
        except:
            passed = None
    m_fail = re.search(r'(\d+)\s+failed', output_text, re.IGNORECASE)
    if m_fail:
        try:
            failed = int(m_fail.group(1))
        except:
            failed = None
    # If no explicit 'failed' but 'failed' word exists (like "FAILED"), assume >0 failed.
    if failed is None:
        if re.search(r'\bfailed\b', output_text, re.IGNORECASE):
            # try to pick up singular "FAILED" or failure summary without counts -> mark unknown
            failed = 1
    return passed, failed

# ---------- Grading logic ----------

def grade_submission(submission, answer_key):
    """
    Apply rubric and return a detailed result dict.
    The rubric weights (out of 100):
      - baseline execution: 15
      - all tests passing after changes: 25
      - correct diagnosis (decision_log): 15
      - quality of test edits: 20
      - added test correctness & style: 10
      - documentation & reproducibility: 15
    """
    # Initialize
    results = {}
    deductions = []
    total_earned = 0.0
    total_max = 100.0

    # Safely extract fields from submission (handle missing keys)
    s = submission if isinstance(submission, dict) else {}
    candidate_name = s.get('candidate_name', '')
    time_spent = s.get('time_spent_minutes', None)
    commands_run = s.get('commands_run', [])
    test_output_before = s.get('test_output_before', '')
    test_output_after = s.get('test_output_after', '')
    modified_files = s.get('modified_files', [])
    added_files = s.get('added_files', [])
    decision_log = s.get('decision_log', [])
    reproduce_instructions = s.get('reproduce_instructions', '')

    # ---- 1) Baseline execution (15 pts) ----
    max_pts = 15.0
    earned = 0.0
    comments = []
    # Check commands_run contains at least one pytest invocation
    found_pytest_cmd = False
    if isinstance(commands_run, list):
        for cmd in commands_run:
            if isinstance(cmd, str) and re.search(r'\bpytest\b', cmd):
                found_pytest_cmd = True
                break
            if isinstance(cmd, str) and re.search(r'python\s+-m\s+pytest', cmd):
                found_pytest_cmd = True
                break
    if not commands_run or len(commands_run) == 0:
        comments.append("commands_run is empty or missing.")
    else:
        comments.append(f"{len(commands_run)} commands recorded in commands_run.")
    if found_pytest_cmd:
        comments.append("Found pytest invocation in commands_run.")
        earned += 5.0  # partial for recording commands
    else:
        comments.append("No pytest invocation found in commands_run; expected at least one pytest command.")
    # Check test_output_before exists and indicates failing tests
    if isinstance(test_output_before, str) and test_output_before.strip():
        comments.append("test_output_before present.")
        # Check for evidence of failing tests
        if re.search(r'\bfailed\b', test_output_before, re.IGNORECASE) or re.search(r'FAILED', test_output_before):
            comments.append("test_output_before shows failing tests (contains 'failed'/'FAILED').")
            earned += 10.0
        else:
            comments.append("test_output_before does not clearly show failing tests.")
            # partial: if non-empty but no 'failed', give small credit
            earned += 2.5
    else:
        comments.append("test_output_before missing or empty.")
    # Cap and assign
    results['baseline_execution'] = {
        'earned': round(earned, 2),
        'max': max_pts,
        'comments': comments
    }
    total_earned += earned

    # ---- 2) All tests passing after changes (25 pts) ----
    max_pts = 25.0
    earned = 0.0
    comments = []
    if isinstance(test_output_after, str) and test_output_after.strip():
        comments.append("test_output_after present.")
        passed, failed = find_number_of_pass_fail(test_output_after)
        if failed is None:
            # If 'failed' not found but 'passed' appears -> assume all passed
            if passed is not None:
                comments.append(f"Detected {passed} passed and no failed count; assuming all passed.")
                earned = max_pts
            else:
                # fallback: if word "passed" present and "failed" not present, assume pass
                if re.search(r'\bpassed\b', test_output_after, re.IGNORECASE) and not re.search(r'\bfailed\b', test_output_after, re.IGNORECASE):
                    comments.append("Output contains 'passed' and no 'failed'; awarding full credit.")
                    earned = max_pts
                else:
                    comments.append("Could not parse pass/fail counts from test_output_after.")
                    earned = 0.0
        else:
            # Both counts parsed (or failed detected)
            if failed == 0:
                comments.append(f"All tests passed according to summary (passed={passed}, failed=0).")
                earned = max_pts
            else:
                # partial credit proportional to passed/(passed+failed) if passed known
                if passed is not None and (passed + failed) > 0:
                    ratio = passed / float(passed + failed)
                    earned = max_pts * ratio
                    comments.append(f"Partial pass: {passed} passed, {failed} failed; awarding proportional credit ({ratio:.2%}).")
                else:
                    comments.append(f"Some tests failed (failed={failed}); awarding no points for full pass requirement.")
                    earned = 0.0
    else:
        comments.append("test_output_after missing or empty.")
        earned = 0.0
    results['all_tests_passing'] = {
        'earned': round(earned, 2),
        'max': max_pts,
        'comments': comments
    }
    total_earned += earned

    # ---- 3) Correct diagnosis of failures (decision_log) (15 pts) ----
    max_pts = 15.0
    earned = 0.0
    comments = []
    expected_topics = {
        'date': False,
        'status': False,
        'processed_at': False,
        'sleep': False,
        'discount': False,
        'added_test': False  # mention of adding test or include_discount
    }
    if isinstance(decision_log, list) and len(decision_log) > 0:
        comments.append(f"decision_log has {len(decision_log)} entries.")
        # Scan decision_log entries for expected topics
        count_mentions = 0
        total_topics = len(expected_topics)
        alt_present_count = 0
        for entry in decision_log:
            # entry expected to be dict with 'decision' and 'alternatives_considered'
            dec_text = str(entry.get('decision', '')).lower() if isinstance(entry, dict) else ''
            alt_text = str(entry.get('alternatives_considered', '')).lower() if isinstance(entry, dict) else ''
            # mark topics
            if 'date' in dec_text or 'iso' in dec_text or 'yyyy' in dec_text:
                expected_topics['date'] = True
            if 'status' in dec_text:
                expected_topics['status'] = True
            if 'processed_at' in dec_text or 'processed at' in dec_text:
                expected_topics['processed_at'] = True
            if 'sleep' in dec_text or 'time.sleep' in dec_text:
                expected_topics['sleep'] = True
            if 'discount' in dec_text or 'include_discount' in dec_text:
                expected_topics['discount'] = True
            if 'add' in dec_text and ('test' in dec_text or 'discount' in dec_text):
                expected_topics['added_test'] = True
            # alternatives considered presence
            if alt_text and alt_text.strip() and alt_text.lower() not in ('none', 'n/a', 'na'):
                alt_present_count += 1
        # tally
        for k, v in expected_topics.items():
            if v:
                count_mentions += 1
        comments.append(f"Detected mention of {count_mentions}/{total_topics} expected diagnostic topics in decision_log.")
        # award proportional credit
        earned = max_pts * (count_mentions / float(total_topics))
        # small bonus if many alternatives_considered present: if at least half entries had alternatives_considered non-empty, add up to 10% of this section
        if len(decision_log) > 0 and alt_present_count / len(decision_log) >= 0.5:
            bonus = 0.10 * max_pts
            earned += bonus
            comments.append(f"Alternatives considered present in {alt_present_count}/{len(decision_log)} entries; applied bonus {bonus:.2f} pts.")
    else:
        comments.append("decision_log missing or empty; no diagnosis credit.")
        earned = 0.0
    # Cap to max_pts + bonus cap (we allow up to 110% if bonus added, but clamp)
    if earned > max_pts:
        earned = max_pts
    results['decision_log'] = {
        'earned': round(earned, 2),
        'max': max_pts,
        'comments': comments,
        'detected_topics': expected_topics
    }
    total_earned += earned

    # ---- 4) Quality of test edits (20 pts) ----
    max_pts = 20.0
    earned = 0.0
    comments = []
    # First basic validations
    if not isinstance(modified_files, list) or len(modified_files) == 0:
        comments.append("No modified_files provided.")
        # No edits -> zero on this section
        earned = 0.0
    else:
        comments.append(f"{len(modified_files)} modified file(s) submitted.")
        # Check if any modified files are outside tests/
        non_test_mods = [mf for mf in modified_files if not mf.get('path', '').startswith('tests/')]
        if non_test_mods:
            comments.append(f"Found {len(non_test_mods)} modified file(s) outside tests/: " +
                            ", ".join([mf.get('path', '') for mf in non_test_mods]) +
                            ". This is discouraged; tests-only edits required.")
        # Assess content improvements:
        improvements = {
            'date_iso_check': False,
            'partial_string_checks': False,
            'removed_sleep': False,
            'status_checks': False,
            'no_strict_whole_dict_equality': False
        }
        # Iterate through modified_files contents
        combined_text = ""
        for mf in modified_files:
            content = mf.get('content', '') or ''
            combined_text += "\n" + content.lower()
        # date ISO example '2025-01-15'
        if '2025-01-15' in combined_text or re.search(r'\d{4}-\d{2}-\d{2}', combined_text):
            improvements['date_iso_check'] = True
        # partial string checks: look for ' in ' usage on summaries or 'contains' style
        if re.search(r'assert\s+.*in\s+.*summary', combined_text) or ' in summary' in combined_text or 'contains' in combined_text:
            improvements['partial_string_checks'] = True
        # removed sleep: ensure time.sleep not present
        if 'time.sleep' not in combined_text:
            improvements['removed_sleep'] = True
        # status checks: presence of '[status:' or 'status' asserts
        if '[status:' in combined_text or 'status: processing' in combined_text or 'status: pending' in combined_text or 'status' in combined_text:
            # ensure it's in context of assertions or checks
            if 'status' in combined_text:
                improvements['status_checks'] = True
        # no strict whole-dict equality: detect if 'assert order == expected' or 'processed_at' still present (bad)
        if 'assert order == expected' not in combined_text and 'processed_at' not in combined_text:
            improvements['no_strict_whole_dict_equality'] = True
        # Count improvements
        improvements_found = sum(1 for v in improvements.values() if v)
        comments.append(f"Improvements detected: {improvements_found}/5 -> {improvements}.")
        # Base points proportional to improvements_found
        earned = max_pts * (improvements_found / 5.0)
        # If modifications include edits outside tests/, deduct half of this section
        if non_test_mods:
            earned *= 0.5
            comments.append("Modifications outside tests/ detected -> applying 50% penalty to this section.")
    results['quality_test_edits'] = {
        'earned': round(earned, 2),
        'max': max_pts,
        'comments': comments
    }
    total_earned += earned

    # ---- 5) Added test correctness & style (10 pts) ----
    max_pts = 10.0
    earned = 0.0
    comments = []
    # Candidate should add a test that calls calculate_total(order, include_discount=True) for O1001 == 32.5
    if isinstance(added_files, list) and len(added_files) > 0:
        comments.append(f"{len(added_files)} added file(s).")
        # Look for at least one added file under tests/ containing required assertion
        found_good = False
        found_partial = False
        for af in added_files:
            path = af.get('path', '')
            content = (af.get('content', '') or '').lower()
            if not path.startswith('tests/'):
                comments.append(f"Added file {path} is not under tests/ (ignored for this check).")
                continue
            # Check for calculate_total with include_discount and expected value 32.5
            if 'calculate_total' in content and 'include_discount=True'.lower() in content:
                # check for 32.5 or 32.50
                if '32.5' in content or '32.50' in content:
                    found_good = True
                    comments.append(f"Found a correct added test in {path} that asserts include_discount=True yields 32.5.")
                    break
                else:
                    found_partial = True
            elif 'calculate_total' in content:
                found_partial = True
        if found_good:
            earned = max_pts
        elif found_partial:
            earned = max_pts * 0.5
            comments.append("Added test references calculate_total but missing include_discount=True or expected value; partial credit awarded.")
        else:
            comments.append("No added tests matching required discount check found in tests/ added files.")
            earned = 0.0
    else:
        comments.append("No added_files provided; required discount test is missing.")
        earned = 0.0
    results['added_test'] = {
        'earned': round(earned, 2),
        'max': max_pts,
        'comments': comments
    }
    total_earned += earned

    # ---- 6) Documentation & reproducibility (15 pts) ----
    max_pts = 15.0
    earned = 0.0
    comments = []
    # Criteria: commands_run non-empty (5), reproduce_instructions includes pytest (5), modified/added file reasons present (3), before/after outputs show failed/passed (2)
    subpoints = {
        'commands_run': 5.0,
        'reproduce_instructions': 5.0,
        'file_reasons': 3.0,
        'before_after_outputs': 2.0
    }
    # commands_run non-empty
    if isinstance(commands_run, list) and len(commands_run) > 0:
        earned += subpoints['commands_run']
        comments.append("commands_run provided.")
    else:
        comments.append("commands_run missing or empty.")
    # reproduce_instructions includes pytest
    if isinstance(reproduce_instructions, str) and reproduce_instructions.strip():
        if re.search(r'\bpytest\b', reproduce_instructions):
            earned += subpoints['reproduce_instructions']
            comments.append("reproduce_instructions provided and includes pytest command.")
        else:
            comments.append("reproduce_instructions provided but does not include pytest command.")
    else:
        comments.append("reproduce_instructions missing or empty.")
    # modified_files and added_files reasons
    reasons_ok = True
    if isinstance(modified_files, list) and len(modified_files) > 0:
        for mf in modified_files:
            if not mf.get('reason') or not isinstance(mf.get('reason'), str) or not mf.get('reason').strip():
                reasons_ok = False
                comments.append(f"Modified file {mf.get('path','?')} missing reason.")
                break
    if isinstance(added_files, list) and len(added_files) > 0:
        for af in added_files:
            if not af.get('reason') or not isinstance(af.get('reason'), str) or not af.get('reason').strip():
                reasons_ok = False
                comments.append(f"Added file {af.get('path','?')} missing reason.")
                break
    if reasons_ok and ( (modified_files and len(modified_files) > 0) or (added_files and len(added_files) > 0) ):
        earned += subpoints['file_reasons']
        comments.append("All modified/added files include brief reasons.")
    else:
        # no files changed -> still acceptable? If no modified_files, then earlier sections would fail. We don't award these points.
        comments.append("Some modified/added files are missing reasons or no files were modified/added.")
    # before/after outputs show failing/passing
    before_ok = isinstance(test_output_before, str) and re.search(r'\bfailed\b', test_output_before, re.IGNORECASE)
    after_ok = isinstance(test_output_after, str) and ( (re.search(r'\b0\s+failed\b', test_output_after, re.IGNORECASE)) or (re.search(r'\bpassed\b', test_output_after, re.IGNORECASE) and not re.search(r'\bfailed\b', test_output_after, re.IGNORECASE)) )
    if before_ok and after_ok:
        earned += subpoints['before_after_outputs']
        comments.append("test_output_before shows failures and test_output_after shows passing tests.")
    else:
        if not before_ok:
            comments.append("test_output_before did not clearly show failing tests.")
        if not after_ok:
            comments.append("test_output_after did not clearly show all tests passing.")
    results['documentation_and_reproducibility'] = {
        'earned': round(earned, 2),
        'max': max_pts,
        'comments': comments
    }
    total_earned += earned

    # ---- Additional checks and final adjustments ----
    # Ensure that modified_files and added_files content are present (full content)
    extra_comments = []
    missing_content = False
    for mf in (modified_files or []):
        if not mf.get('content') or not isinstance(mf.get('content'), str) or not mf.get('content').strip():
            extra_comments.append(f"Modified file {mf.get('path','?')} missing content.")
            missing_content = True
    for af in (added_files or []):
        if not af.get('content') or not isinstance(af.get('content'), str) or not af.get('content').strip():
            extra_comments.append(f"Added file {af.get('path','?')} missing content.")
            missing_content = True
    if missing_content:
        # Penalize documentation points slightly if content missing
        results['documentation_and_reproducibility']['comments'].append("Some modified/added files lacked full content; this may impede reproducibility.")
        # Deduct a small fixed penalty from total_earned (but not below 0)
        penalty = 3.0
        total_earned = max(0.0, total_earned - penalty)
        extra_comments.append(f"Applied penalty of {penalty} points for missing file content.")
    if extra_comments:
        results.setdefault('extra_checks', {'comments': []})
        results['extra_checks']['comments'].extend(extra_comments)

    # Sum up totals and compute percentage
    # total_max already 100
    total_earned = float(total_earned)
    percentage = round((total_earned / total_max) * 100.0, 2) if total_max > 0 else 0.0

    # Build final results structure
    final = {
        'candidate_name': candidate_name,
        'time_spent_minutes': time_spent,
        'scores': results,
        'total_points_earned': round(total_earned, 2),
        'total_points_possible': total_max,
        'percentage': percentage,
        'overall_score': percentage
    }

    return final

# ---------- Main execution ----------

def main():
    # Validate command-line args
    if len(sys.argv) < 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)

    submission_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    # Load submission JSON
    submission, err = safe_load_json(submission_path)
    if err:
        # Write error output to test_results.json and exit gracefully
        out = {
            'error': f"Failed to load submission JSON from '{submission_path}': {err}",
            'overall_score': 0.0
        }
        with open('test_results.json', 'w', encoding='utf-8') as f:
            json.dump(out, f, indent=2)
        print(f"Error: could not load submission JSON: {err}. Wrote test_results.json with error.")
        sys.exit(1)

    # Load answer key JSON (not heavily relied upon, but we validate presence)
    answer_key, err2 = safe_load_json(answer_key_path)
    if err2:
        out = {
            'error': f"Failed to load answer key JSON from '{answer_key_path}': {err2}",
            'overall_score': 0.0
        }
        with open('test_results.json', 'w', encoding='utf-8') as f:
            json.dump(out, f, indent=2)
        print(f"Error: could not load answer key JSON: {err2}. Wrote test_results.json with error.")
        sys.exit(1)

    # Apply grading logic
    results = grade_submission(submission, answer_key)

    # Save results to test_results.json
    try:
        with open('test_results.json', 'w', encoding='utf-8') as f:
            json.dump(results, f, indent=2)
        print("Grading complete. Results saved to test_results.json")
    except Exception as e:
        print(f"Error writing test_results.json: {e}")
        sys.exit(1)

if __name__ == '__main__':
    main()
