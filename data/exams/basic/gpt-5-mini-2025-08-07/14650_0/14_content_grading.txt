# task_evaluation.py
import json
import os
import re
import sys

"""
Automated grader for the "Updating Automated Tests..." basic practical exam.

Usage:
    python task_evaluation.py candidate_submission.json answer_key.json

Outputs:
    test_results.json (written to the same directory as this script)

The grader applies the rubric described in the exam materials and evaluator notes.
It compares the candidate submission JSON to expectations (from the answer key if present,
otherwise uses built-in heuristics) and produces a breakdown of scores, deductions,
and an overall percentage score.

Implementation notes:
- Uses only standard library modules.
- Is tolerant of minor formatting differences (whitespace/case).
- Provides clear explanations for deductions and partial credit decisions.
"""

# --- Utility helpers -------------------------------------------------------

def load_json(path):
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except Exception as e:
        raise RuntimeError(f"Failed to load JSON from {path}: {e}")

def contains_pytest_command(commands):
    """Return True if any command string looks like a pytest invocation."""
    if not isinstance(commands, list):
        return False
    pattern = re.compile(r'\b(pytest|python\s+-m\s+pytest)\b', re.IGNORECASE)
    return any(pattern.search(cmd) for cmd in commands)

def parse_failed_count(output):
    """Return integer failed count if detected, else None."""
    if not output:
        return None
    # Look for patterns like 'X failed' or '= X failed' or 'FAILED (failures=X)'
    m = re.search(r'(\d+)\s+failed', output)
    if m:
        try:
            return int(m.group(1))
        except:
            return None
    # Fallback: check for 'failed' and ': ' lines (pytest summary style)
    # If 'failed' word exists but number not parseable, return 1 as conservative indicator
    if 'failed' in output.lower():
        return 1
    return None

def parse_passed_count(output):
    """Return integer passed count if detected, else None."""
    if not output:
        return None
    m = re.search(r'(\d+)\s+passed', output)
    if m:
        try:
            return int(m.group(1))
        except:
            return None
    return None

def safe_get(dct, key, default=None):
    try:
        return dct.get(key, default)
    except:
        return default

# --- Grading component functions -------------------------------------------

def grade_baseline_execution(submission):
    """
    Baseline execution (15 points)
    Criteria:
      - Candidate ran pytest (commands_run contains pytest invocation) [5 pts]
      - test_output_before shows failing tests (>=1 failed) [10 pts]
    """
    max_points = 15
    score = 0
    comments = []

    commands_run = safe_get(submission, 'commands_run', [])
    test_output_before = safe_get(submission, 'test_output_before', '') or ''

    # Check for pytest command
    if contains_pytest_command(commands_run):
        score += 5
        comments.append("Found pytest invocation in commands_run (+5).")
    else:
        comments.append("No pytest invocation found in commands_run (-5).")

    # Check that the captured "before" output shows failing tests
    failed_before = parse_failed_count(test_output_before)
    if failed_before is None:
        # No evidence of failure; check for the word 'FAIL' or 'FAILED' in output
        if test_output_before and re.search(r'\bFAIL(?:URES)?\b|\bFAILED\b', test_output_before, re.IGNORECASE):
            failed_before = 1
    if failed_before and failed_before > 0:
        score += 10
        comments.append(f"test_output_before indicates failing tests (failed={failed_before}) (+10).")
    else:
        comments.append("test_output_before does not show failing tests or missing (-10).")

    return {'category': 'Baseline execution', 'score': score, 'max': max_points, 'comments': comments}

def grade_all_passing_after(submission):
    """
    All tests passing after changes (25 points)
    Criteria:
      - test_output_after shows tests passing with 0 failures [25 pts]
    """
    max_points = 25
    score = 0
    comments = []

    test_output_after = safe_get(submission, 'test_output_after', '') or ''
    failed_after = parse_failed_count(test_output_after)
    passed_after = parse_passed_count(test_output_after)

    # If output explicitly shows 0 failed or does not mention failures and shows passed
    if failed_after == 0:
        score = max_points
        comments.append("test_output_after explicitly shows 0 failed (+25).")
    else:
        # attempt to catch summary like "X passed" and no "failed" word
        if ('failed' not in test_output_after.lower()) and (passed_after is not None):
            # assume green if passed count present and no 'failed'
            score = max_points
            comments.append("test_output_after shows passed count and no failures word (+25).")
        else:
            # If there are failures, no credit
            if failed_after is None:
                comments.append("Could not parse failure count from test_output_after (-25).")
            else:
                comments.append(f"test_output_after shows failures (failed={failed_after}) (-25).")

    return {'category': 'All tests passing after changes', 'score': score, 'max': max_points, 'comments': comments}

def grade_diagnosis(submission, answer_key):
    """
    Correct diagnosis of failures (15 points)
    Approach:
      - Use expected decision_log steps from answer_key if available.
      - Otherwise expect the candidate decision_log decisions to mention keywords:
        'date', 'status', 'processed_at', 'sleep', 'discount'
      - Allocate equal points across expected items found in candidate decision_log.
    """
    max_points = 15
    score = 0
    comments = []

    # Extract expected steps either from answer_key or default keywords
    expected_steps = []
    ak_decision_log = safe_get(answer_key, 'decision_log', None)
    if isinstance(ak_decision_log, list) and len(ak_decision_log) > 0:
        expected_steps = [safe_get(d, 'step', '') for d in ak_decision_log if safe_get(d, 'step', '')]
    else:
        # Default expected diagnosis items (derived from README / materials)
        expected_steps = [
            'fix-load-order-assertion',        # date format/processed_at
            'fix-format-summary-assertions',   # includes status and ISO date
            'remove-sleep',                    # remove unnecessary sleep
            'add-discount-test',               # new test for discount include
            'baseline-run'                     # ran baseline
        ]

    candidate_log = safe_get(submission, 'decision_log', [])
    candidate_steps = []
    if isinstance(candidate_log, list):
        for entry in candidate_log:
            step = safe_get(entry, 'step', '')
            decision_text = safe_get(entry, 'decision', '')
            candidate_steps.append({'step': step.lower() if isinstance(step, str) else step,
                                    'decision': decision_text.lower() if isinstance(decision_text, str) else decision_text})

    # For each expected step, check presence in candidate steps or presence of expected keywords in decisions
    points_per = max_points / len(expected_steps) if expected_steps else 0
    matched = 0
    missed = []
    for expected in expected_steps:
        expected_norm = expected.lower()
        found = False
        for cs in candidate_steps:
            if isinstance(cs.get('step'), str) and expected_norm in cs.get('step'):
                found = True
                break
        # If not found by step name, check for keyword matches in decision text
        if not found:
            # Map expected step to keywords
            keywords = []
            if 'load' in expected_norm or 'order' in expected_norm or 'date' in expected_norm:
                keywords = ['date', 'processed_at', 'iso', 'format']
            elif 'format' in expected_norm or 'summary' in expected_norm:
                keywords = ['status', 'status]', 'status:', 'date', 'iso', 'summary']
            elif 'sleep' in expected_norm or 'remove' in expected_norm:
                keywords = ['sleep', 'time.sleep', 'unnecessary']
            elif 'discount' in expected_norm or 'add' in expected_norm:
                keywords = ['discount', 'include_discount', 'include-discount']
            elif 'baseline' in expected_norm or 'run' in expected_norm:
                keywords = ['run pytest', 'pytest', 'failing', 'baseline']
            # check presence of any keyword in any decision text
            for cs in candidate_steps:
                decision_text = cs.get('decision', '') or ''
                for kw in keywords:
                    if kw in decision_text:
                        found = True
                        break
                if found:
                    break

        if found:
            matched += 1
        else:
            missed.append(expected)

    score = round(points_per * matched, 2)
    comments.append(f"Found {matched}/{len(expected_steps)} expected diagnosis items; matched: {matched}, missed: {len(missed)}.")
    if missed:
        comments.append(f"Missed diagnosis items: {missed}")
    else:
        comments.append("All expected diagnosis steps present.")

    return {'category': 'Correct diagnosis (decision_log)', 'score': score, 'max': max_points, 'comments': comments}

def grade_test_edits(submission):
    """
    Quality of test edits (20 points)
    Heuristics:
      - All modified files should be under tests/ (if not, penalize heavily)
      - Modified file(s) include tests/test_orders.py (5 pts)
      - 'time.sleep' removed (5 pts)
      - ISO dates present in test content (5 pts)
      - Use of 'in' assertions / partial checks present (5 pts)
    """
    max_points = 20
    score = 0
    comments = []

    modified_files = safe_get(submission, 'modified_files', [])
    if not isinstance(modified_files, list) or len(modified_files) == 0:
        comments.append("No modified_files submitted (-20).")
        return {'category': 'Quality of test edits', 'score': 0, 'max': max_points, 'comments': comments}

    # Check for edits outside tests/
    sut_edits = []
    for mf in modified_files:
        path = safe_get(mf, 'path', '')
        if not isinstance(path, str):
            continue
        if not path.startswith('tests/'):
            sut_edits.append(path)
    if sut_edits:
        # Penalize: edits to SUT are discouraged for the basic exam
        comments.append(f"Modified non-test files (SUT edits): {sut_edits} (will reduce score).")
        # We'll still evaluate other heuristics but subtract 10 points later
        sut_edit_penalty = True
    else:
        sut_edit_penalty = False

    # Heuristic checks
    got_orders_test = False
    removed_sleep = True  # assume removed unless we find it present
    iso_date_present = False
    partial_checks_present = False

    for mf in modified_files:
        content = safe_get(mf, 'content', '') or ''
        path = safe_get(mf, 'path', '') or ''
        lower = content.lower()
        if path.endswith('test_orders.py') or 'test_orders' in path:
            got_orders_test = True
        if 'time.sleep' in lower:
            removed_sleep = False
        # Check for ISO date tokens like '2025-01-15'
        if re.search(r'20\d{2}-\d{2}-\d{2}', content):
            iso_date_present = True
        # Check for partial assertions using ' in ' or 'get(' or 'contains' or 'startswith' etc.
        if re.search(r'\bin\b', content) or 'get(' in content or 'startswith(' in content or 'in summary' in content:
            partial_checks_present = True

    if got_orders_test:
        score += 5
        comments.append("Modified tests/test_orders.py or similar (+5).")
    else:
        comments.append("Did not modify tests/test_orders.py (+0).")

    if removed_sleep:
        score += 5
        comments.append("No 'time.sleep' present in modified tests (sleep removed) (+5).")
    else:
        comments.append("'time.sleep' still present in modified tests (-5).")

    if iso_date_present:
        score += 5
        comments.append("ISO date format (YYYY-MM-DD) present in modified tests (+5).")
    else:
        comments.append("ISO date format not found in modified tests (-5).")

    if partial_checks_present:
        score += 5
        comments.append("Partial/containment/key-based assertions present in modified tests (+5).")
    else:
        comments.append("No evidence of partial/containment assertions (e.g., 'in', 'get(') (-5).")

    # Apply SUT edit penalty
    if sut_edit_penalty:
        # Deduct 10 points but ensure score does not go negative here
        prev = score
        score = max(score - 10, 0)
        comments.append(f"Penalty for editing SUT files: -10 (score {prev} -> {score}).")

    return {'category': 'Quality of test edits', 'score': score, 'max': max_points, 'comments': comments}

def grade_added_test(submission, answer_key):
    """
    Added test correctness & style (10 points)
    Criteria:
      - At least one added file under tests/
      - Added test asserts calculate_total(..., include_discount=True) equals expected discounted value (32.5)
      - Accept numeric variants (32.5, 32.50)
    """
    max_points = 10
    score = 0
    comments = []

    added_files = safe_get(submission, 'added_files', [])
    if not isinstance(added_files, list) or len(added_files) == 0:
        comments.append("No added_files submitted (-10).")
        return {'category': 'Added test correctness & style', 'score': 0, 'max': max_points, 'comments': comments}

    # Look for a test file under tests/ that contains include_discount and expected numeric result
    found_test = False
    discount_assert_ok = False
    for af in added_files:
        path = safe_get(af, 'path', '') or ''
        content = safe_get(af, 'content', '') or ''
        if not path.startswith('tests/'):
            continue
        found_test = True
        lower = content.lower()
        # Check for include_discount usage
        if 'include_discount' in lower or 'include-discount' in lower or 'include_discount=True'.lower() in lower:
            # Find numeric literal 32.5 or 32.50 or 32.50 formatted
            if re.search(r'(32(?:\.50|\.5)?)', content):
                discount_assert_ok = True
                break
            # also check for 32.50 with formatting
            if re.search(r'32\.50', content):
                discount_assert_ok = True
                break

    if not found_test:
        comments.append("No added test files under tests/ (+0).")
        return {'category': 'Added test correctness & style', 'score': 0, 'max': max_points, 'comments': comments}

    if discount_assert_ok:
        score = max_points
        comments.append("Found added test verifying include_discount behavior with expected discounted total (32.5) (+10).")
    else:
        # Partial credit if include_discount used but numeric differs or not found
        any_include = any(('include_discount' in (safe_get(af, 'content', '') or '').lower()) for af in added_files)
        if any_include:
            score = 5
            comments.append("Added test uses include_discount but expected discounted value not clearly asserted; partial credit (+5).")
        else:
            comments.append("Added tests present but no include_discount usage found (-10).")

    return {'category': 'Added test correctness & style', 'score': score, 'max': max_points, 'comments': comments}

def grade_documentation_repro(submission):
    """
    Documentation & reproducibility (15 points)
    Criteria:
      - reproduce_instructions present and mentions pytest command (7 pts)
      - commands_run includes both a before and after pytest invocation (8 pts)
      - modified_files and added_files include reason and content (minor checks included elsewhere)
    """
    max_points = 15
    score = 0
    comments = []

    reproduce_instructions = safe_get(submission, 'reproduce_instructions', '') or ''
    commands_run = safe_get(submission, 'commands_run', []) or []
    modified_files = safe_get(submission, 'modified_files', []) or []
    added_files = safe_get(submission, 'added_files', []) or []

    # Check for reproduce_instructions presence and mention of pytest
    if reproduce_instructions and re.search(r'\bpytest\b', reproduce_instructions, re.IGNORECASE):
        score += 7
        comments.append("reproduce_instructions contains pytest mention (+7).")
    elif reproduce_instructions:
        score += 3
        comments.append("reproduce_instructions present but does not explicitly mention pytest (+3).")
    else:
        comments.append("No reproduce_instructions provided (-7).")

    # Check for at least two pytest runs in commands_run (before and after)
    pytest_cmds = [cmd for cmd in commands_run if re.search(r'\b(pytest|python\s+-m\s+pytest)\b', cmd, re.IGNORECASE)]
    if len(pytest_cmds) >= 2:
        score += 8
        comments.append("commands_run includes at least two pytest invocations (before & after) (+8).")
    elif len(pytest_cmds) == 1:
        score += 4
        comments.append("commands_run includes one pytest invocation; expected before and after (+4).")
    else:
        comments.append("commands_run lacks pytest invocations (-8).")

    # Check modified_files and added_files include reason and content
    missing_reason = []
    missing_content = []
    for mf in (modified_files + added_files):
        if not safe_get(mf, 'reason', ''):
            missing_reason.append(safe_get(mf, 'path', '<unknown>'))
        if not safe_get(mf, 'content', ''):
            missing_content.append(safe_get(mf, 'path', '<unknown>'))
    if missing_reason:
        comments.append(f"Some modified/added files missing 'reason' field: {missing_reason} (no direct point penalty, but noted).")
    if missing_content:
        comments.append(f"Some modified/added files missing 'content' field: {missing_content} (this will affect other checks).")

    return {'category': 'Documentation & reproducibility', 'score': score, 'max': max_points, 'comments': comments}

# --- Main grading orchestration --------------------------------------------

def grade_submission(submission_path, answer_key_path):
    # Load input files (with robust error handling)
    try:
        submission = load_json(submission_path)
    except Exception as e:
        result = {
            'error': f"Failed to load submission JSON: {e}"
        }
        return result

    try:
        answer_key = load_json(answer_key_path)
    except Exception as e:
        # Not fatal: proceed with empty answer_key and heuristics
        answer_key = {}
        # proceed but note it
        # (we'll include message in results)
        ak_load_error = str(e)
    else:
        ak_load_error = None

    # Prepare result structure
    breakdown = []
    total_score = 0.0
    total_max = 0.0
    deductions = []

    # Run grading components
    try:
        comp = grade_baseline_execution(submission)
        breakdown.append(comp)
        total_score += comp['score']
        total_max += comp['max']
    except Exception as e:
        breakdown.append({'category': 'Baseline execution', 'score': 0, 'max': 15, 'comments': [f"Error grading baseline: {e}"]})
        total_max += 15

    try:
        comp = grade_all_passing_after(submission)
        breakdown.append(comp)
        total_score += comp['score']
        total_max += comp['max']
    except Exception as e:
        breakdown.append({'category': 'All tests passing after changes', 'score': 0, 'max': 25, 'comments': [f"Error grading after-run: {e}"]})
        total_max += 25

    try:
        comp = grade_diagnosis(submission, answer_key)
        breakdown.append(comp)
        total_score += comp['score']
        total_max += comp['max']
    except Exception as e:
        breakdown.append({'category': 'Correct diagnosis (decision_log)', 'score': 0, 'max': 15, 'comments': [f"Error grading diagnosis: {e}"]})
        total_max += 15

    try:
        comp = grade_test_edits(submission)
        breakdown.append(comp)
        total_score += comp['score']
        total_max += comp['max']
    except Exception as e:
        breakdown.append({'category': 'Quality of test edits', 'score': 0, 'max': 20, 'comments': [f"Error grading test edits: {e}"]})
        total_max += 20

    try:
        comp = grade_added_test(submission, answer_key)
        breakdown.append(comp)
        total_score += comp['score']
        total_max += comp['max']
    except Exception as e:
        breakdown.append({'category': 'Added test correctness & style', 'score': 0, 'max': 10, 'comments': [f"Error grading added test: {e}"]})
        total_max += 10

    try:
        comp = grade_documentation_repro(submission)
        breakdown.append(comp)
        total_score += comp['score']
        total_max += comp['max']
    except Exception as e:
        breakdown.append({'category': 'Documentation & reproducibility', 'score': 0, 'max': 15, 'comments': [f"Error grading documentation: {e}"]})
        total_max += 15

    # Compose final results
    # Ensure totals align to 100 max if any rounding issues
    total_score = round(total_score, 2)
    total_max = int(total_max) if total_max == int(total_max) else round(total_max, 2)
    overall_percentage = round((total_score / total_max) * 100, 2) if total_max > 0 else 0.0

    results = {
        'candidate_file': os.path.basename(submission_path),
        'answer_key_file': os.path.basename(answer_key_path),
        'breakdown': breakdown,
        'total_score': total_score,
        'total_max': total_max,
        'percentage': overall_percentage,
        'overall_score': overall_percentage,
        'notes': []
    }

    if ak_load_error:
        results['notes'].append(f"Warning: failed to load answer_key.json ({ak_load_error}). Grading used internal heuristics.")

    # Include some sanity checks and helpful messages
    # - Ensure test_output_before contains failing tests and test_output_after shows passing tests
    tbefore = safe_get(submission, 'test_output_before', '') or ''
    tafter = safe_get(submission, 'test_output_after', '') or ''
    failed_before = parse_failed_count(tbefore)
    failed_after = parse_failed_count(tafter)
    if failed_before is None:
        results['notes'].append("Note: Could not parse failing count from test_output_before; ensure you included raw pytest output.")
    if failed_after is None:
        results['notes'].append("Note: Could not parse failing count from test_output_after; ensure you included raw pytest output.")
    else:
        if failed_after != 0:
            results['notes'].append("Candidate's final test_output_after indicates remaining failures; verify test suite is fully passing.")

    # Save results to test_results.json in same directory as this script
    out_path = os.path.join(os.path.dirname(os.path.abspath(__file__)), 'test_results.json')
    try:
        with open(out_path, 'w', encoding='utf-8') as f:
            json.dump(results, f, indent=2)
    except Exception as e:
        return {'error': f"Failed to write results to {out_path}: {e}", 'results': results}

    return results

# --- Script entry point ----------------------------------------------------

def main():
    if len(sys.argv) < 3:
        print("Usage: python task_evaluation.py <submission_json> <answer_key_json>")
        sys.exit(2)

    submission_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    try:
        results = grade_submission(submission_path, answer_key_path)
    except Exception as e:
        print(f"Fatal grading error: {e}")
        sys.exit(1)

    # Print a concise summary to stdout
    if isinstance(results, dict) and 'error' in results:
        print("Grading failed:", results['error'])
        if 'results' in results:
            print("Partial results saved to test_results.json")
        sys.exit(1)

    print("Grading completed.")
    print(f"Total: {results['total_score']} / {results['total_max']}  ({results['percentage']}%)")
    print("Breakdown:")
    for item in results['breakdown']:
        cat = item.get('category', '<unknown>')
        sc = item.get('score', 0)
        mx = item.get('max', 0)
        print(f" - {cat}: {sc}/{mx}")
    print("\nDetailed results written to test_results.json")

if __name__ == '__main__':
    main()
