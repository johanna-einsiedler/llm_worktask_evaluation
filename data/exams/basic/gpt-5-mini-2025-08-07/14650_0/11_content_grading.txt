# task_evaluation.py
"""
Automated grading script for the "Updating Automated Tests..." basic practical exam.

Usage:
    python task_evaluation.py <candidate_submission.json> <answer_key.json>

Produces:
    test_results.json in the same directory as this script.

Notes:
- Uses only Python standard library.
- Grading logic implements the rubric described in the exam materials.
- The script is defensive: it checks for missing/invalid keys and reports clear deduction reasons.
"""

import json
import os
import sys
import re

# Helper: safe read JSON
def load_json(path):
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f), None
    except Exception as e:
        return None, str(e)

# Helper: normalize text for basic substring checks
def norm(s):
    if s is None:
        return ""
    return s.lower()

# Grading logic
def grade(candidate, answer_key):
    # Rubric weights (max points)
    weights = {
        "baseline_execution": 15,
        "all_tests_passing": 25,
        "diagnosis_decision_log": 15,
        "quality_test_edits": 20,
        "added_test": 10,
        "documentation_repro": 15
    }

    # Result structure
    results = {
        "items": {},
        "total_awarded": 0,
        "total_possible": sum(weights.values()),
        "overall_score": 0.0,
        "pass": False
    }

    deductions_notes = []  # global notes

    # Utility: get fields safely
    def get_field(obj, key, expected_type=None):
        v = obj.get(key) if isinstance(obj, dict) else None
        if v is None:
            return None, f"Missing key '{key}'"
        if expected_type and not isinstance(v, expected_type):
            return None, f"Key '{key}' has wrong type (expected {expected_type.__name__})"
        return v, None

    # Validate top-level presence
    needed_keys = [
        "candidate_name", "time_spent_minutes", "commands_run",
        "test_output_before", "test_output_after",
        "modified_files", "added_files", "decision_log",
        "reproduce_instructions"
    ]
    missing_top = [k for k in needed_keys if k not in candidate]
    if missing_top:
        # If major keys missing, we still try to grade what we can.
        deductions_notes.append(f"Submission missing top-level keys: {missing_top}")

    # 1) Baseline execution (15)
    be_score = 0
    be_comments = []
    # Check commands_run exists and includes at least one pytest invocation
    commands_run = candidate.get("commands_run", [])
    if not isinstance(commands_run, list) or len(commands_run) == 0:
        be_comments.append("No commands_run provided or empty.")
    else:
        # look for pytest invocation
        pytest_invocations = [c for c in commands_run if isinstance(c, str) and ("pytest" in c or "python -m pytest" in c)]
        if pytest_invocations:
            be_comments.append("Found pytest invocations in commands_run.")
        else:
            be_comments.append("No pytest invocation found in commands_run.")
    # Check test_output_before shows failing tests
    tb = norm(candidate.get("test_output_before", ""))
    shows_failures = False
    if tb:
        # Basic heuristics: presence of 'failed' and 'failures' or 'FAILED' lines, and not showing '0 failed' only
        # If the before-output contains 'failed' but also '0 failed', be careful: we want some failures (>0)
        if "failed" in tb:
            # ensure it's not only "0 failed"
            if "0 failed" in tb:
                shows_failures = False
            else:
                shows_failures = True
        # also check for 'FAILED' from pytest output lines
        if "failed" in candidate.get("test_output_before", "") and shows_failures:
            shows_failures = True
    if shows_failures:
        be_comments.append("Initial test output indicates failing tests.")
    else:
        be_comments.append("Initial test output does not clearly show failures (expected failing tests).")

    # Scoring rule for baseline:
    # Full points if commands_run contains pytest and test_output_before shows failures.
    if pytest_invocations and shows_failures:
        be_score = weights["baseline_execution"]
    else:
        # partial awarding:
        # if pytest invocation present but output missing failures -> half points
        if pytest_invocations and not shows_failures:
            be_score = int(round(weights["baseline_execution"] * 0.5))
            be_comments.append("Partial credit: pytest command present but failing tests not clearly shown in test_output_before.")
        # if no pytest invocation but test_output_before shows failures -> partial
        elif shows_failures and not pytest_invocations:
            be_score = int(round(weights["baseline_execution"] * 0.5))
            be_comments.append("Partial credit: failing tests shown but no pytest command recorded in commands_run.")
        else:
            be_score = 0
            be_comments.append("No pytest invocation and failing output not shown; baseline execution not demonstrated.")

    results["items"]["baseline_execution"] = {
        "awarded": be_score,
        "possible": weights["baseline_execution"],
        "comments": be_comments
    }
    results["total_awarded"] += be_score

    # 2) All tests passing after changes (25)
    atp_score = 0
    atp_comments = []
    ta = norm(candidate.get("test_output_after", ""))
    if not ta:
        atp_comments.append("No test_output_after provided.")
        atp_score = 0
    else:
        # Determine if zero failures. Heuristic:
        # - If 'failed' not present in lowercase test_output_after -> good (but ensure 'passed' present)
        # - Or explicitly contains '0 failed' or '0 failed' substring.
        has_passed = ("passed" in ta)
        has_failed = ("failed" in ta and "0 failed" not in ta)
        zero_failures = False
        if has_passed and not has_failed:
            zero_failures = True
        # Also accept explicit "0 failed" even if 'passed' missing
        if "0 failed" in ta:
            zero_failures = True

        if zero_failures:
            atp_score = weights["all_tests_passing"]
            atp_comments.append("Test output after edits indicates zero failures (tests passing).")
        else:
            atp_score = 0
            atp_comments.append("Test output after edits does not clearly show all tests passing (no 'passed' summary without failures).")
            # Partial credit if 'passed' found but also 'failed' present - maybe some passing tests
            if "passed" in ta and "failed" in ta:
                # Award small partial credit proportional to ratio if counts available (optional)
                # Attempt to parse counts like "6 passed, 0 failed" or "2 passed, 3 failed"
                try:
                    # find patterns like "X passed" and "Y failed"
                    p = re.search(r'(\d+)\s+passed', candidate.get("test_output_after", ""))
                    f = re.search(r'(\d+)\s+failed', candidate.get("test_output_after", ""))
                    if p:
                        pnum = int(p.group(1))
                    else:
                        pnum = None
                    if f:
                        fnum = int(f.group(1))
                    else:
                        fnum = None
                    if pnum is not None and fnum is not None:
                        ratio = pnum / (pnum + fnum)
                        atp_score = int(round(weights["all_tests_passing"] * ratio))
                        atp_comments.append(f"Partial credit based on parsed pass/fail counts: {pnum} passed, {fnum} failed.")
                except Exception:
                    pass

    results["items"]["all_tests_passing"] = {
        "awarded": atp_score,
        "possible": weights["all_tests_passing"],
        "comments": atp_comments
    }
    results["total_awarded"] += atp_score

    # 3) Correct diagnosis (decision_log) (15)
    dd_score = 0
    dd_comments = []
    decision_log = candidate.get("decision_log", [])
    if not isinstance(decision_log, list) or len(decision_log) == 0:
        dd_comments.append("No decision_log entries provided.")
        dd_score = 0
    else:
        # Look for keywords indicating they identified the main root causes:
        # - date format / ISO / YYYY-MM-DD / processed_at removed / status added
        # - sleep removal
        # - adding discount test (include_discount)
        text_blob = " ".join([norm(entry.get("decision", "") if isinstance(entry, dict) else "") for entry in decision_log])
        found_date = any(k in text_blob for k in ["iso", "yyyy", "yyyy-mm-dd", "date", "mm/dd", "mm/dd/yyyy", "iso format", "iso date"])
        found_status_or_processed = any(k in text_blob for k in ["status", "processed_at", "processed at", "processedat", "processed"])
        found_sleep = "sleep" in text_blob or "time.sleep" in text_blob
        found_discount = any(k in text_blob for k in ["discount", "include_discount", "include discount", "discount test"])
        # Award up to 15: 5 pts each major point
        dd_score = 0
        if found_date or found_status_or_processed:
            dd_score += 7  # give slightly higher weight to date/status
            dd_comments.append("Identified date format/status/legacy field issues.")
        else:
            dd_comments.append("Did not clearly identify date/status/legacy field issues.")
        if found_sleep:
            dd_score += 4
            dd_comments.append("Identified and chose to remove unnecessary time.sleep.")
        else:
            dd_comments.append("No mention of removing sleep or timing anti-patterns.")
        if found_discount:
            dd_score += 4
            dd_comments.append("Planned to add/verify discount behavior (include_discount).")
        else:
            dd_comments.append("No clear mention of adding a discount test or verifying include_discount behavior.")
        # Cap to possible
        dd_score = min(dd_score, weights["diagnosis_decision_log"])

    results["items"]["diagnosis_decision_log"] = {
        "awarded": dd_score,
        "possible": weights["diagnosis_decision_log"],
        "comments": dd_comments
    }
    results["total_awarded"] += dd_score

    # 4) Quality of test edits (20)
    qte_score = weights["quality_test_edits"]
    qte_comments = []
    modified_files = candidate.get("modified_files", [])
    if not isinstance(modified_files, list) or len(modified_files) == 0:
        qte_comments.append("No modified_files provided.")
        qte_score = 0
    else:
        # Deductions:
        # - big deduction if any modified file outside tests/ (e.g., app/)
        deduct = 0
        outside_tests = []
        modified_sut = []
        for mf in modified_files:
            path = mf.get("path") if isinstance(mf, dict) else None
            if not path:
                continue
            if not path.startswith("tests/"):
                outside_tests.append(path)
            if path.startswith("app/"):
                modified_sut.append(path)
        if outside_tests:
            # heavy deduction
            qte_comments.append(f"Modified files outside tests/: {outside_tests}. Tests-only edits expected; SUT edits may reduce score.")
            deduct += 10
        if modified_sut:
            qte_comments.append(f"Modified application/SUT files: {modified_sut}. This is discouraged.")
            # Already counted in outside_tests; ensure extra note but not double-deduct beyond allowed
        # - check content for time.sleep still present
        sleep_found = False
        strict_eq_found = False
        for mf in modified_files:
            content = mf.get("content", "") if isinstance(mf, dict) else ""
            lc = norm(content)
            if "time.sleep" in lc or "sleep(" in lc:
                sleep_found = True
            # Detect legacy strict dict equality checks like 'assert order == expected' or 'expected =' followed by assert order == expected
            if re.search(r"assert\s+order\s*==\s*expected", content) or ("expected =" in content and "assert order ==" in content):
                strict_eq_found = True
        if sleep_found:
            qte_comments.append("time.sleep found in modified test content (should be removed).")
            deduct += 4
        if strict_eq_found:
            qte_comments.append("Detected strict whole-dict equality assertion still present in modified tests.")
            deduct += 6

        # Apply deductions
        qte_score = max(0, qte_score - deduct)

    results["items"]["quality_test_edits"] = {
        "awarded": qte_score,
        "possible": weights["quality_test_edits"],
        "comments": qte_comments
    }
    results["total_awarded"] += qte_score

    # 5) Added test correctness & style (10)
    atc_score = 0
    atc_comments = []
    added_files = candidate.get("added_files", [])
    if not isinstance(added_files, list):
        atc_comments.append("added_files key missing or not a list.")
        atc_score = 0
    elif len(added_files) == 0:
        atc_comments.append("No added_files; expected a test verifying include_discount behavior for O1001.")
        atc_score = 0
    else:
        # Look for any added test that asserts include_discount True for O1001 yields 32.5
        found = False
        found_details = []
        for af in added_files:
            path = af.get("path", "")
            content = af.get("content", "")
            lc = norm(content)
            # check the path looks like tests/test_*.py
            if not path.startswith("tests/"):
                found_details.append(f"Added file {path} is not under tests/.")
            # Check for include_discount usage and expected numeric value
            if "include_discount" in lc and ("o1001" in lc or "o1001" in norm(path)):
                # check for expected numeric 32.5 or 32.50
                if re.search(r"32\.5|32\.50", content):
                    found = True
                    found_details.append(f"Found candidate added test {path} with include_discount and expected 32.5.")
                    break
                else:
                    found_details.append(f"Found include_discount test in {path} but expected value 32.5 not present.")
        if found:
            atc_score = weights["added_test"]
            atc_comments.append("Added test correctly verifies include_discount behavior for O1001 (32.5).")
        else:
            atc_score = 0
            atc_comments.append("No added test found that verifies calculate_total(..., include_discount=True) == 32.5 for O1001.")
            atc_comments.extend(found_details)

    results["items"]["added_test"] = {
        "awarded": atc_score,
        "possible": weights["added_test"],
        "comments": atc_comments
    }
    results["total_awarded"] += atc_score

    # 6) Documentation & reproducibility (15)
    dr_score = 0
    dr_comments = []
    reproduce_instr = candidate.get("reproduce_instructions", "")
    candidate_name = candidate.get("candidate_name", "")
    time_spent = candidate.get("time_spent_minutes", None)
    # Check reproduce_instructions non-empty and contains pytest command
    if reproduce_instr and isinstance(reproduce_instr, str) and "pytest" in norm(reproduce_instr):
        dr_comments.append("reproduce_instructions present and contains pytest command.")
        dr_score += 5
    else:
        dr_comments.append("reproduce_instructions missing or does not contain a pytest command (partial/no reproducibility).")
    # Check each modified_file and added_file has 'reason'
    mf_list = candidate.get("modified_files", [])
    af_list = candidate.get("added_files", [])
    missing_reasons = []
    ok_reasons = True
    for mf in mf_list:
        if not isinstance(mf, dict) or not mf.get("reason"):
            missing_reasons.append(mf.get("path", "<unknown>"))
    for af in af_list:
        if not isinstance(af, dict) or not af.get("reason"):
            missing_reasons.append(af.get("path", "<unknown>"))
    if missing_reasons:
        dr_comments.append(f"Some modified/added files missing 'reason' field: {missing_reasons}")
        # partial credit: give 2/5
        dr_score += 2
    else:
        dr_comments.append("All modified and added files include brief reasons.")
        dr_score += 5
    # Check commands_run includes at least two pytest runs (before and after)
    commands = candidate.get("commands_run", [])
    pytest_cmds = [c for c in commands if isinstance(c, str) and ("pytest" in c or "python -m pytest" in c)]
    if len(pytest_cmds) >= 2:
        dr_comments.append("commands_run contains at least two pytest invocations (likely before and after).")
        dr_score += 5
    elif len(pytest_cmds) == 1:
        dr_comments.append("commands_run contains a single pytest invocation (should include both before and after runs).")
        dr_score += 3
    else:
        dr_comments.append("No pytest commands recorded in commands_run.")
        dr_score += 0

    # Check candidate_name and time_spent_minutes presence and reasonable time
    if not candidate_name:
        dr_comments.append("candidate_name not provided.")
    if isinstance(time_spent, int) or isinstance(time_spent, float):
        if time_spent > 90:
            dr_comments.append(f"time_spent_minutes is {time_spent}, exceeding allowed 90 minutes.")
        else:
            dr_comments.append(f"time_spent_minutes recorded as {time_spent}.")
    else:
        dr_comments.append("time_spent_minutes missing or not numeric.")

    # Cap dr_score to possible
    dr_score = max(0, min(dr_score, weights["documentation_repro"]))

    results["items"]["documentation_repro"] = {
        "awarded": dr_score,
        "possible": weights["documentation_repro"],
        "comments": dr_comments
    }
    results["total_awarded"] += dr_score

    # Summary
    results["overall_score"] = round((results["total_awarded"] / results["total_possible"]) * 100, 2) if results["total_possible"] > 0 else 0.0
    results["pass"] = results["overall_score"] >= 80.0

    # Include some helpful diagnostics comparing to answer_key (if provided)
    ak_notes = []
    if isinstance(answer_key, dict):
        # If answer_key includes an added_files expectation, check candidate added_files path matches expected (optional)
        ak_added = answer_key.get("added_files", [])
        if isinstance(ak_added, list) and len(ak_added) > 0:
            ak_paths = [a.get("path", "") for a in ak_added if isinstance(a, dict)]
            cand_added_paths = [a.get("path", "") for a in added_files if isinstance(a, dict)]
            # If expected path present in answer_key, check candidate included something similar
            for expected_path in ak_paths:
                if expected_path and expected_path not in cand_added_paths:
                    ak_notes.append(f"Answer key expects added file '{expected_path}', candidate did not add a file with that path.")
                else:
                    ak_notes.append(f"Answer key expected added file '{expected_path}' found in candidate submission.")
    results["answer_key_notes"] = ak_notes

    # Attach any global deductions notes
    results["deductions_notes"] = deductions_notes

    return results

def main():
    if len(sys.argv) < 3:
        print("Usage: python task_evaluation.py <candidate_submission.json> <answer_key.json>")
        sys.exit(2)

    candidate_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    candidate_json, err = load_json(candidate_path)
    if err:
        print(f"Error reading candidate submission JSON '{candidate_path}': {err}")
        # produce minimal test_results.json
        out = {
            "error": f"Could not read candidate submission JSON: {err}"
        }
        with open("test_results.json", "w", encoding="utf-8") as f:
            json.dump(out, f, indent=2)
        sys.exit(1)

    answer_key_json, ak_err = load_json(answer_key_path)
    if ak_err:
        print(f"Error reading answer key JSON '{answer_key_path}': {ak_err}")
        # proceed without answer key but include warning
        answer_key_json = {}
        # continue

    # Perform grading
    results = grade(candidate_json, answer_key_json)

    # Save detailed results to test_results.json
    try:
        with open("test_results.json", "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
    except Exception as e:
        print(f"Failed to write test_results.json: {e}")
        sys.exit(1)

    # Print summary to stdout
    print("Grading complete.")
    print(f"Total awarded: {results['total_awarded']} / {results['total_possible']}")
    print(f"Overall score: {results['overall_score']}%")
    print(f"Pass (>=80%): {results['pass']}")
    print("Detailed results written to test_results.json")

if __name__ == "__main__":
    main()
