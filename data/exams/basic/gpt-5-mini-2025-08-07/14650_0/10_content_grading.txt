# task_evaluation.py
"""
Automated grader for the "Updating Automated Tests" basic practical exam.

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

The script compares the candidate submission JSON against the answer key JSON and
produces a detailed grading result saved as test_results.json in the script's folder.

Requirements: only standard library modules are used.
"""

import json
import os
import re
import sys

# Constants for maximum points per rubric item
MAX_POINTS = {
    "baseline_execution": 15,
    "all_tests_passing": 25,
    "decision_log": 15,
    "test_edit_quality": 20,
    "added_test": 10,
    "documentation": 15
}

RESULTS_FILENAME = "test_results.json"


def safe_load_json(path):
    """Load JSON from path with basic error handling."""
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f), None
    except Exception as e:
        return None, f"Failed to load JSON from {path}: {e}"


def contains_pytest_command(commands):
    """Return True if commands list contains a pytest invocation."""
    if not isinstance(commands, list):
        return False
    for c in commands:
        if not isinstance(c, str):
            continue
        cstr = c.strip().lower()
        # common pytest invocations
        if cstr.startswith("pytest") or "pytest" in cstr or cstr.startswith("python -m pytest"):
            return True
    return False


def find_pass_fail_info(output_text):
    """Analyze pytest output text to find occurrences of passed/failed counts.

    Returns a dict with keys:
     - has_passed_phrase (bool): any "passed" occurrence
     - has_failed_phrase (bool): any "failed" occurrence (including 'FAILED')
     - passed_count (int|null): number extracted from 'X passed' or None
     - failed_count (int|null): number extracted from 'X failed' or None
    """
    res = {"has_passed_phrase": False, "has_failed_phrase": False, "passed_count": None, "failed_count": None}
    if not output_text or not isinstance(output_text, str):
        return res
    text = output_text.lower()
    # Detect 'X passed' and 'X failed'
    m_pass = re.search(r"(\d+)\s+passed", text)
    if m_pass:
        res["has_passed_phrase"] = True
        try:
            res["passed_count"] = int(m_pass.group(1))
        except Exception:
            res["passed_count"] = None
    m_fail = re.search(r"(\d+)\s+failed", text)
    if m_fail:
        res["has_failed_phrase"] = True
        try:
            res["failed_count"] = int(m_fail.group(1))
        except Exception:
            res["failed_count"] = None
    # Also consider presence of words 'FAILED' or 'FAILURES' as failing indicator
    if "failed" in text or "failure" in text:
        res["has_failed_phrase"] = res["has_failed_phrase"] or ("failed" in text)
    return res


def score_baseline_execution(submission):
    """Score baseline execution (15 points)."""
    pts = 0.0
    max_pts = MAX_POINTS["baseline_execution"]
    reasons = []

    test_output_before = submission.get("test_output_before", "")
    commands_run = submission.get("commands_run", [])

    # 1) test_output_before non-empty
    if test_output_before and isinstance(test_output_before, str) and test_output_before.strip():
        pts += 5.0
    else:
        reasons.append("test_output_before is empty or missing (no baseline output captured).")

    # 2) test_output_before should show failing tests (look for 'failed' or 'FAILURES' indicators)
    info = find_pass_fail_info(test_output_before)
    if info["has_failed_phrase"]:
        # reward more when explicit failed count exists
        if info.get("failed_count") and info["failed_count"] > 0:
            pts += 6.0
        else:
            # there is indication of failure words
            pts += 4.0
    else:
        reasons.append("Initial test output did not contain clear failure indicators ('failed'/'FAILURES').")

    # 3) commands_run should include pytest invocation
    if contains_pytest_command(commands_run):
        pts += 4.0
    else:
        reasons.append("commands_run does not include a pytest invocation (e.g., 'pytest -q').")

    return min(pts, max_pts), reasons


def score_all_tests_passing(submission):
    """Score all tests passing after edits (25 points)."""
    pts = 0.0
    max_pts = MAX_POINTS["all_tests_passing"]
    reasons = []

    test_output_after = submission.get("test_output_after", "")

    if not test_output_after or not isinstance(test_output_after, str) or not test_output_after.strip():
        reasons.append("test_output_after is empty or missing (no final test output captured).")
        return 0.0, reasons

    info = find_pass_fail_info(test_output_after)

    # Must show passed tests
    if info["has_passed_phrase"]:
        # If there is a failed_count explicitly >0, mark partial
        if info.get("failed_count") is not None:
            if info["failed_count"] == 0:
                # best case: explicit "0 failed"
                pts = max_pts
            else:
                # has failed > 0
                pts = 10.0
                reasons.append(f"test_output_after indicates {info['failed_count']} failed tests; expected 0 failed.")
        else:
            # No explicit failed count; check for 'failed' word presence
            if info["has_failed_phrase"]:
                # ambiguous: contains 'failed' but no numeric count
                pts = 12.0
                reasons.append("test_output_after contains failure keywords; expected zero failed.")
            else:
                # passed present and no failure words
                pts = max_pts
    else:
        reasons.append("test_output_after does not show 'passed' results; expected tests to pass.")
        pts = 0.0

    return min(pts, max_pts), reasons


def score_decision_log(submission):
    """Score decision_log quality (15 points)."""
    pts = 0.0
    max_pts = MAX_POINTS["decision_log"]
    reasons = []

    decision_log = submission.get("decision_log")
    if not isinstance(decision_log, list) or len(decision_log) == 0:
        reasons.append("decision_log missing or empty.")
        return 0.0, reasons

    # Basic credit for providing entries
    if len(decision_log) >= 1:
        pts += 5.0
    # Encourage >=3 entries
    if len(decision_log) >= 3:
        pts += 3.0
    # Now check for presence of keywords indicating correct diagnosis
    text_blob = " ".join(
        [ (entry.get("decision","") + " " + entry.get("alternatives_considered","") + " " + entry.get("step","")) for entry in decision_log if isinstance(entry, dict)]
    ).lower()

    keywords = ["date", "status", "processed_at", "processed", "legacy", "sleep", "time.sleep", "discount", "pytest", "format", "summary"]
    # Count unique matched high-value keywords among a subset
    matched = set()
    for kw in keywords:
        if kw in text_blob:
            matched.add(kw)
    # We consider up to 4 matches for additional credit
    matched_count = len(matched)
    # allocate up to 7 points for matched diagnostics (since we already gave 8)
    diagnostic_pts = min(matched_count, 4) * (7.0 / 4.0)  # up to 7 pts
    pts += diagnostic_pts

    # Provide feedback
    if matched_count == 0:
        reasons.append("decision_log does not mention expected root-cause keywords (date, status, discount, sleep, processed_at).")
    else:
        reasons.append(f"decision_log mentions {matched_count} relevant keywords: {sorted(list(matched))}.")

    return min(pts, max_pts), reasons


def score_test_edit_quality(submission, answer_key=None):
    """Score quality of test edits (20 points).

    Checks:
    - modified files are test files (in tests/)
    - no app/ or SUT modifications (penalize)
    - time.sleep removed (i.e., absence in modified content)
    - assertions are robust: presence of 'in' checks, use of order.get, or calculate_total checks
    - presence of numeric total checks (37.5 or 32.5) in test content (ensures not loosened excessively)
    """
    pts = 0.0
    max_pts = MAX_POINTS["test_edit_quality"]
    reasons = []

    modified_files = submission.get("modified_files", [])
    if not isinstance(modified_files, list) or len(modified_files) == 0:
        reasons.append("No modified_files provided.")
        return 0.0, reasons

    # 1) All modified files should be under tests/
    all_under_tests = all(isinstance(f, dict) and isinstance(f.get("path",""), str) and f.get("path","").startswith("tests/") for f in modified_files)
    if all_under_tests:
        pts += 5.0
    else:
        reasons.append("One or more modified files are not under 'tests/'. Modifying SUT code typically reduces score.")
        # still allow partial credit: award 2 points if some test files were modified
        if any(isinstance(f, dict) and isinstance(f.get("path",""), str) and f.get("path","").startswith("tests/") for f in modified_files):
            pts += 2.0

    # 2) Ensure modified content does not contain time.sleep (anti-pattern)
    removed_sleep = True
    for f in modified_files:
        content = f.get("content", "") if isinstance(f, dict) else ""
        if "time.sleep" in content or "sleep(" in content:
            removed_sleep = False
            break
    if removed_sleep:
        pts += 5.0
    else:
        reasons.append("Found time.sleep in modified test content (should be removed).")

    # 3) Robust assertion patterns: presence of 'in' checks, 'get(' usage, or partial string checks
    robust_found = False
    numeric_total_found = False
    calculate_total_checks = False
    for f in modified_files:
        content = f.get("content", "") if isinstance(f, dict) else ""
        c_lower = content.lower()
        # check for common robust patterns
        if re.search(r"assert\s+.*in\s+.*", content):
            robust_found = True
        if "order.get(" in content or "order.get \"" in content or "order.get '" in content:
            robust_found = True
        if "in summary" in c_lower or "contains" in c_lower:
            robust_found = True
        if re.search(r"calculate_total\(.+include_discount\s*=\s*True", content) or "include_discount=True" in content:
            calculate_total_checks = True
        # numeric totals: look for 37.5, 32.5, 37.50, 32.50
        if re.search(r"32\.5|32\.50|37\.5|37\.50", content):
            numeric_total_found = True

    if robust_found:
        pts += 5.0
    else:
        reasons.append("Modified tests do not appear to use robust assertions (e.g., 'in' checks or order.get).")

    if numeric_total_found:
        pts += 5.0
    else:
        reasons.append("Modified tests do not include explicit numeric total checks (e.g., 37.5 or 32.5). This may indicate overly-loose assertions.")

    # 4) Avoid editing SUT: penalize if modified file path starts with 'app/' or 'testdata/'
    sut_modified = any(isinstance(f, dict) and isinstance(f.get("path",""), str) and (f.get("path","").startswith("app/") or f.get("path","").startswith("testdata/")) for f in modified_files)
    if sut_modified:
        # Deduct: don't award full points; subtract significant amount
        reasons.append("One or more SUT files (app/ or testdata/) were modified; tests-only edits are preferred and expected.")
        # Deduct by removing some of earned points (we'll just cap max achievable to 10 here)
        if pts > 10.0:
            pts = 10.0

    return min(pts, max_pts), reasons


def score_added_test(submission):
    """Score the added test correctness & style (10 points).

    Look for an added file that asserts calculate_total(order, include_discount=True) == 32.5
    """
    pts = 0.0
    max_pts = MAX_POINTS["added_test"]
    reasons = []

    added_files = submission.get("added_files", [])
    if not isinstance(added_files, list) or len(added_files) == 0:
        reasons.append("No added_files found; required test for discount flag missing.")
        return 0.0, reasons

    # Search added files for calculate_total with include_discount=True and an assert for 32.5
    found_good = False
    found_some = False
    for f in added_files:
        content = f.get("content", "") if isinstance(f, dict) else ""
        if not isinstance(content, str):
            continue
        content_lower = content.lower()
        if "calculate_total" in content_lower:
            found_some = True
            # Check for include_discount=True and numeric assert 32.5
            if "include_discount=True".lower() in content_lower or "include_discount = True".lower() in content_lower:
                # Check for 32.5 or 32.50
                if re.search(r"32\.5|32\.50", content_lower):
                    found_good = True
                    break

    if found_good:
        pts = max_pts
    elif found_some:
        pts = max_pts * 0.5
        reasons.append("Added test(s) reference calculate_total but did not clearly assert include_discount=True with expected 32.5.")
    else:
        reasons.append("Added test(s) do not reference calculate_total; required discount test appears missing.")

    return pts, reasons


def score_documentation_and_repro(submission):
    """Score documentation & reproducibility (15 points)."""
    pts = 0.0
    max_pts = MAX_POINTS["documentation"]
    reasons = []

    commands_run = submission.get("commands_run", [])
    reproduce_instructions = submission.get("reproduce_instructions", "")
    modified_files = submission.get("modified_files", [])
    added_files = submission.get("added_files", [])

    # 1) commands_run includes pytest and at least 2 commands (before and after)
    if isinstance(commands_run, list) and contains_pytest_command(commands_run):
        # encourage at least two entries (baseline run and final run)
        pts += 3.0
        if len(commands_run) >= 2:
            pts += 2.0
    else:
        reasons.append("commands_run does not include pytest or is missing.")

    # 2) reproduce_instructions contains pytest -q (or pytest)
    if isinstance(reproduce_instructions, str) and "pytest" in reproduce_instructions.lower():
        pts += 5.0
    else:
        reasons.append("reproduce_instructions does not mention how to run pytest (e.g., 'pytest -q').")

    # 3) modified_files and added_files content included and non-empty
    good_files = True
    if not isinstance(modified_files, list) or len(modified_files) == 0:
        good_files = False
        reasons.append("modified_files missing or empty in submission.")
    else:
        for f in modified_files:
            if not isinstance(f, dict) or not f.get("content"):
                good_files = False
                reasons.append(f"One modified file missing content or improperly formatted: {f.get('path') if isinstance(f, dict) else str(f)}")
                break
    if not isinstance(added_files, list):
        good_files = False
        reasons.append("added_files field missing or not an array.")
    else:
        # It's acceptable to have empty array; but we expect at least the added test per spec
        pass

    if good_files:
        pts += 5.0
    else:
        reasons.append("Some modified/added file contents are missing or malformed in submission.")

    return min(pts, max_pts), reasons


def aggregate_results(scores_dict, deductions_list):
    """Aggregate results into a summary dict for JSON output."""
    total_earned = sum(scores_dict.values())
    total_max = sum(MAX_POINTS.values())
    percentage = round((total_earned / total_max) * 100.0, 2) if total_max > 0 else 0.0
    return {
        "scores": scores_dict,
        "max_points": total_max,
        "total_points_earned": round(total_earned, 2),
        "percentage_score": percentage,
        "overall_score": percentage,
        "deductions": deductions_list
    }


def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)

    submission_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    submission, err = safe_load_json(submission_path)
    if err:
        print(err)
        # produce minimal results file with error
        out = {
            "error": err,
            "overall_score": 0.0
        }
        with open(RESULTS_FILENAME, "w", encoding="utf-8") as f:
            json.dump(out, f, indent=2)
        sys.exit(1)

    answer_key, err2 = safe_load_json(answer_key_path)
    if err2:
        print(err2)
        out = {
            "error": err2,
            "overall_score": 0.0
        }
        with open(RESULTS_FILENAME, "w", encoding="utf-8") as f:
            json.dump(out, f, indent=2)
        sys.exit(1)

    # Basic validation of required fields in submission
    required_fields = ["candidate_name", "time_spent_minutes", "commands_run", "test_output_before",
                       "test_output_after", "modified_files", "added_files", "decision_log", "reproduce_instructions"]
    missing = [f for f in required_fields if f not in submission]
    deductions = []
    if missing:
        deductions.append(f"Submission missing required fields: {missing}")

    # Optional: check time_spent_minutes within limit
    t_spent = submission.get("time_spent_minutes")
    try:
        t_spent_val = int(t_spent) if t_spent is not None else None
        if t_spent_val is not None and t_spent_val > 90:
            deductions.append(f"time_spent_minutes is {t_spent_val} which exceeds the 90 minute limit.")
    except Exception:
        deductions.append("time_spent_minutes is not an integer or missing.")

    # Run scoring functions
    scores = {}
    reasons_all = []

    b_pts, b_reasons = score_baseline_execution(submission)
    scores["baseline_execution"] = round(b_pts, 2)
    reasons_all.extend([("baseline_execution", r) for r in b_reasons])

    p_pts, p_reasons = score_all_tests_passing(submission)
    scores["all_tests_passing"] = round(p_pts, 2)
    reasons_all.extend([("all_tests_passing", r) for r in p_reasons])

    d_pts, d_reasons = score_decision_log(submission)
    scores["decision_log"] = round(d_pts, 2)
    reasons_all.extend([("decision_log", r) for r in d_reasons])

    q_pts, q_reasons = score_test_edit_quality(submission, answer_key)
    scores["test_edit_quality"] = round(q_pts, 2)
    reasons_all.extend([("test_edit_quality", r) for r in q_reasons])

    a_pts, a_reasons = score_added_test(submission)
    scores["added_test"] = round(a_pts, 2)
    reasons_all.extend([("added_test", r) for r in a_reasons])

    doc_pts, doc_reasons = score_documentation_and_repro(submission)
    scores["documentation"] = round(doc_pts, 2)
    reasons_all.extend([("documentation", r) for r in doc_reasons])

    # Include any earlier missing fields penalty info into deductions list
    if missing:
        reasons_all.append(("submission_validation", f"Missing required fields: {missing}"))

    # Compose deductions list (flattened strings)
    deductions_msgs = []
    for (section, msg) in reasons_all:
        if msg:
            deductions_msgs.append({"section": section, "issue": msg})

    # Add any global deductions noted earlier
    for d in deductions:
        deductions_msgs.append({"section": "general", "issue": d})

    aggregated = aggregate_results(scores, deductions_msgs)

    # Add some metadata from submission and answer key for reviewer convenience
    aggregated["candidate_name"] = submission.get("candidate_name")
    aggregated["time_spent_minutes"] = submission.get("time_spent_minutes")
    aggregated["commands_run"] = submission.get("commands_run")
    aggregated["test_output_before_exists"] = bool(submission.get("test_output_before"))
    aggregated["test_output_after_exists"] = bool(submission.get("test_output_after"))

    # Save results JSON
    try:
        with open(RESULTS_FILENAME, "w", encoding="utf-8") as f:
            json.dump(aggregated, f, indent=2)
        print(f"Grading complete. Results written to {RESULTS_FILENAME}")
    except Exception as e:
        print(f"Failed to write results to {RESULTS_FILENAME}: {e}")
        sys.exit(1)


if __name__ == "__main__":
    main()
