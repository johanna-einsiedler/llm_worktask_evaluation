# task_evaluation.py
import json
import os
import re
import sys

"""
Automated grading script for the "Updating Automated Tests to Match Evolving Application Behavior" basic exam.

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

Outputs:
    - Creates 'test_results.json' in the same directory as this script with detailed grading breakdown.
    - Variable overall_score inside the JSON contains the numeric percentage (0-100).

Notes:
- This script is deterministic and uses only Python standard library.
- It tolerantly inspects text (case-insensitive where appropriate), and uses heuristics to judge
  whether the candidate met the exam requirements.
- The grading rubric (weights) is embedded in this script to follow the evaluator instructions.
"""

# ----- Helper utilities -----


def safe_load_json(path):
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except Exception as e:
        raise RuntimeError(f"Failed to load JSON from {path}: {e}")


def find_number_from_summary(text, key):
    """
    Try to find occurrences like 'X passed' or 'Y failed' in pytest output text.
    Returns the integer value if found, otherwise None.
    """
    if not isinstance(text, str):
        return None
    # search patterns like '12 passed', '1 passed', possibly with commas, e.g., '1 passed, 0 failed'
    m = re.search(r'(\d+)\s+' + re.escape(key), text, flags=re.IGNORECASE)
    if m:
        try:
            return int(m.group(1))
        except:
            return None
    return None


def contains_failed_indicator(text):
    if not isinstance(text, str):
        return False
    # Accept 'FAILED' or 'failed' or 'failures' etc.
    return bool(re.search(r'\bfailed\b|\bFAILED\b|\bFAILURES\b', text))


def contains_passed_indicator(text):
    if not isinstance(text, str):
        return False
    return bool(re.search(r'\bpassed\b|\bPASSED\b', text))


def text_contains_any(text, keywords):
    if not isinstance(text, str):
        return False
    lower = text.lower()
    for kw in keywords:
        if kw.lower() in lower:
            return True
    return False


def score_fractional(total_points, numerator, denominator):
    if denominator <= 0:
        return 0
    return round(total_points * (numerator / denominator), 2)


# ----- Main grading logic -----


def grade_submission(candidate_path, answer_key_path):
    results = {
        "per_task": {},
        "deductions": [],
        "total_points": 0.0,
        "max_points": 100.0,
        "overall_score": 0.0,
    }

    # Load JSON inputs
    try:
        candidate = safe_load_json(candidate_path)
    except Exception as e:
        results["error"] = str(e)
        results["overall_score"] = 0.0
        # write results and exit
        out_path = os.path.join(os.path.dirname(__file__), "test_results.json")
        with open(out_path, "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
        raise SystemExit(f"Error loading candidate submission: {e}")

    try:
        answer_key = safe_load_json(answer_key_path)
    except Exception as e:
        results["error"] = str(e)
        results["overall_score"] = 0.0
        out_path = os.path.join(os.path.dirname(__file__), "test_results.json")
        with open(out_path, "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
        raise SystemExit(f"Error loading answer key: {e}")

    # Validate minimal candidate fields
    required_fields = [
        "candidate_name",
        "time_spent_minutes",
        "commands_run",
        "test_output_before",
        "test_output_after",
        "modified_files",
        "added_files",
        "decision_log",
        "reproduce_instructions",
    ]
    missing_fields = [f for f in required_fields if f not in candidate]
    if missing_fields:
        results["error"] = f"Submission JSON missing required fields: {missing_fields}"
        results["overall_score"] = 0.0
        out_path = os.path.join(os.path.dirname(__file__), "test_results.json")
        with open(out_path, "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
        raise SystemExit(f"Submission missing fields: {missing_fields}")

    # Extract values with defaults
    commands_run = candidate.get("commands_run", []) or []
    test_output_before = candidate.get("test_output_before", "") or ""
    test_output_after = candidate.get("test_output_after", "") or ""
    modified_files = candidate.get("modified_files", []) or []
    added_files = candidate.get("added_files", []) or []
    decision_log = candidate.get("decision_log", []) or []
    reproduce_instructions = candidate.get("reproduce_instructions", "") or ""
    time_spent = candidate.get("time_spent_minutes", None)

    # Scoring weights (per evaluator guideline)
    weights = {
        "baseline_execution": 15,
        "all_tests_passing": 25,
        "decision_log": 15,
        "quality_test_edits": 20,
        "added_test": 10,
        "documentation": 15,
    }

    total_awarded = 0.0
    deductions = []

    # ---- 1) Baseline execution (15 pts) ----
    baseline_score = 0.0
    baseline_comments = []
    ran_pytest = any("pytest" in cmd.lower() or "python -m pytest" in cmd.lower() for cmd in commands_run)
    before_has_failed = contains_failed_indicator(test_output_before)
    before_has_passed = contains_passed_indicator(test_output_before)

    if ran_pytest and before_has_failed:
        baseline_score = float(weights["baseline_execution"])
        baseline_comments.append("Detected pytest run and failing tests in test_output_before.")
    elif ran_pytest and not before_has_failed and before_has_passed:
        # They ran pytest but didn't capture failing output - partial credit
        baseline_score = round(weights["baseline_execution"] * 0.6, 2)
        baseline_comments.append("Pytest was run but test_output_before does not show failures; partial credit.")
    elif ran_pytest:
        baseline_score = round(weights["baseline_execution"] * 0.4, 2)
        baseline_comments.append("Pytest appears in commands_run but test_output_before lacks clear pass/fail summary; minimal credit.")
    else:
        baseline_score = 0.0
        baseline_comments.append("No pytest invocation found in commands_run; no credit awarded.")

    results["per_task"]["baseline_execution"] = {
        "score": baseline_score,
        "max_score": weights["baseline_execution"],
        "comments": baseline_comments,
    }
    total_awarded += baseline_score

    # ---- 2) All tests passing after changes (25 pts) ----
    passing_score = 0.0
    passing_comments = []
    # Try to extract passed/failed numbers
    passed_num = find_number_from_summary(test_output_after, "passed")
    failed_num = find_number_from_summary(test_output_after, "failed")

    # If we couldn't parse numbers, fall back to text checks
    if failed_num is not None and passed_num is not None:
        total = passed_num + failed_num
        if total == 0:
            passing_score = 0.0
            passing_comments.append("No tests run or could not detect test counts in test_output_after.")
        else:
            if failed_num == 0 and passed_num > 0:
                passing_score = float(weights["all_tests_passing"])
                passing_comments.append(f"All tests passed according to test_output_after ({passed_num} passed, 0 failed).")
            else:
                # Partial credit proportional to passing fraction
                frac = passed_num / total
                passing_score = round(weights["all_tests_passing"] * frac, 2)
                passing_comments.append(f"Detected {passed_num} passed and {failed_num} failed; awarded proportional credit.")
    else:
        # Try simpler heuristics
        after_has_failed = contains_failed_indicator(test_output_after)
        after_has_passed = contains_passed_indicator(test_output_after)
        if after_has_passed and not after_has_failed:
            passing_score = float(weights["all_tests_passing"])
            passing_comments.append("test_output_after indicates passed tests and no failed tests (heuristic).")
        elif after_has_passed and after_has_failed:
            # Unknown counts but both present -> partial
            passing_score = round(weights["all_tests_passing"] * 0.5, 2)
            passing_comments.append("test_output_after shows both 'passed' and 'failed'; granted partial credit.")
        else:
            passing_score = 0.0
            passing_comments.append("test_output_after does not show clear passing summary; no credit for this category.")

    results["per_task"]["all_tests_passing"] = {
        "score": passing_score,
        "max_score": weights["all_tests_passing"],
        "comments": passing_comments,
    }
    total_awarded += passing_score

    # ---- 3) Decision log quality (15 pts) ----
    decision_score = 0.0
    decision_comments = []
    # Look for key diagnoses in decision_log text fields
    required_keywords = [
        "date",  # date format change
        "status",  # presence of status
        "processed_at",  # legacy key removed
        "sleep",  # removal of sleep
        "discount",  # mention include_discount/new test
    ]
    found_keywords = set()
    if isinstance(decision_log, list):
        for entry in decision_log:
            # Combine the fields into one string to search
            combined = " ".join(
                [str(entry.get(k, "")).lower() for k in ("step", "decision", "alternatives_considered")]
            )
            for kw in required_keywords:
                if kw.lower() in combined:
                    found_keywords.add(kw.lower())
    else:
        decision_comments.append("decision_log is not a list; no points awarded for decision_log quality.")

    # Score proportionally to number of distinct required keywords found (max 5)
    found_count = len(found_keywords)
    decision_score = round(weights["decision_log"] * min(found_count / len(required_keywords), 1.0), 2)
    if found_count == len(required_keywords):
        decision_comments.append("Decision log identifies all expected root causes (date, status, removed legacy key, sleep, discount).")
    else:
        missing = [k for k in required_keywords if k not in found_keywords]
        decision_comments.append(f"Decision log found {found_count}/{len(required_keywords)} expected issues; missing: {missing}.")

    results["per_task"]["decision_log"] = {
        "score": decision_score,
        "max_score": weights["decision_log"],
        "comments": decision_comments,
        "found_keywords": sorted(list(found_keywords)),
    }
    total_awarded += decision_score

    # ---- 4) Quality of test edits (20 pts) ----
    quality_score = 0.0
    quality_comments = []
    # Heuristics to evaluate quality of edits:
    heuristics = {
        "only_tests_modified": True,
        "no_legacy_date_format": True,
        "presence_iso_date": False,
        "status_check_present": False,
        "sleep_removed": True,
    }
    # Check modified_files structure
    if not isinstance(modified_files, list) or len(modified_files) == 0:
        quality_comments.append("No modified_files provided; cannot evaluate test edit quality.")
        heuristics["only_tests_modified"] = False
        heuristics["no_legacy_date_format"] = False
        heuristics["presence_iso_date"] = False
        heuristics["status_check_present"] = False
        heuristics["sleep_removed"] = False
    else:
        for mf in modified_files:
            path = mf.get("path", "")
            content = mf.get("content", "") or ""
            # Only tests/ files should be modified for full credit
            if not path.startswith("tests/"):
                heuristics["only_tests_modified"] = False
                quality_comments.append(f"Modified file outside tests/: {path} (reduces score).")
            # legacy date formats (MM/DD/YYYY) should not appear in updated tests
            if re.search(r'0[1-9]|1[0-2]\/\d{2}\/\d{4}', content) or "01/15/2025" in content or "02/20/2025" in content:
                heuristics["no_legacy_date_format"] = False
            # ensure ISO date appears
            if "2025-01-15" in content or "2025-02-20" in content:
                heuristics["presence_iso_date"] = True
            # check for status assertion or '[status:' text
            if "[status:" in content or "status" in content:
                heuristics["status_check_present"] = True
            # sleep removed: ensure no 'sleep(' remains
            if "sleep(" in content:
                heuristics["sleep_removed"] = False

    # Compute quality score: 5 heuristics, equal weight
    true_count = sum(1 for v in heuristics.values() if v)
    quality_score = round(weights["quality_test_edits"] * (true_count / len(heuristics)), 2)

    # If SUT files modified (i.e., only_tests_modified is False), penalize by reducing quality score by 50%
    if not heuristics["only_tests_modified"]:
        quality_score = round(quality_score * 0.5, 2)
        quality_comments.append("Some modified files were outside tests/ — this reduces maintainability and score.")

    # Append heuristics detail to comments
    quality_comments.append(f"Heuristics results: {heuristics}.")
    results["per_task"]["quality_test_edits"] = {
        "score": quality_score,
        "max_score": weights["quality_test_edits"],
        "comments": quality_comments,
        "heuristics": heuristics,
    }
    total_awarded += quality_score

    # ---- 5) Added test correctness & style (10 pts) ----
    added_score = 0.0
    added_comments = []
    # Candidate must add test that verifies calculate_total(..., include_discount=True) for O1001 -> 32.5
    found_good_added_test = False
    found_partial = False
    added_files_listed = isinstance(added_files, list) and len(added_files) > 0
    if added_files_listed:
        for af in added_files:
            path = af.get("path", "")
            content = af.get("content", "") or ""
            # must be in tests/
            if not path.startswith("tests/"):
                continue
            # Look for calculate_total call with include_discount=True and O1001 and 32.5
            content_lower = content.lower()
            has_calc = "calculate_total" in content_lower
            has_include = "include_discount" in content_lower
            has_o1001 = "o1001" in content_lower
            has_32_5 = re.search(r'32(?:\.5|\.50)|32\.50|32\.5', content) is not None
            if has_calc and has_include and has_o1001 and has_32_5:
                found_good_added_test = True
                break
            if has_calc and (has_o1001 or has_32_5):
                found_partial = True
    else:
        added_comments.append("No added_files present in submission.")

    if found_good_added_test:
        added_score = float(weights["added_test"])
        added_comments.append("Added test verifies calculate_total(..., include_discount=True) for O1001 with expected value 32.5.")
    elif found_partial:
        added_score = round(weights["added_test"] * 0.5, 2)
        added_comments.append("Added test includes some relevant checks for calculate_total but missing explicit include_discount=True or exact expected value; partial credit.")
    else:
        added_score = 0.0
        added_comments.append("No appropriate added test found that verifies include_discount behavior for O1001.")

    results["per_task"]["added_test"] = {
        "score": added_score,
        "max_score": weights["added_test"],
        "comments": added_comments,
    }
    total_awarded += added_score

    # ---- 6) Documentation & reproducibility (15 pts) ----
    doc_score = 0.0
    doc_comments = []
    # Conditions for full credit:
    # - commands_run contains pytest executed at least twice (one before, one after) OR contains pytest and there is reproduce_instructions
    pytest_runs = sum(1 for c in commands_run if "pytest" in c.lower() or "python -m pytest" in c.lower())
    has_repro = isinstance(reproduce_instructions, str) and len(reproduce_instructions.strip()) > 0
    has_modified_files_content = isinstance(modified_files, list) and len(modified_files) > 0
    has_added_files_content = isinstance(added_files, list)  # may be empty

    if pytest_runs >= 2 and has_repro and has_modified_files_content:
        doc_score = float(weights["documentation"])
        doc_comments.append("Commands include pytest runs (>=2), reproduce_instructions provided, and modified_files included — full credit.")
    else:
        # Partial scoring:
        subtotal = 0
        if pytest_runs >= 1:
            subtotal += 6  # approx half of doc points
            doc_comments.append("At least one pytest invocation found in commands_run.")
        else:
            doc_comments.append("No pytest command found in commands_run.")

        if has_repro:
            subtotal += 5
            doc_comments.append("reproduce_instructions provided.")
        else:
            doc_comments.append("No reproduce_instructions or it's empty.")

        if has_modified_files_content:
            subtotal += 4
            doc_comments.append("modified_files content included in submission.")
        else:
            doc_comments.append("modified_files missing or empty.")

        # Cap to max
        doc_score = min(round(subtotal, 2), weights["documentation"])

    results["per_task"]["documentation"] = {
        "score": doc_score,
        "max_score": weights["documentation"],
        "comments": doc_comments,
    }
    total_awarded += doc_score

    # ---- Additional mandatory checks and warnings ----
    mandatory_warnings = []
    mandatory_fail_reasons = []

    # Check requirement: test_output_before must show failing tests
    if not before_has_failed:
        mandatory_warnings.append("test_output_before does not show failing tests. The exam expects an initial failing run.")
        # Deduct up to 5 points from baseline if they captured something else
        if results["per_task"]["baseline_execution"]["score"] == weights["baseline_execution"]:
            # they previously got full; now reduce baseline score
            reduction = 5.0
            results["per_task"]["baseline_execution"]["score"] = max(0.0, results["per_task"]["baseline_execution"]["score"] - reduction)
            total_awarded -= reduction
            deductions.append(f"Baseline capture did not show failing tests; deducted {reduction} points from baseline_execution.")
    # Check requirement: test_output_after must show all tests passing (0 failed)
    # We consider full passing only when failed_num == 0 OR heuristic sees 'passed' without 'failed'
    after_failed = find_number_from_summary(test_output_after, "failed")
    if after_failed is not None:
        if after_failed != 0:
            mandatory_warnings.append("test_output_after shows failing tests. The exam expects the final suite to pass (0 failed).")
    else:
        # If no numeric parsed, be strict and require '0 failed' or absence of 'failed' after passes
        if contains_passed_indicator(test_output_after) and contains_failed_indicator(test_output_after):
            mandatory_warnings.append("test_output_after shows both passed and failed indicators; expected final pass with 0 failed.")

    # Check that added test requirement is met
    if not found_good_added_test:
        mandatory_warnings.append("Required added test verifying calculate_total(..., include_discount=True) for O1001 with expected 32.5 not present or not detectable.")

    # Check that modifications were limited to tests/ files ideally
    non_test_mods = [mf.get("path") for mf in modified_files if not mf.get("path", "").startswith("tests/")]
    if non_test_mods:
        mandatory_warnings.append(f"Modified files outside tests/ detected: {non_test_mods}. The basic exam expects changes limited to tests/ only.")
        # Deduct a fixed penalty from total_awarded (e.g., 5 points)
        penalty = 5.0
        total_awarded = max(0.0, total_awarded - penalty)
        deductions.append(f"Modified non-tests files penalty: -{penalty} points for editing SUT or non-test files.")

    # Compose final results
    results["deductions"] = deductions
    results["warnings"] = mandatory_warnings

    # Totals
    results["total_points"] = round(total_awarded, 2)
    results["max_points"] = float(weights["baseline_execution"] + weights["all_tests_passing"] + weights["decision_log"]
                                 + weights["quality_test_edits"] + weights["added_test"] + weights["documentation"])
    # overall percentage
    if results["max_points"] > 0:
        overall_pct = round((results["total_points"] / results["max_points"]) * 100.0, 2)
    else:
        overall_pct = 0.0
    results["overall_score"] = overall_pct

    # Add some metadata from submission for evaluator convenience
    results["candidate_name"] = candidate.get("candidate_name")
    results["time_spent_minutes"] = candidate.get("time_spent_minutes")
    results["commands_run"] = commands_run
    results["summary"] = {
        "passed_checks": {
            "before_has_failed": before_has_failed,
            "after_has_no_failed_detected": (after_failed == 0 if after_failed is not None else (contains_passed_indicator(test_output_after) and not contains_failed_indicator(test_output_after))),
            "added_discount_test_present": found_good_added_test,
        }
    }

    # Save results as JSON next to this script
    out_path = os.path.join(os.path.dirname(__file__), "test_results.json")
    try:
        with open(out_path, "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
    except Exception as e:
        raise SystemExit(f"Failed to write test_results.json: {e}")

    # Print a short summary to stdout
    print(f"Grading complete for candidate: {results.get('candidate_name')}")
    print(f"Total: {results['total_points']}/{results['max_points']} points  ({results['overall_score']}%)")
    if results["warnings"]:
        print("Warnings:")
        for w in results["warnings"]:
            print(" -", w)
    if results["deductions"]:
        print("Deductions:")
        for d in results["deductions"]:
            print(" -", d)
    print(f"Detailed results written to: {out_path}")

    return results


# ----- Entrypoint -----


if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)
    candidate_path = sys.argv[1]
    answer_key_path = sys.argv[2]
    try:
        grade_submission(candidate_path, answer_key_path)
    except SystemExit as e:
        # Already printed a message in the function; ensure exit code non-zero
        sys.exit(1)
    except Exception as e:
        print(f"Unexpected error during grading: {e}")
        sys.exit(1)
