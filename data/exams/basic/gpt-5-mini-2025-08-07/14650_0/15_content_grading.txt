#!/usr/bin/env python3
"""
task_evaluation.py

Automated grader for the SQA practical exam (basic). Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

Produces test_results.json in the same directory as this script.

Requirements:
- Uses only Python standard library.
- Robust to common formatting differences (whitespace, capitalization).
- Attempts partial credit using the rubric described in the exam materials.

Author: Generated by assistant
"""

import json
import os
import sys
import traceback

# === Configuration: point allocation based on the rubric ===
POINTS = {
    "task1_reproduce": 15,
    "task2_selector": 15,
    "task2_helper": 10,
    "task3_timing": 20,
    "task4_config": 20,
    "documentation": 10,
    "bonus": 10
}
MAX_POINTS = sum(POINTS.values())


def load_json_file(path):
    with open(path, 'r', encoding='utf-8') as f:
        return json.load(f)


def safe_lower(s):
    if s is None:
        return ""
    return str(s).lower()


def find_changed_file(changed_files, target_path_suffix):
    """Return the first changed_files entry whose file_path ends with target_path_suffix."""
    if not isinstance(changed_files, list):
        return None
    for entry in changed_files:
        fp = entry.get("file_path", "")
        # match by suffix to be tolerant of relative paths
        if fp.endswith(target_path_suffix):
            return entry
    return None


def presence_check(substring, text):
    """Case-insensitive presence check."""
    return substring.lower() in safe_lower(text)


def evaluate_submission(submission, answer_key):
    """
    Main grading logic. Returns a results dict for output to test_results.json.
    """
    results = {
        "scores": {},
        "deductions": [],
        "notes": [],
        "total_points_earned": 0,
        "total_points_possible": MAX_POINTS,
        "overall_score": 0.0
    }

    # Basic validation of submission structure
    required_top_fields = [
        "candidate_name", "start_time_utc", "end_time_utc", "total_minutes",
        "commands_run", "initial_test_output", "final_test_output",
        "changed_files", "summary_of_changes", "risk_notes_and_next_steps"
    ]
    missing_fields = [f for f in required_top_fields if f not in submission]
    if missing_fields:
        results["deductions"].append(
            "Missing required top-level submission fields: " + ", ".join(missing_fields)
        )
        # Continue but note deduction: will affect documentation score
    # Normalize some commonly used strings
    initial_out = submission.get("initial_test_output", "")
    final_out = submission.get("final_test_output", "")
    changed_files = submission.get("changed_files", [])
    tasks_section = submission.get("tasks", {})

    # --- Task 1: Reproduce failing tests (15 points) ---
    # We expect failures for: KeyError: 'base_url', AttributeError (old_find_by_id), FileNotFoundError for data/users.json
    task1_total = POINTS["task1_reproduce"]
    signatures = []
    # Try to extract expected failure signatures from answer key initial_test_output if present
    ak_initial = answer_key.get("initial_test_output", "") if isinstance(answer_key, dict) else ""
    if ak_initial:
        # heuristically pick distinct signature lines from the answer key by splitting lines that contain "KeyError", "AttributeError", "FileNotFoundError"
        for sig in ["KeyError", "AttributeError", "FileNotFoundError", "FileNotFoundError:"]:
            # collect lines containing the sig
            for line in ak_initial.splitlines():
                if sig.lower() in line.lower():
                    # trim whitespace
                    trimmed = line.strip()
                    if trimmed and trimmed not in signatures:
                        signatures.append(trimmed)
    # Fallback default signatures if none found in answer key
    if not signatures:
        signatures = [
            "KeyError: 'base_url'",
            "AttributeError: module 'tests.helpers' has no attribute 'old_find_by_id'",
            "FileNotFoundError: 'data/users.json'"
        ]

    found = 0
    missing_sigs = []
    for sig in signatures:
        if presence_check(sig, initial_out):
            found += 1
        else:
            missing_sigs.append(sig)
    if signatures:
        earned = int(round(task1_total * (found / len(signatures))))
    else:
        earned = 0
    results["scores"]["task1_reproduce"] = {"earned": earned, "possible": task1_total}
    results["total_points_earned"] += earned
    if missing_sigs:
        results["deductions"].append(
            "Task1: initial_test_output missing expected failure signatures: " + "; ".join(missing_sigs)
        )
    else:
        results["notes"].append("Task1: All expected failure signatures present in initial_test_output.")

    # --- Task 2: Selector fix (15) and Helper deprecation fix (10) ---
    # Look for edits in tests/test_suite.py and tests/helpers.py
    t2_selector_total = POINTS["task2_selector"]
    t2_helper_total = POINTS["task2_helper"]
    earned_selector = 0
    earned_helper = 0

    tf_entry = find_changed_file(changed_files, "tests/test_suite.py")
    helpers_entry = find_changed_file(changed_files, "tests/helpers.py")

    # Selector check: updated test should reference 'btn-login' and not keep 'login-btn'
    if tf_entry:
        updated_tf = tf_entry.get("updated_content", "")
        original_tf = tf_entry.get("original_content", "")
        # Ensure test still exists: check that 'def test_login_button' is present in updated (not removed)
        if not presence_check("def test_login_button", updated_tf):
            # candidate may have removed the test - heavy deduction
            results["deductions"].append("Task2-selector: 'def test_login_button' missing from updated tests/test_suite.py (test may have been removed).")
            earned_selector = 0
        else:
            has_new_id = presence_check("btn-login", updated_tf)
            has_old_id_remaining = presence_check("login-btn", updated_tf)
            uses_find_by_id = presence_check("helpers.find_by_id", updated_tf) or presence_check(".find_by_id(", updated_tf)
            # allocate criteria: must use new id and not use old id ideally, and use current helper or compatibility wrapper
            if has_new_id and not has_old_id_remaining and (uses_find_by_id or (helpers_entry and presence_check("def old_find_by_id", helpers_entry.get("updated_content", "")))):
                earned_selector = t2_selector_total
            else:
                # partial credit heuristics
                if has_new_id:
                    earned_selector = int(round(t2_selector_total * 0.7))
                    results["deductions"].append("Task2-selector: new id 'btn-login' found but either old id still present or helper usage not updated; awarded partial credit.")
                elif not has_new_id and presence_check("login-btn", updated_tf):
                    # candidate left old id unchanged
                    earned_selector = int(round(t2_selector_total * 0.2))
                    results["deductions"].append("Task2-selector: test still targets old id 'login-btn'; minimal credit awarded.")
                else:
                    earned_selector = 0
                    results["deductions"].append("Task2-selector: Could not verify selector update for login button.")
    else:
        results["deductions"].append("Task2-selector: No changed entry for tests/test_suite.py found in changed_files.")
        earned_selector = 0

    # Helper deprecation: either tests updated to use helpers.find_by_id or helpers.py adds compatibility old_find_by_id
    helper_fixed = False
    if tf_entry:
        updated_tf = tf_entry.get("updated_content", "")
        if presence_check("helpers.find_by_id", updated_tf) or presence_check(".find_by_id(", updated_tf):
            helper_fixed = True
    if helpers_entry:
        # check if helpers added a compatibility function old_find_by_id
        updated_helpers = helpers_entry.get("updated_content", "")
        if presence_check("def old_find_by_id", updated_helpers) or presence_check("def old_find_by_id(", updated_helpers):
            helper_fixed = True

    if helper_fixed:
        earned_helper = t2_helper_total
    else:
        # partial credit if tests switched to a different approach (e.g., direct string search for button)
        if tf_entry and (presence_check("<button id=", tf_entry.get("updated_content", ""))):
            earned_helper = int(round(t2_helper_total * 0.5))
            results["deductions"].append("Task2-helper: helper deprecation not fully addressed, but test uses a direct button lookup; partial credit.")
        else:
            earned_helper = 0
            results["deductions"].append("Task2-helper: Deprecated helper usage not addressed (no helpers.old_find_by_id compatibility and tests not using helpers.find_by_id).")

    results["scores"]["task2_selector"] = {"earned": earned_selector, "possible": t2_selector_total}
    results["scores"]["task2_helper"] = {"earned": earned_helper, "possible": t2_helper_total}
    results["total_points_earned"] += (earned_selector + earned_helper)

    # --- Task 3: Timing / stability (20) ---
    t3_total = POINTS["task3_timing"]
    earned_t3 = 0
    if tf_entry:
        updated_tf = tf_entry.get("updated_content", "")
        original_tf = tf_entry.get("original_content", "")
        # Conditions: original had time.sleep(2); updated should remove exact hard sleep and either implement polling (os.path.exists) or use updated data path users_v2.json
        originally_had_sleep = presence_check("time.sleep(2)", original_tf)
        updated_has_sleep = presence_check("time.sleep(2)", updated_tf)
        updated_has_poll = presence_check("os.path.exists", updated_tf) or presence_check("while not", updated_tf) or presence_check("time.sleep(0.", updated_tf)
        updated_path_fixed = presence_check("users_v2.json", updated_tf) or presence_check("users_v2", updated_tf)
        # scoring logic
        if originally_had_sleep and (not updated_has_sleep) and (updated_path_fixed or updated_has_poll):
            earned_t3 = t3_total
        else:
            # partial cases
            points = 0
            if (not updated_has_sleep):
                points += t3_total * 0.5
            if updated_path_fixed:
                points += t3_total * 0.5
            # cap
            earned_t3 = int(round(min(points, t3_total)))
            if earned_t3 == 0:
                results["deductions"].append("Task3: No evidence of replacing the hard sleep or fixing data path (time.sleep(2) may still be present or data path unchanged).")
            else:
                # add specific deduction note
                if updated_has_sleep and updated_path_fixed:
                    results["deductions"].append("Task3: Data path updated but hard-coded sleep remains; partial credit awarded.")
                elif (not updated_has_sleep) and (not updated_path_fixed):
                    results["deductions"].append("Task3: Sleep removed but data path still references old file; partial credit awarded.")
    else:
        results["deductions"].append("Task3: tests/test_suite.py change not found; cannot verify timing/stability fix.")
        earned_t3 = 0

    results["scores"]["task3_timing"] = {"earned": earned_t3, "possible": t3_total}
    results["total_points_earned"] += earned_t3

    # --- Task 4: Config/API/deprecation fix (20) ---
    t4_total = POINTS["task4_config"]
    earned_t4 = 0
    config_fixed = False
    # Look for cfg['app_url'] used in updated test or helpers.load_config adapted
    if tf_entry:
        updated_tf = tf_entry.get("updated_content", "")
        if presence_check("cfg['app_url']", updated_tf) or presence_check('cfg.get("app_url"', updated_tf) or presence_check("cfg.get('app_url'", updated_tf):
            config_fixed = True
    if helpers_entry:
        updated_helpers = helpers_entry.get("updated_content", "")
        # check if helpers.load_config normalizes keys to include base_url
        if presence_check("load_config", updated_helpers) and (presence_check("base_url", updated_helpers) or presence_check("'base_url'", updated_helpers) or presence_check('"base_url"', updated_helpers)):
            # e.g., mapping base_url to app_url
            config_fixed = True

    if config_fixed:
        earned_t4 = t4_total
    else:
        # partial credit if candidate adjusted tests to use config file path directly (e.g., hardcoded app/index.html)
        if tf_entry and (presence_check("app/index.html", tf_entry.get("updated_content", "")) or presence_check("app/index.html", tf_entry.get("original_content", ""))):
            earned_t4 = int(round(t4_total * 0.5))
            results["deductions"].append("Task4: Candidate may have hardcoded the app path instead of using config key; partial credit.")
        else:
            earned_t4 = 0
            results["deductions"].append("Task4: No evidence config key was adjusted to use 'app_url' or otherwise handled.")

    results["scores"]["task4_config"] = {"earned": earned_t4, "possible": t4_total}
    results["total_points_earned"] += earned_t4

    # --- Documentation & submission quality (10) ---
    docs_total = POINTS["documentation"]
    required_doc_fields = [
        "candidate_name", "start_time_utc", "end_time_utc", "total_minutes",
        "commands_run", "initial_test_output", "final_test_output",
        "changed_files", "summary_of_changes", "risk_notes_and_next_steps"
    ]
    present_count = 0
    missing_doc_fields = []
    for f in required_doc_fields:
        if f in submission and submission.get(f) not in [None, "", [], {}]:
            present_count += 1
        else:
            missing_doc_fields.append(f)
    docs_score = int(round(docs_total * (present_count / len(required_doc_fields))))
    results["scores"]["documentation"] = {"earned": docs_score, "possible": docs_total}
    results["total_points_earned"] += docs_score
    if missing_doc_fields:
        results["deductions"].append("Documentation: missing or empty fields: " + ", ".join(missing_doc_fields))
    else:
        results["notes"].append("Documentation: all required high-level fields present.")

    # Also validate changed_files entries have required subfields
    cf_problem = False
    if isinstance(changed_files, list) and changed_files:
        for cf in changed_files:
            for key in ["file_path", "original_content", "updated_content", "short_rationale"]:
                if key not in cf or cf.get(key) in [None, ""]:
                    cf_problem = True
                    results["deductions"].append("Documentation: changed_files entry missing field '{}' in file '{}'.".format(key, cf.get("file_path", "<unknown>")))
    else:
        cf_problem = True
        results["deductions"].append("Documentation: changed_files array missing or empty.")
    # Slight penalty already applied via required_doc_fields above

    # --- Bonus (10) ---
    bonus_total = POINTS["bonus"]
    bonus_earned = 0
    bonus_field = submission.get("optional_bonus", "")
    task5 = tasks_section.get("task_5_optional_improvement", {}) if isinstance(tasks_section, dict) else {}
    did_bonus = False
    if isinstance(task5, dict):
        did_bonus = str(task5.get("did_bonus_work", "")).strip().lower() in ["yes", "y", "true"]
    if (isinstance(bonus_field, str) and bonus_field.strip()) or did_bonus:
        # Simple heuristic: award full bonus if optional_bonus described or task_5 indicated
        bonus_earned = bonus_total
        results["notes"].append("Bonus: candidate reported optional improvement; bonus awarded.")
    else:
        bonus_earned = 0

    results["scores"]["bonus"] = {"earned": bonus_earned, "possible": bonus_total}
    results["total_points_earned"] += bonus_earned

    # Finalize totals and percentage
    total_earned = results["total_points_earned"]
    results["overall_score"] = round((total_earned / MAX_POINTS) * 100.0, 2) if MAX_POINTS > 0 else 0.0
    results["total_points_earned"] = total_earned

    # Add a human-readable summary section
    results["summary"] = {
        "max_points": MAX_POINTS,
        "points_earned": total_earned,
        "percentage": results["overall_score"],
        "pass_threshold_percent": 80.0,
        "passed": results["overall_score"] >= 80.0
    }

    return results


def main():
    # Prepare default output structure in case of early failure
    default_result = {
        "scores": {},
        "deductions": ["Grader failed before completing; check error field."],
        "notes": [],
        "total_points_earned": 0,
        "total_points_possible": MAX_POINTS,
        "overall_score": 0.0
    }

    # Determine script directory for writing results
    script_dir = os.path.dirname(os.path.abspath(__file__))
    output_path = os.path.join(script_dir, "test_results.json")

    # Validate CLI args
    if len(sys.argv) < 3:
        default_result["deductions"].append("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        with open(output_path, "w", encoding="utf-8") as f:
            json.dump(default_result, f, indent=2)
        print("Insufficient arguments. Wrote default result to", output_path)
        return

    submission_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    try:
        submission = load_json_file(submission_path)
    except Exception as e:
        default_result["deductions"].append("Failed to load submission JSON '{}': {}".format(submission_path, str(e)))
        with open(output_path, "w", encoding="utf-8") as f:
            json.dump(default_result, f, indent=2)
        print("Error loading submission. Wrote default result to", output_path)
        return

    try:
        answer_key = load_json_file(answer_key_path)
    except Exception as e:
        # Non-fatal: proceed with empty answer_key but note it
        answer_key = {}
        # We'll include a note but continue grading with fallback heuristics
        print("Warning: failed to load answer key '{}': {}. Proceeding with fallback expectations.".format(answer_key_path, str(e)))

    try:
        results = evaluate_submission(submission, answer_key)
    except Exception as e:
        tb = traceback.format_exc()
        default_result["deductions"].append("Exception during grading: {}".format(str(e)))
        default_result["deductions"].append(tb)
        with open(output_path, "w", encoding="utf-8") as f:
            json.dump(default_result, f, indent=2)
        print("Exception during grading. Wrote error result to", output_path)
        return

    # Save results to test_results.json
    try:
        with open(output_path, "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
        print("Grading complete. Results written to:", output_path)
    except Exception as e:
        print("Failed to write results file:", str(e))
        print("Results (brief):", results)


if __name__ == "__main__":
    main()
