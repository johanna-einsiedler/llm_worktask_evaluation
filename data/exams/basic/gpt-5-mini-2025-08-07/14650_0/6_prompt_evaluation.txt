# small application module for the exam

import time

def greet(name: str) -> str:
    """
    Return a greeting for the given name.
    Current application behavior includes an exclamation mark.
    """
    return f"Hello, {name}!"

def wait_and_report(duration: float) -> float:
    """
    Sleep for `duration` seconds and return the measured elapsed time
    rounded to two decimal places.
    This intentionally uses time.time() and time.sleep() so tests that
    depend on exact timing must be made deterministic via test code.
    """
    start = time.time()
    time.sleep(duration)
    end = time.time()
    return round(end - start, 2)

def has_digit(s: str) -> bool:
    """
    Return True if the string contains at least one ASCII digit, otherwise False.
    """
    return any(ch.isdigit() for ch in s)

Purpose and design choices
