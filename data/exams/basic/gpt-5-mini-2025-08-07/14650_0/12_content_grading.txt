# task_evaluation.py
import json
import os
import re
import sys

"""
Automated grader for the "Updating Automated Tests" basic practical exam.

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

Output:
    Writes detailed grading results to test_results.json in the script directory.

Notes:
- Uses only the Python standard library.
- Attempts to be tolerant to whitespace/capitalization differences.
- Implements rubric and partial credit logic described in the exam materials.
"""

# ---------------------------
# Helper parsing & utilities
# ---------------------------

def load_json(path):
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except FileNotFoundError:
        raise
    except Exception as e:
        raise RuntimeError(f"Failed to load JSON from {path}: {e}")

def safe_get(dct, key, default=None):
    return dct.get(key, default) if isinstance(dct, dict) else default

def find_number_failed(output):
    """Return integer number of failed tests if found in pytest output; return None if not found."""
    if not isinstance(output, str):
        return None
    # Look for patterns like "X failed", "X failed, Y passed", or "Y passed, X failed"
    m = re.search(r'(\d+)\s+failed', output, re.IGNORECASE)
    if m:
        try:
            return int(m.group(1))
        except:
            return None
    # Some outputs may say "FAILED (failures=X)" - handle less common formats
    m = re.search(r'failures?:\s*(\d+)', output, re.IGNORECASE)
    if m:
        try:
            return int(m.group(1))
        except:
            return None
    return None

def find_number_passed(output):
    """Return integer number of passed tests if found; else None."""
    if not isinstance(output, str):
        return None
    m = re.search(r'(\d+)\s+passed', output, re.IGNORECASE)
    if m:
        try:
            return int(m.group(1))
        except:
            return None
    return None

def contains_pytest_command(commands):
    """Return True if commands list contains at least one pytest invocation."""
    if not isinstance(commands, list):
        return False
    for c in commands:
        if isinstance(c, str) and re.search(r'\b(pytest|python\s+-m\s+pytest)\b', c):
            return True
    return False

def normalize_text(s):
    return s.strip().lower() if isinstance(s, str) else ''

# ---------------------------
# Grading logic
# ---------------------------

def grade_submission(submission, answer_key):
    """
    Returns results dict with detailed scoring and explanations.
    Uses rubric:
    - Baseline execution: 15 pts
    - All tests passing after changes: 25 pts
    - Correct diagnosis (decision_log): 15 pts
    - Quality of test edits: 20 pts
    - Added test correctness & style: 10 pts
    - Documentation & reproducibility: 15 pts
    Total: 100
    """
    # Initialize
    max_points = {
        'baseline_execution': 15,
        'all_tests_passing': 25,
        'diagnosis': 15,
        'test_edits_quality': 20,
        'added_test': 10,
        'documentation': 15
    }
    scores = {k: 0 for k in max_points}
    deductions = {k: [] for k in max_points}
    notes = []

    # Basic validation of required fields
    required_fields = ['candidate_name', 'time_spent_minutes', 'commands_run',
                       'test_output_before', 'test_output_after',
                       'modified_files', 'added_files', 'decision_log',
                       'reproduce_instructions']
    missing_fields = [f for f in required_fields if f not in submission]
    if missing_fields:
        # Fatal-ish: still proceed but record zero for many sections
        notes.append(f"Submission is missing required fields: {missing_fields}")

    # Helpful local copies
    commands_run = submission.get('commands_run', [])
    test_output_before = submission.get('test_output_before', '') or ''
    test_output_after = submission.get('test_output_after', '') or ''
    modified_files = submission.get('modified_files', []) or []
    added_files = submission.get('added_files', []) or []
    decision_log = submission.get('decision_log', []) or []
    reproduce_instructions = submission.get('reproduce_instructions', '') or ''

    # Use answer_key outputs if available to determine expected initial failure counts
    ak_before = answer_key.get('test_output_before') if isinstance(answer_key, dict) else None
    ak_after = answer_key.get('test_output_after') if isinstance(answer_key, dict) else None

    # Parse failure counts
    # Prefer answer_key baseline if present; else use candidate before output
    initial_failed = None
    if isinstance(ak_before, str) and ak_before.strip():
        initial_failed = find_number_failed(ak_before)
    if initial_failed is None:
        initial_failed = find_number_failed(test_output_before)
    # If still None, set to 0 (unknown)
    if initial_failed is None:
        initial_failed = 0

    final_failed = find_number_failed(test_output_after)
    if final_failed is None:
        # if no explicit 'failed' found but 'passed' found, assume 0 failed
        passed = find_number_passed(test_output_after)
        final_failed = 0 if passed is not None else None

    # ===============================
    # 1) Baseline execution (15 pts)
    # Criteria:
    # - commands_run includes pytest invocation AND
    # - test_output_before contains evidence of failing tests (e.g., "failed" present) and/or pytest failure traces.
    # Give full credit if both; partial if one present.
    # ===============================
    baseline_pts = max_points['baseline_execution']
    reasons = []
    has_pytest_cmd = contains_pytest_command(commands_run)
    if not has_pytest_cmd:
        deductions['baseline_execution'].append("No pytest invocation found in commands_run.")
    else:
        reasons.append("Found pytest command in commands_run.")
    # Check that test_output_before shows failures
    before_has_fail = False
    if isinstance(test_output_before, str) and test_output_before.strip():
        # Look for 'failed' or 'FAILURES' keywords typical of pytest
        if re.search(r'\bfailed\b', test_output_before, re.IGNORECASE) or re.search(r'FAILURES', test_output_before):
            before_has_fail = True
            reasons.append("test_output_before contains pytest failure summary/trace.")
    if not before_has_fail:
        deductions['baseline_execution'].append("test_output_before does not appear to show failing tests (no 'failed'/'FAILURES' detected).")
    # Score allocation
    if has_pytest_cmd and before_has_fail:
        scores['baseline_execution'] = baseline_pts
    elif has_pytest_cmd or before_has_fail:
        scores['baseline_execution'] = baseline_pts * 0.6  # partial credit
    else:
        scores['baseline_execution'] = 0
    # Add notes
    notes.append("Baseline execution: " + ("; ".join(reasons) if reasons else "No evidence of proper baseline run."))

    # ===============================
    # 2) All tests passing after changes (25 pts)
    # Criteria:
    # - test_output_after shows 0 failed and some passed.
    # - If some failures remain, give proportional credit relative to initial_failed.
    # ===============================
    all_pts = max_points['all_tests_passing']
    if final_failed is None:
        # Could not parse final; be conservative
        deductions['all_tests_passing'].append("Could not parse final test output for failures/passes.")
        scores['all_tests_passing'] = 0
    else:
        if final_failed == 0:
            # full credit if tests passing
            # also ensure there is at least one passed test count
            passed_num = find_number_passed(test_output_after)
            if passed_num is None:
                # still award full but note uncertainty
                scores['all_tests_passing'] = all_pts
                notes.append("Final output shows 0 failures; passed count not detected but awarding full points.")
            else:
                scores['all_tests_passing'] = all_pts
        else:
            # Partial credit proportionally if initial_failed > 0
            if initial_failed > 0:
                # If they reduced failures, give proportional credit
                reduced = max(0, initial_failed - final_failed)
                prop = reduced / initial_failed
                scores['all_tests_passing'] = round(all_pts * prop, 2)
                deductions['all_tests_passing'].append(
                    f"Final run still has {final_failed} failures (initially {initial_failed}); partial credit proportional to reduction."
                )
            else:
                # No baseline info; give zero
                scores['all_tests_passing'] = 0
                deductions['all_tests_passing'].append("Final run shows failures but initial failure count unknown; no credit.")

    notes.append(f"All-tests-passing: initial_failed={initial_failed}, final_failed={final_failed}, awarded={scores['all_tests_passing']} pts.")

    # ===============================
    # 3) Correct diagnosis of failures (15 pts)
    # Criteria:
    # - decision_log should mention key root causes:
    #   expected keywords: 'date' (ISO change), 'status' (added), 'processed_at' (legacy removed),
    #   'sleep'/'time.sleep' (removed), 'discount' (for added test)
    # - Award points per matched concept.
    # ===============================
    diag_pts = max_points['diagnosis']
    expected_keywords = {
        'date': "date format changed / ISO date",
        'status': "status added/changed",
        'processed_at': "legacy 'processed_at' removed",
        'sleep': "test sleep/unnecessary wait removed",
        'discount': "discount/include_discount requirement"
    }
    matched = {k: False for k in expected_keywords}
    # Combine all decision_log text into single string for searching
    dl_text = ""
    if isinstance(decision_log, list):
        for entry in decision_log:
            # entry may be dict with 'decision' and 'alternatives_considered'
            if isinstance(entry, dict):
                dl_text += " " + safe_get(entry, 'decision', '') + " " + safe_get(entry, 'alternatives_considered', '')
            elif isinstance(entry, str):
                dl_text += " " + entry
    dl_lower = normalize_text(dl_text)
    # Search for keyword presence heuristically
    if 'date' in dl_lower or 'iso' in dl_lower or 'yyyy-mm-dd' in dl_lower or 'mm/dd' in dl_lower:
        matched['date'] = True
    if 'status' in dl_lower:
        matched['status'] = True
    if 'processed_at' in dl_lower or 'processed at' in dl_lower or 'processedat' in dl_lower:
        matched['processed_at'] = True
    if 'sleep' in dl_lower or 'time.sleep' in dl_lower or 'wait' in dl_lower:
        matched['sleep'] = True
    if 'discount' in dl_lower or 'include_discount' in dl_lower or 'discount' in dl_text:
        matched['discount'] = True

    # Score: evenly distribute points across expected keywords (5 items).
    per_item = diag_pts / len(expected_keywords)
    diag_score = 0.0
    for k, present in matched.items():
        if present:
            diag_score += per_item
        else:
            deductions['diagnosis'].append(f"Decision log did not mention: {expected_keywords[k]}")
    # Minimum check: ensure decision_log exists and has entries
    if not decision_log:
        deductions['diagnosis'].append("No decision_log entries provided.")
        diag_score = 0.0
    scores['diagnosis'] = round(diag_score, 2)
    notes.append(f"Diagnosis: matched items {', '.join([k for k,v in matched.items() if v])}.")

    # ===============================
    # 4) Quality of test edits (20 pts)
    # Criteria:
    # - Changes limited to tests/ files (modified_files paths start with 'tests/')
    # - No edits to SUT (app/) unless explicitly justified
    # - No remaining obvious brittle patterns: 'time.sleep' removed
    # - No legacy exact-date strings (MM/DD/YYYY) remain in modified tests
    # - No whole-dict equality assertions like 'assert order == expected' remain
    # - Presence of meaningful assertions (presence of 'assert' lines)
    # ===============================
    edits_pts = max_points['test_edits_quality']
    reasons = []
    penalty = 0
    max_penalty = 0
    # Start with full points and subtract for issues
    quality_score = edits_pts

    if not modified_files:
        deductions['test_edits_quality'].append("No modified_files provided.")
        quality_score = 0.0
    else:
        # Check each modified file path
        for mf in modified_files:
            path = mf.get('path', '')
            content = mf.get('content', '') or ''
            if not path.startswith('tests/'):
                # Penalty: editing non-test file (likely SUT)
                deductions['test_edits_quality'].append(f"Modified file outside tests/: {path}")
                # deduct up to 10 points for SUT edits
                penalty += 10
                max_penalty += 10
            # Check for time.sleep usages
            if re.search(r'\btime\.sleep\(', content):
                deductions['test_edits_quality'].append(f"Found time.sleep in modified file {path}; should be removed.")
                penalty += 5
                max_penalty += 5
            # Check for legacy date strings like 01/15/2025 or 02/20/2025 in content
            if re.search(r'\b0[1-9]/\d{2}/\d{4}\b', content) or re.search(r'\b0[1-9]/\d{1,2}/\d{4}\b', content):
                # More general approach: look for patterns like MM/DD/YYYY
                if re.search(r'\b\d{1,2}/\d{1,2}/\d{4}\b', content):
                    # Likely legacy date format remained
                    deductions['test_edits_quality'].append(f"Found legacy MM/DD/YYYY date format in {path}; tests should expect ISO YYYY-MM-DD.")
                    penalty += 5
                    max_penalty += 5
            # Check for whole-dict equality assertion (very brittle)
            if re.search(r'assert\s+.*==\s*{', content):
                deductions['test_edits_quality'].append(f"Found whole-dict equality style assertion in {path}; prefer focused key assertions.")
                penalty += 5
                max_penalty += 5
            # Check presence of assert statements to ensure test still validates behavior
            if not re.search(r'^\s*assert\s+', content, re.MULTILINE):
                deductions['test_edits_quality'].append(f"No assert statements detected in modified file {path}.")
                penalty += 10
                max_penalty += 10
    # Cap penalty to edits_pts
    if penalty > edits_pts:
        penalty = edits_pts
    quality_score = max(0.0, edits_pts - penalty)
    scores['test_edits_quality'] = round(quality_score, 2)
    if penalty == 0:
        notes.append("Test edits quality: no obvious brittle artifacts detected; changes confined to tests/ files.")
    else:
        notes.append(f"Test edits quality: applied penalties totalling {penalty} points for issues: {deductions['test_edits_quality']}")

    # ===============================
    # 5) Added test correctness & style (10 pts)
    # Criteria:
    # - Added file(s) should include a test verifying calculate_total(order, include_discount=True) for O1001 equals 32.5.
    # - Award full points if exact check present; partial credit if present but approximate or placed in modified_files instead of added_files.
    # ===============================
    add_pts = max_points['added_test']
    added_score = 0.0
    # First check added_files list content for presence
    found_discount_test = False
    found_in_modified = False
    # Search in added_files
    for af in added_files:
        path = af.get('path', '')
        content = af.get('content', '') or ''
        # Look for load_order("O1001") and calculate_total(..., include_discount=True) and assert == 32.5
        if re.search(r'load_order\(\s*[\'"]O1001[\'"]\s*\)', content) and re.search(r'calculate_total\(.+include_discount\s*=\s*True', content) or re.search(r'calculate_total\(.+,\s*True', content):
            # Look for numeric assertion 32.5 or 32.50
            if re.search(r'==\s*32\.5+|==\s*32\.50', content) or re.search(r'==\s*32\.50', content):
                found_discount_test = True
                break
            # maybe they used 32.50 or float arithmetic - accept 32.5 as substring
            if '32.5' in content or '32.50' in content:
                found_discount_test = True
                break
    # If not in added_files, search modified_files (they might have placed new test there)
    if not found_discount_test:
        for mf in modified_files:
            content = mf.get('content', '') or ''
            if re.search(r'load_order\(\s*[\'"]O1001[\'"]\s*\)', content) and re.search(r'calculate_total\(.+include_discount\s*=\s*True', content) or re.search(r'calculate_total\(.+,\s*True', content):
                if re.search(r'==\s*32\.5+|==\s*32\.50', content) or '32.5' in content or '32.50' in content:
                    found_in_modified = True
                    break
    if found_discount_test:
        added_score = add_pts
    elif found_in_modified:
        added_score = add_pts * 0.7  # partial credit if added but not in new file
        deductions['added_test'].append("Discount test present in modified files rather than added_files; partial credit.")
    else:
        # Possibly they added a test but used slightly different numeric representation or forgot include_discount
        # Try to be generous: search for any mention of include_discount and O1001
        mentions = False
        for collection in (added_files + modified_files):
            content = collection.get('content', '') or ''
            if 'include_discount' in content and 'O1001' in content:
                mentions = True
                break
        if mentions:
            added_score = add_pts * 0.5
            deductions['added_test'].append("Found mention of discount and O1001 but couldn't detect a clear numeric assertion for 32.5.")
        else:
            added_score = 0.0
            deductions['added_test'].append("No added test verifying include_discount=True for O1001 with expected total 32.5 found.")
    scores['added_test'] = round(added_score, 2)
    notes.append(f"Added test check: found_discount_test={found_discount_test}, found_in_modified={found_in_modified} -> awarded {scores['added_test']} pts.")

    # ===============================
    # 6) Documentation & reproducibility (15 pts)
    # Criteria:
    # - commands_run includes pytest invocations (ideally both before and after)
    # - test_output_before and test_output_after provided and non-empty
    # - modified_files and added_files content included (non-empty)
    # - reproduce_instructions present and mentions pytest commands
    # Award points across these items.
    # ===============================
    doc_pts = max_points['documentation']
    doc_score = 0.0
    # 3 subitems: commands_run clarity (5), before/after outputs included (5), reproduce instructions & file contents (5)
    # commands_run clarity
    sub1 = 0
    if contains_pytest_command(commands_run):
        # Prefer at least two pytest runs (before and after). Check count
        pytest_count = sum(1 for c in commands_run if isinstance(c, str) and re.search(r'\b(pytest|python\s+-m\s+pytest)\b', c))
        if pytest_count >= 2:
            sub1 = 5.0
        else:
            sub1 = 3.0
            deductions['documentation'].append("Only one pytest command found in commands_run; recommended to show both before and after runs.")
    else:
        deductions['documentation'].append("No pytest command found in commands_run.")
    # before/after outputs included
    sub2 = 0
    if isinstance(test_output_before, str) and test_output_before.strip():
        if isinstance(test_output_after, str) and test_output_after.strip():
            sub2 = 5.0
        else:
            sub2 = 2.5
            deductions['documentation'].append("test_output_after is missing or empty.")
    else:
        deductions['documentation'].append("test_output_before is missing or empty.")
    # reproduce instructions & files
    sub3 = 0
    ri_lower = normalize_text(reproduce_instructions)
    if ri_lower and ('pytest' in ri_lower or 'python -m pytest' in ri_lower):
        sub3 = 3.0
    else:
        # If reproduce_instructions missing pytest but modified_files and added_files content present, give partial
        if reproduce_instructions.strip():
            sub3 = 1.5
            deductions['documentation'].append("reproduce_instructions provided but does not mention pytest explicitly.")
        else:
            deductions['documentation'].append("No reproduce_instructions provided.")
    # Check content presence
    # Ensure modified_files and added_files content fields are non-empty
    files_ok = True
    for mf in modified_files:
        if not mf.get('content'):
            files_ok = False
            deductions['documentation'].append(f"Modified file {mf.get('path','?')} content missing in submission.")
    for af in added_files:
        if not af.get('content'):
            files_ok = False
            deductions['documentation'].append(f"Added file {af.get('path','?')} content missing in submission.")
    if not files_ok:
        sub3 = max(0.0, sub3 - 1.0)
    # Combine
    doc_score = sub1 + sub2 + sub3
    # Cap to doc_pts
    doc_score = min(doc_score, doc_pts)
    scores['documentation'] = round(doc_score, 2)
    notes.append(f"Documentation: commands_pytest_count={sum(1 for c in commands_run if isinstance(c, str) and re.search(r'\\b(pytest|python\\s+-m\\s+pytest)\\b', c))}, outputs_present={(bool(test_output_before.strip()), bool(test_output_after.strip()))}.")

    # -----------------------------
    # Aggregate results
    # -----------------------------
    total_earned = sum(scores.values())
    total_possible = sum(max_points.values())
    percentage = round((total_earned / total_possible) * 100, 2) if total_possible > 0 else 0.0

    # Build detailed report structure
    results = {
        "candidate_name": submission.get('candidate_name', '<unknown>'),
        "time_spent_minutes": submission.get('time_spent_minutes'),
        "scores": {
            "baseline_execution": {
                "earned": round(scores['baseline_execution'], 2),
                "max": max_points['baseline_execution'],
                "deductions": deductions['baseline_execution']
            },
            "all_tests_passing": {
                "earned": round(scores['all_tests_passing'], 2),
                "max": max_points['all_tests_passing'],
                "deductions": deductions['all_tests_passing']
            },
            "diagnosis": {
                "earned": round(scores['diagnosis'], 2),
                "max": max_points['diagnosis'],
                "deductions": deductions['diagnosis']
            },
            "test_edits_quality": {
                "earned": round(scores['test_edits_quality'], 2),
                "max": max_points['test_edits_quality'],
                "deductions": deductions['test_edits_quality']
            },
            "added_test": {
                "earned": round(scores['added_test'], 2),
                "max": max_points['added_test'],
                "deductions": deductions['added_test']
            },
            "documentation": {
                "earned": round(scores['documentation'], 2),
                "max": max_points['documentation'],
                "deductions": deductions['documentation']
            }
        },
        "total_earned": round(total_earned, 2),
        "total_possible": total_possible,
        "percentage": percentage,
        "overall_score": percentage,  # backward-compatible variable requested
        "notes": notes,
        "summary": {
            "initial_failed_parsed": initial_failed,
            "final_failed_parsed": final_failed,
        }
    }

    return results

# ---------------------------
# Main execution
# ---------------------------

def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)
    sub_path = sys.argv[1]
    ak_path = sys.argv[2]
    # Load JSON inputs
    try:
        submission = load_json(sub_path)
    except FileNotFoundError:
        print(f"Submission file not found: {sub_path}")
        sys.exit(1)
    except Exception as e:
        print(f"Error loading submission JSON: {e}")
        sys.exit(1)
    try:
        answer_key = load_json(ak_path)
    except FileNotFoundError:
        print(f"Answer key file not found: {ak_path}")
        sys.exit(1)
    except Exception as e:
        print(f"Error loading answer key JSON: {e}")
        sys.exit(1)

    try:
        results = grade_submission(submission, answer_key)
    except Exception as e:
        # Ensure deterministic failure message
        print(f"Grading failed due to an unexpected error: {e}")
        sys.exit(1)

    # Write results to test_results.json in same directory as script
    out_path = os.path.join(os.path.dirname(os.path.abspath(__file__)), 'test_results.json')
    try:
        with open(out_path, 'w', encoding='utf-8') as f:
            json.dump(results, f, indent=2, ensure_ascii=False)
    except Exception as e:
        print(f"Failed to write results to {out_path}: {e}")
        sys.exit(1)

    # Also print a brief summary to stdout
    print(f"Grading complete. Candidate: {results.get('candidate_name')}")
    print(f"Score: {results.get('total_earned')}/{results.get('total_possible')} ({results.get('percentage')}%)")
    print(f"Detailed results written to: {out_path}")

if __name__ == '__main__':
    main()
