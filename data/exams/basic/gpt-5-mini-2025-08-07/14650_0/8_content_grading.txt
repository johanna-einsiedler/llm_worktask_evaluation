# task_evaluation.py
import json
import os
import re
import sys

# Grading script for the "Updating Automated Tests" practical exam.
# Usage: python task_evaluation.py <submission_json> <answer_key_json>
#
# The script loads the candidate submission and the answer key, grades the submission
# according to the rubric in the exam materials, and writes a detailed results JSON
# (test_results.json) in the script's directory.
#
# Implementation notes:
# - Uses only Python standard library.
# - Performs tolerant string matching (case-insensitive where appropriate).
# - Produces per-category scores, explanations, and an overall percentage (overall_score).


def load_json(path):
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception as e:
        raise RuntimeError(f"Failed to load JSON from '{path}': {e}")


def lower(s):
    return s.lower() if isinstance(s, str) else ""


def contains_any(haystack, needles):
    h = lower(haystack)
    for n in needles:
        if n.lower() in h:
            return True
    return False


def find_in_files_list(files_list, path_to_find):
    if not isinstance(files_list, list):
        return None
    for f in files_list:
        if not isinstance(f, dict):
            continue
        if f.get("path") == path_to_find:
            return f
    return None


def extract_failed_count(output):
    if not isinstance(output, str):
        return None
    # Look for patterns like: "2 failed" or "FAILED (failures=2)" or "FAILED" lines
    m = re.search(r"(\d+)\s+failed", output, flags=re.IGNORECASE)
    if m:
        try:
            return int(m.group(1))
        except:
            pass
    # If explicit "failed" not followed by number, detect the word "FAILED" and treat as ambiguous failure
    if re.search(r"\bFAILED\b", output, flags=re.IGNORECASE):
        return -1
    return 0


def contains_zero_failed(output):
    if not isinstance(output, str):
        return False
    # Look for "0 failed" explicit
    if re.search(r"\b0\s+failed\b", output, flags=re.IGNORECASE):
        return True
    # Or presence of "passed" and absence of "failed" -> treat as 0 failed
    if re.search(r"\bpassed\b", output, flags=re.IGNORECASE) and not re.search(r"\bfailed\b", output, flags=re.IGNORECASE):
        return True
    return False


def score_baseline(test_output_before):
    # Baseline run should show failing tests.
    max_points = 15
    notes = []
    pts = 0
    if not test_output_before:
        notes.append("No test_output_before provided.")
        return pts, max_points, notes
    low = lower(test_output_before)
    # Check for 'failed' occurrences
    failed_count = extract_failed_count(test_output_before)
    if failed_count is None:
        notes.append("Could not parse test_output_before.")
        return pts, max_points, notes
    if failed_count == 0:
        notes.append("test_output_before does not show failing tests (no 'failed' detected).")
        pts = 0
    elif failed_count == -1:
        # ambiguous 'FAILED' without number -> treat as failing evidence
        pts = max_points
        notes.append("test_output_before contains failing test evidence ('FAILED').")
    else:
        pts = max_points
        notes.append(f"test_output_before shows {failed_count} failed test(s).")
    return pts, max_points, notes


def score_all_passed(test_output_after):
    max_points = 25
    notes = []
    pts = 0
    if not test_output_after:
        notes.append("No test_output_after provided.")
        return pts, max_points, notes
    if contains_zero_failed(test_output_after):
        pts = max_points
        notes.append("test_output_after shows tests passing with no failures.")
    else:
        # Attempt to extract failed count
        fc = extract_failed_count(test_output_after)
        if fc is None:
            notes.append("Could not parse test_output_after.")
            pts = 0
        elif fc == 0:
            pts = max_points
            notes.append("test_output_after indicates 0 failed tests.")
        elif fc == -1:
            notes.append("test_output_after contains 'FAILED' (some failures present).")
            pts = 0
        else:
            notes.append(f"test_output_after shows {fc} failed test(s).")
            pts = 0
    return pts, max_points, notes


def score_diagnosis(decision_log):
    # Evaluate decision_log: expect it to mention key root causes:
    # - date format change (ISO / YYYY-MM-DD)
    # - status added / status default
    # - legacy 'processed_at' removed (legacy/processed)
    # - removal of sleep/time.sleep
    # - addition of discount test / include_discount behavior
    max_points = 15
    notes = []
    if not isinstance(decision_log, list) or len(decision_log) == 0:
        return 0, max_points, ["decision_log missing or empty."]
    combined_text = ""
    for entry in decision_log:
        # combine decision and step and alternatives for search
        decision = entry.get("decision", "")
        alternatives = entry.get("alternatives_considered", "")
        step = entry.get("step", "")
        combined_text += " " + " ".join([str(step), str(decision), str(alternatives)])
    found = {
        "date": bool(re.search(r"\bdate\b|\bYYYY-MM-DD\b|\biso\b|\b\d{4}-\d{2}-\d{2}\b", combined_text, flags=re.IGNORECASE)),
        "status": bool(re.search(r"\bstatus\b|\bpending\b|\bprocessing\b", combined_text, flags=re.IGNORECASE)),
        "processed": bool(re.search(r"processed_at|\bprocessed\b|\blegacy\b", combined_text, flags=re.IGNORECASE)),
        "sleep": bool(re.search(r"time\.sleep|\bsleep\b|\bremove sleep\b", combined_text, flags=re.IGNORECASE)),
        "discount": bool(re.search(r"discount|include_discount|include discount", combined_text, flags=re.IGNORECASE)),
    }
    matched = sum(1 for v in found.values() if v)
    notes.append(f"Detected diagnosis mentions: {', '.join([k for k, v in found.items() if v]) or 'none'}")
    pts = int(round(max_points * (matched / len(found))))
    # Partial message
    if matched == len(found):
        notes.append("Decision log covers all expected root causes.")
    else:
        notes.append(f"Decision log covered {matched} of {len(found)} expected items; partial credit awarded.")
    return pts, max_points, notes


def score_quality_modifications(modified_files, added_files):
    max_points = 20
    pts = 0
    notes = []
    if not isinstance(modified_files, list) or len(modified_files) == 0:
        notes.append("No modified_files submitted.")
        return 0, max_points, notes

    # Criteria:
    # - At least one modified file under tests/ -> 5 pts
    # - Modified content includes assert usages and changed assertions (presence of 'assert') -> 5 pts
    # - No modifications to app/ (if present -> negative impact). If no app modifications -> +5 pts
    # - Removed sleep (no 'time.sleep' in modified files or added files) -> +5 pts

    has_tests_mod = any(isinstance(f, dict) and f.get("path", "").startswith("tests/") for f in modified_files)
    if has_tests_mod:
        pts += 5
        notes.append("Found modifications in tests/ (5 pts).")
    else:
        notes.append("No modifications under tests/ detected (0/5).")

    # Check assert presence
    assert_count = 0
    for f in modified_files:
        content = f.get("content", "")
        if isinstance(content, str) and re.search(r"\bassert\b", content):
            assert_count += 1
    if assert_count >= 1:
        pts += 5
        notes.append("Modified test files contain assertions (5 pts).")
    else:
        notes.append("Modified files appear to lack assertions (0/5).")

    # Check app/ modifications
    app_mods = [f for f in modified_files if isinstance(f, dict) and f.get("path", "").startswith("app/")]
    if len(app_mods) == 0:
        pts += 5
        notes.append("No edits to app/ detected (5 pts).")
    else:
        notes.append(f"Edits to app/ detected in paths: {[m.get('path') for m in app_mods]}. Modifying SUT reduces score (0/5).")

    # Check removal of sleep
    sleep_present = False
    for f in modified_files + (added_files or []):
        if not isinstance(f, dict):
            continue
        content = f.get("content", "")
        if isinstance(content, str) and re.search(r"time\.sleep|sleep\(", content):
            sleep_present = True
            break
    if not sleep_present:
        pts += 5
        notes.append("No time.sleep occurrences found in submitted test files (5 pts).")
    else:
        notes.append("time.sleep usage still present in submitted tests (0/5).")

    return pts, max_points, notes


def score_added_test(modified_files, added_files, answer_key):
    max_points = 10
    pts = 0
    notes = []
    # We expect a new test verifying calculate_total(..., include_discount=True) for O1001 -> result 32.5
    # Search for evidence in added_files first, then in modified_files.
    target_found = False
    partial_found = False

    search_candidates = []
    if isinstance(added_files, list):
        search_candidates.extend(added_files)
    if isinstance(modified_files, list):
        search_candidates.extend(modified_files)

    for f in search_candidates:
        if not isinstance(f, dict):
            continue
        content = f.get("content", "") or ""
        # Look for calculate_total( ... include_discount=True...) and order id O1001 and expected numeric 32.5
        has_include_flag = re.search(r"calculate_total\([^)]*include_discount\s*=\s*True", content, flags=re.IGNORECASE)
        has_o1001 = re.search(r"load_order\([^)]*['\"]O1001['\"]", content, flags=re.IGNORECASE)
        has_expected_value = re.search(r"32\.5{1,2}|32\.50", content)
        if has_include_flag and has_o1001 and has_expected_value:
            target_found = True
            notes.append(f"Found a test with include_discount=True, O1001, and expected 32.5 in file {f.get('path')}.")
            break
        # Partial: has include_discount and O1001 but different/absent expected value
        if has_include_flag and has_o1001:
            partial_found = True
            notes.append(f"Found a test that uses include_discount=True and O1001 in file {f.get('path')}, but expected value may be missing or different.")
    # Additionally, check answer_key for expected added_files and expected substrings to compare
    # Use answer_key to find expected keywords if available
    if not target_found and isinstance(answer_key, dict):
        # Try to find expected substrings from answer_key added_files content if present
        try:
            ak_added = answer_key.get("added_files", []) or []
            required_terms = set()
            for af in ak_added:
                content = af.get("content", "")
                # pull common tokens
                if "include_discount" in content:
                    required_terms.add("include_discount")
                if "O1001" in content:
                    required_terms.add("O1001")
                if "32.5" in content or "32.50" in content:
                    required_terms.add("32.5")
            # Check if candidate has these terms somewhere
            joined = " ".join([f.get("content", "") for f in search_candidates if isinstance(f, dict)])
            matched_terms = [t for t in required_terms if t in joined]
            if required_terms and len(matched_terms) == len(required_terms):
                target_found = True
                notes.append("Candidate submission contains expected terms from answer key added test content.")
        except Exception:
            pass

    if target_found:
        pts = max_points
    elif partial_found:
        pts = int(max_points * 0.5)
        notes.append("Partial evidence of added discount test found; partial credit awarded.")
    else:
        pts = 0
        notes.append("No added test for include_discount behavior found; required by README.")
    return pts, max_points, notes


def score_documentation(commands_run, reproduce_instructions, modified_files, added_files):
    max_points = 15
    pts = 0
    notes = []
    # Requirements:
    # - commands_run present and includes pytest runs (preferably before and after)
    # - reproduce_instructions present and mentions pytest
    # - modified_files and added_files contents included (non-empty)
    if not isinstance(commands_run, list) or len(commands_run) == 0:
        notes.append("commands_run missing or empty.")
    else:
        # Count how many pytest occurrences in commands_run
        pytest_hits = sum(1 for c in commands_run if isinstance(c, str) and "pytest" in c.lower())
        if pytest_hits >= 2:
            pts += 7
            notes.append(f"commands_run contains pytest calls ({pytest_hits}) including expected before/after runs (7 pts).")
        elif pytest_hits == 1:
            pts += 4
            notes.append("commands_run contains pytest but only once (4/7 pts).")
        else:
            notes.append("commands_run does not contain pytest commands (0/7 pts).")

    # Reproduce instructions
    if isinstance(reproduce_instructions, str) and reproduce_instructions.strip():
        if "pytest" in reproduce_instructions.lower():
            pts += 5
            notes.append("reproduce_instructions provided and mentions pytest (5 pts).")
        else:
            pts += 2
            notes.append("reproduce_instructions provided but does not mention pytest explicitly (2/5 pts).")
    else:
        notes.append("reproduce_instructions missing or empty (0/5 pts).")

    # Content inclusion
    if isinstance(modified_files, list) and len(modified_files) > 0:
        if all(isinstance(f, dict) and isinstance(f.get("content", None), str) for f in modified_files):
            pts += 3
            notes.append("modified_files contents included in submission (3 pts).")
        else:
            notes.append("modified_files present but some entries lack content (0/3 pts).")
    else:
        notes.append("modified_files empty or missing (0/3 pts).")

    # Note: added_files may validly be empty; we don't require added_files non-empty here,
    # but modified_files must be present and include full contents per exam instructions.

    return pts, max_points, notes


def apply_time_penalty(time_spent_minutes):
    # If time_spent_minutes > 90, apply a penalty of 10 points (as per grader guidance)
    # Return penalty points (positive number to subtract) and note.
    if time_spent_minutes is None:
        return 0, "time_spent_minutes not provided; no time penalty applied."
    try:
        t = int(time_spent_minutes)
    except Exception:
        return 0, "Invalid time_spent_minutes value; no time penalty applied."
    if t > 90:
        return 10, f"time_spent_minutes is {t} (>90); applying 10 point penalty."
    return 0, f"time_spent_minutes is {t}; no penalty."


def main():
    if len(sys.argv) < 3:
        print("Usage: python task_evaluation.py <submission_json> <answer_key_json>")
        sys.exit(2)

    submission_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    try:
        submission = load_json(submission_path)
    except Exception as e:
        print(f"Error loading submission JSON: {e}")
        sys.exit(2)
    try:
        answer_key = load_json(answer_key_path)
    except Exception as e:
        print(f"Error loading answer key JSON: {e}")
        sys.exit(2)

    # Prepare fields from submission with safe defaults
    candidate_name = submission.get("candidate_name", "<unknown>")
    time_spent_minutes = submission.get("time_spent_minutes", None)
    commands_run = submission.get("commands_run", [])
    test_output_before = submission.get("test_output_before", "")
    test_output_after = submission.get("test_output_after", "")
    modified_files = submission.get("modified_files", [])
    added_files = submission.get("added_files", [])
    decision_log = submission.get("decision_log", [])
    reproduce_instructions = submission.get("reproduce_instructions", "")

    results = {
        "candidate_name": candidate_name,
        "checks": {},
        "notes": [],
    }

    # Score each rubric category
    cat_scores = {}

    # 1. Baseline execution
    pts, mx, notes = score_baseline(test_output_before)
    cat_scores["baseline_execution"] = {"score": pts, "max": mx, "notes": notes}

    # 2. All tests passing after changes
    pts2, mx2, notes2 = score_all_passed(test_output_after)
    cat_scores["all_tests_passing"] = {"score": pts2, "max": mx2, "notes": notes2}

    # 3. Diagnosis quality
    pts3, mx3, notes3 = score_diagnosis(decision_log)
    cat_scores["diagnosis"] = {"score": pts3, "max": mx3, "notes": notes3}

    # 4. Quality of test edits
    pts4, mx4, notes4 = score_quality_modifications(modified_files, added_files)
    cat_scores["quality_test_edits"] = {"score": pts4, "max": mx4, "notes": notes4}

    # 5. Added test correctness & style
    pts5, mx5, notes5 = score_added_test(modified_files, added_files, answer_key)
    cat_scores["added_test"] = {"score": pts5, "max": mx5, "notes": notes5}

    # 6. Documentation & reproducibility
    pts6, mx6, notes6 = score_documentation(commands_run, reproduce_instructions, modified_files, added_files)
    cat_scores["documentation_reproducibility"] = {"score": pts6, "max": mx6, "notes": notes6}

    # Sum up
    total_score = sum(cat_scores[c]["score"] for c in cat_scores)
    total_max = sum(cat_scores[c]["max"] for c in cat_scores)

    # Time penalty
    penalty, penalty_note = apply_time_penalty(time_spent_minutes)
    if penalty > 0:
        total_score -= penalty
        if total_score < 0:
            total_score = 0
    # Compose final notes
    final_notes = []
    final_notes.append(penalty_note)
    # Add per-category notes
    for k, v in cat_scores.items():
        final_notes.append(f"{k}: " + "; ".join(v.get("notes", [])))
    # Check for required structural items presence warnings
    missing = []
    required_keys = ["candidate_name", "time_spent_minutes", "commands_run", "test_output_before", "test_output_after", "modified_files", "decision_log", "reproduce_instructions"]
    for rk in required_keys:
        if rk not in submission:
            missing.append(rk)
    if missing:
        final_notes.append("Missing required submission fields: " + ", ".join(missing))

    # Normalize total_max to 100 if mismatch
    if total_max != 100:
        # Normally sum is 100; but ensure percentages are computed relative to 100.
        # We will scale to 100.
        scaling_note = f"Total rubric maximum ({total_max}) differs from expected 100; scaling to 100 for reporting."
        final_notes.append(scaling_note)
        if total_max > 0:
            scaled_score = (total_score / total_max) * 100.0
        else:
            scaled_score = 0.0
        overall_percentage = round(scaled_score, 2)
    else:
        overall_percentage = round(float(total_score), 2)

    # Pass/fail based on >=80
    passed = overall_percentage >= 80.0

    # Prepare output structure
    results["checks"] = cat_scores
    results["time_spent_minutes"] = time_spent_minutes
    results["total_points"] = total_score
    results["total_max_points"] = total_max
    results["percentage"] = overall_percentage
    results["overall_score"] = overall_percentage  # numeric per requirement
    results["passed"] = passed
    results["notes"] = final_notes

    # Provide quick human-readable deductions explanation
    deductions = []
    # Example deductions from categories with less than max
    for k, v in cat_scores.items():
        if v["score"] < v["max"]:
            deductions.append({
                "category": k,
                "score": v["score"],
                "max": v["max"],
                "reason_summary": v.get("notes", [])
            })
    results["deductions"] = deductions

    # Save test_results.json in same directory as this script
    try:
        script_dir = os.path.dirname(os.path.abspath(__file__))
    except NameError:
        script_dir = os.getcwd()
    out_path = os.path.join(script_dir, "test_results.json")
    try:
        with open(out_path, "w", encoding="utf-8") as outf:
            json.dump(results, outf, indent=2)
        print(f"Grading complete. Results written to: {out_path}")
    except Exception as e:
        print(f"Failed to write results JSON: {e}")
        sys.exit(2)


if __name__ == "__main__":
    main()
