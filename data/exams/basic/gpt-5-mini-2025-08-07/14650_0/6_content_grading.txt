# task_evaluation.py
import json
import os
import sys
import re

"""
Automated grader for the basic practical exam "Updating Automated Tests to Match Evolving Application Behavior".

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

The script produces a JSON file 'test_results.json' in the same directory as this script.

It grades according to the rubric described in the exam materials:
 - Baseline execution (15)
 - All tests passing after changes (25)
 - Correct diagnosis (15)
 - Quality of test edits (20)
 - Added test correctness & style (10)
 - Documentation & reproducibility (15)

Total: 100 points.

The grader is tolerant of whitespace/capitalization differences and provides detailed
reasons for deductions.

Author: Generated for exam evaluation
"""

# --- Helper functions -------------------------------------------------------

def load_json_file(path):
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except FileNotFoundError:
        raise
    except Exception as e:
        raise RuntimeError(f"Failed to load/parse JSON file '{path}': {e}")

def safe_lower(s):
    return s.lower() if isinstance(s, str) else ""

def contains_pytest_command(cmds):
    if not isinstance(cmds, list):
        return False
    for c in cmds:
        if isinstance(c, str) and re.search(r'\b(pytest|python -m pytest)\b', c, re.IGNORECASE):
            return True
    return False

def summary_has_failures(output):
    """Return (has_failures:bool, failure_count:int or None)"""
    if not isinstance(output, str) or not output.strip():
        return (False, None)
    o = output.lower()
    # Look for patterns like "X failed" or "failed in"
    m = re.search(r'(\d+)\s+failed', o)
    if m:
        return (True, int(m.group(1)))
    # some outputs may show "FAILED" lines per test; if 'failed' present but no number, assume failures
    if 'failed' in o and 'failed to' not in o:
        # attempt to parse summary line like "2 passed, 3 failed in"
        m2 = re.search(r'(\d+)\s+passed.*?(\d+)\s+failed', o)
        if m2:
            return (True, int(m2.group(2)))
        return (True, None)
    return (False, 0)

def summary_has_all_passed(output):
    """Return True if output indicates tests passed and no failures reported."""
    if not isinstance(output, str) or not output.strip():
        return False
    o = output.lower()
    # If 'failed' appears, it's not all-passing
    if 'failed' in o:
        return False
    # require at least one 'passed' occurrence
    return 'passed' in o

def find_added_discount_test(added_files):
    """Check added_files for a test asserting calculate_total(..., include_discount=True) == 32.5 for O1001."""
    if not isinstance(added_files, list):
        return (False, "added_files missing or not a list")
    for f in added_files:
        content = f.get('content', '') if isinstance(f, dict) else ''
        path = f.get('path', '') if isinstance(f, dict) else ''
        if not isinstance(content, str):
            continue
        # Look for calculate_total usage with include_discount or explicit include_discount=True
        # and an assertion for 32.5 (32.5 or 32.50)
        if re.search(r'calculate_total\([^)]*include_discount\s*=\s*True', content) or re.search(r'calculate_total\s*\([^,)]*,\s*True\)', content):
            if re.search(r'assert\s+.*calculate_total\([^)]*include_discount.*\)\s*==\s*32(?:\.0+)?|assert\s+.*==\s*32(?:\.0+)?|assert\s+.*==\s*32\.5', content):
                # try also to ensure it uses load_order("O1001")
                if re.search(r'load_order\s*\(\s*[\'"]O1001[\'"]\s*\)', content) or re.search(r'order\s*=.*load_order\(', content):
                    return (True, f"Found discount test in added file {path}")
                else:
                    return (True, f"Found discount test assertion in {path} but could not find explicit load_order(\"O1001\") call")
        # alternative: maybe they call calculate_total(order, True) and assert 32.5
        if re.search(r'calculate_total\([^)]*,\s*True\)', content) and re.search(r'assert\s+.*==\s*32(?:\.0+)?', content):
            if re.search(r'load_order\s*\(\s*[\'"]O1001[\'"]\s*\)', content):
                return (True, f"Found discount test in added file {path}")
    return (False, "No added test found asserting include_discount behavior for O1001")

def modified_files_only_tests(modified_files):
    """Return (only_tests:bool, non_test_files: list)"""
    non_tests = []
    for f in modified_files:
        path = f.get('path', '') if isinstance(f, dict) else ''
        if not path.startswith('tests/'):
            non_tests.append(path)
    return (len(non_tests) == 0, non_tests)

def any_sleep_present_in_content(contents):
    """Return True if time.sleep occurrences exist."""
    return bool(re.search(r'\btime\.sleep\s*\(', contents))

def strict_dict_equality_present(contents):
    """Detect patterns like 'assert order == expected' which indicate brittle strict comparisons."""
    # crude checks
    if re.search(r'assert\s+[^=]+\s*==\s*expected', contents):
        return True
    # check for exact equality comparisons of returned dicts
    if re.search(r'assert\s+load_order\([^)]*\)\s*==', contents):
        return True
    # other pattern: assert order == { ... }
    if re.search(r'assert\s+[a-zA-Z0-9_]+\s*==\s*\{', contents):
        return True
    return False

def check_decision_log_for_keywords(decision_log):
    """
    Inspect decision_log entries for presence of key diagnosis points.
    Return dict of booleans for keys: date, status, processed_at, sleep, discount_add_test
    """
    flags = {
        'date': False,
        'status': False,
        'processed_at': False,
        'sleep': False,
        'discount_add_test': False,
        'final_run': False
    }
    if not isinstance(decision_log, list):
        return flags
    joined = " ".join([ (item.get('decision','') + " " + item.get('alternatives_considered','')) for item in decision_log if isinstance(item, dict) ])
    j = joined.lower()
    if 'date' in j or 'iso' in j or 'yyyy-mm-dd' in j or 'mm/dd' in j:
        flags['date'] = True
    if 'status' in j:
        flags['status'] = True
    if 'processed_at' in j or 'processed' in j or 'legacy' in j:
        flags['processed_at'] = True
    if 'sleep' in j or 'time.sleep' in j or 'remove' in j and 'sleep' in j:
        if 'sleep' in j:
            flags['sleep'] = True
    if 'discount' in j or 'include_discount' in j or 'add' in j and 'discount' in j:
        flags['discount_add_test'] = True
    if 're-ran' in j or 're ran' in j or 'final-run' in j or 'final run' in j or 'ran the full test suite' in j:
        flags['final_run'] = True
    return flags

# --- Grading logic ----------------------------------------------------------

def grade(submission, answer_key):
    """
    Grades the submission dict against expectations in answer_key.
    Returns a results dict with breakdown and explanations.
    """
    # Initialize result structure
    results = {
        'scores': {},
        'deductions': [],
        'total_points': 0,
        'max_points': 100,
        'overall_score': 0.0,
    }

    # Sanity checks for required fields in submission
    required_keys = ['candidate_name', 'time_spent_minutes', 'commands_run', 'test_output_before',
                     'test_output_after', 'modified_files', 'added_files', 'decision_log', 'reproduce_instructions']
    missing = [k for k in required_keys if k not in submission]
    if missing:
        results['deductions'].append(f"Submission is missing required fields: {', '.join(missing)}")
        # Still proceed with best-effort grading using present keys

    # Extract submission fields with safe defaults
    candidate_name = submission.get('candidate_name', '')
    time_spent = submission.get('time_spent_minutes', None)
    commands_run = submission.get('commands_run', [])
    test_output_before = submission.get('test_output_before', '')
    test_output_after = submission.get('test_output_after', '')
    modified_files = submission.get('modified_files', [])
    added_files = submission.get('added_files', [])
    decision_log = submission.get('decision_log', [])
    reproduce_instructions = submission.get('reproduce_instructions', '')

    # Track earned points
    total = 0.0

    # 1) Baseline execution (15 pts)
    points_baseline = 15
    baseline_awarded = 0.0
    baseline_reasons = []
    has_pytest_cmd = contains_pytest_command(commands_run)
    before_has_failures, before_failure_count = summary_has_failures(test_output_before)
    if has_pytest_cmd and before_has_failures:
        baseline_awarded = points_baseline
        baseline_reasons.append("Found pytest command in commands_run and test_output_before shows failures.")
    else:
        # Partial credit rules:
        if has_pytest_cmd or before_has_failures:
            baseline_awarded = points_baseline * 0.6
            reason_parts = []
            if has_pytest_cmd:
                reason_parts.append("pytest command present")
            else:
                reason_parts.append("pytest command missing from commands_run")
            if before_has_failures:
                reason_parts.append("test_output_before shows failures")
            else:
                reason_parts.append("test_output_before does not clearly show failures")
            baseline_reasons.append("Partial: " + "; ".join(reason_parts))
        else:
            baseline_awarded = 0.0
            baseline_reasons.append("No pytest command found and initial output does not show failures.")
    total += baseline_awarded
    results['scores']['baseline_execution'] = {
        'awarded': baseline_awarded,
        'max': points_baseline,
        'notes': baseline_reasons
    }

    # 2) All tests passing after changes (25 pts)
    points_passing = 25
    passing_awarded = 0.0
    passing_reasons = []
    after_all_passed = summary_has_all_passed(test_output_after)
    after_has_failures, after_failure_count = summary_has_failures(test_output_after)
    if after_all_passed:
        passing_awarded = points_passing
        passing_reasons.append("test_output_after indicates tests passed with no failures.")
    else:
        # Partial credit: if some 'passed' present but also 'failed', give partial credit
        o = safe_lower(test_output_after)
        if 'passed' in o:
            passing_awarded = points_passing * 0.5
            passing_reasons.append("Some tests passed according to test_output_after, but failures still present.")
        else:
            passing_awarded = 0.0
            passing_reasons.append("test_output_after does not indicate passing tests.")
    total += passing_awarded
    results['scores']['all_tests_passing_after_changes'] = {
        'awarded': passing_awarded,
        'max': points_passing,
        'notes': passing_reasons
    }

    # 3) Correct diagnosis (15 pts)
    points_diagnosis = 15
    diagnosis_awarded = 0.0
    diagnosis_notes = []
    flags = check_decision_log_for_keywords(decision_log)
    # We'll award for presence of key diagnosis items: date, status, processed_at, sleep, discount_add_test
    key_items = ['date', 'status', 'processed_at', 'sleep', 'discount_add_test']
    points_per_item = points_diagnosis / len(key_items)  # 3 points each
    found_items = []
    missing_items = []
    for k in key_items:
        if flags.get(k, False):
            diagnosis_awarded += points_per_item
            found_items.append(k)
        else:
            missing_items.append(k)
    # require at least some entries in decision_log
    if not isinstance(decision_log, list) or len(decision_log) == 0:
        diagnosis_awarded = 0.0
        diagnosis_notes.append("decision_log missing or empty; no diagnosis information provided.")
    else:
        diagnosis_notes.append(f"Found diagnosis mentions for: {', '.join(found_items) if found_items else 'none'}; missing: {', '.join(missing_items) if missing_items else 'none'}.")
    total += diagnosis_awarded
    results['scores']['correct_diagnosis'] = {
        'awarded': round(diagnosis_awarded, 2),
        'max': points_diagnosis,
        'notes': diagnosis_notes
    }

    # 4) Quality of test edits (20 pts)
    points_quality = 20
    quality_awarded = 0.0
    quality_notes = []
    # Checks:
    # - modified files only under tests/
    # - no time.sleep present in modified tests
    # - no strict dict equality checks like assert order == expected
    # - presence of ISO date checks and status assertion or use of 'in' substring checks for summaries
    if not isinstance(modified_files, list) or len(modified_files) == 0:
        quality_notes.append("No modified_files provided.")
        quality_awarded = 0.0
    else:
        only_tests, non_tests = modified_files_only_tests(modified_files)
        if not only_tests:
            quality_notes.append(f"Some modified files are not in tests/: {non_tests}. Editing SUT files reduces score.")
            # We'll penalize: award at most 50% of this category if SUT edited
            penalty_non_test = True
        else:
            penalty_non_test = False
        # Check combined contents
        combined = "\n".join([ f.get('content','') for f in modified_files if isinstance(f, dict) ])
        # removed sleep?
        if any_sleep_present_in_content(combined):
            quality_notes.append("Found time.sleep usage in modified tests (should be removed).")
            sleep_ok = False
        else:
            quality_notes.append("No time.sleep found in modified tests.")
            sleep_ok = True
        # strict dict equality?
        if strict_dict_equality_present(combined):
            quality_notes.append("Found strict whole-dict equality assertions (e.g., assert order == expected) still present.")
            strict_ok = False
        else:
            quality_notes.append("No strict whole-dict equality assertions detected.")
            strict_ok = True
        # Check for date ISO and status assertions or use of 'in' with status
        date_ok = bool(re.search(r'2025-01-15|2025-02-20|yyyy-mm-dd|iso', combined.lower()))
        status_ok = bool(re.search(r'\[status:|status\"\s*,|status\'\s*,|\bstatus\b', combined))
        contains_checks_ok = bool(re.search(r'\bin\b.*summary|\bin summary|\" in summary|assert .* in .*summary', combined, re.IGNORECASE))
        # Award points: 5 pts each for (only_tests), (no_sleep), (no_strict_equality), (checks for date/status/contains)
        fragment_scores = 0.0
        fragment_details = []
        # only_tests
        if only_tests:
            fragment_scores += 5.0
            fragment_details.append("Modified files limited to tests/ (5 pts).")
        else:
            # partial credit: 2.5 if changed mostly tests but also SUT; but we already detected non-tests
            fragment_scores += 2.5
            fragment_details.append("Modified files included non-test paths; partial credit (2.5/5).")
        # sleep removal
        if sleep_ok:
            fragment_scores += 5.0
            fragment_details.append("No time.sleep in modified tests (5 pts).")
        else:
            # if still present, 0 for this subcriterion
            fragment_details.append("time.sleep still present (0/5).")
        # strict equality
        if strict_ok:
            fragment_scores += 5.0
            fragment_details.append("No strict whole-dict equality assertions (5 pts).")
        else:
            fragment_details.append("Found strict equality assertions (0/5).")
        # date/status/contains: award up to 5 depending on presence of date or status and/or contains checks
        date_status_points = 0.0
        if date_ok:
            date_status_points += 2.5
        if status_ok or contains_checks_ok:
            date_status_points += 2.5
        fragment_scores += date_status_points
        fragment_details.append(f"Date/status/contains checks score: {date_status_points}/5.")
        # If SUT edited, reduce this category by half (penalty)
        if penalty_non_test:
            fragment_scores = fragment_scores * 0.5
            fragment_details.append("Penalty applied: modified non-test files; quality score halved.")
        quality_awarded = fragment_scores
        quality_notes.extend(fragment_details)
    total += quality_awarded
    results['scores']['quality_of_test_edits'] = {
        'awarded': round(quality_awarded, 2),
        'max': points_quality,
        'notes': quality_notes
    }

    # 5) Added test correctness & style (10 pts)
    points_added_test = 10
    added_awarded = 0.0
    added_notes = []
    ok, message = find_added_discount_test(added_files)
    if ok:
        added_awarded = points_added_test
        added_notes.append("Added test verifies calculate_total with include_discount for O1001 and asserts expected result (32.5).")
    else:
        # partial: if any added file references calculate_total but missing proper assertion, partial credit 50%
        any_calc = False
        for f in added_files if isinstance(added_files, list) else []:
            cont = f.get('content','')
            if isinstance(cont, str) and 'calculate_total' in cont:
                any_calc = True
                break
        if any_calc:
            added_awarded = points_added_test * 0.5
            added_notes.append("Found calculate_total usage in added files but no clear include_discount=True assertion to 32.5; partial credit.")
        else:
            added_awarded = 0.0
            added_notes.append("No appropriate added test found for include_discount behavior; required by README.")
    total += added_awarded
    results['scores']['added_test_correctness_and_style'] = {
        'awarded': added_awarded,
        'max': points_added_test,
        'notes': added_notes + [message]
    }

    # 6) Documentation & reproducibility (15 pts)
    points_docs = 15
    docs_awarded = 0.0
    docs_notes = []
    # Subchecks:
    # - commands_run includes pytest before and after (we check at least one pytest command present)
    # - reproduce_instructions present and includes pytest commands
    # - modified_files.reason and added_files.reason present and non-empty strings
    # - candidate_name non-empty and time_spent_minutes <= 90
    cmd_ok = contains_pytest_command(commands_run)
    if cmd_ok:
        docs_awarded += 5.0
        docs_notes.append("commands_run includes pytest command (5 pts).")
    else:
        docs_notes.append("commands_run does not include pytest command (0/5).")
    if isinstance(reproduce_instructions, str) and 'pytest' in reproduce_instructions.lower():
        docs_awarded += 5.0
        docs_notes.append("reproduce_instructions include pytest invocation (5 pts).")
    else:
        docs_notes.append("reproduce_instructions do not include pytest invocation (0/5).")
    # Check reasons non-empty
    mod_reasons_ok = True
    for mf in modified_files if isinstance(modified_files, list) else []:
        reason = mf.get('reason','') if isinstance(mf, dict) else ''
        if not isinstance(reason, str) or not reason.strip():
            mod_reasons_ok = False
            break
    add_reasons_ok = True
    for af in added_files if isinstance(added_files, list) else []:
        reason = af.get('reason','') if isinstance(af, dict) else ''
        if not isinstance(reason, str) or not reason.strip():
            add_reasons_ok = False
            break
    if mod_reasons_ok and add_reasons_ok:
        docs_awarded += 3.0
        docs_notes.append("modified_files and added_files include brief reasons (3 pts).")
    else:
        docs_notes.append("Missing or empty reason in modified_files/added_files (0/3).")
    # candidate_name and time_spent
    name_ok = isinstance(candidate_name, str) and candidate_name.strip() != ''
    time_ok = isinstance(time_spent, int) and (0 <= time_spent <= 90)
    if name_ok and time_ok:
        docs_awarded += 2.0
        docs_notes.append("candidate_name present and time_spent_minutes recorded <= 90 (2 pts).")
    else:
        if not name_ok:
            docs_notes.append("candidate_name missing or empty.")
        if not time_ok:
            docs_notes.append("time_spent_minutes missing or not a valid integer <= 90.")
    total += docs_awarded
    results['scores']['documentation_and_reproducibility'] = {
        'awarded': round(docs_awarded, 2),
        'max': points_docs,
        'notes': docs_notes
    }

    # Finalize totals
    results['total_points'] = round(total, 2)
    results['max_points'] = 100
    percent = (total / 100.0) * 100.0
    results['overall_score'] = round(percent, 2)

    # Provide a human-friendly summary of deductions / notes
    summary_notes = []
    if baseline_awarded < points_baseline:
        summary_notes.append(f"Baseline execution awarded {baseline_awarded}/{points_baseline}. Details: {'; '.join(baseline_reasons)}")
    if passing_awarded < points_passing:
        summary_notes.append(f"Passing tests awarded {passing_awarded}/{points_passing}. Details: {'; '.join(passing_reasons)}")
    if diagnosis_awarded < points_diagnosis:
        summary_notes.append(f"Diagnosis awarded {round(diagnosis_awarded,2)}/{points_diagnosis}. Details: {'; '.join(diagnosis_notes)}")
    if quality_awarded < points_quality:
        summary_notes.append(f"Quality of edits awarded {round(quality_awarded,2)}/{points_quality}. Details: {'; '.join(quality_notes)}")
    if added_awarded < points_added_test:
        summary_notes.append(f"Added test awarded {added_awarded}/{points_added_test}. Details: {'; '.join(added_notes)}")
    if docs_awarded < points_docs:
        summary_notes.append(f"Documentation awarded {round(docs_awarded,2)}/{points_docs}. Details: {'; '.join(docs_notes)}")
    if not summary_notes:
        summary_notes.append("All categories fully satisfied.")

    results['deductions'].extend(summary_notes)

    # Attach some parsed evidence for evaluator convenience
    results['evidence'] = {
        'commands_run': commands_run,
        'test_output_before_snippet': (test_output_before[:400] + '...') if isinstance(test_output_before, str) and len(test_output_before) > 400 else test_output_before,
        'test_output_after_snippet': (test_output_after[:400] + '...') if isinstance(test_output_after, str) and len(test_output_after) > 400 else test_output_after,
        'modified_files_paths': [mf.get('path','') for mf in modified_files if isinstance(mf, dict)],
        'added_files_paths': [af.get('path','') for af in added_files if isinstance(af, dict)],
        'decision_log_summary_flags': flags
    }

    return results

# --- Main execution ---------------------------------------------------------

def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)

    submission_path = sys.argv[1]
    answer_key_path = sys.argv[2]
    script_dir = os.path.dirname(os.path.abspath(__file__))
    output_path = os.path.join(script_dir, 'test_results.json')

    # Load files with error handling
    try:
        submission = load_json_file(submission_path)
    except FileNotFoundError:
        print(f"Submission file not found: {submission_path}")
        sys.exit(2)
    except Exception as e:
        print(str(e))
        sys.exit(2)

    try:
        answer_key = load_json_file(answer_key_path)
    except FileNotFoundError:
        print(f"Answer key file not found: {answer_key_path}")
        sys.exit(2)
    except Exception as e:
        print(str(e))
        sys.exit(2)

    # Grade
    try:
        results = grade(submission, answer_key)
    except Exception as e:
        # Catch unexpected grading errors and save minimal diagnostics
        err = {
            'error': f"Grading failed with exception: {e}"
        }
        with open(output_path, 'w', encoding='utf-8') as f:
            json.dump(err, f, indent=2)
        print(f"Grading encountered an error. Details written to {output_path}")
        sys.exit(1)

    # Save results
    try:
        with open(output_path, 'w', encoding='utf-8') as f:
            json.dump(results, f, indent=2)
    except Exception as e:
        print(f"Failed to write results to {output_path}: {e}")
        sys.exit(1)

    # Print basic summary to stdout
    print(f"Grading complete. Overall score: {results['overall_score']}% ({results['total_points']}/{results['max_points']} points).")
    print(f"Detailed results written to {output_path}")

if __name__ == '__main__':
    main()
