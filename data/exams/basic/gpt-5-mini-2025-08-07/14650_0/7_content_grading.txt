#!/usr/bin/env python3
"""
task_evaluation.py

Automated grader for the "Updating Automated Tests" basic practical exam.

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

This script compares the candidate submission against an answer key and
produces a grading report saved to test_results.json in the current directory.

Implementation notes:
- Uses only Python standard library.
- Scoring follows the rubric defined in the evaluator materials.
- Outputs detailed breakdown and explanations for deductions.
"""

import json
import os
import sys
import re

# Maximum points per category (from evaluator's suggested rubric)
WEIGHTS = {
    "baseline_execution": 15,
    "all_tests_passing": 25,
    "diagnosis": 15,
    "test_edits_quality": 20,
    "added_test": 10,
    "documentation": 15
}
MAX_TOTAL = sum(WEIGHTS.values())


def safe_load_json(path):
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except FileNotFoundError:
        raise
    except Exception as e:
        raise ValueError(f"Error loading JSON from {path}: {e}")


def contains_failed_summary(output_text):
    """Heuristic: detect failing pytest run in output."""
    if not output_text:
        return False
    low = output_text.lower()
    # look for common pytest failure indicators
    if "failed" in low or "failure" in low or "failures" in low:
        # ensure it's a summary like 'X failed' or 'FAILED'
        return True
    return False


def detects_all_passing(output_text):
    """Heuristic: detect passing pytest run in output."""
    if not output_text:
        return False
    low = output_text.lower()
    # Prefer explicit 'passed' and absence of 'failed'
    if "failed" in low:
        return False
    if "passed" in low:
        # Basic sanity: ensure there's an occurrence like 'x passed' or 'passed in'
        return True
    # fallback: if output contains '0 failed' explicitly
    if "0 failed" in low:
        return True
    return False


def any_pytest_command(commands):
    """Return True if any command looks like it runs pytest."""
    if not commands:
        return False
    for c in commands:
        c_clean = c.strip()
        if not c_clean:
            continue
        # common ways: pytest, python -m pytest, py.test etc.
        if re.search(r'(^|\s)(pytest|python\s+-m\s+pytest|py\.test)(\s|$)', c_clean):
            return True
    return False


def find_discount_test_presence(modified_files, added_files):
    """Search modified and added file contents for required discount test.

    The README requires a test verifying calculate_total(..., include_discount=True)
    works correctly for order 'O1001' (expected 32.5).
    """
    needle_order = "O1001"
    needle_flag_pattern = r"include_discount\s*=\s*True|include_discount=True"
    needle_expected = re.compile(r"32\.5|32\.50")
    # Search in all provided file contents
    files = []
    if modified_files:
        files.extend(modified_files)
    if added_files:
        files.extend(added_files)
    for f in files:
        content = (f.get("content") or "").lower()
        if needle_order.lower() in content and re.search(needle_flag_pattern, content) and needle_expected.search(content):
            return True
    return False


def summarize_modified_files_scope(modified_files):
    """Check that all modified files are under tests/ and that SUT not modified."""
    reasons = []
    non_test_files = []
    for f in modified_files:
        path = f.get("path", "")
        if not path.startswith("tests/"):
            non_test_files.append(path)
    if non_test_files:
        reasons.append(f"Modified non-test files: {non_test_files}")
    return non_test_files, reasons


def count_asserts_in_content(content):
    # crude count of 'assert' as a token (ignores commented asserts)
    if not content:
        return 0
    # remove python comments to avoid counting commented asserts
    # naive: strip lines that start with #
    lines = content.splitlines()
    count = 0
    for ln in lines:
        stripped = ln.strip()
        if stripped.startswith("#"):
            continue
        # count 'assert ' tokens
        count += len(re.findall(r'\bassert\b', ln))
    return count


def any_time_sleep_present(files):
    """Return True if any provided file contents contain time.sleep usage."""
    for f in files:
        content = f.get("content", "") if f else ""
        if "time.sleep" in content:
            return True
    return False


def any_trivial_assert_only(files):
    """Detect if any test file contains only trivial assertions like 'assert True' or 'pass'."""
    trivial_files = []
    for f in files:
        content = f.get("content", "") if f else ""
        # If file has no assert (or only 'assert True') and contains 'pass' and little else, flag it
        num_asserts = count_asserts_in_content(content)
        if num_asserts == 0:
            # if file content small and contains 'pass' or no assertions, flag
            stripped = content.strip()
            if not stripped:
                trivial_files.append(f.get("path", "unknown"))
            else:
                # check if only 'pass' or module-level docstring
                lines = [l for l in stripped.splitlines() if l.strip() and not l.strip().startswith("#")]
                if len(lines) <= 2:
                    if any("pass" in l for l in lines):
                        trivial_files.append(f.get("path", "unknown"))
                    else:
                        # could be a helper file without asserts; still flag cautiously
                        trivial_files.append(f.get("path", "unknown"))
    return trivial_files


def any_partial_checks_present(files):
    """Heuristic: detect presence of partial string checks like 'in summary' or 'in' assertions for substrings."""
    patterns = [r'in summary', r' in summary', r'assert .*in .*summary', r'\bin\b', r'contains', r'get\(\s*[\'"]status[\'"]']
    for f in files:
        content = f.get("content", "") if f else ""
        lc = content.lower()
        for p in patterns:
            if re.search(p, lc):
                return True
    return False


def ensure_modified_contents_present(modified_files, added_files):
    """Check that content fields are present and non-empty for all modified/added files."""
    missing = []
    for f in (modified_files or []) + (added_files or []):
        path = f.get("path")
        content = f.get("content")
        if path is None or content is None or (isinstance(content, str) and content.strip() == ""):
            missing.append(path or "(missing path)")
    return missing


def grade(submission, answer_key):
    """Compute grading results given candidate submission and answer key (both dicts)."""
    results = {
        "scores": {},
        "explanations": [],
        "total_points": 0,
        "max_points": MAX_TOTAL
    }

    # Basic validation of submission schema presence
    required_keys = ["candidate_name", "time_spent_minutes", "commands_run", "test_output_before",
                     "test_output_after", "modified_files", "added_files", "decision_log", "reproduce_instructions"]
    missing_keys = [k for k in required_keys if k not in submission]
    if missing_keys:
        results["explanations"].append(f"Submission is missing required fields: {missing_keys}")
        # Return zero score but still create a structure
        for k in WEIGHTS:
            results["scores"][k] = 0
        results["total_points"] = 0
        results["overall_score"] = 0.0
        return results

    # Extract fields
    commands_run = submission.get("commands_run", []) or []
    test_output_before = submission.get("test_output_before", "") or ""
    test_output_after = submission.get("test_output_after", "") or ""
    modified_files = submission.get("modified_files", []) or []
    added_files = submission.get("added_files", []) or []
    decision_log = submission.get("decision_log", []) or []
    reproduce_instructions = submission.get("reproduce_instructions", "") or ""

    # 1) Baseline execution (15 pts)
    pts_baseline = 0
    baseline_msgs = []
    if test_output_before and isinstance(test_output_before, str):
        if contains_failed_summary(test_output_before):
            pts_baseline = WEIGHTS["baseline_execution"]
            baseline_msgs.append("Initial failing test output present and indicates failures.")
        else:
            pts_baseline = int(WEIGHTS["baseline_execution"] * 0.5)
            baseline_msgs.append("Initial test output present but failing summary not clearly detected; partial credit given.")
    else:
        baseline_msgs.append("No initial test output captured; 0 points for baseline execution.")
    results["scores"]["baseline_execution"] = pts_baseline
    if baseline_msgs:
        results["explanations"].append({"baseline_execution": baseline_msgs})

    # 2) All tests passing after changes (25 pts)
    pts_passing = 0
    passing_msgs = []
    if test_output_after and isinstance(test_output_after, str):
        if detects_all_passing(test_output_after):
            pts_passing = WEIGHTS["all_tests_passing"]
            passing_msgs.append("Final test output indicates passing tests (no failures detected).")
        else:
            # Partial: if there's 'passed' but 'failed' too, no credit
            pts_passing = 0
            passing_msgs.append("Final test output does not indicate all tests passing; 0 points awarded.")
    else:
        passing_msgs.append("No final test output provided; 0 points.")
    results["scores"]["all_tests_passing"] = pts_passing
    results["explanations"].append({"all_tests_passing": passing_msgs})

    # 3) Correct diagnosis of failures (15 pts)
    # Look for keywords in decision_log entries: date, status, processed_at, sleep, discount
    pts_diagnosis = 0
    diag_msgs = []
    # weights for sub-issues (sum must be 15)
    diag_weights = {
        "date": 4,
        "status": 4,
        "processed_at": 3,
        "sleep": 2,
        "add_discount_test": 2
    }
    found = {k: False for k in diag_weights}
    # Aggregate text from decision_log fields
    aggregated = ""
    if isinstance(decision_log, list):
        for entry in decision_log:
            for fld in ("step", "decision", "alternatives_considered"):
                if entry.get(fld):
                    aggregated += " " + str(entry.get(fld))
    agg_low = aggregated.lower()
    # check keywords
    if "date" in agg_low or "iso" in agg_low or "yyyy-mm-dd" in agg_low or "iso format" in agg_low or "mm/dd" in agg_low:
        found["date"] = True
    if "status" in agg_low:
        found["status"] = True
    if "processed_at" in agg_low or "processed at" in agg_low or "legacy field" in agg_low:
        found["processed_at"] = True
    if "sleep" in agg_low or "time.sleep" in agg_low:
        found["sleep"] = True
    if "discount" in agg_low or "include_discount" in agg_low or "discount test" in agg_low:
        found["add_discount_test"] = True

    for k, w in diag_weights.items():
        if found[k]:
            pts_diagnosis += w
            diag_msgs.append(f"Detected diagnosis mention: '{k}' (+{w} pts).")
        else:
            diag_msgs.append(f"No clear mention of '{k}' (0 pts for this sub-item).")

    results["scores"]["diagnosis"] = pts_diagnosis
    results["explanations"].append({"diagnosis": diag_msgs})

    # 4) Quality of test edits (20 pts)
    pts_edits = 0
    edits_msgs = []

    # 4a: modifications limited to tests (5 pts)
    non_test_files, scope_reasons = summarize_modified_files_scope(modified_files)
    if non_test_files:
        edits_msgs.append(f"Modified non-test file(s) detected: {non_test_files} (0/5 pts for test-only edits).")
        pts_edits += 0
    else:
        edits_msgs.append("All modified files are under tests/ (5/5 pts).")
        pts_edits += 5

    # 4b: removal of time.sleep (5 pts)
    # Check both modified and added files for time.sleep presence
    sleep_present = any_time_sleep_present(modified_files + added_files)
    if sleep_present:
        edits_msgs.append("time.sleep usage still present in tests (0/5 pts).")
    else:
        edits_msgs.append("No time.sleep usage found in provided test files (5/5 pts).")
        pts_edits += 5

    # 4c: assertions meaningful (5 pts)
    # Count asserts across modified files
    total_asserts = 0
    for f in modified_files:
        total_asserts += count_asserts_in_content(f.get("content", ""))
    # require at least 2-3 asserts across modified files to consider meaningful (heuristic)
    if total_asserts >= 2:
        edits_msgs.append(f"Found {total_asserts} meaningful assert(s) in modified test files (5/5 pts).")
        pts_edits += 5
    else:
        edits_msgs.append(f"Only {total_asserts} asserts found in modified tests (0/5 pts).")

    # 4d: partial checks used (5 pts)
    partial_present = any_partial_checks_present(modified_files)
    if partial_present:
        edits_msgs.append("Partial/substring checks or status handling detected (5/5 pts).")
        pts_edits += 5
    else:
        edits_msgs.append("No evidence of partial substring checks or improved resiliance detected (0/5 pts).")

    results["scores"]["test_edits_quality"] = pts_edits
    results["explanations"].append({"test_edits_quality": edits_msgs})

    # 5) Added test correctness & style (10 pts)
    pts_added = 0
    added_msgs = []
    added_present = find_discount_test_presence(modified_files, added_files)
    if added_present:
        pts_added = WEIGHTS["added_test"]
        added_msgs.append("Found test asserting calculate_total(..., include_discount=True) for O1001 with expected value (full points).")
    else:
        # maybe candidate added such test in modified_files rather than added_files - find already checked both
        added_msgs.append("No test found verifying include_discount=True for order O1001 with expected discounted total (0 pts).")
    results["scores"]["added_test"] = pts_added
    results["explanations"].append({"added_test": added_msgs})

    # 6) Documentation & reproducibility (15 pts)
    pts_doc = 0
    doc_msgs = []
    # 6a commands_run presence and pytest invocation (5 pts)
    if commands_run and isinstance(commands_run, list) and len(commands_run) > 0:
        if any_pytest_command(commands_run):
            pts_doc += 5
            doc_msgs.append("commands_run includes pytest invocation (5/5 pts).")
        else:
            pts_doc += 2  # partial credit for listing commands but not pytest
            doc_msgs.append("commands_run present but no explicit pytest invocation found (2/5 pts).")
    else:
        doc_msgs.append("commands_run missing or empty (0/5 pts).")

    # 6b reproduce_instructions mention pytest and steps (5 pts)
    ri_low = reproduce_instructions.lower()
    if "pytest" in ri_low and ("overwrite" in ri_low or "copy" in ri_low or "run" in ri_low or "1)" in ri_low or "step" in ri_low):
        pts_doc += 5
        doc_msgs.append("reproduce_instructions mention pytest and basic steps (5/5 pts).")
    elif "pytest" in ri_low:
        pts_doc += 3
        doc_msgs.append("reproduce_instructions mention pytest but lack clear step structure (3/5 pts).")
    else:
        doc_msgs.append("reproduce_instructions do not mention pytest or are missing (0/5 pts).")

    # 6c modified/added file contents included and non-empty (5 pts)
    missing_contents = ensure_modified_contents_present(modified_files, added_files)
    if not missing_contents:
        pts_doc += 5
        doc_msgs.append("All modified/added files include non-empty content (5/5 pts).")
    else:
        doc_msgs.append(f"Missing or empty content for files: {missing_contents} (0/5 pts).")

    results["scores"]["documentation"] = pts_doc
    results["explanations"].append({"documentation": doc_msgs})

    # Tally total
    total = pts_baseline + pts_passing + pts_diagnosis + pts_edits + pts_added + pts_doc
    results["total_points"] = total
    percent = (total / MAX_TOTAL) * 100 if MAX_TOTAL else 0.0
    # Round to one decimal for clarity
    overall_score = round(percent, 1)

    results["overall_score"] = overall_score
    results["grading_summary"] = {
        "weights": WEIGHTS,
        "max_total": MAX_TOTAL,
        "computed_total": total,
        "percentage": overall_score
    }

    # Add some top-level metadata
    results["submission_candidate"] = submission.get("candidate_name", "(no name)")
    results["time_spent_minutes"] = submission.get("time_spent_minutes")

    return results


def main():
    if len(sys.argv) < 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)

    submission_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    # Load JSON inputs with error handling
    try:
        submission = safe_load_json(submission_path)
    except FileNotFoundError:
        print(f"Submission file not found: {submission_path}")
        sys.exit(2)
    except ValueError as e:
        print(str(e))
        sys.exit(2)

    try:
        answer_key = safe_load_json(answer_key_path)
    except FileNotFoundError:
        print(f"Answer key file not found: {answer_key_path}")
        sys.exit(2)
    except ValueError as e:
        print(str(e))
        sys.exit(2)

    # Perform grading
    try:
        results = grade(submission, answer_key)
    except Exception as e:
        # Defensive: ensure we still write a minimal results file
        results = {
            "error": f"Grading failed with exception: {e}",
            "overall_score": 0.0
        }

    # Save to test_results.json in same directory as this script (current working dir)
    out_path = os.path.join(os.getcwd(), "test_results.json")
    try:
        with open(out_path, "w", encoding="utf-8") as out_f:
            json.dump(results, out_f, indent=2, ensure_ascii=False)
        print(f"Grading complete. Results written to {out_path}")
    except Exception as e:
        print(f"Failed to write results to {out_path}: {e}")
        sys.exit(2)


if __name__ == "__main__":
    main()
