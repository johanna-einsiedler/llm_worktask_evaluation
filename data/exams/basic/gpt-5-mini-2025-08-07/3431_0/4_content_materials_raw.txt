#!/usr/bin/env python3
"""
Run tests and a small benchmark. Use --low-mem to run the low-memory validation.
"""
import sys
import time
import subprocess
from pathlib import Path

REPEATS = 5000  # repeat count to amplify the benchmark (adjustable but kept fixed)

def run_tests():
    print("Running tests (tests/test_runner.py)...")
    p = subprocess.run([sys.executable, "tests/test_runner.py"], capture_output=True, text=True)
    print(p.stdout)
    if p.returncode != 0:
        print("TESTS FAILED (exit code {})".format(p.returncode))
    else:
        print("ALL TESTS PASSED")
    return p.returncode, p.stdout + p.stderr

def benchmark():
    print("Running benchmark (compute_cooccurrence) with REPEATS={}...".format(REPEATS))
    # import processor after tests (in case code changed)
    import importlib
    import src.processor as processor
    importlib.reload(processor)
    events = processor.load_all_events("data/events.csv")
    start = time.time()
    # run compute_cooccurrence REPEATS times
    for _ in range(REPEATS):
        _ = processor.compute_cooccurrence(events)
    elapsed = time.time() - start
    print("BENCHMARK_TIME_S: {:.6f}".format(elapsed))
    return elapsed

def low_mem_validation():
    print("Running low-memory validation (streaming aggregation)...")
    import importlib
    import src.processor as processor
    importlib.reload(processor)
    try:
        result = processor.aggregate_by_event_code("data/events.csv", stream=True)
    except NotImplementedError:
        print("LOW-MEM MODE: NOT IMPLEMENTED in src/processor.py")
        return 1, "NotImplemented"
    # expected results (pre-computed for provided dataset)
    expected = {"C1": 100, "C2": 20, "C3": 25}
    ok = result == expected
    print("LOW-MEM MODE RESULT: {}".format(result))
    print("LOW-MEM MODE OK: {}".format(ok))
    return (0 if ok else 2), str(result)

def main():
    args = sys.argv[1:]
    rc, test_out = run_tests()
    if rc != 0:
        print("Skipping benchmark because tests failed.")
        # still allow low-mem flag to be executed separately
        if "--low-mem" in args:
            lm_rc, lm_out = low_mem_validation()
            sys.exit(lm_rc)
        sys.exit(rc)
    # run benchmark
    baseline = benchmark()
    print("BASELINE_TIME_S: {:.6f}".format(baseline))
    if "--low-mem" in args:
        lm_rc, lm_out = low_mem_validation()
        sys.exit(lm_rc)
    sys.exit(0)

if __name__ == "__main__":
    main()

--- src/processor.py ---
"""
Processor module for the exam.
Contains simple implementations; some parts intentionally incomplete or suboptimal
for the candidate to modify.

Public functions used by tests and run_all.py:
- parse_line(line) -> (event_id, user_id, event_code, value_int)
- load_all_events(path) -> list of tuples (event_id, user_id, event_code, value)
- aggregate_by_event_code(path, stream=False) -> dict event_code -> total value
- compute_cooccurrence(events) -> integer count of unordered user pairs that share event codes
"""

from typing import List, Tuple, Dict

def parse_line(line: str) -> Tuple[str, str, str, int]:
    """
    Parse a CSV line into fields.
    BUG: does not strip whitespace around fields (intentionally).
    Candidate must fix this so whitespace is handled correctly.
    """
    parts = line.strip().split(",")
    if len(parts) != 4:
        raise ValueError("Bad line: {}".format(line))
    event_id = parts[0]
    user_id = parts[1]
    event_code = parts[2]
    value = int(parts[3])
    return event_id, user_id, event_code, value

def load_all_events(path: str) -> List[Tuple[str, str, str, int]]:
    """
    Load all events into memory and return a list of parsed tuples.
    This is the non-streaming (default) implementation.
    """
    with open(path, "r", encoding="utf-8") as f:
        lines = f.read().splitlines()
    # skip header
    lines = lines[1:]
    events = [parse_line(l) for l in lines]
    return events

def aggregate_by_event_code(path: str, stream: bool = False) -> Dict[str, int]:
    """
    Aggregate total Value per EventCode.
    If stream=True is requested, this function currently raises NotImplementedError.
    Candidate must implement a streaming version that does not load entire file into memory.
    """
    if stream:
        raise NotImplementedError("Low-memory streaming mode not implemented")
    events = load_all_events(path)
    totals = {}
    for _eid, _uid, code, value in events:
        totals.setdefault(code, 0)
        totals[code] += value
    return totals

def compute_cooccurrence(events: List[Tuple[str, str, str, int]]) -> int:
    """
    Compute the total number of unique unordered user pairs that share an EventCode.
    Current implementation is a naive O(n^2) approach (slow on larger data).
    Candidate will optimize this function in Task C while preserving correctness.
    """
    # build set of pairs (frozenset of two user ids) where both events have same code
    pairs = set()
    n = len(events)
    for i in range(n):
        _eid_i, user_i, code_i, _v_i = events[i]
        for j in range(i+1, n):
            _eid_j, user_j, code_j, _v_j = events[j]
            if code_i == code_j and user_i != user_j:
                pairs.add(frozenset([user_i, user_j]))
    return len(pairs)

# end of src/processor.py

#!/usr/bin/env python3
"""
A minimal test runner using only the standard library.
Exits with non-zero code on failures. Prints details to stdout.
"""

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))  # add repo root to path
from src import processor

def assert_equal(a, b, msg=""):
    if a != b:
        raise AssertionError("Assertion failed: {} != {}. {}".format(a, b, msg))

def test_parse_line_whitespace():
    line = "E1, U1, C1, 10"
    parsed = processor.parse_line(line)
    # expected: whitespace should be removed from user and code
    assert_equal(parsed, ("E1", "U1", "C1", 10), "parse_line must strip fields")

def test_aggregate_top_event():
    totals = processor.aggregate_by_event_code("data/events.csv", stream=False)
    # expected totals for provided data
    expected = {"C1": 100, "C2": 20, "C3": 25}
    assert_equal(totals, expected, "aggregate_by_event_code returned wrong totals")

def test_cooccurrence_count():
    events = processor.load_all_events("data/events.csv")
    count = processor.compute_cooccurrence(events)
    # For provided data:
    # C1 users: U1,U2,U3,U4,U5 => 5C2 = 10
    # C2 users: U1,U2,U5,U7 => 4C2 = 6
    # C3 users: U4,U6 => 1
    # total = 17
    assert_equal(count, 17, "compute_cooccurrence returned wrong count")

def run_all():
    tests = [test_parse_line_whitespace, test_aggregate_top_event, test_cooccurrence_count]
    failures = []
    for t in tests:
        try:
            t()
            print("PASS:", t.__name__)
        except Exception as e:
            print("FAIL:", t.__name__)
            print("  ", e)
            failures.append((t.__name__, str(e)))
    if failures:
        print("FAILED {} tests".format(len(failures)))
        sys.exit(1)
    else:
        print("ALL TESTS PASSED")
        sys.exit(0)

if __name__ == "__main__":
    run_all()

How materials were constructed and purpose
- Overall goal: small deterministic Python project (no external libs) that exercises reading an unfamiliar codebase and making three types of changes:
  - Task A (bug fix): fix incorrect parsing (whitespace handling) so tests pass.
  - Task B (hardware adaptation): implement streaming aggregation to avoid loading the full file into memory when --low-mem / stream=True is requested.
  - Task C (performance improvement): replace the O(n^2) co-occurrence computation with an O(n) approach that counts unique users per event code and computes pair counts via combinatorics.

Files and relationships
- data/events.csv
  - 12 deterministic rows. Header plus events E1..E12 with clearly labeled fields EventID,UserID,EventCode,Value.
  - Pre-computed totals:
    - C1 -> 100
    - C2 -> 20
    - C3 -> 25
  - Pre-computed co-occurrence count: 17 (explanation included in test file comments).

- src/processor.py
  - parse_line(line): intentionally does not strip whitespace — this creates a failing unit test (Task A). This simulates a typical small parsing bug that can be fixed locally without changing test inputs.
  - load_all_events(path): reads entire file into memory (read().splitlines()) — acceptable initially, but Task B will require a streaming alternative.
  - aggregate_by_event_code(path, stream=False): uses in-memory events when stream=False; when stream=True raises NotImplementedError. Candidate must implement streaming aggregation (read file line-by-line, parse and accumulate) to succeed on low-memory validation.
  - compute_cooccurrence(events): computes unique unordered user pairs that share an event code using a naive O(n^2) approach. This is correct but slow; candidate will optimize this to compute counts per code and sum k*(k-1)//2 to get O(n) behavior.

- tests/test_runner.py
  - Three deterministic tests:
    1. test_parse_line_whitespace: constructs a line with spaces ("E1, U1, C1, 10") and asserts parse_line normalizes whitespace. This fails initially because parse_line doesn't strip — this is Task A.
    2. test_aggregate_top_event: checks aggregate_by_event_code returns the exact expected dict for the provided dataset.

  - Tests use only the standard library and exit with non-zero if any test fails.

- run_all.py
  - Orchestrates tests and a small benchmark.
  - On normal run: runs tests (calls tests/test_runner.py as a subprocess); if tests pass, runs benchmark that calls processor.load_all_events() and then calls processor.compute_cooccurrence(events) REPEATS times to measure elapsed time and prints BENCHMARK_TIME_S and BASELINE_TIME_S. Candidate should run this before making changes to capture baseline time.
  - Supports --low-mem flag: when provided, after tests the script calls processor.aggregate_by_event_code(..., stream=True) and compares the returned dict to the expected totals; if stream=True is unimplemented it prints a NotImplemented message. This is the Task B validation target.

Design choices and how they map to exam tasks
- Task A (Bug Fix)
  - The parse_line whitespace bug is small, localizable, and realistic. Candidates must run tests, inspect failing assertion from test_parse_line_whitespace, and fix parse_line by stripping whitespace from each field. This tests reading unfamiliar code, running tests, and applying a minimal safe fix.

- Task B (Hardware Adaptation)
  - The code currently reads the entire file in load_all_events. The run_all.py --low-mem path calls aggregate_by_event_code with stream=True, but the function currently raises NotImplementedError. Candidate should implement streaming aggregation by reading the file line-by-line, calling parse_line for each line, and updating totals without storing the entire dataset. The run_all.py low-memory validation compares the dict to the expected totals, so correctness is verifiable. This tests adapting to constrained memory by replacing in-memory approach with streaming.

- Task C (Performance Improvement)
  - The naive compute_cooccurrence is O(n^2). The optimal approach is to:
    - Build a mapping from event_code -> set of users (single pass over events).
    - For each code, compute pairs = k * (k-1) // 2 where k is number of unique users.
    - Sum over codes to produce the same result.
  - This yields a substantial improvement for repeated benchmarks because run_all.py calls compute_cooccurrence REPEATS times. Candidates must ensure the optimized result equals the original for the provided data and that tests still pass.
  - The benchmark uses REPEATS = 5000 to amplify runtime difference deterministically. The evaluator can compare baseline and optimized times printed by run_all.py.

Traps, common errors, and grader guidance
- Possible naive but incorrect fixes for Task A:
  - Hard-coding stripped values only for the test line or special-casing a test input. The grader should verify parse_line is generally fixed by trying a few additional lines or by inspecting the code change (must use strip()).
  - Changing tests instead of code — tests are part of material; candidate must not alter tests.

- Task B pitfalls:
  - Implementing a streaming function that still accumulates all events into memory (e.g., reading lines but appending to a list) defeats the purpose. The evaluator should review code to ensure the stream=True path does not create a full listcopy.
  - Not preserving correctness: the aggregator must produce the exact expected dict ordering-independent. The run_all.py script compares dict equality.

- Task C pitfalls:
  - Implementing an optimization that changes semantics (e.g., counting pair occurrences per event instance instead of unique user pairs) will break tests. Tests check cooccurrence number; final code must match 17 for provided dataset.
  - Forgetting to keep function name compute_cooccurrence so the benchmark picks up the optimized implementation.

What correct answers depend on
- Correctness for Task A depends on trimming whitespace; parse_line should apply .strip() to each CSV field and still convert Value to int.
- Correctness for Task B depends on a streaming implementation that iterates file lines and updates totals without building a full events list.
- Correctness for Task C depends on computing the same unique unordered user-pair count as the naive algorithm, but doing so without nested loops (use mapping code->set(users) then sum combinations).
