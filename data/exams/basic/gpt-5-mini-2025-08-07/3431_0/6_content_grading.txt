Here's the requested grading script. Save it as task_evaluation.py. It reads the candidate's submission JSON and the answer key JSON, applies the grading rubric described in the exam materials, and writes a detailed test_results.json file in the same directory.

Script: task_evaluation.py
```python
#!/usr/bin/env python3
"""
Automated grader for the Practical Maintenance & Optimization — Basic exam.

Usage:
    python task_evaluation.py <candidate_submission.json> <answer_key.json>

Produces:
    test_results.json in the current working directory with detailed scoring.

Notes:
- Uses only the Python standard library.
- Tries to be robust to missing/incorrect fields and reports clear deduction reasons.
"""

import json
import sys
import os
from typing import Any, Dict, List, Tuple

# --- Helper utilities ------------------------------------------------------

def safe_load_json(path: str) -> Tuple[Dict[str, Any], List[str]]:
    """Load JSON from path. Return (data, errors). Errors list is empty on success."""
    errors = []
    if not os.path.exists(path):
        errors.append(f"File not found: {path}")
        return {}, errors
    try:
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        return data, errors
    except Exception as e:
        errors.append(f"Failed to read/parse JSON {path}: {e}")
        return {}, errors

def get_field(d: Dict[str, Any], path: List[str], default=None):
    """Safely get nested field from dict; return default if missing."""
    cur = d
    for p in path:
        if not isinstance(cur, dict) or p not in cur:
            return default
        cur = cur[p]
    return cur

def safe_float(v) -> Tuple[float, bool]:
    """Try to convert v to float. Return (value, ok)."""
    try:
        return float(v), True
    except Exception:
        return 0.0, False

def contains_all_tests_passed(commands_run: List[Dict[str, Any]]) -> bool:
    """Detect if any command stdout contains 'ALL TESTS PASSED' and exit_code == 0."""
    if not isinstance(commands_run, list):
        return False
    for entry in commands_run:
        stdout = str(entry.get("stdout", "")).lower()
        ec = entry.get("exit_code", None)
        if "all tests passed" in stdout and (ec == 0 or ec is None):
            return True
    return False

def contains_low_mem_ok(commands_run: List[Dict[str, Any]]) -> bool:
    """Detect if any command stdout contains 'LOW-MEM MODE OK: True' (case-insensitive) and exit_code == 0."""
    if not isinstance(commands_run, list):
        return False
    for entry in commands_run:
        stdout = str(entry.get("stdout", "")).lower()
        ec = entry.get("exit_code", None)
        if "low-mem mode ok: true" in stdout and (ec == 0 or ec is None):
            return True
    return False

# --- Main grading logic ---------------------------------------------------

def grade_submission(candidate: Dict[str, Any], answer_key: Dict[str, Any]) -> Dict[str, Any]:
    """
    Grade the candidate submission against answer_key using the rubric in the exam materials.
    Returns a detailed results dict suitable for JSON output.
    """
    results = {
        "tasks": {},
        "total_score": 0.0,
        "max_score": 100.0,
        "percentage": 0.0,
        "overall_score": 0.0,
        "notes": []
    }

    total_score = 0.0
    max_total = 100.0

    # Points per task
    TASK_POINTS = {"A-bugfix": 35.0, "B-lowmem": 30.0, "C-performance": 35.0}

    # --- Task A: Bug Fix ---
    task_id = "A-bugfix"
    task_res = {"score": 0.0, "max_score": TASK_POINTS[task_id], "breakdown": {}, "reasons": []}
    t = None
    # Find candidate task entry by id
    cand_tasks = candidate.get("tasks", [])
    for entry in cand_tasks:
        if isinstance(entry, dict) and entry.get("id") == task_id:
            t = entry
            break
    if t is None:
        task_res["reasons"].append("Task entry missing.")
        # zero score
    else:
        # Extract evidence/tests_passed
        evidence = t.get("evidence", {})
        tests_passed = evidence.get("tests_passed", None)
        # correctness (25 pts)
        correctness_pts = 25.0 if tests_passed is True else 0.0
        if correctness_pts == 0.0:
            task_res["reasons"].append("Unit tests not reported as passing (evidence.tests_passed != true).")
        # quality of fix (5 pts): presence of files_changed non-empty
        files_changed = t.get("files_changed", [])
        quality_pts = 5.0 if isinstance(files_changed, list) and len(files_changed) > 0 else 0.0
        if quality_pts == 0.0:
            task_res["reasons"].append("No files_changed reported for Task A; cannot award quality points.")
        # explanation (5 pts): presence and reasonable length
        explanation = t.get("explanation", "")
        expl_len = len(explanation.strip()) if isinstance(explanation, str) else 0
        if expl_len == 0:
            explanation_pts = 0.0
            task_res["reasons"].append("No explanation provided for Task A.")
        elif expl_len <= 200:
            explanation_pts = 5.0
        elif expl_len <= 500:
            explanation_pts = 3.0
            task_res["reasons"].append("Explanation for Task A exceeds 200 chars; partial credit for explanation.")
        else:
            explanation_pts = 0.0
            task_res["reasons"].append("Explanation for Task A too long or unsuitable; no explanation points awarded.")
        # Sum and clamp
        task_score = correctness_pts + quality_pts + explanation_pts
        if task_score > TASK_POINTS[task_id]:
            task_score = TASK_POINTS[task_id]
        task_res["score"] = task_score
        task_res["breakdown"] = {
            "correctness_pts": correctness_pts,
            "quality_pts": quality_pts,
            "explanation_pts": explanation_pts
        }
    results["tasks"][task_id] = task_res
    total_score += task_res["score"]

    # --- Task B: Low-memory adaptation ---
    task_id = "B-lowmem"
    task_res = {"score": 0.0, "max_score": TASK_POINTS[task_id], "breakdown": {}, "reasons": []}
    t = None
    for entry in cand_tasks:
        if isinstance(entry, dict) and entry.get("id") == task_id:
            t = entry
            break
    if t is None:
        task_res["reasons"].append("Task entry missing.")
    else:
        evidence = t.get("evidence", {})
        low_mem_mode_ran = evidence.get("low_mem_mode_ran", None)
        notes = evidence.get("notes", "")
        # adaptation correctness (15 pts): low_mem_mode_ran == True
        adap_pts = 15.0 if low_mem_mode_ran is True else 0.0
        if adap_pts == 0.0:
            task_res["reasons"].append("Low-memory mode not reported as run/correct (evidence.low_mem_mode_ran != true).")
        # evidence (10 pts): look for command output indicating LOW-MEM MODE OK: True or evidence flag true
        commands_run = t.get("commands_run", [])
        lowmem_ok_in_stdout = contains_low_mem_ok(commands_run)
        evidence_pts = 0.0
        if low_mem_mode_ran is True or lowmem_ok_in_stdout:
            evidence_pts = 10.0
        else:
            task_res["reasons"].append("No command output showing 'LOW-MEM MODE OK: True' and evidence.low_mem_mode_ran not true; cannot award evidence points.")
        # explanation (5 pts)
        explanation = t.get("explanation", "")
        expl_len = len(explanation.strip()) if isinstance(explanation, str) else 0
        if expl_len == 0:
            explanation_pts = 0.0
            task_res["reasons"].append("No explanation provided for Task B.")
        elif expl_len <= 200:
            explanation_pts = 5.0
        elif expl_len <= 500:
            explanation_pts = 3.0
            task_res["reasons"].append("Explanation for Task B exceeds 200 chars; partial credit for explanation.")
        else:
            explanation_pts = 0.0
            task_res["reasons"].append("Explanation for Task B too long or unsuitable; no explanation points awarded.")
        task_score = adap_pts + evidence_pts + explanation_pts
        if task_score > TASK_POINTS[task_id]:
            task_score = TASK_POINTS[task_id]
        task_res["score"] = task_score
        task_res["breakdown"] = {
            "adaptation_correctness_pts": adap_pts,
            "evidence_pts": evidence_pts,
            "explanation_pts": explanation_pts,
            "evidence_notes": notes
        }
    results["tasks"][task_id] = task_res
    total_score += task_res["score"]

    # --- Task C: Performance improvement ---
    task_id = "C-performance"
    task_res = {"score": 0.0, "max_score": TASK_POINTS[task_id], "breakdown": {}, "reasons": []}
    t = None
    for entry in cand_tasks:
        if isinstance(entry, dict) and entry.get("id") == task_id:
            t = entry
            break
    if t is None:
        task_res["reasons"].append("Task entry missing.")
    else:
        evidence = t.get("evidence", {})
        baseline = evidence.get("baseline_time_s", None)
        optimized = evidence.get("optimized_time_s", None)
        provided_factor = evidence.get("improvement_factor", None)

        baseline_val, ok_b = safe_float(baseline)
        optimized_val, ok_o = safe_float(optimized)
        computed_factor = None
        factor_ok = False
        factor_value = 0.0

        if ok_b and ok_o and optimized_val > 0:
            computed_factor = baseline_val / optimized_val
            factor_value = computed_factor
            factor_ok = True
        else:
            # If candidate supplied improvement_factor numeric, try to use it
            pf, pf_ok = safe_float(provided_factor)
            if pf_ok and pf > 0:
                computed_factor = pf
                factor_value = pf
                factor_ok = True
            else:
                factor_ok = False

        # Performance target scoring (20 pts)
        perf_pts = 0.0
        if factor_ok:
            if factor_value >= 2.0:
                perf_pts = 20.0
            elif factor_value > 1.0:
                perf_pts = 10.0  # partial credit
            else:
                perf_pts = 0.0
                task_res["reasons"].append("No measurable performance improvement (improvement_factor <= 1.0).")
        else:
            task_res["reasons"].append("Could not determine improvement factor from evidence (invalid or missing baseline/optimized times).")

        # Regression-free tests (10 pts): ensure tests still pass after optimization
        # Look in Task C commands_run for 'ALL TESTS PASSED' and exit_code 0
        commands_run = t.get("commands_run", [])
        tests_ok = contains_all_tests_passed(commands_run)
        tests_pts = 10.0 if tests_ok else 0.0
        if tests_pts == 0.0:
            task_res["reasons"].append("No evidence that tests passed after optimization (Task C commands_run missing 'ALL TESTS PASSED').")

        # Explanation (5 pts)
        explanation = t.get("explanation", "")
        expl_len = len(explanation.strip()) if isinstance(explanation, str) else 0
        if expl_len == 0:
            explanation_pts = 0.0
            task_res["reasons"].append("No explanation provided for Task C.")
        elif expl_len <= 200:
            explanation_pts = 5.0
        elif expl_len <= 500:
            explanation_pts = 3.0
            task_res["reasons"].append("Explanation for Task C exceeds 200 chars; partial credit for explanation.")
        else:
            explanation_pts = 0.0
            task_res["reasons"].append("Explanation for Task C too long or unsuitable; no explanation points awarded.")

        task_score = perf_pts + tests_pts + explanation_pts
        if task_score > TASK_POINTS[task_id]:
            task_score = TASK_POINTS[task_id]

        task_res["score"] = task_score
        task_res["breakdown"] = {
            "performance_pts": perf_pts,
            "regression_free_tests_pts": tests_pts,
            "explanation_pts": explanation_pts,
            "baseline_time_s": baseline_val if ok_b else None,
            "optimized_time_s": optimized_val if ok_o else None,
            "computed_improvement_factor": computed_factor
        }
    results["tasks"][task_id] = task_res
    total_score += task_res["score"]

    # Final aggregation
    results["total_score"] = round(total_score, 2)
    results["max_score"] = max_total
    percent = (total_score / max_total) * 100.0 if max_total > 0 else 0.0
    results["percentage"] = round(percent, 2)
    results["overall_score"] = round(percent, 2)

    return results

# --- Main entrypoint ------------------------------------------------------

def main():
    # Basic argument validation
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py <candidate_submission.json> <answer_key.json>", file=sys.stderr)
        sys.exit(2)

    cand_path = sys.argv[1]
    key_path = sys.argv[2]

    candidate, cerrs = safe_load_json(cand_path)
    answer_key, kerrs = safe_load_json(key_path)

    errors = []
    if cerrs:
        errors.extend(cerrs)
    if kerrs:
        errors.extend(kerrs)

    if errors:
        # still produce a partial test_results.json with errors listed
        out = {
            "error": True,
            "messages": errors
        }
        try:
            with open("test_results.json", "w", encoding="utf-8") as f:
                json.dump(out, f, indent=2)
            print("Errors encountered. Wrote test_results.json with error details.", file=sys.stderr)
        except Exception as e:
            print(f"Failed to write test_results.json: {e}", file=sys.stderr)
        sys.exit(1)

    # Grade
    try:
        results = grade_submission(candidate, answer_key)
    except Exception as e:
        # Unexpected exception during grading; write trace info to results file
        results = {
            "error": True,
            "messages": [f"Exception during grading: {e}"]
        }

    # Save results
    try:
        with open("test_results.json", "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
        print("Grading complete. Results written to test_results.json")
        # Also print overall score to stdout for quick look
        print(f"Overall percentage score: {results.get('overall_score', 0.0)}%")
    except Exception as e:
        print(f"Failed to write test_results.json: {e}", file=sys.stderr)
        sys.exit(1)

if __name__ == "__main__":
    main()
```

What the script checks and how it scores (summary)
- Task A (35 points):
  - correctness (25 pts): evidence.tests_passed must be true.
  - quality (5 pts): files_changed list non-empty.
  - explanation (5 pts): explanation present and <=200 chars for full points, partial if longer.
- Task B (30 points):
  - adaptation correctness (15 pts): evidence.low_mem_mode_ran true.
  - evidence (10 pts): command output contains "LOW-MEM MODE OK: True" or evidence flag true.
  - explanation (5 pts): present and reasonable length.
- Task C (35 points):
  - performance (20 pts): improvement_factor >=2.0 for full points, >1.0 for partial.
  - regression-free tests (10 pts): Task C commands_run contains "ALL TESTS PASSED".
  - explanation (5 pts): present and reasonable length.

The script is tolerant of missing/extra whitespace/capitalization in the important markers (case-insensitive checks for key outputs). It writes clear reasons for deductions into test_results.json so the evaluator can see why points were awarded or withheld.

If you want the script to also re-run the supplied commands to verify authenticity, that can be added later — this script performs static grading of the submission JSON only, per the requested requirements.
