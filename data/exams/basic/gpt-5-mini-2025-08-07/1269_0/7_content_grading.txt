# task_evaluation.py
import json
import os
import sys
import traceback

# Grading script for the Basic Practical Programming Exam ("sales_agg").
# Usage:
#   python task_evaluation.py <candidate_submission.json> <answer_key.json>
#
# This script loads the candidate submission JSON and the answer key JSON,
# performs automated grading across the defined criteria and weights, and
# writes a detailed JSON report named test_results.json in the same directory
# as this script.
#
# Implementation notes:
# - Uses only Python standard library.
# - Tolerant to small formatting differences in outputs (trims whitespace, normalizes line endings).
# - Produces per-criterion breakdown, explanations for deductions, and an overall_score (0-100).
# - Robust error handling: missing/invalid JSON or missing required fields are handled gracefully,
#   with deductions and clear messages.

# ========== Configuration: scoring weights (must sum to 100) ==========
WEIGHTS = {
    "correctness": 50,     # includes three tests (normal, malformed, min-revenue)
    "documentation": 20,   # README, build_instructions, changelog
    "comments": 15,        # inline comments & candidate self-assessment
    "tests_evidence": 10,  # test entries presence and captures
    "revision": 5          # --min-revenue implemented and documented
}

# Detailed breakdown within correctness (sub-weights sum to WEIGHTS['correctness'])
CORRECTNESS_SUB = {
    "normal_sample": 20,
    "malformed_sample": 20,
    "min_revenue_filter": 10
}

# Required test names expected in tests array
REQUIRED_TEST_NAMES = set(CORRECTNESS_SUB.keys())


# ========== Helper utility functions ==========

def load_json_file(path):
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f), None
    except Exception as e:
        return None, str(e)


def normalize_output(s):
    """Normalize stdout/stderr-like strings for tolerant comparison.
    - Convert None to empty string.
    - Normalize CRLF to LF.
    - Strip leading/trailing whitespace and trailing blank lines.
    - Strip trailing spaces on each line.
    """
    if s is None:
        return ""
    if not isinstance(s, str):
        s = str(s)
    s = s.replace('\r\n', '\n').replace('\r', '\n')
    # Split into lines, strip trailing spaces on each, then remove leading/trailing blank lines
    lines = [line.rstrip() for line in s.split('\n')]
    # Remove leading blank lines
    while lines and lines[0] == '':
        lines.pop(0)
    # Remove trailing blank lines
    while lines and lines[-1] == '':
        lines.pop()
    return '\n'.join(lines)


def safe_get(d, key, default=None):
    return d.get(key, default) if isinstance(d, dict) else default


def find_test_by_name(tests, name):
    if not isinstance(tests, list):
        return None
    for t in tests:
        if safe_get(t, 'test_name', '') == name:
            return t
    return None


def count_comment_lines_in_source(content, language_hint):
    """Heuristic: count lines that look like comments for common languages.
    Supports Python (#, triple-quote docstrings), Java/C/JS (//, /* */), and shell (#).
    Returns (comment_lines, total_lines).
    """
    if not isinstance(content, str):
        return 0, 0
    lines = content.splitlines()
    total = len(lines)
    comment_lines = 0
    # Very simple state machine for block comments (/* ... */) and triple-quotes (""")
    in_block = False
    block_end = None
    for line in lines:
        stripped = line.strip()
        if in_block:
            comment_lines += 1
            if block_end and block_end in stripped:
                in_block = False
            continue
        if stripped.startswith('#'):
            comment_lines += 1
            continue
        if stripped.startswith('//'):
            comment_lines += 1
            continue
        # block comment start for /* or triple quotes
        if stripped.startswith('/*'):
            in_block = True
            block_end = '*/'
            comment_lines += 1
            continue
        if stripped.startswith('"""') or stripped.startswith("'''"):
            # Might be docstring; count as comment block start. If it's a one-line docstring, count it as comment line.
            if (stripped.endswith('"""') and len(stripped) >= 6) or (stripped.endswith("'''") and len(stripped) >= 6):
                # one-line docstring
                comment_lines += 1
                continue
            in_block = True
            block_end = stripped[:3]
            comment_lines += 1
            continue
    return comment_lines, total


def string_contains_any(s, tokens):
    if not isinstance(s, str):
        return False
    s_lower = s.lower()
    for t in tokens:
        if t.lower() in s_lower:
            return True
    return False


# ========== Main grading logic ==========

def grade_submission(candidate_path, answer_key_path):
    results = {
        "criteria": {},
        "deductions": [],
        "errors": [],
    }

    # Load JSON files
    candidate_json, err = load_json_file(candidate_path)
    if err:
        results["errors"].append(f"Failed to load candidate submission JSON: {err}")
        # Prepare zeroed result
        return finalize_results(results, 0.0, "Could not load candidate submission")

    answer_json, err = load_json_file(answer_key_path)
    if err:
        results["errors"].append(f"Failed to load answer key JSON: {err}")
        return finalize_results(results, 0.0, "Could not load answer key")

    # Basic sanity checks for required top-level fields
    required_fields = [
        "candidate_name", "language", "time_spent_minutes", "source_files", "build_instructions",
        "readme", "changelog", "tests", "assumptions", "comments_coverage", "signature"
    ]
    missing_fields = [f for f in required_fields if f not in candidate_json]
    if missing_fields:
        results["errors"].append(f"Candidate submission missing required fields: {missing_fields}")
        # Continue grading but many deductions will apply.

    # Keep running tallies
    total_awarded = 0.0
    total_max = sum(WEIGHTS.values())

    # ---------- Correctness (50 pts) ----------
    correctness_details = []
    correctness_awarded = 0.0
    correctness_max = WEIGHTS["correctness"]

    candidate_tests = candidate_json.get("tests", [])
    answer_tests = answer_json.get("tests", [])

    # Build a mapping in the answer key for expected outputs by test_name
    answer_expected_map = {}
    if isinstance(answer_tests, list):
        for t in answer_tests:
            tn = safe_get(t, 'test_name', None)
            if tn:
                answer_expected_map[tn] = safe_get(t, 'expected_output', '')

    # For each required test, check presence and matching outputs
    for test_name, weight in CORRECTNESS_SUB.items():
        test_result = find_test_by_name(candidate_tests, test_name)
        reason_msgs = []
        awarded = 0.0
        if test_result is None:
            reason_msgs.append(f"Test '{test_name}' missing from submission tests array.")
            # award 0 for missing test
        else:
            # Check presence of command
            cmd = safe_get(test_result, 'command', '')
            if not cmd:
                reason_msgs.append("No command recorded for this test.")
            # Compare actual_output to expected_output from answer key if present
            expected = answer_expected_map.get(test_name)
            candidate_actual = safe_get(test_result, 'actual_output', '')
            if expected is None:
                # No expected in answer key (unlikely). Use candidate expected_output if provided.
                expected = safe_get(test_result, 'expected_output', '')
            # Normalize and compare
            norm_expected = normalize_output(expected)
            norm_actual = normalize_output(candidate_actual)
            if norm_expected == norm_actual:
                awarded = weight
                reason_msgs.append("Output matches expected output exactly (after normalization). Full credit for this test.")
            else:
                # Partial credit heuristic:
                # If candidate produced any output (non-empty) award small partial credit,
                # otherwise zero. Also include difference message.
                if norm_actual.strip() == "":
                    reason_msgs.append("Actual output is empty or missing. No credit for this test.")
                    awarded = 0.0
                else:
                    # Provide partial credit proportionally to how many lines match in order
                    exp_lines = norm_expected.split('\n') if norm_expected != "" else []
                    act_lines = norm_actual.split('\n') if norm_actual != "" else []
                    if exp_lines:
                        # count matching lines at same positions
                        matches = sum(1 for i in range(min(len(exp_lines), len(act_lines))) if exp_lines[i].strip() == act_lines[i].strip())
                        proportion = matches / len(exp_lines)
                        awarded = round(proportion * weight, 2)
                        reason_msgs.append(f"Output does not match exactly. Matched {matches}/{len(exp_lines)} expected lines -> partial credit {awarded}/{weight}.")
                    else:
                        # no expected lines to compare, give minimal credit for non-empty output
                        awarded = round(0.1 * weight, 2)
                        reason_msgs.append(f"No expected output to compare against; gave {awarded}/{weight} for presence of output.")
            # Also check the 'passed' boolean if present and truthful
            passed_flag = safe_get(test_result, 'passed', None)
            if isinstance(passed_flag, bool) and passed_flag and awarded < weight:
                # Candidate marked passed but outputs differ -> note inconsistency
                reason_msgs.append("Candidate marked this test as passed but actual_output does not match expected_output.")
        correctness_details.append({
            "test_name": test_name,
            "weight": weight,
            "awarded": awarded,
            "messages": reason_msgs
        })
        correctness_awarded += awarded

    results["criteria"]["correctness"] = {
        "score": round(correctness_awarded, 2),
        "max": correctness_max,
        "details": correctness_details
    }
    total_awarded += correctness_awarded

    # ---------- Documentation (20 pts) ----------
    doc_awarded = 0.0
    doc_max = WEIGHTS["documentation"]
    doc_details = []
    # README presence and helpfulness
    readme = candidate_json.get("readme", "")
    if isinstance(readme, str) and readme.strip() != "":
        # Give up to 10 pts for README content (presence + includes instructions)
        # Heuristic checks: contains 'run' or 'usage', contains 'sample_sales.csv' or mentions filenames, mentions assumptions or behaviors
        score_readme = 10.0
        msgs = ["README present."]
        if not string_contains_any(readme, ["run", "usage", "how to", "python", "command"]):
            score_readme -= 3.0
            msgs.append("README does not include obvious run/build instructions or usage keywords.")
        if not string_contains_any(readme, ["sample_sales.csv", "malformed_sales.csv", "min-revenue", "--min-revenue", "assum"]):
            score_readme -= 2.0
            msgs.append("README does not mention sample filenames, filter flag, or assumptions explicitly.")
        if score_readme < 0:
            score_readme = 0.0
        doc_awarded += score_readme
        doc_details.append({"aspect": "README", "awarded": round(score_readme, 2), "max": 10, "messages": msgs})
    else:
        doc_details.append({"aspect": "README", "awarded": 0.0, "max": 10, "messages": ["README missing or empty."]})

    # build_instructions
    build_instructions = candidate_json.get("build_instructions", "")
    score_build = 0.0
    msgs_build = []
    if isinstance(build_instructions, str) and build_instructions.strip() != "":
        # Expect commands to reference sample_sales.csv and malformed_sales.csv and mention --min-revenue
        score_build = 6.0
        if "sample_sales.csv" not in build_instructions:
            score_build -= 2.0
            msgs_build.append("build_instructions do not reference sample_sales.csv.")
        if "malformed_sales.csv" not in build_instructions:
            score_build -= 2.0
            msgs_build.append("build_instructions do not reference malformed_sales.csv.")
        if not string_contains_any(build_instructions, ["--min-revenue", "-m"]):
            score_build -= 2.0
            msgs_build.append("build_instructions do not show how to run the --min-revenue filter.")
        if score_build < 0:
            score_build = 0.0
        if not msgs_build:
            msgs_build.append("build_instructions reference required sample files and filter flag.")
    else:
        msgs_build.append("build_instructions missing or empty.")
    doc_awarded += score_build
    doc_details.append({"aspect": "build_instructions", "awarded": round(score_build, 2), "max": 6, "messages": msgs_build})

    # changelog presence and content
    changelog = candidate_json.get("changelog", [])
    score_changelog = 0.0
    msgs_changelog = []
    if isinstance(changelog, list) and len(changelog) >= 1:
        # Reward more for multiple entries and inclusion of a revision mention
        score_changelog = 4.0
        if len(changelog) < 2:
            score_changelog = 2.0
            msgs_changelog.append("Changelog has only one entry; expected a sequence including a follow-up revision.")
        # If any changelog entry mentions min-revenue or filter, better.
        has_min = any(string_contains_any(json.dumps(entry), ["--min-revenue", "min-revenue", "-m", "min_revenue", "filter"]) for entry in changelog)
        if not has_min:
            msgs_changelog.append("Changelog does not mention the follow-up --min-revenue revision explicitly.")
    else:
        msgs_changelog.append("Changelog missing or not a list.")
    doc_awarded += score_changelog
    doc_details.append({"aspect": "changelog", "awarded": round(score_changelog, 2), "max": 4, "messages": msgs_changelog})

    results["criteria"]["documentation"] = {
        "score": round(doc_awarded, 2),
        "max": doc_max,
        "details": doc_details
    }
    total_awarded += doc_awarded

    # ---------- Comments & Code clarity (15 pts) ----------
    comments_awarded = 0.0
    comments_max = WEIGHTS["comments"]
    comments_details = []

    comments_coverage_self = candidate_json.get("comments_coverage", "")
    if isinstance(comments_coverage_self, str) and comments_coverage_self.strip() != "":
        comments_awarded += 3.0
        comments_details.append({"aspect": "self_assessment_present", "awarded": 3.0, "max": 3.0, "messages": ["Candidate provided comments_coverage self-assessment."]})
    else:
        comments_details.append({"aspect": "self_assessment_present", "awarded": 0.0, "max": 3.0, "messages": ["comments_coverage field missing or empty."]})

    # Check source_files for comment density
    source_files = candidate_json.get("source_files", [])
    if not isinstance(source_files, list) or len(source_files) == 0:
        comments_details.append({"aspect": "source_files_present", "awarded": 0.0, "max": 2.0, "messages": ["No source_files included in submission. Cannot analyze code comments."]})
    else:
        comments_awarded += 2.0  # for presence of source files
        comments_details.append({"aspect": "source_files_present", "awarded": 2.0, "max": 2.0, "messages": ["Source files present."]})
        # Aggregate comment lines across all source files
        total_comment_lines = 0
        total_code_lines = 0
        for sf in source_files:
            content = safe_get(sf, 'content', '')
            lang = safe_get(sf, 'language', '') or safe_get(sf, 'filename', '')
            clines, tlines = count_comment_lines_in_source(content, lang)
            total_comment_lines += clines
            total_code_lines += tlines
        # Heuristic thresholds:
        # - Minimum 3 comment lines AND comment ratio >= 3 / total_lines OR >= 5% of lines -> full points for main comment metric
        main_metric_award = 0.0
        if total_code_lines == 0:
            main_metric_award = 0.0
            comments_details.append({"aspect": "comment_density", "awarded": 0.0, "max": 10.0,
                                     "messages": ["Source files empty; cannot assess comment density."]})
        else:
            comment_ratio = total_comment_lines / total_code_lines
            msgs = [f"Total lines: {total_code_lines}, comment lines: {total_comment_lines}, ratio: {comment_ratio:.3f}"]
            # Desired >= 5% and at least 3 comment lines
            if total_comment_lines >= 3 and comment_ratio >= 0.05:
                main_metric_award = 10.0
                msgs.append("Sufficient comment coverage (>=3 comment lines and >=5% of lines).")
            else:
                # Partial credit scaled by ratio up to 10 points
                scaled = min(comment_ratio / 0.05, 1.0) * 10.0
                # But require at least 1 comment line to give any credit
                if total_comment_lines < 1:
                    scaled = 0.0
                main_metric_award = round(scaled, 2)
                msgs.append("Insufficient comment coverage; partial credit awarded based on ratio.")
            comments_details.append({"aspect": "comment_density", "awarded": main_metric_award, "max": 10.0, "messages": msgs})
            comments_awarded += main_metric_award

    # Cap to max
    if comments_awarded > comments_max:
        comments_awarded = comments_max

    results["criteria"]["comments"] = {
        "score": round(comments_awarded, 2),
        "max": comments_max,
        "details": comments_details
    }
    total_awarded += comments_awarded

    # ---------- Tests & Evidence (10 pts) ----------
    tests_awarded = 0.0
    tests_max = WEIGHTS["tests_evidence"]
    tests_details = []

    # Requirement: at least three tests executed: normal, malformed, min-revenue
    if not isinstance(candidate_tests, list):
        tests_details.append({"aspect": "tests_array", "awarded": 0.0, "max": 4.0, "messages": ["tests field missing or not an array."]})
    else:
        # Count presence
        present_required = sum(1 for name in REQUIRED_TEST_NAMES if find_test_by_name(candidate_tests, name) is not None)
        awarded_presence = round((present_required / len(REQUIRED_TEST_NAMES)) * 4.0, 2)
        tests_awarded += awarded_presence
        tests_details.append({"aspect": "required_tests_present", "awarded": awarded_presence, "max": 4.0, "messages": [f"{present_required}/{len(REQUIRED_TEST_NAMES)} required tests present."]})

        # Check that actual_output fields are present for each required test
        actuals_present = 0
        for name in REQUIRED_TEST_NAMES:
            t = find_test_by_name(candidate_tests, name)
            if t and safe_get(t, 'actual_output', '') != "":
                actuals_present += 1
        awarded_actuals = round((actuals_present / len(REQUIRED_TEST_NAMES)) * 3.0, 2)
        tests_awarded += awarded_actuals
        tests_details.append({"aspect": "actual_output_captured", "awarded": awarded_actuals, "max": 3.0, "messages": [f"actual_output present for {actuals_present}/{len(REQUIRED_TEST_NAMES)} required tests."]})

        # Check passed booleans - small portion
        passed_count = 0
        for name in REQUIRED_TEST_NAMES:
            t = find_test_by_name(candidate_tests, name)
            if t and safe_get(t, 'passed', False):
                passed_count += 1
        awarded_passed = round((passed_count / len(REQUIRED_TEST_NAMES)) * 3.0, 2)
        tests_awarded += awarded_passed
        tests_details.append({"aspect": "passed_flags", "awarded": awarded_passed, "max": 3.0, "messages": [f"{passed_count}/{len(REQUIRED_TEST_NAMES)} required tests marked passed."]})

    # Cap and record
    if tests_awarded > tests_max:
        tests_awarded = tests_max
    results["criteria"]["tests_evidence"] = {
        "score": round(tests_awarded, 2),
        "max": tests_max,
        "details": tests_details
    }
    total_awarded += tests_awarded

    # ---------- Minor revision implemented & documented (5 pts) ----------
    rev_awarded = 0.0
    rev_max = WEIGHTS["revision"]
    rev_details = []

    # Check changelog mentions min-revenue or filter
    changelog_entries = candidate_json.get("changelog", [])
    changelog_mentions_min = False
    if isinstance(changelog_entries, list):
        for entry in changelog_entries:
            entry_text = json.dumps(entry)
            if string_contains_any(entry_text, ["--min-revenue", "min-revenue", "-m", "min_revenue", "filter"]):
                changelog_mentions_min = True
                break

    # Check source_files or build_instructions include min flag mention
    code_mentions_min = False
    for sf in source_files:
        content = safe_get(sf, 'content', '')
        if string_contains_any(content, ["--min-revenue", "min-revenue", "-m", "min_revenue"]):
            code_mentions_min = True
            break
    build_mentions_min = False
    if isinstance(build_instructions, str) and string_contains_any(build_instructions, ["--min-revenue", "-m", "min-revenue"]):
        build_mentions_min = True

    if changelog_mentions_min or code_mentions_min or build_mentions_min:
        rev_awarded = rev_max
        msgs = ["Follow-up --min-revenue revision implemented and/or documented (found in changelog, source, or build instructions)."]
    else:
        msgs = ["No evidence found for --min-revenue revision in changelog, source files, or build_instructions."]

    rev_details.append({"aspect": "min_revenue_revision", "awarded": rev_awarded, "max": rev_max, "messages": msgs})
    results["criteria"]["revision"] = {
        "score": round(rev_awarded, 2),
        "max": rev_max,
        "details": rev_details
    }
    total_awarded += rev_awarded

    # ---------- Final checks and adjustments ----------
    # Check time_spent_minutes <= 90 (if present)
    time_spent = candidate_json.get("time_spent_minutes", None)
    time_msgs = []
    if isinstance(time_spent, int):
        if time_spent > 90:
            time_msgs.append(f"time_spent_minutes = {time_spent} exceeds 90 minute limit. No automatic penalty applied but noted.")
    else:
        time_msgs.append("time_spent_minutes missing or not an integer.")
    if time_msgs:
        results["errors"].extend(time_msgs)

    # Put together summary and deductions (simple list from each criterion analysis where partial credit < max)
    deductions = []
    for crit, info in results["criteria"].items():
        if info["score"] < info["max"]:
            deductions.append({
                "criterion": crit,
                "awarded": info["score"],
                "max": info["max"],
                "reason_excerpt": "; ".join([d.get("messages", [""])[0] if isinstance(d, dict) else str(d) for d in info.get("details", [])[:2]])
            })
    results["deductions"] = deductions

    # Compose overall summary
    summary_lines = []
    summary_lines.append(f"Total awarded points: {round(total_awarded, 2)} / {total_max}")
    summary_lines.append("Per-criterion scores:")
    for crit, info in results["criteria"].items():
        summary_lines.append(f" - {crit}: {info['score']} / {info['max']}")
    summary = "\n".join(summary_lines)

    # Finalize and return
    return finalize_results(results, total_awarded, summary, total_max)


def finalize_results(results, total_awarded, summary_text, total_max=None):
    """Prepare final JSON structure and write test_results.json to script directory."""
    if total_max is None:
        total_max = sum(WEIGHTS.values())
    percentage = round((total_awarded / total_max) * 100, 2) if total_max > 0 else 0.0
    final = {
        "criteria": results.get("criteria", {}),
        "deductions": results.get("deductions", []),
        "errors": results.get("errors", []),
        "summary": summary_text,
        "total_score": round(total_awarded, 2),
        "max_score": total_max,
        "percentage": percentage,
        "overall_score": percentage
    }
    # Save to test_results.json in the same directory as this script
    try:
        script_dir = os.path.abspath(os.path.dirname(__file__))
    except Exception:
        script_dir = os.getcwd()
    out_path = os.path.join(script_dir, "test_results.json")
    try:
        with open(out_path, 'w', encoding='utf-8') as f:
            json.dump(final, f, indent=2, ensure_ascii=False)
    except Exception as e:
        # If writing fails, append error and still return the final structure
        results.setdefault("errors", []).append(f"Failed to write test_results.json: {e}")
    return final


# ========== Entry point ==========
def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py <candidate_submission.json> <answer_key.json>", file=sys.stderr)
        sys.exit(2)
    candidate_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    # Validate files exist
    if not os.path.isfile(candidate_path):
        print(f"Candidate submission file not found: {candidate_path}", file=sys.stderr)
        sys.exit(2)
    if not os.path.isfile(answer_key_path):
        print(f"Answer key file not found: {answer_key_path}", file=sys.stderr)
        sys.exit(2)

    try:
        final = grade_submission(candidate_path, answer_key_path)
        # Print a concise summary to stdout
        pct = final.get("overall_score", 0.0)
        print(f"Grading complete. overall_score = {pct}%")
        print("Detailed results written to test_results.json in the script directory.")
    except Exception:
        print("An unexpected error occurred during grading:", file=sys.stderr)
        traceback.print_exc()
        sys.exit(1)


if __name__ == "__main__":
    main()
