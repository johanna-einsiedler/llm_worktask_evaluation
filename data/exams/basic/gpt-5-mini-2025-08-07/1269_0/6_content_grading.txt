# task_evaluation.py
#!/usr/bin/env python3
"""
Automated grader for the "sales_agg" basic practical exam.

Usage:
    python task_evaluation.py <candidate_submission.json> <answer_key.json>

The script reads the candidate submission and the answer key, compares test outputs,
and grades the submission according to the evaluator rules described in the exam.
It writes results to test_results.json in the current working directory.

Output (test_results.json) contains:
- overall_score (numeric percentage 0-100)
- total_points_earned, total_points_possible
- breakdown per category with earned/max and comments
- detailed test comparisons and deduction reasons

Implementation notes:
- Uses only standard library modules.
- Attempts to be tolerant of minor whitespace differences when comparing outputs.
- Provides clear explanations for deductions.
"""

import json
import os
import sys
from typing import Dict, Any, List, Tuple

# -----------------------------------------------------------------------------
# Helpers
# -----------------------------------------------------------------------------
def load_json(path: str) -> Any:
    """Load JSON from path with error handling."""
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {path}")
    except json.JSONDecodeError as e:
        raise ValueError(f"Invalid JSON in {path}: {e}")

def normalize_output(s: str) -> str:
    """
    Normalize stdout/stderr strings to reduce false mismatches:
    - Strip leading/trailing whitespace
    - Normalize line endings
    - Trim trailing spaces on each line
    """
    if s is None:
        return ""
    # Ensure it's a string
    if not isinstance(s, str):
        s = str(s)
    # Normalize line endings and strip leading/trailing whitespace
    lines = s.replace('\r\n', '\n').replace('\r', '\n').strip().split('\n')
    # Trim trailing/leading spaces from each line
    norm_lines = [ln.rstrip() for ln in lines if ln is not None]
    # Remove possible empty strings at start/end produced after strip
    return "\n".join(norm_lines).strip()

def find_matching_candidate_test(candidate_tests: List[Dict[str, Any]],
                                 answer_test: Dict[str, Any]) -> Tuple[int, Dict[str, Any]]:
    """
    Try to locate a matching test in candidate_tests for a given answer_test.
    Matching strategy:
      1. Match by test_name (exact, case-insensitive)
      2. Match by command (exact)
      3. Match by command tokens presence (e.g., sample_sales.csv, malformed_sales.csv)
    Returns index and test dict if found, otherwise (-1, {}).
    """
    # 1) by name
    ans_name = (answer_test.get('test_name') or "").strip().lower()
    ans_cmd = (answer_test.get('command') or "").strip()
    if ans_name:
        for idx, t in enumerate(candidate_tests):
            if (t.get('test_name') or "").strip().lower() == ans_name:
                return idx, t
    # 2) by exact command
    if ans_cmd:
        for idx, t in enumerate(candidate_tests):
            if (t.get('command') or "").strip() == ans_cmd:
                return idx, t
    # 3) by partial command tokens
    tokens = []
    if 'sample_sales.csv' in ans_cmd:
        tokens.append('sample_sales.csv')
    if 'malformed_sales.csv' in ans_cmd:
        tokens.append('malformed_sales.csv')
    if '--min-revenue' in ans_cmd or '-m' in ans_cmd:
        tokens.append('--min-revenue')
    if not tokens:
        # Fall back to matching by expected_output length or name
        for idx, t in enumerate(candidate_tests):
            if (t.get('test_name') or "").strip().lower() == ans_name:
                return idx, t
        return -1, {}
    for idx, t in enumerate(candidate_tests):
        cmd = (t.get('command') or "").strip()
        if all(tok in cmd for tok in tokens):
            return idx, t
    return -1, {}

def safe_get(d: Dict[str, Any], key: str, default=None):
    return d.get(key, default) if isinstance(d, dict) else default

# -----------------------------------------------------------------------------
# Grading logic
# -----------------------------------------------------------------------------
def grade(candidate: Dict[str, Any], answer_key: Dict[str, Any]) -> Dict[str, Any]:
    """
    Compute grading results based on candidate submission and answer key.
    Returns a detailed results dict to be serialized to JSON.
    """
    results = {
        "total_points_possible": 100.0,
        "total_points_earned": 0.0,
        "breakdown": [],
        "test_details": [],
        "issues": [],
    }

    # Validate presence of top-level required fields in candidate submission
    required_fields = [
        "candidate_name", "language", "time_spent_minutes", "source_files",
        "build_instructions", "readme", "changelog", "tests",
        "run_output_summary", "assumptions", "comments_coverage",
        "self_score_estimate", "signature"
    ]
    missing_fields = [f for f in required_fields if f not in candidate]
    if missing_fields:
        results['issues'].append(f"Missing required fields in submission: {missing_fields}")

    # Extract candidate tests and answer_key tests
    cand_tests = candidate.get('tests', []) if isinstance(candidate.get('tests'), list) else []
    ans_tests = answer_key.get('tests', []) if isinstance(answer_key.get('tests'), list) else []

    # --- Category 1: Correctness (50 points) ---
    correctness_max = 50.0
    correctness_score = 0.0
    correctness_comments = []
    # Distribute points: normal_sample 20, malformed_sample 20, min_revenue_filter 10
    per_test_weight = {
        "normal_sample": 20.0,
        "malformed_sample": 20.0,
        "min_revenue_filter": 10.0
    }
    # Build a lookup in answer key by test_name
    answer_tests_by_name = { (t.get('test_name') or '').strip(): t for t in ans_tests }

    for key_name, pts in per_test_weight.items():
        ans_test = answer_tests_by_name.get(key_name)
        if not ans_test:
            # If answer key lacks this test, cannot award points; log issue
            correctness_comments.append(f"Answer key missing expected test '{key_name}'; cannot grade this subtask.")
            continue
        idx, cand_test = find_matching_candidate_test(cand_tests, ans_test)
        if idx == -1:
            correctness_comments.append(f"Candidate missing test '{key_name}' (expected command: '{ans_test.get('command')}'). 0/{pts} points.")
            continue
        # Compare outputs
        expected = normalize_output(ans_test.get('expected_output', ''))
        actual = normalize_output(cand_test.get('actual_output', ''))
        passed_flag = bool(cand_test.get('passed') is True)
        # Primary check: compare normalized outputs exactly
        if expected == actual:
            correctness_score += pts
            correctness_comments.append(f"Test '{key_name}': PASS ({pts}/{pts})")
        else:
            # If outputs differ but candidate claimed passed, still record mismatch
            # Allow tolerance: whitespace normalization already done. No fuzzy numeric tolerance applied.
            correctness_comments.append(f"Test '{key_name}': FAIL (0/{pts}). Expected vs Actual differ.")
            correctness_comments.append({
                "test_name": key_name,
                "expected": expected,
                "actual": actual
            })
        # Record per-test detail
        results['test_details'].append({
            "test_name": key_name,
            "expected_output": expected,
            "actual_output": actual,
            "passed_flag": passed_flag,
            "points_awarded": pts if expected == actual else 0.0,
            "points_possible": pts
        })

    results['breakdown'].append({
        "category": "Correctness",
        "earned": round(correctness_score, 2),
        "max": correctness_max,
        "comments": correctness_comments
    })
    results['total_points_earned'] += correctness_score

    # --- Category 2: Documentation (20 points) ---
    doc_max = 20.0
    doc_score = 0.0
    doc_comments = []

    # Sub-allocations:
    # README with usage and assumptions: 8
    # build_instructions includes three commands / sample filenames and --min-revenue: 6
    # changelog present and includes at least 1 entry: 3
    # assumptions present non-empty: 3

    # README check
    readme = candidate.get('readme', '')
    if isinstance(readme, str) and readme.strip():
        lower = readme.lower()
        if 'sample_sales.csv' in lower or 'how to run' in lower or 'usage' in lower or 'run the script' in lower:
            doc_score += 8.0
            doc_comments.append("README: contains usage/sample info (8/8).")
        else:
            doc_score += 4.0
            doc_comments.append("README: present but missing explicit usage/sample info (4/8).")
    else:
        doc_comments.append("README missing or empty (0/8).")

    # build_instructions check
    build_instr = candidate.get('build_instructions', '')
    if isinstance(build_instr, str) and build_instr.strip():
        tokens = ['sample_sales.csv', 'malformed_sales.csv', '--min-revenue']
        found = [tok for tok in tokens if tok in build_instr]
        if len(found) == 3:
            doc_score += 6.0
            doc_comments.append("build_instructions: includes sample, malformed, and --min-revenue commands (6/6).")
        elif len(found) >= 1:
            doc_score += 3.0
            doc_comments.append(f"build_instructions: partial coverage (found tokens: {found}) (3/6).")
        else:
            doc_comments.append("build_instructions present but did not include required tokens (0/6).")
    else:
        doc_comments.append("build_instructions missing or empty (0/6).")

    # changelog check
    changelog = candidate.get('changelog', [])
    if isinstance(changelog, list) and len(changelog) >= 1:
        # Prefer to see at least two entries and mention of revision (but award partial if one)
        if any(('min-revenue' in json.dumps(entry).lower() or 'filter' in json.dumps(entry).lower() or '-m' in json.dumps(entry).lower()) for entry in changelog):
            doc_score += 3.0
            doc_comments.append("changelog: present and documents revision/feature (3/3).")
        else:
            doc_score += 2.0
            doc_comments.append("changelog: present but no explicit mention of revision/filter (2/3).")
    else:
        doc_comments.append("changelog missing or empty (0/3).")

    # assumptions check
    assumptions = candidate.get('assumptions', [])
    if isinstance(assumptions, list) and len(assumptions) >= 1:
        doc_score += 3.0
        doc_comments.append("assumptions: provided (3/3).")
    else:
        doc_comments.append("assumptions missing or empty (0/3).")

    results['breakdown'].append({
        "category": "Documentation",
        "earned": round(doc_score, 2),
        "max": doc_max,
        "comments": doc_comments
    })
    results['total_points_earned'] += doc_score

    # --- Category 3: Code comments & clarity (15 points) ---
    comments_max = 15.0
    comments_score = 0.0
    comments_notes = []

    # Check comments_coverage field
    comments_coverage = (candidate.get('comments_coverage') or "")
    if isinstance(comments_coverage, str) and comments_coverage.strip():
        comments_score += 5.0
        comments_notes.append("comments_coverage: self-assessment provided (5/5).")
    else:
        comments_notes.append("comments_coverage missing or empty (0/5).")

    # Check source_files presence and comment markers in source content
    source_files = candidate.get('source_files', [])
    if isinstance(source_files, list) and len(source_files) >= 1:
        # Look for comment markers in source content based on language heuristics
        found_comment = False
        total_comment_count = 0
        for sf in source_files:
            content = sf.get('content', '') or ''
            lang = (sf.get('language') or '').lower()
            if not content:
                continue
            # Heuristics: for Python look for '#' or triple quotes; for others look for '//' or '/*'
            if lang.startswith('py') or sf.get('filename', '').endswith('.py'):
                count_hash = content.count('#')
                count_doc = content.count('"""') + content.count("'''")
                total_comment_count += count_hash + count_doc
                if count_hash + count_doc > 0:
                    found_comment = True
            else:
                # generic search
                if ('//' in content) or ('/*' in content) or ('#' in content):
                    total_comment_count += content.count('//') + content.count('/*') + content.count('#')
                    found_comment = True
        if found_comment and total_comment_count >= 3:
            comments_score += 10.0
            comments_notes.append(f"source_files: found {total_comment_count} comment markers (10/10).")
        elif found_comment:
            comments_score += 5.0
            comments_notes.append(f"source_files: comments present but sparse ({total_comment_count} markers) (5/10).")
        else:
            comments_notes.append("source_files: no comment markers found (0/10).")
    else:
        comments_notes.append("source_files missing or empty (0/10).")

    results['breakdown'].append({
        "category": "Code comments & clarity",
        "earned": round(comments_score, 2),
        "max": comments_max,
        "comments": comments_notes
    })
    results['total_points_earned'] += comments_score

    # --- Category 4: Tests and evidence of execution (10 points) ---
    tests_max = 10.0
    tests_score = 0.0
    tests_comments = []

    # At least two tests executed (normal + malformed) -> 3 points
    if isinstance(cand_tests, list) and len(cand_tests) >= 2:
        tests_score += 3.0
        tests_comments.append("At least two test runs present (3/3).")
    else:
        tests_comments.append("Fewer than two test runs provided (0/3).")

    # Inclusion of three specific tests (presence of commands or names) -> 3 points
    required_test_names = ['normal_sample', 'malformed_sample', 'min_revenue_filter']
    found_required = []
    for rt in required_test_names:
        # Try to find by name or by command containing expected file/token
        match_idx, _ = find_matching_candidate_test(cand_tests, {"test_name": rt, "command": ""})
        if match_idx != -1:
            found_required.append(rt)
    if len(found_required) == 3:
        tests_score += 3.0
        tests_comments.append("All three required tests present (normal, malformed, min-revenue) (3/3).")
    elif len(found_required) >= 1:
        tests_score += 1.5
        tests_comments.append(f"Partial required tests present: {found_required} (1.5/3).")
    else:
        tests_comments.append("Required tests not found (0/3).")

    # actual_output included for each found test -> 2 points
    actuals_present = 0
    for t in cand_tests:
        ao = t.get('actual_output')
        if ao is not None and str(ao).strip() != '':
            actuals_present += 1
    if actuals_present >= 3:
        tests_score += 2.0
        tests_comments.append("Actual outputs captured for >=3 tests (2/2).")
    elif actuals_present >= 1:
        tests_score += 1.0
        tests_comments.append(f"Actual outputs captured for {actuals_present} tests (1/2).")
    else:
        tests_comments.append("No actual outputs captured (0/2).")

    # passed booleans true for the required tests -> 2 points
    passed_count = 0
    for rt in required_test_names:
        ans_test = answer_tests_by_name.get(rt)
        if not ans_test:
            continue
        idx, ct = find_matching_candidate_test(cand_tests, ans_test)
        if idx != -1:
            if bool(ct.get('passed') is True):
                passed_count += 1
    if passed_count == 3:
        tests_score += 2.0
        tests_comments.append("Candidate marked all required tests as passed (2/2).")
    elif passed_count >= 1:
        tests_score += 1.0
        tests_comments.append(f"Candidate marked {passed_count} required tests as passed (1/2).")
    else:
        tests_comments.append("Candidate did not mark required tests as passed (0/2).")

    results['breakdown'].append({
        "category": "Tests and evidence of execution",
        "earned": round(tests_score, 2),
        "max": tests_max,
        "comments": tests_comments
    })
    results['total_points_earned'] += tests_score

    # --- Category 5: Minor revision implemented & documented (5 points) ---
    revision_max = 5.0
    revision_score = 0.0
    revision_comments = []

    # Check changelog for mention of min-revenue or filter or -m
    changelog_text = ""
    if isinstance(changelog, list):
        # Join changelog entries into a single string for searching
        try:
            changelog_text = " ".join(json.dumps(entry).lower() for entry in changelog)
        except Exception:
            changelog_text = " ".join(str(entry).lower() for entry in changelog)
    if any(tok in changelog_text for tok in ['min-revenue', '--min-revenue', '-m', 'filter']):
        revision_score += revision_max
        revision_comments.append("Changelog documents the --min-revenue revision (5/5).")
    else:
        revision_comments.append("Changelog does not explicitly mention the --min-revenue revision (0/5).")

    results['breakdown'].append({
        "category": "Minor revision implemented & documented",
        "earned": round(revision_score, 2),
        "max": revision_max,
        "comments": revision_comments
    })
    results['total_points_earned'] += revision_score

    # Finalize totals and overall score
    results['total_points_possible'] = 100.0
    # Protect against floating small fractions
    total_earned = round(results['total_points_earned'], 2)
    results['total_points_earned'] = total_earned
    overall_score = round((total_earned / results['total_points_possible']) * 100.0, 2) if results['total_points_possible'] > 0 else 0.0
    results['overall_score'] = overall_score
    results['pass'] = overall_score >= 80.0

    # Provide summary messages and highlight mismatches from correctness
    if correctness_score < correctness_max:
        results['issues'].append("One or more required tests did not match expected outputs exactly. See test_details for diffs.")
    if doc_score < doc_max:
        results['issues'].append("Documentation incomplete or missing elements (README, build_instructions, changelog, assumptions).")
    if comments_score < comments_max:
        results['issues'].append("Code comments coverage appears insufficient based on submission metadata and source files.")
    if revision_score < revision_max:
        results['issues'].append("Revision (--min-revenue) not clearly documented in changelog; ensure you document follow-up changes.")

    # Attach candidate metadata for convenience
    results['candidate_summary'] = {
        "candidate_name": candidate.get('candidate_name'),
        "language": candidate.get('language'),
        "time_spent_minutes": candidate.get('time_spent_minutes'),
        "signature": candidate.get('signature')
    }

    return results

# -----------------------------------------------------------------------------
# Main entry
# -----------------------------------------------------------------------------
def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py <candidate_submission.json> <answer_key.json>")
        sys.exit(2)

    candidate_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    # Load files
    try:
        candidate = load_json(candidate_path)
    except Exception as e:
        print(f"Error loading candidate submission: {e}", file=sys.stderr)
        sys.exit(2)

    try:
        answer_key = load_json(answer_key_path)
    except Exception as e:
        print(f"Error loading answer key: {e}", file=sys.stderr)
        sys.exit(2)

    # Grade
    try:
        results = grade(candidate, answer_key)
    except Exception as e:
        # Catch unexpected grading errors and write a minimal results file
        tb_msg = f"Grading error: {e}"
        results = {
            "total_points_possible": 100.0,
            "total_points_earned": 0.0,
            "overall_score": 0.0,
            "breakdown": [],
            "issues": [tb_msg],
            "candidate_summary": {
                "candidate_name": candidate.get('candidate_name') if isinstance(candidate, dict) else None
            }
        }
        print(f"Error during grading: {e}", file=sys.stderr)

    # Save results to test_results.json in current directory (same directory as script)
    out_path = os.path.join(os.getcwd(), "test_results.json")
    try:
        with open(out_path, 'w', encoding='utf-8') as f:
            json.dump(results, f, indent=2, ensure_ascii=False)
        # Also print summary to stdout
        print(f"Grading complete. Overall score: {results.get('overall_score')}%")
        print(f"Detailed results written to {out_path}")
    except Exception as e:
        print(f"Failed to write results file: {e}", file=sys.stderr)
        sys.exit(2)

if __name__ == "__main__":
    main()
