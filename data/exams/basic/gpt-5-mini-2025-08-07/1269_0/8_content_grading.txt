# task_evaluation.py
#!/usr/bin/env python3
"""
Automated grader for the "sales_agg" basic practical exam.

Usage:
    python task_evaluation.py <candidate_submission.json> <answer_key.json>

Produces:
    test_results.json  -- detailed scoring breakdown and overall_score (0-100)

This grader:
- Loads candidate submission JSON and the official answer key JSON.
- Performs automated checks according to the evaluation rubric in the exam spec.
- Is tolerant to minor whitespace differences in stdout comparisons.
- Reports per-criterion scores and explanations for deductions.

Implementation notes:
- Uses only Python standard library modules.
- Deterministic and contains defensive error handling for malformed submissions.
"""

import json
import os
import sys
from typing import Any, Dict, List

# ---------- Utility helpers ----------

def load_json_file(path: str) -> Any:
    """Load JSON from path with basic error handling."""
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except FileNotFoundError:
        raise
    except json.JSONDecodeError as e:
        raise ValueError(f"Invalid JSON in {path}: {e}")

def normalize_output(s: str) -> str:
    """Normalize stdout/stderr strings for tolerant comparison.

    - Convert None -> empty string.
    - Split into lines, strip each line of leading/trailing whitespace.
    - Remove empty lines at start/end (but keep interior blank lines if any).
    - Join with '\n'.
    """
    if s is None:
        return ""
    # Ensure string
    if not isinstance(s, str):
        s = str(s)
    lines = s.splitlines()
    # Strip whitespace from each line
    stripped = [ln.strip() for ln in lines]
    # Remove leading/trailing empty lines
    # Find first non-empty index
    first = 0
    last = len(stripped)
    while first < last and stripped[first] == "":
        first += 1
    while last > first and stripped[last - 1] == "":
        last -= 1
    mid = stripped[first:last]
    return "\n".join(mid)

def find_candidate_test(candidate_tests: List[Dict], expected_test: Dict) -> Dict:
    """Try to locate the best-matching candidate test entry for a given expected test (by name or command)."""
    exp_name = expected_test.get("test_name", "").lower()
    exp_cmd = expected_test.get("command", "")
    # First pass: exact test_name match (case-insensitive)
    for t in candidate_tests:
        if isinstance(t.get("test_name", None), str) and t.get("test_name", "").lower() == exp_name:
            return t
    # Second pass: command contains the same key file names or keywords
    exp_tokens = []
    if exp_cmd:
        exp_tokens = [token.strip() for token in exp_cmd.split() if token.strip()]
    for t in candidate_tests:
        cmd = t.get("command", "") or ""
        # If both commands mention at least one token in common, consider it a match
        for tok in exp_tokens:
            if tok in cmd:
                return t
    # Third pass: try to find by presence of sample filename strings
    for t in candidate_tests:
        cmd = t.get("command", "") or ""
        if "sample_sales.csv" in cmd and "sample_sales.csv" in exp_cmd:
            return t
    # No match found; return empty dict
    return {}

# ---------- Main grading routine ----------

def grade(candidate_path: str, answer_key_path: str) -> Dict[str, Any]:
    """Main function performing grading and returning a results dict to be written to JSON."""
    results: Dict[str, Any] = {
        "candidate_file": candidate_path,
        "answer_key_file": answer_key_path,
        "errors": [],
        "breakdown": {},
        "total_points": 0.0,
        "max_points": 100.0,
        "overall_score": 0.0  # will be set later
    }

    # Try loading files
    try:
        candidate = load_json_file(candidate_path)
    except FileNotFoundError:
        results["errors"].append(f"Candidate submission file not found: {candidate_path}")
        return results
    except Exception as e:
        results["errors"].append(f"Failed reading candidate JSON: {e}")
        return results

    try:
        answer_key = load_json_file(answer_key_path)
    except FileNotFoundError:
        results["errors"].append(f"Answer key file not found: {answer_key_path}")
        return results
    except Exception as e:
        results["errors"].append(f"Failed reading answer key JSON: {e}")
        return results

    # Validate candidate basic keys presence
    required_fields = [
        "candidate_name", "language", "time_spent_minutes", "source_files", "build_instructions",
        "readme", "changelog", "tests", "run_output_summary", "assumptions",
        "comments_coverage", "self_score_estimate", "signature"
    ]
    missing = [f for f in required_fields if f not in candidate]
    if missing:
        results["errors"].append(f"Missing required fields in submission JSON: {missing}")
        # Continue but deductions will be applied
    # Extract candidate fields safely
    candidate_tests = candidate.get("tests", [])
    candidate_source_files = candidate.get("source_files", [])
    build_instructions = candidate.get("build_instructions", "")
    readme = candidate.get("readme", "")
    changelog = candidate.get("changelog", [])
    comments_coverage_field = candidate.get("comments_coverage", "")
    assumptions = candidate.get("assumptions", [])
    time_spent = candidate.get("time_spent_minutes", None)

    # Prepare answer key tests expected outputs
    answer_tests = answer_key.get("tests", [])
    # Ensure answer key has tests
    if not answer_tests:
        results["errors"].append("Answer key contains no tests to compare against.")
        return results

    # ---------- Scoring configuration ----------
    # Weights and max points per criterion (total 100)
    weights = {
        "correctness": 50.0,
        "documentation": 20.0,
        "comments": 15.0,
        "tests_evidence": 10.0,
        "revision": 5.0
    }

    # Breakdown initializations
    breakdown = {}

    # ---------- 1) Correctness (50 pts) ----------
    # Split correctness across three tests:
    # - normal_sample: 25 pts
    # - malformed_sample: 15 pts
    # - min_revenue_filter: 10 pts
    correctness_points = 0.0
    correctness_details = []
    # Map expected test names to allocated points
    expected_test_point_map = {
        "normal_sample": 25.0,
        "malformed_sample": 15.0,
        "min_revenue_filter": 10.0
    }

    # Build a lookup of answer key expected outputs by test_name (lowercased)
    answer_test_lookup = {}
    for t in answer_tests:
        name = t.get("test_name", "").lower()
        answer_test_lookup[name] = t

    # For each expected test, find candidate test and compare outputs
    for expected_name, pts in expected_test_point_map.items():
        expected = answer_test_lookup.get(expected_name)
        detail = {"expected_test_name": expected_name, "max_points": pts, "points_awarded": 0.0, "reasons": []}
        if not expected:
            detail["reasons"].append(f"Answer key missing expected test: {expected_name}")
            correctness_details.append(detail)
            continue
        # Find matching candidate test
        found = find_candidate_test(candidate_tests, expected)
        if not found:
            detail["reasons"].append("No corresponding test run found in candidate submission.")
            correctness_details.append(detail)
            continue
        # Compare actual_output to expected_output (use tolerant normalize)
        exp_out = expected.get("expected_output", "")
        cand_actual = found.get("actual_output", "")
        norm_exp = normalize_output(exp_out)
        norm_act = normalize_output(cand_actual)
        if norm_exp == norm_act:
            detail["points_awarded"] = pts
            detail["reasons"].append("Output matches expected output exactly after normalization.")
            correctness_points += pts
        else:
            # Partial credit: attempt line-wise match ratio
            exp_lines = [ln for ln in norm_exp.splitlines() if ln != ""]
            act_lines = [ln for ln in norm_act.splitlines() if ln != ""]
            matches = 0
            for ln in exp_lines:
                if ln in act_lines:
                    matches += 1
            if exp_lines:
                ratio = matches / len(exp_lines)
            else:
                ratio = 0.0
            awarded = round(pts * ratio, 2)
            detail["points_awarded"] = awarded
            correctness_points += awarded
            detail["reasons"].append(
                f"Output mismatch. {matches}/{len(exp_lines)} expected lines found in actual output. "
                f"Partial credit {awarded}/{pts} awarded."
            )
            detail["expected_normalized"] = norm_exp
            detail["actual_normalized"] = norm_act
        correctness_details.append(detail)

    breakdown["correctness"] = {
        "max_points": weights["correctness"],
        "points_awarded": round(correctness_points, 2),
        "details": correctness_details
    }

    # ---------- 2) Documentation (20 pts) ----------
    # Sub-allocations:
    # - README present & non-empty: 10 pts
    # - README contains run examples / builds for three tests: 5 pts
    # - Changelog presence and contains revision entries: 5 pts
    doc_points = 0.0
    doc_details = []

    # README non-empty
    if isinstance(readme, str) and readme.strip():
        doc_points += 10.0
        doc_details.append({"item": "README_present", "points": 10.0, "reason": "README present and non-empty."})
    else:
        doc_details.append({"item": "README_present", "points": 0.0, "reason": "README missing or empty."})

    # README contains run examples for sample and malformed and min-revenue
    run_cmd_tokens = ["sample_sales.csv", "malformed_sales.csv", "--min-revenue"]
    found_tokens = []
    for tok in run_cmd_tokens:
        if tok in (readme or "") or tok in (build_instructions or ""):
            found_tokens.append(tok)
    # Award partial points proportional to tokens found
    token_pts = 5.0 * (len(found_tokens) / len(run_cmd_tokens))
    token_pts = round(token_pts, 2)
    doc_points += token_pts
    doc_details.append({"item": "README_and_build_examples", "points": token_pts,
                        "reason": f"Found tokens for run examples: {found_tokens}"})

    # Changelog presence and at least 2 entries and mention of revision (--min-revenue)
    changelog_pts = 0.0
    if isinstance(changelog, list) and len(changelog) >= 1:
        # Award half if any entries present
        changelog_pts += 2.5
        # Additional half if >=2 entries
        if len(changelog) >= 2:
            changelog_pts += 1.5
        # Additional if one entry mentions '--min-revenue' or 'min-revenue' or 'filter'
        mentions_revision = False
        for entry in changelog:
            text = (entry.get("change", "") + " " + entry.get("reason", "")).lower()
            if "--min-revenue" in text or "min-revenue" in text or "min revenue" in text or "filter" in text:
                mentions_revision = True
                break
        if mentions_revision:
            changelog_pts += 1.0
        # Cap at 5.0
        changelog_pts = min(changelog_pts, 5.0)
        doc_details.append({"item": "changelog", "points": changelog_pts,
                            "reason": f"{len(changelog)} entries; mentions_revision={mentions_revision}"})
        doc_points += changelog_pts
    else:
        doc_details.append({"item": "changelog", "points": 0.0, "reason": "Changelog missing or empty."})

    # Ensure doc_points doesn't exceed weight
    doc_points = min(doc_points, weights["documentation"])
    breakdown["documentation"] = {"max_points": weights["documentation"], "points_awarded": round(doc_points, 2),
                                 "details": doc_details}

    # ---------- 3) Comments & clarity (15 pts) ----------
    # Sub-allocations:
    # - comments_coverage field present & non-empty: 5 pts
    # - source file(s) contain inline comments or docstrings: 10 pts
    comments_points = 0.0
    comments_details = []
    if isinstance(comments_coverage_field, str) and comments_coverage_field.strip():
        comments_points += 5.0
        comments_details.append({"item": "comments_coverage_field", "points": 5.0,
                                 "reason": "comments_coverage provided and non-empty."})
    else:
        comments_details.append({"item": "comments_coverage_field", "points": 0.0,
                                 "reason": "comments_coverage field missing or empty."})

    # Scan source_files for comments or docstrings
    found_comment_like = False
    total_source_files = 0
    comment_files = 0
    for sf in candidate_source_files:
        if not isinstance(sf, dict):
            continue
        content = sf.get("content", "")
        total_source_files += 1
        if not isinstance(content, str):
            continue
        # Heuristics: look for '#' (python style), '/*' or '//' (c/java/go), or triple quotes for docstrings
        if ("#" in content) or ("/*" in content) or ("//" in content) or ("\"\"\"" in content) or ("'''"
                                                                                        in content):
            comment_files += 1
            found_comment_like = True
    if total_source_files == 0:
        comments_details.append({"item": "source_files", "points": 0.0,
                                 "reason": "No source files provided."})
    else:
        # award proportionally: up to 10 points if most files have comments
        ratio = comment_files / total_source_files
        awarded = round(10.0 * ratio, 2)
        comments_points += awarded
        comments_details.append({"item": "source_files_comments", "points": awarded,
                                 "reason": f"{comment_files}/{total_source_files} source files contain comment-like tokens."})

    # Cap comments_points to allocated weight
    comments_points = min(comments_points, weights["comments"])
    breakdown["comments"] = {"max_points": weights["comments"], "points_awarded": round(comments_points, 2),
                             "details": comments_details}

    # ---------- 4) Tests & evidence (10 pts) ----------
    # Allocate equally among three required runs: normal, malformed, min_revenue (3.333... each)
    tests_points = 0.0
    tests_details = []
    required_test_names = ["normal_sample", "malformed_sample", "min_revenue_filter"]
    per_test_pts = round(weights["tests_evidence"] / len(required_test_names), 4)  # approx 3.3333
    candidate_tests_len = len(candidate_tests) if isinstance(candidate_tests, list) else 0

    if candidate_tests_len == 0:
        tests_details.append({"item": "tests_present", "points": 0.0, "reason": "No tests array or empty tests in submission."})
    else:
        tests_details.append({"item": "tests_present", "points": min(4.0, weights["tests_evidence"]),
                              "reason": f"{candidate_tests_len} test run(s) provided."})
    # For each required test, check existence and fields and whether passed per our normalized check against answer key
    for test_name in required_test_names:
        detail = {"test_name": test_name, "max_points": per_test_pts, "points_awarded": 0.0, "reasons": []}
        expected = answer_test_lookup.get(test_name)
        if not expected:
            detail["reasons"].append("No expected test in answer key for this test.")
            tests_details.append(detail)
            continue
        cand_test = find_candidate_test(candidate_tests, expected)
        if not cand_test:
            detail["reasons"].append("Candidate did not include this test run.")
            tests_details.append(detail)
            continue
        # Check presence of required fields in candidate test object
        missing_fields = [f for f in ("command", "expected_output", "actual_output", "passed") if f not in cand_test]
        if missing_fields:
            detail["reasons"].append(f"Candidate test missing fields: {missing_fields}")
            tests_details.append(detail)
            continue
        # Compare actual -> expected (from answer key)
        norm_exp = normalize_output(expected.get("expected_output", ""))
        norm_act = normalize_output(cand_test.get("actual_output", ""))
        if norm_exp == norm_act:
            detail["points_awarded"] = per_test_pts
            detail["reasons"].append("Actual output matches expected output (after normalization).")
            tests_points += per_test_pts
        else:
            # no partial credit here beyond correctness logic handled earlier (but give small credit if some lines match)
            exp_lines = [ln for ln in norm_exp.splitlines() if ln != ""]
            act_lines = [ln for ln in norm_act.splitlines() if ln != ""]
            matches = sum(1 for ln in exp_lines if ln in act_lines) if exp_lines else 0
            if exp_lines:
                ratio = matches / len(exp_lines)
            else:
                ratio = 0.0
            partial = round(per_test_pts * ratio, 3)
            if partial > 0:
                detail["points_awarded"] = partial
                tests_points += partial
                detail["reasons"].append(f"Partial match: {matches}/{len(exp_lines)} expected lines found; awarded {partial:.3f}/{per_test_pts}.")
            else:
                detail["reasons"].append("Outputs do not match expected output.")
            detail["expected_normalized"] = norm_exp
            detail["actual_normalized"] = norm_act
        tests_details.append(detail)

    # Cap tests_points to the weight
    tests_points = min(tests_points, weights["tests_evidence"])
    breakdown["tests_evidence"] = {"max_points": weights["tests_evidence"], "points_awarded": round(tests_points, 3),
                                  "details": tests_details}

    # ---------- 5) Minor revision implemented & documented (5 pts) ----------
    revision_points = 0.0
    revision_details = []
    # Check for --min-revenue in source files content
    minrev_found_in_source = False
    for sf in candidate_source_files:
        content = sf.get("content", "")
        if isinstance(content, str) and ("--min-revenue" in content or "min_revenue" in content or "min-revenue" in content or " -m " in content):
            minrev_found_in_source = True
            break
    # Check build_instructions includes min-revenue usage
    minrev_in_build = "--min-revenue" in (build_instructions or "") or "-m" in (build_instructions or "")
    # Check tests include min_revenue_filter (and that it passed)
    minrev_test = None
    for t in candidate_tests:
        if t.get("test_name", "").lower() == "min_revenue_filter" or ("--min-revenue" in (t.get("command") or "")):
            minrev_test = t
            break
    minrev_test_passed = False
    if minrev_test:
        # Compare to answer key expected
        expected = answer_test_lookup.get("min_revenue_filter")
        if expected:
            norm_exp = normalize_output(expected.get("expected_output", ""))
            norm_act = normalize_output(minrev_test.get("actual_output", ""))
            if norm_exp == norm_act:
                minrev_test_passed = True

    reasons = []
    if minrev_found_in_source:
        revision_points += 3.0
        reasons.append("Found min-revenue handling in source files (+3).")
    if minrev_in_build:
        revision_points += 1.0
        reasons.append("Found min-revenue mention in build/run instructions (+1).")
    if minrev_test_passed:
        revision_points += 1.0
        reasons.append("Min-revenue test exists and passed (+1).")

    # Cap to max 5
    revision_points = min(revision_points, weights["revision"])
    if revision_points == 0.0:
        revision_details.append({"item": "min_revenue_revision", "points": 0.0,
                                 "reason": "No evidence of implementing/testing --min-revenue found in source, build_instructions or tests."})
    else:
        revision_details.append({"item": "min_revenue_revision", "points": revision_points, "reason": "; ".join(reasons)})

    breakdown["revision"] = {"max_points": weights["revision"], "points_awarded": round(revision_points, 2),
                             "details": revision_details}

    # ---------- Aggregate points ----------
    total_awarded = (
        breakdown["correctness"]["points_awarded"]
        + breakdown["documentation"]["points_awarded"]
        + breakdown["comments"]["points_awarded"]
        + breakdown["tests_evidence"]["points_awarded"]
        + breakdown["revision"]["points_awarded"]
    )
    # Round totals
    total_awarded = round(total_awarded, 3)
    results["breakdown"] = breakdown
    results["total_points"] = total_awarded
    results["max_points"] = 100.0
    # overall_score numeric percentage (0-100)
    overall_score = round((total_awarded / results["max_points"]) * 100.0, 2) if results["max_points"] else 0.0
    results["overall_score"] = overall_score

    # Human-friendly summary of deductions
    deductions = []
    # Missing required fields
    if missing:
        deductions.append(f"Missing required JSON fields: {missing}")
    # If any of the three core correctness tests had zero points, add note
    zero_correctness_tests = [d["expected_test_name"] for d in correctness_details if d.get("points_awarded", 0.0) == 0.0]
    if zero_correctness_tests:
        deductions.append(f"No correct outputs for expected tests: {zero_correctness_tests}")
    # If README missing or changelog missing
    if not (isinstance(readme, str) and readme.strip()):
        deductions.append("README missing or empty.")
    if not (isinstance(changelog, list) and len(changelog) >= 1):
        deductions.append("Changelog missing or empty.")
    # Comments missing
    if breakdown["comments"]["points_awarded"] < weights["comments"] * 0.5:
        deductions.append("Insufficient code comments or comments_coverage self-assessment missing/weak.")

    results["deduction_reasons"] = deductions

    return results

# ---------- Script entry point ----------

def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py <candidate_submission.json> <answer_key.json>", file=sys.stderr)
        sys.exit(2)
    candidate_path = sys.argv[1]
    answer_key_path = sys.argv[2]
    try:
        results = grade(candidate_path, answer_key_path)
    except Exception as e:
        # Catch-all: ensure test_results.json is produced with the error
        results = {
            "errors": [f"Uncaught grader error: {e}"],
            "breakdown": {},
            "total_points": 0.0,
            "max_points": 100.0,
            "overall_score": 0.0
        }
    # Write test_results.json in the same directory as the script (current working directory)
    out_path = os.path.join(os.getcwd(), "test_results.json")
    try:
        with open(out_path, 'w', encoding='utf-8') as f:
            json.dump(results, f, indent=2, ensure_ascii=False)
    except Exception as e:
        print(f"Failed to write results file {out_path}: {e}", file=sys.stderr)
        sys.exit(3)

    # Also print a short summary to stdout
    overall = results.get("overall_score", 0.0)
    total_points = results.get("total_points", 0.0)
    print(f"Grading complete. Overall score: {overall}% ({total_points}/100 points).")
    print(f"Detailed results written to {out_path}")

if __name__ == "__main__":
    main()
