# task_evaluation.py
"""
Automated grader for the Practical Data Ingestion, Querying, Transformation, and Analysis (Basic level) exam.

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

Outputs:
    Writes test_results.json next to this script containing a detailed score breakdown,
    explanations for deductions, and overall_score (percentage 0-100).

Implementation notes:
- Uses only Python standard library.
- Tolerant comparisons for numeric values (floats compared with small tolerance).
- Provides partial credit per rubric described in the exam materials.
"""

import json
import sys
import os
import math
from collections import Counter

# --- Helper utilities ------------------------------------------------------

def load_json(path):
    """Load JSON file robustly with error messages."""
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {path}")
    except json.JSONDecodeError as e:
        raise ValueError(f"Invalid JSON in {path}: {e}")

def is_number(val):
    """Return True if val can be parsed as a float (and is not NaN)."""
    if val is None:
        return False
    try:
        f = float(val)
        return not math.isnan(f)
    except Exception:
        return False

def to_number(val):
    """Convert to float if possible, else return original."""
    if val is None:
        return None
    try:
        f = float(val)
        if math.isnan(f):
            return None
        return f
    except Exception:
        return val

def normalize_value(v):
    """Normalize a value for comparison: strip strings, convert numerics to floats."""
    if isinstance(v, str):
        s = v.strip()
        # Attempt numeric conversion
        if is_number(s):
            return float(s)
        # normalize booleans / null-like
        low = s.lower()
        if low in ('null', 'none', 'na', ''):
            return None
        return s
    elif isinstance(v, (int, float)):
        if isinstance(v, float) and math.isnan(v):
            return None
        return float(v) if isinstance(v, (int, float)) else v
    else:
        return v

def normalize_row(row):
    """Return a normalized dictionary where values are normalized for comparison."""
    if not isinstance(row, dict):
        return row
    return {str(k).strip(): normalize_value(v) for k, v in row.items()}

def rows_match(expected_row, candidate_row, float_tol=0.02):
    """
    Compare two normalized rows.
    For numeric fields, allow relative absolute tolerance (float_tol absolute),
    or for large numbers use relative tolerance.
    Strings compared case-sensitively after stripping.
    None matches None.
    """
    # both should be dicts
    if not isinstance(expected_row, dict) or not isinstance(candidate_row, dict):
        return expected_row == candidate_row

    # keys may be same but ordering may differ; require expected keys are subset of candidate keys
    for k, exp_val in expected_row.items():
        if k not in candidate_row:
            return False
        cand_val = candidate_row[k]
        # Normalize again
        exp = normalize_value(exp_val)
        cand = normalize_value(cand_val)
        # Compare
        if exp is None and cand is None:
            continue
        if isinstance(exp, (int, float)) and isinstance(cand, (int, float)):
            # numeric tolerance: absolute tolerance first
            try:
                exp_f = float(exp)
                cand_f = float(cand)
            except Exception:
                return False
            # If either is zero or small, compare absolute difference
            if abs(exp_f - cand_f) <= float_tol:
                continue
            # relative tolerance
            denom = max(abs(exp_f), abs(cand_f), 1.0)
            if abs(exp_f - cand_f) / denom <= (float_tol):
                continue
            return False
        else:
            # Compare string/other
            if exp == cand:
                continue
            # allow comparison where numbers are strings and vice versa
            if is_number(exp) and is_number(cand):
                # both parseable numbers
                try:
                    if abs(float(exp) - float(cand)) <= float_tol:
                        continue
                except Exception:
                    pass
            # last resort: case-insensitive string compare if both are strings
            if isinstance(exp, str) and isinstance(cand, str):
                if exp.strip().lower() == cand.strip().lower():
                    continue
            return False
    return True

def compare_row_lists(expected_rows, candidate_rows):
    """
    Compute how many expected rows are matched by candidate rows.
    Returns matched_count, expected_count, candidate_count.
    Matching is one-to-one (candidate row used once).
    """
    if expected_rows is None:
        expected_rows = []
    if candidate_rows is None:
        candidate_rows = []

    # Normalize rows
    exp_norm = [normalize_row(r) for r in expected_rows]
    cand_norm = [normalize_row(r) for r in candidate_rows]

    matched = 0
    used = [False] * len(cand_norm)

    for exp in exp_norm:
        found_idx = None
        for i, cand in enumerate(cand_norm):
            if used[i]:
                continue
            if rows_match(exp, cand):
                found_idx = i
                break
        if found_idx is not None:
            matched += 1
            used[found_idx] = True

    return matched, len(exp_norm), len(cand_norm)

def safe_get(dct, *keys, default=None):
    """Get nested keys in dict safely; returns default if any missing."""
    cur = dct
    for k in keys:
        if not isinstance(cur, dict) or k not in cur:
            return default
        cur = cur[k]
    return cur

# --- Grading logic ---------------------------------------------------------

def grade_submission(sub_path, key_path):
    """Main grading orchestration. Returns results dict to be written to JSON."""
    # Load files
    try:
        candidate = load_json(sub_path)
    except Exception as e:
        return {"error": str(e)}

    try:
        answer = load_json(key_path)
    except Exception as e:
        return {"error": str(e)}

    results = {
        "candidate_file": os.path.abspath(sub_path),
        "answer_key_file": os.path.abspath(key_path),
        "scores": {},
        "deductions": [],
        "total_points": 100,
        "points_awarded": 0,
    }

    # --- Check top-level required keys presence (Reproducibility & Packaging small check) ---
    required_top_keys = ["candidate", "time_spent_minutes", "steps", "summary", "reproducibility", "justification"]
    missing_top = [k for k in required_top_keys if k not in candidate]
    if missing_top:
        results["deductions"].append(f"Missing top-level keys in submission: {missing_top}. This reduces reproducibility/packaging score.")
    # Basic structure checks:
    steps = candidate.get("steps", [])
    # We'll expect 4 steps named accordingly; not strictly required but we check presence
    expected_step_names = ["Setup & Import", "Core Queries", "Transformation & Export", "Summary"]
    step_map = {s.get("name",""): s for s in steps if isinstance(s, dict)}

    # --- 1) Setup & Import (20 points) -------------------------------------
    setup_max = 20
    setup_score = 0
    setup_feedback = []

    step_setup = step_map.get("Setup & Import")
    if not step_setup:
        setup_feedback.append("Setup & Import step missing.")
    else:
        artifacts = step_setup.get("artifacts", {})
        code = artifacts.get("code", "")
        commands = artifacts.get("commands", [])
        outputs = artifacts.get("outputs", {})

        # 1a Schema is appropriate and justified (8 pts)
        schema_pts = 0
        schema_reasons = []
        # Heuristic: presence of CREATE TABLE or mention of table names / created_tables
        created_tables = outputs.get("created_tables") or candidate.get("reproducibility", {}).get("files_created") or []
        create_stmt = outputs.get("create_table_statements", "") or code or ""
        if "CREATE TABLE" in create_stmt.upper() or "CREATE TABLE" in (code or "").upper() or created_tables:
            schema_pts = 8
            schema_reasons.append("Found CREATE TABLE statements or declared created_tables.")
        else:
            # partial credit if code mentions events or components insert
            if "events" in create_stmt.lower() or "component" in create_stmt.lower():
                schema_pts = 5
                schema_reasons.append("No explicit CREATE TABLE found, but code mentions target tables (partial credit).")
            else:
                schema_pts = 1
                schema_reasons.append("No CREATE TABLE or table mention found; very low credit.")
        setup_score += schema_pts
        setup_feedback.append(f"Schema points: {schema_pts}/8 - {'; '.join(schema_reasons)}")

        # 1b Data ingestion and handling malformed rows (7 pts)
        ingestion_pts = 0
        ingestion_reasons = []
        total_rows = outputs.get("total_rows_in_csv")
        if total_rows == 60:
            ingestion_pts += 3
            ingestion_reasons.append("Reported total_rows_in_csv == 60.")
        else:
            ingestion_reasons.append(f"Reported total_rows_in_csv = {total_rows}; expected 60.")

        mal_metric = outputs.get("malformed_metric_row_ids", [])
        mal_ts = outputs.get("malformed_timestamp_row_ids", [])
        # award points for correctly reporting expected ids
        exp_mal_metric = {7, 18, 55}
        exp_mal_ts = {10, 58}
        try:
            mal_metric_set = set(int(x) for x in (mal_metric or []))
        except Exception:
            mal_metric_set = set()
        try:
            mal_ts_set = set(int(x) for x in (mal_ts or []))
        except Exception:
            mal_ts_set = set()

        # metric list correctness (2 pts)
        if exp_mal_metric == mal_metric_set:
            ingestion_pts += 2
            ingestion_reasons.append("Correct malformed metric_value row ids reported.")
        else:
            # partial credit proportional to matches
            matched = len(exp_mal_metric & mal_metric_set)
            ingestion_pts += round(2 * matched / len(exp_mal_metric))
            ingestion_reasons.append(f"Malformed metric ids reported {mal_metric_set}; expected {exp_mal_metric}.")

        # timestamp list correctness (2 pts)
        if exp_mal_ts == mal_ts_set:
            ingestion_pts += 2
            ingestion_reasons.append("Correct malformed timestamp row ids reported.")
        else:
            matched_ts = len(exp_mal_ts & mal_ts_set)
            ingestion_pts += round(2 * matched_ts / len(exp_mal_ts))
            ingestion_reasons.append(f"Malformed timestamp ids reported {mal_ts_set}; expected {exp_mal_ts}.")

        # cap to 7
        ingestion_pts = min(7, ingestion_pts)
        setup_score += ingestion_pts
        setup_feedback.append(f"Ingestion points: {ingestion_pts}/7 - {'; '.join(ingestion_reasons)}")

        # 1c Runnable commands/scripts to reproduce import (5 pts)
        reproducible_pts = 0
        repro_reasons = []
        # check presence of commands containing 'sqlite' or 'python' or 'ingest'
        all_cmds = []
        if isinstance(commands, list):
            all_cmds.extend(commands)
        # also check reproducibility.commands top-level
        top_cmds = candidate.get("reproducibility", {}).get("commands", [])
        if isinstance(top_cmds, list):
            all_cmds.extend(top_cmds)

        cmd_text = " ".join(all_cmds).lower()
        if any(tok in cmd_text for tok in ("sqlite3", "python", "ingest.py", ".db", "create.sql")):
            reproducible_pts = 5
            repro_reasons.append("Found reproducibility commands mentioning sqlite3 or python/ingest.")
        elif all_cmds:
            reproducible_pts = 3
            repro_reasons.append("Commands present but do not mention expected tools; partial credit.")
        else:
            reproducible_pts = 0
            repro_reasons.append("No reproducibility commands found.")

        setup_score += reproducible_pts
        setup_feedback.append(f"Reproducibility points: {reproducible_pts}/5 - {'; '.join(repro_reasons)}")

    # finalize setup
    results["scores"]["setup_import"] = {"score": setup_score, "max": setup_max, "notes": setup_feedback}
    results["points_awarded"] += setup_score

    # --- 2) Core Retrieval Queries (40 points) -------------------------------
    core_max = 40
    core_score = 0
    core_feedback = []

    # Expect step 'Core Queries'
    step_core = step_map.get("Core Queries")
    if not step_core:
        core_feedback.append("Core Queries step missing.")
    else:
        artifacts = step_core.get("artifacts", {})
        outputs = artifacts.get("outputs", {})
        # Answer key likely has expected outputs under answer['steps'][1] ... try to find corresponding
        # Search in answer for a Core Queries step
        answer_core_step = None
        for s in safe_get(answer, "steps", default=[]):
            if isinstance(s, dict) and s.get("name") == "Core Queries":
                answer_core_step = s
                break
        # If not found, try answer['steps'][1]
        if not answer_core_step:
            ans_steps = answer.get("steps", [])
            if len(ans_steps) > 1 and isinstance(ans_steps[1], dict):
                answer_core_step = ans_steps[1]

        # For grading correctness (25 pts): 5 queries * 5 pts each
        query_correctness_pts = 0
        qc_notes = []
        if answer_core_step:
            ans_outputs = safe_get(answer_core_step, "artifacts", "outputs", default={})
            # iterate queries 1..5
            for i in range(1, 6):
                q_key = f"query_{i}"
                expected = ans_outputs.get(q_key)
                candidate_q = outputs.get(q_key) or {}
                # Candidate may include nested fields; expected likely is the array under 'result_rows' or the array itself.
                expected_rows = None
                candidate_rows = None

                # Extract expected rows array
                if isinstance(expected, dict) and "result_rows" in expected:
                    expected_rows = expected.get("result_rows", [])
                elif isinstance(expected, list):
                    expected_rows = expected
                elif isinstance(expected, dict) and "export_full_if_included" in expected:
                    expected_rows = expected.get("export_full_if_included", [])
                else:
                    # try answer top-level
                    expected_rows = expected if isinstance(expected, list) else []

                # Extract candidate rows similarly
                if isinstance(candidate_q, dict) and "result_rows" in candidate_q:
                    candidate_rows = candidate_q.get("result_rows", [])
                elif isinstance(candidate_q, dict) and "result_rows" not in candidate_q and isinstance(candidate_q.get("result_rows", None), list):
                    candidate_rows = candidate_q.get("result_rows", [])
                elif isinstance(candidate_q, list):
                    candidate_rows = candidate_q
                else:
                    # candidate may have put rows directly under outputs.query_1 etc.
                    # try outputs[q_key] as dict of rows
                    # Try candidate_q.get('result_rows') or candidate_q.get('rows') or candidate_q.get('query_result')
                    if isinstance(candidate_q, dict):
                        candidate_rows = candidate_q.get("result_rows") or candidate_q.get("rows") or candidate_q.get("query_results") or []
                    else:
                        candidate_rows = []

                # If expected is empty (answer key missing), give basic credit for presence
                if not expected_rows:
                    # If candidate provided something, award small partial credit
                    if candidate_rows:
                        pts = 3  # presence
                        qc_notes.append(f"Query {i}: no expected rows in key; candidate provided rows => partial credit {pts}/5.")
                    else:
                        pts = 0
                        qc_notes.append(f"Query {i}: no expected rows and candidate has none => 0.")
                    query_correctness_pts += pts
                    continue

                matched, exp_count, cand_count = compare_row_lists(expected_rows, candidate_rows)
                # Fraction matched
                fraction = matched / exp_count if exp_count > 0 else (1.0 if cand_count == 0 else 0.0)
                pts = round(fraction * 5, 2)  # up to 5 per query
                query_correctness_pts += pts
                qc_notes.append(f"Query {i}: matched {matched}/{exp_count} expected rows => {pts}/5.")
        else:
            core_feedback.append("Answer key Core Queries step not found; cannot auto-validate query correctness.")

        core_score += query_correctness_pts
        core_feedback.append(f"Query correctness subtotal: {query_correctness_pts}/25. Details: {qc_notes}")

        # 2b SQL style/readability and comments (5 pts)
        sql_pts = 0
        sql_reasons = []
        code = artifacts.get("code", "") or ""
        codelow = code.lower() if isinstance(code, str) else ""
        if code and ("select" in codelow and ("group by" in codelow or "having" in codelow or "join" in codelow)):
            sql_pts = 5
            sql_reasons.append("SQL includes SELECT and GROUP BY / JOIN or HAVING.")
        elif code and "select" in codelow:
            sql_pts = 3
            sql_reasons.append("SQL includes SELECT but missing some grouping/join keywords (partial).")
        else:
            sql_pts = 0
            sql_reasons.append("No SQL code or not recognizable; 0 points.")
        core_score += sql_pts
        core_feedback.append(f"SQL style points: {sql_pts}/5 - {'; '.join(sql_reasons)}")

        # 2c Result outputs included & match what SQL would return (10 pts)
        output_pts = 0
        output_reasons = []
        # award 5 points if outputs for all five queries are present and are arrays (non-empty or empty)
        present_count = 0
        for i in range(1, 6):
            qk = f"query_{i}"
            qval = outputs.get(qk)
            if qval is not None:
                # accept dict or list
                present_count += 1
        if present_count == 5:
            output_pts += 5
            output_reasons.append("All five query outputs present.")
        elif present_count >= 3:
            output_pts += 3
            output_reasons.append(f"{present_count}/5 query outputs present (partial).")
        else:
            output_reasons.append(f"{present_count}/5 query outputs present (low).")

        # additional 5 points for outputs matching expected (we already gave correctness pts up to 25)
        # measure whether the candidate result rows matched majority of expected rows overall
        # Compute aggregate matched fraction if answer_core_step exists
        if answer_core_step:
            ans_outputs = safe_get(answer_core_step, "artifacts", "outputs", default={})
            total_expected_rows = 0
            total_matched = 0
            for i in range(1, 6):
                qk = f"query_{i}"
                expected = ans_outputs.get(qk)
                if isinstance(expected, dict) and "result_rows" in expected:
                    exp_rows = expected.get("result_rows", [])
                elif isinstance(expected, list):
                    exp_rows = expected
                else:
                    exp_rows = expected.get("result_rows", []) if isinstance(expected, dict) else []
                candidate_q = outputs.get(qk) or {}
                if isinstance(candidate_q, dict) and "result_rows" in candidate_q:
                    cand_rows = candidate_q.get("result_rows", [])
                elif isinstance(candidate_q, list):
                    cand_rows = candidate_q
                else:
                    cand_rows = candidate_q.get("result_rows") or candidate_q.get("rows") or []
                m, ecount, ccount = compare_row_lists(exp_rows, cand_rows)
                total_expected_rows += ecount
                total_matched += m
            frac = total_matched / total_expected_rows if total_expected_rows > 0 else 0.0
            # award up to 5 points proportional to fraction
            match_bonus = round(frac * 5, 2)
            output_pts += match_bonus
            output_reasons.append(f"Overall query rows matched {total_matched}/{total_expected_rows} => {match_bonus}/5 bonus.")
        else:
            output_reasons.append("No answer key expected rows to compare aggregate matching; no bonus points.")

        core_score += output_pts
        core_feedback.append(f"Output points: {output_pts}/10 - {'; '.join(output_reasons)}")

    results["scores"]["core_queries"] = {"score": round(core_score, 2), "max": core_max, "notes": core_feedback}
    results["points_awarded"] += core_score

    # --- 3) Data Transformation & Export (20 points) ------------------------
    trans_max = 20
    trans_score = 0
    trans_feedback = []

    step_trans = step_map.get("Transformation & Export")
    if not step_trans:
        trans_feedback.append("Transformation & Export step missing.")
    else:
        artifacts = step_trans.get("artifacts", {})
        outputs = artifacts.get("outputs", {})
        code = artifacts.get("code", "")
        commands = artifacts.get("commands", [])

        # 3a Transformation correctness/usefulness (10 pts)
        transform_pts = 0
        transform_reasons = []
        # Try to find expected transform in answer key (steps[2] expected outputs)
        answer_trans_step = None
        for s in safe_get(answer, "steps", default=[]):
            if isinstance(s, dict) and s.get("name") == "Transformation & Export":
                answer_trans_step = s
                break
        if not answer_trans_step:
            # fallback to steps[2]
            ans_steps = answer.get("steps", [])
            if len(ans_steps) > 2 and isinstance(ans_steps[2], dict):
                answer_trans_step = ans_steps[2]

        if answer_trans_step:
            ans_out = safe_get(answer_trans_step, "artifacts", "outputs", default={})
            expected_rows = ans_out.get("export_full_if_included") or ans_out.get("export_sample_if_large") or []
            candidate_rows = outputs.get("export_full_if_included") or outputs.get("export_sample_if_large") or []
            # If both exist and expected non-empty, compare
            if expected_rows:
                matched, exp_count, cand_count = compare_row_lists(expected_rows, candidate_rows)
                frac = matched / exp_count if exp_count > 0 else 0.0
                transform_pts = round(frac * 10, 2)
                transform_reasons.append(f"Matched {matched}/{exp_count} expected export rows => {transform_pts}/10.")
            else:
                # If expected not present in key, award partial if candidate produced a reasonable export_rows
                export_rows = outputs.get("export_rows")
                if isinstance(export_rows, int) and export_rows > 0:
                    transform_pts = 6
                    transform_reasons.append("No expected export in key; candidate produced export_rows => partial credit.")
                else:
                    transform_reasons.append("No expected export and candidate did not include export rows => 0.")
        else:
            # No answer key reference; accept any reasonable export sample
            export_rows = outputs.get("export_rows")
            if isinstance(export_rows, int) and export_rows > 0:
                transform_pts = 8
                transform_reasons.append("No answer key transform; candidate exported rows => good credit.")
            else:
                transform_reasons.append("No transform or export found.")

        trans_score += transform_pts
        trans_feedback.append(f"Transformation points: {transform_pts}/10 - {'; '.join(transform_reasons)}")

        # 3b Exported results included and proper format (5 pts)
        export_pts = 0
        export_reasons = []
        exp_format = outputs.get("export_format") or ""
        export_rows_count = outputs.get("export_rows")
        if exp_format and exp_format.lower() in ("csv", "json"):
            export_pts += 3
            export_reasons.append(f"Export format provided: {exp_format}.")
        if isinstance(export_rows_count, int) and export_rows_count >= 0:
            export_pts += 2
            export_reasons.append(f"Export rows count provided: {export_rows_count}.")
        trans_score += export_pts
        trans_feedback.append(f"Export points: {export_pts}/5 - {'; '.join(export_reasons)}")

        # 3c Explanation of why transformation useful (5 pts)
        explain_pts = 0
        explanation = outputs.get("transformation_description_one_to_two_sentences") or artifacts.get("short_explanation") or ""
        if explanation and isinstance(explanation, str) and len(explanation.strip()) >= 15:
            explain_pts = 5
            trans_feedback.append("Transformation explanation provided.")
        else:
            explain_pts = 0
            trans_feedback.append("No adequate transformation explanation provided.")
        trans_score += explain_pts

    results["scores"]["transformation_export"] = {"score": round(trans_score, 2), "max": trans_max, "notes": trans_feedback}
    results["points_awarded"] += trans_score

    # --- 4) Summary & Interpretation (10 points) ----------------------------
    summ_max = 10
    summ_score = 0
    summ_feedback = []

    summary = candidate.get("summary", {})
    key_findings = summary.get("key_findings", [])
    recommendations = summary.get("recommendations", [])
    assumptions = summary.get("assumptions", [])

    # 4a Clear, accurate conclusions & recommendations (7 pts)
    concl_pts = 0
    concl_reasons = []
    if isinstance(key_findings, list) and len(key_findings) >= 3:
        concl_pts += 5
        concl_reasons.append("At least 3 key findings present.")
    elif isinstance(key_findings, list) and len(key_findings) >= 1:
        concl_pts += 2
        concl_reasons.append("Some key findings present (less than 3).")
    else:
        concl_reasons.append("No key findings present.")

    if isinstance(recommendations, list) and len(recommendations) >= 2:
        concl_pts += 2
        concl_reasons.append("At least 2 recommendations present.")
    elif isinstance(recommendations, list) and len(recommendations) == 1:
        concl_pts += 1
        concl_reasons.append("One recommendation present.")
    else:
        concl_reasons.append("No recommendations present.")

    # cap to 7
    concl_pts = min(7, concl_pts)
    summ_score += concl_pts
    summ_feedback.append(f"Conclusions/recommendations points: {concl_pts}/7 - {'; '.join(concl_reasons)}")

    # 4b Time log + explanation of choices and edge-case handling (3 pts)
    time_pts = 0
    time_reasons = []
    # Check that per-step time_spent_minutes exist and sum up to top-level time_spent_minutes (or at least provided)
    total_reported = candidate.get("time_spent_minutes")
    steps_times = []
    for sname in expected_step_names:
        s = step_map.get(sname)
        if s:
            t = s.get("time_spent_minutes")
            if isinstance(t, int) or isinstance(t, float):
                steps_times.append(t)
    if isinstance(total_reported, (int, float)) and steps_times:
        # check sum approx equal (allow +/-5 minutes)
        if abs(total_reported - sum(steps_times)) <= 5:
            time_pts += 2
            time_reasons.append("Top-level time matches sum of per-step times (within tolerance).")
        else:
            time_pts += 1
            time_reasons.append("Top-level time differs from sum of per-step times (partial credit).")
    elif isinstance(total_reported, (int, float)):
        time_pts += 1
        time_reasons.append("Top-level time reported but per-step times incomplete (partial).")
    else:
        time_reasons.append("No time log reported.")

    # Check that candidate mentioned handling malformed rows explicitly
    step0_outputs = safe_get(step_map.get("Setup & Import", {}), "artifacts", "outputs", default={})
    data_clean_decision = step0_outputs.get("data_cleaning_decision_one_line") or step0_outputs.get("data_cleaning_decision_one_line", "") or step0_outputs.get("rows_dropped") or ""
    if data_clean_decision or ("malformed" in json.dumps(step0_outputs).lower()):
        time_pts += 1
        time_reasons.append("Data-cleaning decisions reported (malformed rows handling).")
    else:
        time_reasons.append("No explicit data-cleaning decision reported.")

    time_pts = min(3, time_pts)
    summ_score += time_pts
    summ_feedback.append(f"Time & edge-case handling points: {time_pts}/3 - {'; '.join(time_reasons)}")

    results["scores"]["summary_interpretation"] = {"score": summ_score, "max": summ_max, "notes": summ_feedback}
    results["points_awarded"] += summ_score

    # --- 5) Reproducibility & Packaging (10 points) -------------------------
    repro_max = 10
    repro_score = 0
    repro_feedback = []

    # 5a test_submission.json present and properly structured (5 pts)
    structure_pts = 0
    struct_reasons = []
    # Check presence of required top-level keys
    missing = [k for k in required_top_keys if k not in candidate]
    if not missing:
        structure_pts = 5
        struct_reasons.append("All required top-level keys present in submission.")
    else:
        # partial for some keys present
        have = len(required_top_keys) - len(missing)
        structure_pts = round(5 * (have / len(required_top_keys)), 2)
        struct_reasons.append(f"Missing top-level keys: {missing}. Partial credit: {structure_pts}/5.")
    repro_score += structure_pts
    repro_feedback.append(f"Structure points: {structure_pts}/5 - {'; '.join(struct_reasons)}")

    # 5b Commands/steps sufficient to rerun everything in <=10 minutes (5 pts)
    runable_pts = 0
    run_reasons = []
    reproducibility = candidate.get("reproducibility", {})
    cmds = reproducibility.get("commands", []) if isinstance(reproducibility, dict) else []
    files_created = reproducibility.get("files_created", []) if isinstance(reproducibility, dict) else []
    cmds_text = " ".join(cmds).lower() if isinstance(cmds, list) else ""
    # Heuristics: must include ingestion and query commands, and mention the DB or export file
    has_ingest = any(tok in cmds_text for tok in ("ingest", "system_events.csv", "python", "sqlite3", "create.sql"))
    has_queries = any(tok in cmds_text for tok in ("select", "sqlite3", "queries.py", "cleaned.db"))
    has_export = any(tok in cmds_text for tok in ("csv", "transformed", "export", "transformed_daily.csv", "-csv"))
    if has_ingest and has_queries and has_export:
        runable_pts = 5
        run_reasons.append("Reproducibility commands include ingestion, queries, and export steps.")
    else:
        # partial credit if at least two of three present
        present = sum([bool(has_ingest), bool(has_queries), bool(has_export)])
        runable_pts = round(5 * (present / 3), 2)
        run_reasons.append(f"Reproducibility commands partially complete (ingest:{has_ingest}, queries:{has_queries}, export:{has_export}).")
    repro_score += runable_pts
    repro_feedback.append(f"Runability points: {runable_pts}/5 - {'; '.join(run_reasons)}")

    results["scores"]["reproducibility_packaging"] = {"score": round(repro_score, 2), "max": repro_max, "notes": repro_feedback}
    results["points_awarded"] += repro_score

    # --- Final aggregation --------------------------------------------------
    total_awarded = results["points_awarded"]
    total_possible = results["total_points"]
    percentage = round((total_awarded / total_possible) * 100, 2) if total_possible > 0 else 0.0

    # Prepare detailed results structure
    results["summary_scores"] = {
        "total_awarded": round(total_awarded, 2),
        "total_possible": total_possible,
        "percentage": percentage,
        "overall_score": percentage
    }

    # Collect deduction explanations (flatten)
    if not results.get("deductions"):
        results["deductions"] = []
    # Add some human-friendly notes
    results["notes"] = {
        "overall_feedback": [
            "Scoring is approximate and automated: evaluators should review borderline/subjective items (e.g., cleaning choices).",
            "This automated grader expects the answer key to contain expected outputs for detailed matching; if the answer key lacks some expected rows, partial credit logic is used."
        ]
    }

    return results

# --- Main execution -------------------------------------------------------

def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)

    sub_path = sys.argv[1]
    key_path = sys.argv[2]

    try:
        results = grade_submission(sub_path, key_path)
    except Exception as e:
        err = {"error": f"Exception during grading: {str(e)}"}
        # Write error results
        out_path = os.path.join(os.path.dirname(os.path.abspath(__file__)), "test_results.json")
        with open(out_path, "w", encoding="utf-8") as f:
            json.dump(err, f, indent=2)
        print(f"Grading failed: {e}. Wrote error file to {out_path}")
        sys.exit(1)

    # Write test_results.json next to this script
    out_path = os.path.join(os.path.dirname(os.path.abspath(__file__)), "test_results.json")
    try:
        with open(out_path, "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
    except Exception as e:
        print(f"Failed to write results to {out_path}: {e}")
        sys.exit(1)

    # Print summary
    overall = results.get("summary_scores", {}).get("overall_score")
    print(f"Grading completed. Overall score: {overall}%")
    print(f"Detailed results written to: {out_path}")

if __name__ == "__main__":
    main()
