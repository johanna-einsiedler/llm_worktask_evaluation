# task_evaluation.py
"""
Automated grader for the Inventory CLI basic practical exam.

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

The script loads the candidate submission JSON and an answer key JSON, applies
the grading rubric described in the exam materials, and writes a detailed
report to test_results.json in the same directory as this script.

Only standard libraries are used.
"""

import json
import os
import sys
import shlex

# Constants for rubric points
RUBRIC = {
    "core": {
        "total": 60,
        "per_command": 8,  # add, list, update, delete, find -> each 8
        "persistence": 15,
        "exit_errors": 5,
    },
    "robustness": {
        "total": 10,
        "duplicate": 2,
        "negative_qty": 2,
        "missing_fields": 2,
        "clear_errors": 4,
    },
    "deterministic": {
        "total": 10,
        "json_output": 6,
        "cli_contract": 4,
    },
    "code_quality": {
        "total": 10,
        "readable": 6,
        "docs": 4,
    },
    "submission": {
        "total": 10,
        "json_wellformed": 5,
        "commands_recorded": 5,
    },
}

MAX_POINTS = (
    RUBRIC["core"]["total"]
    + RUBRIC["robustness"]["total"]
    + RUBRIC["deterministic"]["total"]
    + RUBRIC["code_quality"]["total"]
    + RUBRIC["submission"]["total"]
)


def load_json_file(path):
    """Load JSON and return object, or raise with a helpful message."""
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {path}")
    except json.JSONDecodeError as e:
        raise ValueError(f"Invalid JSON in {path}: {e}")


def parse_command(cmd_str):
    """
    Parse a command string into a dict with:
      - raw_tokens: list of tokens
      - action: command name (add, list, update, delete, find) if present
      - flags: dict mapping flag names (without --) to their values (strings)
    Uses shlex.split to respect quoted strings.
    """
    try:
        tokens = shlex.split(cmd_str)
    except Exception:
        # fallback naive split
        tokens = cmd_str.strip().split()

    action = None
    flags = {}

    # Find action token: it's the first token that equals one of the action names.
    actions = {"add", "list", "update", "delete", "find"}
    for i, t in enumerate(tokens):
        if t in actions:
            action = t
            # flags parsing from the remaining tokens
            j = i + 1
            while j < len(tokens):
                tok = tokens[j]
                if tok.startswith("--"):
                    key = tok[2:]
                    # boolean flags could be present; assume next token is value if it doesn't start with --
                    val = True
                    if (j + 1) < len(tokens) and not tokens[j + 1].startswith("--"):
                        val = tokens[j + 1]
                        j += 1
                    flags[key] = val
                j += 1
            break

    # If no explicit action found, assume first token is action (best-effort)
    if action is None and tokens:
        if tokens[0] in actions:
            action = tokens[0]

    return {"raw_tokens": tokens, "action": action, "flags": flags}


def normalize_whitespace(s):
    return " ".join(s.split()) if isinstance(s, str) else s


def safe_parse_json(s):
    try:
        return json.loads(s)
    except Exception:
        return None


def sku_in_json_stdout(stdout_text, sku):
    """
    Given stdout text (likely JSON array or JSON object), check whether an object with
    key 'sku' == sku exists somewhere in the parsed structure.
    """
    parsed = safe_parse_json(stdout_text)
    if parsed is None:
        # attempt substring match as fallback
        return (sku in stdout_text) if stdout_text else False

    # If it's a list, check items
    if isinstance(parsed, list):
        for item in parsed:
            if isinstance(item, dict) and item.get("sku") == sku:
                return True
    elif isinstance(parsed, dict):
        # maybe a single object
        if parsed.get("sku") == sku:
            return True
        # or nested structure; search recursively
        def search(o):
            if isinstance(o, dict):
                if o.get("sku") == sku:
                    return True
                for v in o.values():
                    if search(v):
                        return True
            elif isinstance(o, list):
                for e in o:
                    if search(e):
                        return True
            elif isinstance(o, str):
                return sku in o
            return False
        return search(parsed)

    return False


def count_success_for_action(commands_run, action):
    """
    Return tuple (has_success, has_any) where has_success means there's at least one command
    with this action and exit_code == 0; has_any means the action appears at all.
    """
    has_any = False
    has_success = False
    for entry in commands_run:
        cmd = entry.get("cmd", "")
        parsed = parse_command(cmd)
        if parsed["action"] == action:
            has_any = True
            if isinstance(entry.get("exit_code"), int) and entry.get("exit_code") == 0:
                has_success = True
                break
    return has_success, has_any


def find_first_command_with_flag(commands_run, action, flag_name, flag_value=None):
    """Find the first command entry where action matches and flags[flag_name] == flag_value (if provided)."""
    for i, entry in enumerate(commands_run):
        cmd = entry.get("cmd", "")
        parsed = parse_command(cmd)
        if parsed["action"] != action:
            continue
        flags = parsed["flags"]
        if flag_name in flags:
            if flag_value is None or str(flags[flag_name]) == str(flag_value):
                return i, entry, parsed
    return None, None, None


def index_of_command(commands_run, entry_to_find):
    """Return index of a particular command dict in commands_run by matching cmd string; -1 if not found."""
    target_cmd = normalize_whitespace(entry_to_find.get("cmd", ""))
    for i, entry in enumerate(commands_run):
        if normalize_whitespace(entry.get("cmd", "")) == target_cmd:
            return i
    return -1


def grade_submission(candidate, answer_key):
    """
    Main grading logic. Returns a dict with detailed breakdown, reasons, and score totals.
    """

    results = {
        "breakdown": {},
        "deductions": [],
        "notes": [],
    }

    # Basic validation of candidate submission JSON structure
    required_keys = [
        "language",
        "entry_point",
        "persistence_file",
        "commands_run",
        "short_description",
        "self_assessment",
        "time_taken_minutes",
    ]
    missing_keys = [k for k in required_keys if k not in candidate]
    if missing_keys:
        results["error"] = f"Candidate submission missing required keys: {missing_keys}"
        # Fill results with zeros and return
        results["breakdown"] = {}
        results["total_points"] = 0
        results["max_points"] = MAX_POINTS
        results["percentage"] = 0.0
        results["overall_score"] = 0.0
        return results

    commands_run = candidate.get("commands_run", [])
    if not isinstance(commands_run, list):
        results["error"] = "candidate.commands_run is not a list"
        results["breakdown"] = {}
        results["total_points"] = 0
        results["max_points"] = MAX_POINTS
        results["percentage"] = 0.0
        results["overall_score"] = 0.0
        return results

    # --- Core Functionality (60 points) ---
    core_points = 0
    core_details = {}
    actions = ["add", "list", "update", "delete", "find"]
    per_command_points = RUBRIC["core"]["per_command"]

    for action in actions:
        success, present = count_success_for_action(commands_run, action)
        if success:
            pts = per_command_points
            remark = f"{action}: present and successful"
        elif present:
            pts = per_command_points // 2  # partial credit
            remark = f"{action}: present but no successful run (non-zero exit or other failure)"
        else:
            pts = 0
            remark = f"{action}: not present"
        core_points += pts
        core_details[action] = {"points": pts, "max": per_command_points, "remark": remark}

    # Persistence (15 points)
    persistence_pts = 0
    persistence_remark = ""
    # Strategy:
    # - find any successful add with an SKU
    # - check if later a list (any) contains that SKU in stdout JSON (or substring)
    added_skus = {}
    for idx, entry in enumerate(commands_run):
        cmd = entry.get("cmd", "")
        parsed = parse_command(cmd)
        if parsed["action"] == "add":
            sku = parsed["flags"].get("sku")
            if sku and entry.get("exit_code") == 0:
                added_skus[sku] = idx

    found_persistence_evidence = False
    for sku, add_idx in added_skus.items():
        # search for any list command after add_idx
        for j in range(add_idx + 1, len(commands_run)):
            e = commands_run[j]
            parsed_l = parse_command(e.get("cmd", ""))
            if parsed_l["action"] == "list":
                stdout = e.get("stdout", "")
                # check if stdout contains SKU (prefer JSON parsing)
                if sku_in_json_stdout(stdout, sku):
                    found_persistence_evidence = True
                    break
                else:
                    # found list but did not include SKU
                    found_persistence_evidence = found_persistence_evidence or False
        if found_persistence_evidence:
            break

    if found_persistence_evidence:
        persistence_pts = RUBRIC["core"]["persistence"]
        persistence_remark = "Persistence demonstrated: item added and later listed in a subsequent invocation."
    else:
        # if any list exists after adds but didn't show sku or list not JSON, give partial credit
        any_list_after_add = False
        for sku, add_idx in added_skus.items():
            for j in range(add_idx + 1, len(commands_run)):
                if parse_command(commands_run[j].get("cmd", "")).get("action") == "list":
                    any_list_after_add = True
                    break
            if any_list_after_add:
                break
        if any_list_after_add:
            persistence_pts = 5  # partial
            persistence_remark = "List occurred after add but SKU not observed in list output (or output not parseable). Partial credit."
        else:
            persistence_pts = 0
            persistence_remark = "No evidence of persistence across invocations (no list after add showing previously added items)."

    core_points += persistence_pts

    # Exit codes & error messages for invalid commands (5 points)
    # Use answer_key to know which error cases were expected if provided; otherwise, check candidate errors
    exit_errors_pts = 0
    expected_errors = []
    if isinstance(answer_key, dict):
        expected_errors = [
            e for e in answer_key.get("commands_run", []) if isinstance(e, dict) and isinstance(e.get("exit_code"), int) and e.get("exit_code", 0) != 0
        ]

    if expected_errors:
        matched_errs = 0
        for expected in expected_errors:
            # parse expected cmd to get action and identifying flags (sku if present)
            exp_parsed = parse_command(expected.get("cmd", ""))
            exp_action = exp_parsed.get("action")
            exp_sku = exp_parsed.get("flags", {}).get("sku")
            # try to find a candidate command with same action and sku and non-zero exit code and stderr
            for c in commands_run:
                c_parsed = parse_command(c.get("cmd", ""))
                if c_parsed.get("action") != exp_action:
                    continue
                if exp_sku and c_parsed.get("flags", {}).get("sku") != exp_sku:
                    continue
                if isinstance(c.get("exit_code"), int) and c.get("exit_code") != 0 and c.get("stderr", ""):
                    matched_errs += 1
                    break
        if expected_errors:
            exit_errors_pts = int(RUBRIC["core"]["exit_errors"] * (matched_errs / len(expected_errors)))
            exit_errors_remark = f"Matched {matched_errs}/{len(expected_errors)} expected error cases."
        else:
            exit_errors_pts = 0
            exit_errors_remark = "No expected error cases in answer key."
    else:
        # No expected errors provided in answer key: check candidate for at least one error handling (duplicate or negative qty)
        error_entries = [e for e in commands_run if isinstance(e.get("exit_code"), int) and e.get("exit_code") != 0]
        if error_entries:
            exit_errors_pts = RUBRIC["core"]["exit_errors"]
            exit_errors_remark = "Candidate reported errors with non-zero exit codes / stderr."
        else:
            exit_errors_pts = 0
            exit_errors_remark = "No error runs observed in candidate submission."

    core_points += exit_errors_pts

    results["breakdown"]["core"] = {
        "points_awarded": core_points,
        "points_max": RUBRIC["core"]["total"],
        "details": core_details,
        "persistence": {"points": persistence_pts, "max": RUBRIC["core"]["persistence"], "remark": persistence_remark},
        "exit_errors": {"points": exit_errors_pts, "max": RUBRIC["core"]["exit_errors"], "remark": exit_errors_remark},
    }

    # --- Robustness & Input validation (10 points) ---
    robustness_points = 0
    robustness_details = {}

    # Duplicate SKU rejection (2 points)
    dup_pts = 0
    dup_found = False
    # Look for candidate error entries that mention 'sku' and 'exists' or 'duplicate'
    for e in commands_run:
        if isinstance(e.get("exit_code"), int) and e.get("exit_code") != 0:
            stderr = (e.get("stderr") or "").lower()
            if "sku" in stderr and ("exist" in stderr or "duplicate" in stderr or "already" in stderr):
                dup_found = True
                break
    if dup_found:
        dup_pts = RUBRIC["robustness"]["duplicate"]
    else:
        dup_pts = 0
    robustness_points += dup_pts
    robustness_details["duplicate_rejection"] = {"points": dup_pts, "max": RUBRIC["robustness"]["duplicate"], "found": dup_found}

    # Negative qty rejection (2 points)
    neg_pts = 0
    neg_found = False
    for e in commands_run:
        parsed = parse_command(e.get("cmd", ""))
        if parsed.get("action") == "add":
            qty = parsed.get("flags", {}).get("qty")
            if qty is not None:
                try:
                    if int(qty) < 0:
                        # if candidate tried to add negative qty, check that it failed
                        if isinstance(e.get("exit_code"), int) and e.get("exit_code") != 0:
                            neg_found = True
                            break
                except Exception:
                    pass
    if neg_found:
        neg_pts = RUBRIC["robustness"]["negative_qty"]
    else:
        neg_pts = 0
    robustness_points += neg_pts
    robustness_details["negative_qty_rejection"] = {"points": neg_pts, "max": RUBRIC["robustness"]["negative_qty"], "found": neg_found}

    # Missing fields rejection (2 points)
    missing_pts = 0
    missing_found = False
    # Look for an 'add' entry with missing sku flag and that resulted in non-zero exit_code
    for e in commands_run:
        parsed = parse_command(e.get("cmd", ""))
        if parsed.get("action") == "add":
            if "sku" not in parsed.get("flags", {}):
                # candidate issued an add with missing sku; prefer that it failed
                if isinstance(e.get("exit_code"), int) and e.get("exit_code") != 0:
                    missing_found = True
                    break
    if missing_found:
        missing_pts = RUBRIC["robustness"]["missing_fields"]
    else:
        missing_pts = 0
    robustness_points += missing_pts
    robustness_details["missing_fields_rejection"] = {"points": missing_pts, "max": RUBRIC["robustness"]["missing_fields"], "found": missing_found}

    # Clear error messages & non-zero exit codes on failures (4 points)
    clear_err_pts = 0
    error_entries = [e for e in commands_run if isinstance(e.get("exit_code"), int) and e.get("exit_code") != 0]
    if error_entries:
        # check proportion of error_entries that include stderr text
        with_stderr = sum(1 for e in error_entries if e.get("stderr", "").strip())
        ratio = with_stderr / len(error_entries)
        clear_err_pts = int(round(RUBRIC["robustness"]["clear_errors"] * ratio))
        clear_err_remark = f"{with_stderr}/{len(error_entries)} error entries included stderr text."
    else:
        clear_err_pts = 0
        clear_err_remark = "No error entries to evaluate."
    robustness_points += clear_err_pts

    robustness_details["clear_errors"] = {"points": clear_err_pts, "max": RUBRIC["robustness"]["clear_errors"], "remark": clear_err_remark}

    results["breakdown"]["robustness"] = {"points_awarded": robustness_points, "points_max": RUBRIC["robustness"]["total"], "details": robustness_details}

    # --- Deterministic & Scriptable outputs (10 points) ---
    deterministic_points = 0
    deterministic_details = {}

    # JSON outputs for list and find (6 points)
    json_pts = 0
    list_json_good = False
    find_json_good = False
    for e in commands_run:
        parsed = parse_command(e.get("cmd", ""))
        if parsed.get("action") == "list":
            if "format" in parsed.get("flags", {}) and str(parsed["flags"]["format"]).lower() == "json":
                if safe_parse_json(e.get("stdout", "") ) is not None:
                    list_json_good = True
        if parsed.get("action") == "find":
            if "format" in parsed.get("flags", {}) and str(parsed["flags"]["format"]).lower() == "json":
                if safe_parse_json(e.get("stdout", "")) is not None:
                    find_json_good = True
    if list_json_good and find_json_good:
        json_pts = RUBRIC["deterministic"]["json_output"]
        json_remark = "Both list and find produced valid JSON when requested."
    elif list_json_good or find_json_good:
        json_pts = int(RUBRIC["deterministic"]["json_output"] / 2)
        json_remark = "Only one of list/find produced valid JSON."
    else:
        json_pts = 0
        json_remark = "No valid JSON outputs observed for list/find with --format json."
    deterministic_points += json_pts
    deterministic_details["json_output"] = {"points": json_pts, "max": RUBRIC["deterministic"]["json_output"], "remark": json_remark}

    # Reproducible CLI contract (4 points)
    cli_pts = 0
    entry_point = candidate.get("entry_point", "").strip()
    if entry_point:
        # Check that candidate's commands_run appear to be full CLI invocations (contain the entry_point or start with it).
        commands_contain_entry = sum(1 for e in commands_run if entry_point in (e.get("cmd", "") or "")) >= max(1, int(len(commands_run) * 0.5))
        if commands_contain_entry and len(commands_run) >= 3:
            cli_pts = RUBRIC["deterministic"]["cli_contract"]
            cli_remark = "Entry point provided and commands appear reproducible/scriptable."
        else:
            cli_pts = int(RUBRIC["deterministic"]["cli_contract"] / 2)
            cli_remark = "Entry point present but commands did not consistently include it or few commands present."
    else:
        cli_pts = 0
        cli_remark = "No entry_point provided."
    deterministic_points += cli_pts
    deterministic_details["cli_contract"] = {"points": cli_pts, "max": RUBRIC["deterministic"]["cli_contract"], "remark": cli_remark}

    results["breakdown"]["deterministic"] = {"points_awarded": deterministic_points, "points_max": RUBRIC["deterministic"]["total"], "details": deterministic_details}

    # --- Code quality & organization (10 points) ---
    code_points = 0
    code_details = {}
    short_desc = candidate.get("short_description", "") or ""
    sa = candidate.get("self_assessment", {}) or {}

    # Readable code, modular functions (heuristic) (6 points)
    readable_pts = 0
    # Heuristic: if short_description mentions 'code', 'modular', 'functions', 'persistence', or 'atomic', give credit
    sd_lower = short_desc.lower()
    if any(k in sd_lower for k in ["persistence", "atomic", "json", "sqlite", "modul", "function", "entry point", "entry_point", "cli"]):
        readable_pts = RUBRIC["code_quality"]["readable"]
        readable_remark = "Short description mentions implementation details suggesting reasonable code organization."
    else:
        # partial if short description is non-trivial
        if len(short_desc.strip()) >= 40:
            readable_pts = int(RUBRIC["code_quality"]["readable"] / 2)
            readable_remark = "Short description present but lacks clues about code organization."
        else:
            readable_pts = 0
            readable_remark = "Short description too short to infer code quality."
    code_points += readable_pts
    code_details["readable"] = {"points": readable_pts, "max": RUBRIC["code_quality"]["readable"], "remark": readable_remark}

    # Minimal inline comments or README describing run instructions (4 points)
    docs_pts = 0
    implemented_required = sa.get("implemented_required")
    if isinstance(implemented_required, bool) and implemented_required:
        docs_pts = RUBRIC["code_quality"]["docs"]
        docs_remark = "Self-assessment claims required features implemented."
    else:
        docs_pts = 0
        docs_remark = "Self-assessment does not claim full implementation."
    code_points += docs_pts
    code_details["docs"] = {"points": docs_pts, "max": RUBRIC["code_quality"]["docs"], "remark": docs_remark}

    results["breakdown"]["code_quality"] = {"points_awarded": code_points, "points_max": RUBRIC["code_quality"]["total"], "details": code_details}

    # --- Submission completeness & clarity (10 points) ---
    submission_points = 0
    submission_details = {}

    # test_submission.json present and well-formed - since we loaded it, give full credit
    json_wellformed_pts = RUBRIC["submission"]["json_wellformed"]
    submission_points += json_wellformed_pts
    submission_details["json_wellformed"] = {"points": json_wellformed_pts, "max": RUBRIC["submission"]["json_wellformed"], "remark": "test_submission.json successfully parsed."}

    # Commands recorded and outputs included (5 points)
    commands_count = len(commands_run)
    if commands_count >= 8:
        cmd_pts = RUBRIC["submission"]["commands_recorded"]
        cmd_remark = f"{commands_count} command runs recorded (meets or exceeds expected demonstration sequence)."
    elif 5 <= commands_count < 8:
        cmd_pts = int(RUBRIC["submission"]["commands_recorded"] * 0.6)
        cmd_remark = f"{commands_count} command runs recorded (partial)."
    else:
        cmd_pts = 0
        cmd_remark = f"Only {commands_count} command runs recorded (insufficient)."
    submission_points += cmd_pts
    submission_details["commands_recorded"] = {"points": cmd_pts, "max": RUBRIC["submission"]["commands_recorded"], "remark": cmd_remark}

    results["breakdown"]["submission"] = {"points_awarded": submission_points, "points_max": RUBRIC["submission"]["total"], "details": submission_details}

    # Sum all points
    total_awarded = (
        core_points + robustness_points + deterministic_points + code_points + submission_points
    )
    percentage = (total_awarded / MAX_POINTS) * 100 if MAX_POINTS else 0.0

    # Build detailed deductions and notes
    deductions = []
    # Example deductions: missing command types
    for action in actions:
        if core_details.get(action, {}).get("points", 0) == 0:
            deductions.append(f"Missing required command: {action} (0/{per_command_points} points).")
        elif core_details.get(action, {}).get("points", 0) < per_command_points:
            deductions.append(f"Command {action} present but not fully successful ({core_details[action]['points']}/{per_command_points} points).")

    # Persistence
    if persistence_pts == 0:
        deductions.append("Persistence not demonstrated across invocations; items added were not observed in subsequent list output.")
    elif persistence_pts < RUBRIC["core"]["persistence"]:
        deductions.append("Persistence partially demonstrated: list occurred after add but did not show added SKUs or output not parseable as JSON.")

    # Robustness notes
    if dup_pts == 0:
        deductions.append("Duplicate SKU rejection not observed in stderr messages.")
    if neg_pts == 0:
        deductions.append("Negative quantity rejection not observed for invalid add attempts.")
    if missing_pts == 0:
        deductions.append("Missing-field (e.g., add without --sku) rejection not observed.")

    # Deterministic notes
    if json_pts < RUBRIC["deterministic"]["json_output"]:
        deductions.append("list/find --format json did not both produce valid JSON outputs as required.")
    if cli_pts < RUBRIC["deterministic"]["cli_contract"]:
        deductions.append("CLI entry_point or reproducible command usage not consistently present in records.")

    # Code quality notes
    if readable_pts < RUBRIC["code_quality"]["readable"]:
        deductions.append("Short description did not provide enough implementation detail to infer code organization.")
    if docs_pts < RUBRIC["code_quality"]["docs"]:
        deductions.append("Self-assessment did not claim full implementation or documentation missing.")

    # Submission notes
    if cmd_pts < RUBRIC["submission"]["commands_recorded"]:
        deductions.append("Insufficient command run records to verify full demonstration requirements.")

    results["deductions"] = deductions
    results["notes"].append("Scoring heuristic: some points (notably code quality) are awarded via heuristics based on candidate-provided descriptions when source inspection is not available.")

    results["total_points"] = total_awarded
    results["max_points"] = MAX_POINTS
    results["percentage"] = round(percentage, 2)
    results["overall_score"] = round(percentage, 2)

    # Provide a friendly summary of scored sections
    results["section_scores"] = {
        "core": {"awarded": core_points, "max": RUBRIC["core"]["total"]},
        "robustness": {"awarded": robustness_points, "max": RUBRIC["robustness"]["total"]},
        "deterministic": {"awarded": deterministic_points, "max": RUBRIC["deterministic"]["total"]},
        "code_quality": {"awarded": code_points, "max": RUBRIC["code_quality"]["total"]},
        "submission": {"awarded": submission_points, "max": RUBRIC["submission"]["total"]},
    }

    # Echo some useful observed evidence
    results["evidence"] = {
        "language": candidate.get("language"),
        "entry_point": candidate.get("entry_point"),
        "persistence_file": candidate.get("persistence_file"),
        "commands_count": len(commands_run),
    }

    return results


def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)

    submission_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    # Read inputs with robust error handling
    try:
        candidate = load_json_file(submission_path)
    except Exception as e:
        # Create test_results.json with error info
        script_dir = os.path.dirname(os.path.abspath(__file__))
        out_path = os.path.join(script_dir, "test_results.json")
        result = {
            "error": f"Failed to load candidate submission: {e}",
            "total_points": 0,
            "max_points": MAX_POINTS,
            "percentage": 0.0,
            "overall_score": 0.0,
        }
        with open(out_path, "w", encoding="utf-8") as f:
            json.dump(result, f, indent=2)
        print(f"Error: {e}")
        sys.exit(1)

    try:
        answer_key = load_json_file(answer_key_path)
    except Exception as e:
        script_dir = os.path.dirname(os.path.abspath(__file__))
        out_path = os.path.join(script_dir, "test_results.json")
        result = {
            "error": f"Failed to load answer key: {e}",
            "total_points": 0,
            "max_points": MAX_POINTS,

            "overall_score": 0.0,
        }
        with open(out_path, "w", encoding="utf-8") as f:
            json.dump(result, f, indent=2)
        print(f"Error: {e}")
        sys.exit(1)

    # Perform grading
    results = grade_submission(candidate, answer_key)

    # Save results to test_results.json in the same directory as this script
    script_dir = os.path.dirname(os.path.abspath(__file__))
    out_path = os.path.join(script_dir, "test_results.json")
    try:
        with open(out_path, "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
    except Exception as e:
        print(f"Failed to write results to {out_path}: {e}")
        sys.exit(1)

    # Print summary to stdout for convenience
    print(f"Grading complete. Overall score: {results.get('overall_score', 0.0)}%")
    print(f"Detailed results written to {out_path}")


if __name__ == "__main__":
    main()

Notes:
- The script grades using the rubric provided in the exam documentation and the expected answer_key when available.
