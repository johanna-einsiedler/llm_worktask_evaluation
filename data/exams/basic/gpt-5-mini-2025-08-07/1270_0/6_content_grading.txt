# task_evaluation.py
import json
import sys
import os
import re

"""
Automated grader for the Basic Inventory CLI practical exam.

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

Produces:
    test_results.json in the current working directory containing a detailed
    breakdown and the overall_score numeric percentage (0-100) as required.

Notes:
- This grader is tolerant of formatting differences (quotes, whitespace).
- It uses simple substring/regex matching to map candidate commands to expected actions.
- It relies primarily on the contents of test_submission.json (candidate) and
  uses answer_key.json when present to compare expected steps.
- Only standard Python libraries are used.
"""

# Scoring rubric constants
MAX_POINTS = 100.0
# Breakdown based on the exam rubric
POINTS = {
    "core": {
        "add": 8,
        "list": 8,
        "find": 8,
        "update": 8,
        "delete": 8,
        "persistence": 15,
        "exit_codes_and_errors": 5
    },  # total 60
    "robustness": {
        "validation": 6,
        "clear_errors": 4
    },  # total 10
    "deterministic": {
        "json_output": 6,
        "cli_contract": 4
    },  # total 10
    "code_quality": 10,
    "submission_completeness": {
        "json_present": 5,
        "commands_recorded": 5
    }  # total 10
}

# Helper functions ------------------------------------------------------------

def safe_load_json(path):
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except Exception as e:
        raise ValueError(f"Failed to load JSON from {path}: {e}")

def normalize_whitespace(s):
    return re.sub(r'\s+', ' ', s.strip()) if isinstance(s, str) else s

def contains_flag(cmd, flag, value):
    """
    Checks whether a command string contains a flag with a given value.
    Accepts patterns:
      --flag value
      --flag="value"
      --flag='value'
      --flag=value
    """
    if not isinstance(cmd, str) or not isinstance(flag, str):
        return False
    patterns = [
        rf'{re.escape(flag)}\s+{re.escape(value)}',
        rf'{re.escape(flag)}="{re.escape(value)}"',
        rf"{re.escape(flag)}='{re.escape(value)}'",
        rf'{re.escape(flag)}={re.escape(value)}'
    ]
    cmd_compact = cmd
    for p in patterns:
        if re.search(p, cmd_compact):
            return True
    return False

def extract_flag_value(cmd, flag):
    """
    Tries to extract the value of a flag from a command string.
    Returns the first match or None.
    """
    if not isinstance(cmd, str):
        return None
    # match --flag value or --flag="value" or --flag='value' or --flag=value
    regexes = [
        rf'{re.escape(flag)}\s+(".*?"|\'.*?\'|\S+)',
        rf'{re.escape(flag)}=(".*?"|\'.*?\'|\S+)',
        rf"{re.escape(flag)}=('.*?'|\S+)"
    ]
    for reg in regexes:
        m = re.search(reg, cmd)
        if m:
            val = m.group(1)
            # strip surrounding quotes if present
            if (val.startswith('"') and val.endswith('"')) or (val.startswith("'") and val.endswith("'")):
                val = val[1:-1]
            return val
    return None

def parse_json_stdout(stdout_str):
    """
    Attempts to parse stdout as JSON. Returns (parsed_object, error_message).
    If parse successful, error_message is None.
    """
    if not isinstance(stdout_str, str):
        return None, "stdout not a string"
    s = stdout_str.strip()
    if not s:
        return None, "empty stdout"
    try:
        parsed = json.loads(s)
        return parsed, None
    except Exception as e:
        return None, f"JSON parse error: {e}"

def find_command_runs_by_action(cmds, action):
    """
    Return list of indices of commands whose 'cmd' contains the action word
    as a standalone token (e.g., ' add ' or starting with 'add').
    """
    indices = []
    for i, entry in enumerate(cmds):
        cmd = entry.get("cmd", "")
        # match action as separate token: ^action | \saction\s
        if re.search(rf'(^|\s){re.escape(action)}(\s|$)', cmd):
            indices.append(i)
    return indices

def list_skus_from_json_items(items):
    """
    Given a list of item dicts, return set of sku strings found.
    """
    skus = set()
    if isinstance(items, list):
        for it in items:
            if isinstance(it, dict) and 'sku' in it:
                skus.add(str(it['sku']))
    return skus

# Main grading logic ----------------------------------------------------------

def grade_submission(candidate, answer_key):
    results = {
        "per_item_scores": {},
        "deductions": [],
        "notes": [],
    }

    total_points_awarded = 0.0

    # Basic sanity checks
    required_keys = ["language", "entry_point", "persistence_file", "commands_run", "short_description", "self_assessment", "time_taken_minutes"]
    missing_keys = [k for k in required_keys if k not in candidate]
    if missing_keys:
        results["deductions"].append(f"Missing required keys in submission JSON: {missing_keys}")
        # Continue but missing items will reduce scores where applicable

    commands_run = candidate.get("commands_run", [])
    if not isinstance(commands_run, list):
        results["deductions"].append("commands_run is not an array.")
        commands_run = []

    # Build helper indexes
    # Map action -> list of (index, entry)
    action_indices = {}
    for action in ["add", "list", "update", "delete", "find"]:
        idxs = find_command_runs_by_action(commands_run, action)
        action_indices[action] = idxs

    # ---- Core Functionality checks (60 points) -----------------------------
    core_awarded = 0.0
    core_details = {}

    # Helper to check if there is at least one successful run for an action
    def has_successful_action(action):
        for i in action_indices.get(action, []):
            entry = commands_run[i]
            exit_code = entry.get("exit_code", 1)
            if exit_code == 0:
                return True, i, entry
        return False, None, None

    # add
    add_ok, add_idx, add_entry = has_successful_action("add")
    if add_ok:
        core_awarded += POINTS["core"]["add"]
        core_details["add"] = {"score": POINTS["core"]["add"], "note": "add command succeeded at least once (exit_code 0)."}
    else:
        core_details["add"] = {"score": 0, "note": "No successful add command recorded (exit_code != 0 or missing)."}

    # list
    list_ok, list_idx, list_entry = has_successful_action("list")
    if list_ok:
        core_awarded += POINTS["core"]["list"]
        core_details["list"] = {"score": POINTS["core"]["list"], "note": "list command succeeded (exit_code 0)."}
    else:
        core_details["list"] = {"score": 0, "note": "No successful list command recorded."}

    # find
    find_ok, find_idx, find_entry = has_successful_action("find")
    if find_ok:
        core_awarded += POINTS["core"]["find"]
        core_details["find"] = {"score": POINTS["core"]["find"], "note": "find command succeeded (exit_code 0)."}
    else:
        core_details["find"] = {"score": 0, "note": "No successful find command recorded."}

    # update
    update_ok, update_idx, update_entry = has_successful_action("update")
    if update_ok:
        core_awarded += POINTS["core"]["update"]
        core_details["update"] = {"score": POINTS["core"]["update"], "note": "update command succeeded (exit_code 0)."}
    else:
        core_details["update"] = {"score": 0, "note": "No successful update command recorded."}

    # delete
    delete_ok, delete_idx, delete_entry = has_successful_action("delete")
    if delete_ok:
        core_awarded += POINTS["core"]["delete"]
        core_details["delete"] = {"score": POINTS["core"]["delete"], "note": "delete command succeeded (exit_code 0)."}
    else:
        core_details["delete"] = {"score": 0, "note": "No successful delete command recorded."}

    # persistence (15 points)
    persistence_award = 0.0
    persistence_note = ""
    # Strategy: find add commands that successfully added some SKUs, then find a later list command
    # that still contains those SKUs (preferably a list that appears after some intervening commands).
    added_skus = set()
    for i in action_indices.get("add", []):
        entry = commands_run[i]
        if entry.get("exit_code", 1) != 0:
            continue
        # try to extract --sku value from cmd or parse stdout JSON for item.sku
        cmd = entry.get("cmd", "")
        sku = extract_flag_value(cmd, "--sku") or extract_flag_value(cmd, "-s")
        if not sku:
            # try parse stdout JSON
            parsed, err = parse_json_stdout(entry.get("stdout", ""))
            if parsed and isinstance(parsed, dict):
                it = parsed.get("item") or parsed
                if isinstance(it, dict) and 'sku' in it:
                    sku = str(it['sku'])
        if sku:
            added_skus.add(sku)
    # find a list command that occurs after at least one add and contains at least one of the added SKUs
    persistence_verified = False
    if added_skus:
        # find earliest add index
        earliest_add_idx = min([i for i in action_indices.get("add", [])]) if action_indices.get("add") else None
        # find list commands that occur after earliest add
        for i in action_indices.get("list", []):
            if earliest_add_idx is not None and i <= earliest_add_idx:
                continue
            entry = commands_run[i]
            stdout = entry.get("stdout", "")
            # if the list was produced with --format json, try parse JSON and get skus
            parsed, err = parse_json_stdout(stdout)
            found_skus = set()
            if parsed is not None:
                # parsed could be list or dict
                if isinstance(parsed, list):
                    found_skus = list_skus_from_json_items(parsed)
                elif isinstance(parsed, dict):
                    # maybe single item
                    if 'sku' in parsed:
                        found_skus = {str(parsed['sku'])}
            else:
                # fallback to simple substring search in stdout
                for sku in added_skus:
                    if sku in stdout:
                        found_skus.add(sku)
            # if any added sku present, persistence verified
            if found_skus & added_skus:
                persistence_verified = True
                persistence_award = POINTS["core"]["persistence"]
                persistence_note = f"Found persisted SKU(s) in later list: {sorted(list(found_skus & added_skus))}"
                break
    if not persistence_verified:
        persistence_award = 0.0
        persistence_note = "Could not verify persistence across invocations from recorded commands."

    core_awarded += persistence_award
    core_details["persistence"] = {"score": persistence_award, "note": persistence_note}

    # exit codes and errors (5 points)
    exit_award = 0.0
    exit_note = ""
    # Check that known error cases (if present) returned non-zero exit codes.
    # We'll search for commands that look like invalid add (missing args) or duplicates/negative qty by looking at stderr messages recorded.
    # Use answer_key expected errors if provided otherwise infer from commands_run content.
    expected_error_cases = []
    if answer_key and isinstance(answer_key, dict):
        expected_error_cases = [e.get("cmd", "") for e in answer_key.get("commands_run", []) if e.get("exit_code", 0) != 0]
    # We'll verify that for each candidate error-like command in their run, non-zero exit code and stderr present.
    error_checks = []
    for i, entry in enumerate(commands_run):
        # treat any entry with non-empty stderr as an error run candidate
        if entry.get("stderr"):
            if entry.get("exit_code", 0) == 0:
                error_checks.append((i, False, "stderr present but exit_code == 0"))
            else:
                error_checks.append((i, True, "stderr and non-zero exit code as expected"))
    # For scoring, if at least one error run exists and uses non-zero exit code and stderr, award full
    if error_checks:
        good = sum(1 for _, ok, _ in error_checks if ok)
        bad = sum(1 for _, ok, _ in error_checks if not ok)
        if good >= 1 and bad == 0:
            exit_award = POINTS["core"]["exit_codes_and_errors"]
            exit_note = f"Found {good} error run(s) with stderr and non-zero exit_code."
        elif good >= 1:
            exit_award = POINTS["core"]["exit_codes_and_errors"] * 0.5
            exit_note = f"Some error runs have correct behavior ({good} good, {bad} bad)."
        else:
            exit_award = 0.0
            exit_note = "Error runs present but none used non-zero exit codes consistently."
    else:
        # No error runs recorded; if expected error cases present in answer_key, this is a deduction
        if expected_error_cases:
            exit_award = 0.0
            exit_note = "No error runs recorded in submission; expected some error demonstrations (e.g., duplicate SKU or negative qty)."
        else:
            # No error runs required; award partial credit (since exit code behavior can't be judged)
            exit_award = POINTS["core"]["exit_codes_and_errors"] * 0.5
            exit_note = "No explicit error runs recorded; partial credit granted."

    core_awarded += exit_award
    core_details["exit_codes_and_errors"] = {"score": exit_award, "note": exit_note}

    total_points_awarded += core_awarded
    results["per_item_scores"]["core_functionality"] = {"score": core_awarded, "max": 60, "details": core_details}

    # ---- Robustness & Input validation (10 points) -------------------------
    robust_awarded = 0.0
    robust_details = {}

    # validation (6 points): check duplicate SKU and negative qty error demonstration
    dup_award = 0.0
    neg_award = 0.0
    dup_found = False
    neg_found = False
    for entry in commands_run:
        cmd = entry.get("cmd", "")
        stderr = entry.get("stderr", "") or ""
        exit_code = entry.get("exit_code", 0)
        # duplicate detection: presence of adding an already-added SKU with non-zero exit_code and stderr mentioning 'exists' or 'duplicate'
        if re.search(r'\badd\b', cmd) and exit_code != 0:
            if re.search(r'(?i)(exists|duplicate|already)', stderr) or re.search(r'(?i)(exists|duplicate|already)', entry.get("stdout", "") or ""):
                dup_found = True
        # negative qty detection: add with negative qty in cmd or stderr mentions qty and non-zero exit
        if re.search(r'\badd\b', cmd):
            qty_val = extract_flag_value(cmd, "--qty") or extract_flag_value(cmd, "-q")
            if qty_val and re.match(r'-\d+', qty_val.strip()):
                # check that it errored
                if exit_code != 0:
                    neg_found = True
            else:
                # maybe they attempted then stderr indicates negative qty
                if exit_code != 0 and re.search(r'(?i)(qty.*(negative|>= 0|must be|must be >=))', stderr):
                    neg_found = True
    if dup_found and neg_found:
        robust_awarded += POINTS["robustness"]["validation"]
        robust_details["validation"] = {"score": POINTS["robustness"]["validation"], "note": "Duplicate SKU and negative-qty validations demonstrated."}
    elif dup_found or neg_found:
        robust_awarded += POINTS["robustness"]["validation"] * 0.5
        robust_details["validation"] = {"score": POINTS["robustness"]["validation"] * 0.5, "note": f"Partial validation: duplicate_found={dup_found}, negative_found={neg_found}."}
    else:
        robust_details["validation"] = {"score": 0, "note": "No evidence of duplicate or negative-qty validation in recorded runs."}

    # clear_errors (4 points): stderr messages and non-zero exit codes
    clear_award = 0.0
    # if we earlier detected good error_runs (error_checks), award full for clear_errors
    good_error_runs = [e for e in error_checks if e[1]]
    if good_error_runs:
        clear_award = POINTS["robustness"]["clear_errors"]
        robust_details["clear_errors"] = {"score": clear_award, "note": f"{len(good_error_runs)} error run(s) with stderr and non-zero exit codes."}
    else:
        # partial credit if there are some stderr entries but inconsistent exit codes
        if error_checks:
            clear_award = POINTS["robustness"]["clear_errors"] * 0.5
            robust_details["clear_errors"] = {"score": clear_award, "note": "Some error runs exist but inconsistent exit codes."}
        else:
            robust_details["clear_errors"] = {"score": 0, "note": "No error runs to judge clear error messaging."}

    robust_awarded += clear_award
    total_points_awarded += robust_awarded
    results["per_item_scores"]["robustness_and_validation"] = {"score": robust_awarded, "max": 10, "details": robust_details}

    # ---- Deterministic & Scriptable outputs (10 points) -------------------
    det_awarded = 0.0
    det_details = {}

    # json_output (6 points): check at least one list/find with --format json and stdout valid JSON
    json_good = False
    json_bad_examples = []
    for i, entry in enumerate(commands_run):
        cmd = entry.get("cmd", "")
        if "--format json" in cmd or "--format=json" in cmd:
            stdout = entry.get("stdout", "")
            parsed, err = parse_json_stdout(stdout)
            if parsed is not None:
                json_good = True
                break
            else:
                json_bad_examples.append((i, err))
    if json_good:
        det_awarded += POINTS["deterministic"]["json_output"]
        det_details["json_output"] = {"score": POINTS["deterministic"]["json_output"], "note": "Found at least one --format json output that is valid JSON."}
    else:
        if json_bad_examples:
            det_details["json_output"] = {"score": 0, "note": f"--format json used but outputs were not valid JSON (examples: {json_bad_examples[:3]})."}
        else:
            det_details["json_output"] = {"score": 0, "note": "No --format json usage recorded; cannot verify machine-parseable output."}

    # cli_contract (4 points): commands recorded and demonstration sequence present
    cli_award = 0.0
    cli_notes = []
    # Check presence of required demonstration: add >=3 distinct SKUs, update >=1, delete >=1, find >=1, list after restart
    skus_added = set()
    for i in action_indices.get("add", []):
        entry = commands_run[i]
        if entry.get("exit_code", 1) != 0:
            continue
        sku = extract_flag_value(entry.get("cmd", ""), "--sku") or extract_flag_value(entry.get("cmd", ""), "-s")
        if not sku:
            # try reading stdout JSON item.sku
            parsed, err = parse_json_stdout(entry.get("stdout", ""))
            if parsed and isinstance(parsed, dict):
                it = parsed.get("item") or parsed
                if isinstance(it, dict) and 'sku' in it:
                    sku = str(it['sku'])
        if sku:
            skus_added.add(sku)
    updated = any(entry.get("exit_code", 1) == 0 for idx, entry in enumerate(commands_run) if re.search(r'(^|\s)update(\s|$)', entry.get("cmd", "")))
    deleted = any(entry.get("exit_code", 1) == 0 for idx, entry in enumerate(commands_run) if re.search(r'(^|\s)delete(\s|$)', entry.get("cmd", "")))
    found = any(entry.get("exit_code", 1) == 0 for idx, entry in enumerate(commands_run) if re.search(r'(^|\s)find(\s|$)', entry.get("cmd", "")))

    # list-after-restart detection: check that there exists add then later a list that includes added sku (already used persistence logic)
    list_after_restart = persistence_verified

    conditions = {
        "added_3_or_more": len(skus_added) >= 3,
        "updated": updated,
        "deleted": deleted,
        "found": found,
        "persistence_demo": list_after_restart
    }
    missing_conditions = [k for k, v in conditions.items() if not v]
    if all(conditions.values()):
        cli_award = POINTS["deterministic"]["cli_contract"]
        cli_notes.append("Recorded commands include add >=3, update, delete, find, and persistence demonstration.")
    else:
        # partial credit proportional to number of conditions met
        met = sum(1 for v in conditions.values() if v)
        cli_award = POINTS["deterministic"]["cli_contract"] * (met / len(conditions))
        cli_notes.append(f"Conditions met: {met}/{len(conditions)}; missing: {missing_conditions}")

    det_awarded += cli_award
    total_points_awarded += det_awarded
    det_details["cli_contract"] = {"score": cli_award, "note": " ".join(cli_notes)}
    results["per_item_scores"]["deterministic_and_scriptable"] = {"score": det_awarded, "max": 10, "details": det_details}

    # ---- Code quality & organization (10 points) --------------------------
    # We cannot inspect code files from test_submission.json. Use short_description & self_assessment heuristics.
    code_award = 0.0
    code_notes = []
    short_desc = candidate.get("short_description", "") or ""
    sa = candidate.get("self_assessment", {}) or {}
    impl_req = sa.get("implemented_required", False)

    # heuristics:
    # - if implemented_required True -> partial credit
    # - if short_description mentions 'modular', 'functions', 'atomic', 'json', 'persistence' -> award more
    sd_lower = short_desc.lower()
    keywords = ["modular", "function", "functions", "atomic", "temp", "json", "persistence", "sqlite", "sqlite3", "readable", "organize", "cli"]
    kcount = sum(1 for k in keywords if k in sd_lower)
    if impl_req:
        code_award += 5.0  # baseline for claiming implemented
        code_notes.append("Self-assessment indicates required features implemented.")
    # award up to remaining points proportional to keyword matches
    code_award += min(5.0, (kcount / len(keywords)) * 5.0)
    if code_award > POINTS["code_quality"]:
        code_award = POINTS["code_quality"]
    code_notes.append(f"Keyword matches in short_description: {kcount}/{len(keywords)}.")
    total_points_awarded += code_award
    results["per_item_scores"]["code_quality"] = {"score": code_award, "max": 10, "details": {"notes": code_notes}}

    # ---- Submission completeness & clarity (10 points) ---------------------
    submission_award = 0.0
    submission_details = {}

    # json present (5 points)
    submission_award_json = 0.0
    if isinstance(candidate, dict):
        # We have already loaded it, so give full points for presence
        submission_award_json = POINTS["submission_completeness"]["json_present"]
        submission_details["json_present"] = {"score": submission_award_json, "note": "test_submission.json successfully parsed."}
    else:
        submission_details["json_present"] = {"score": 0, "note": "test_submission.json missing or unparsable."}
    submission_award += submission_award_json

    # commands recorded (5 points): check each command has cmd, stdout, stderr, exit_code
    if commands_run and isinstance(commands_run, list):
        good_count = 0
        for entry in commands_run:
            if all(k in entry for k in ("cmd", "stdout", "stderr", "exit_code")):
                good_count += 1
        proportion = good_count / len(commands_run) if len(commands_run) > 0 else 0
        submission_award_commands = POINTS["submission_completeness"]["commands_recorded"] * proportion
        submission_award += submission_award_commands
        submission_details["commands_recorded"] = {"score": submission_award_commands, "note": f"{good_count}/{len(commands_run)} command entries include required fields."}
    else:
        submission_details["commands_recorded"] = {"score": 0, "note": "No commands_run recorded."}

    total_points_awarded += submission_award
    results["per_item_scores"]["submission_completeness"] = {"score": submission_award, "max": 10, "details": submission_details}

    # Final aggregation and extra comparisons against answer_key (if provided)
    # If answer_key has expected commands_run, compare presence and list mismatches
    comparison_notes = []
    if answer_key and isinstance(answer_key, dict) and "commands_run" in answer_key:
        expected_cmds = answer_key.get("commands_run", [])
        # For each expected, try to find a candidate match (by checking that candidate recorded a command
        # that contains key tokens present in expected cmd: action and relevant --sku value(s))
        unmatched_expected = []
        matched_count = 0
        for expected in expected_cmds:
            exp_cmd = expected.get("cmd", "")
            # attempt to identify important tokens: action + any --sku values
            action_match = re.search(r'^\s*(\S+)', exp_cmd)
            action = action_match.group(1) if action_match else None
            # collect SKUs mentioned in expected cmd
            skus_in_exp = re.findall(r'--sku\s+(".*?"|\'.*?\'|\S+)|--sku=(".*?"|\'.*?\'|\S+)|--sku=(\S+)', exp_cmd)
            skus = set()
            for tup in skus_in_exp:
                for part in tup:
                    if part:
                        v = part.strip()
                        if (v.startswith('"') and v.endswith('"')) or (v.startswith("'") and v.endswith("'")):
                            v = v[1:-1]
                        skus.add(v)
            # find candidate command that contains action and (if skus present) the sku
            found = False
            for entry in commands_run:
                c = entry.get("cmd", "")
                if action and not re.search(rf'(^|\s){re.escape(action)}(\s|$)', c):
                    continue
                # if expected had --format json, require candidate cmd to have that too (prefer match)
                if "--format json" in exp_cmd and "--format json" not in c and "--format=json" not in c:
                    # still allow match but mark as partial
                    pass
                # check skus
                if skus:
                    all_skus_matched = all(contains_flag(c, "--sku", sku) or (sku in c) for sku in skus)
                    if not all_skus_matched:
                        continue
                # consider this a match
                found = True
                break
            if found:
                matched_count += 1
            else:
                unmatched_expected.append(exp_cmd)
        comparison_notes.append(f"Matched {matched_count}/{len(expected_cmds)} expected answer-key commands.")
        if unmatched_expected:
            comparison_notes.append("Unmatched expected commands (examples): " + "; ".join(unmatched_expected[:5]))
    if comparison_notes:
        results["notes"].extend(comparison_notes)

    # Compose final scores summary
    detailed = results["per_item_scores"]
    # Ensure sum is capped at MAX_POINTS
    total_awarded = round(min(total_points_awarded, MAX_POINTS), 2)
    overall_percentage = (total_awarded / MAX_POINTS) * 100.0

    # Build output structure
    output = {
        "scores": detailed,
        "deductions": results.get("deductions", []),
        "notes": results.get("notes", []),
        "total_points_awarded": total_awarded,
        "max_points": MAX_POINTS,
        "percentage": round(overall_percentage, 2),
        # required top-level variable
        "overall_score": round(overall_percentage, 2)
    }

    # Also include some human-readable explanations for main areas
    output["explanations"] = {
        "core_summary": f"Core functionality: {detailed.get('core_functionality', {}).get('score', 0)} / 60",
        "robustness_summary": f"Robustness: {detailed.get('robustness_and_validation', {}).get('score', 0)} / 10",
        "deterministic_summary": f"Deterministic outputs: {detailed.get('deterministic_and_scriptable', {}).get('score', 0)} / 10",
        "code_quality_summary": f"Code quality heuristics: {detailed.get('code_quality', {}).get('score', 0)} / 10",
        "submission_completeness_summary": f"Submission completeness: {detailed.get('submission_completeness', {}).get('score', 0)} / 10"
    }

    return output

def main():
    if len(sys.argv) < 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)
    submission_path = sys.argv[1]
    answer_key_path = sys.argv[2]

    # Load files with robust error handling
    try:
        candidate = safe_load_json(submission_path)
    except Exception as e:
        err_out = {
            "error": f"Failed to load candidate submission JSON: {e}"
        }
        with open("test_results.json", "w", encoding="utf-8") as f:
            json.dump(err_out, f, indent=2)
        print(err_out["error"])
        sys.exit(1)

    try:
        answer_key = safe_load_json(answer_key_path)
    except Exception as e:
        # Answer key missing or invalid: still attempt grading with candidate only but warn
        answer_key = None
        print(f"Warning: Failed to load answer key JSON ({answer_key_path}): {e}. Grading will proceed with heuristic checks.")

    # Perform grading
    try:
        results = grade_submission(candidate, answer_key)
    except Exception as e:
        err_out = {
            "error": f"Internal grading error: {e}"
        }
        with open("test_results.json", "w", encoding="utf-8") as f:
            json.dump(err_out, f, indent=2)
        print(err_out["error"])
        sys.exit(1)

    # Save results
    try:
        with open("test_results.json", "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2)
    except Exception as e:
        print(f"Failed to write test_results.json: {e}")
        sys.exit(1)

    # Print a short summary to stdout for convenience
    print(f"Grading complete. overall_score = {results.get('overall_score'):.2f}%")
    print("Detailed results written to test_results.json")

if __name__ == "__main__":
    main()
