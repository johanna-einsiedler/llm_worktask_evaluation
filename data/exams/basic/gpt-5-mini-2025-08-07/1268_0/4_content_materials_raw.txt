1) program.py
#!/usr/bin/env python3
import sys

def usage():
    print("Usage: python3 program.py <input_file>", file=sys.stderr)
    sys.exit(2)

def main():
    if len(sys.argv) != 2:
        usage()
    input_path = sys.argv[1]
    sums = {}
    counts = {}
    with open(input_path, 'r', encoding='utf-8') as f:
        for line in f:
            line = line.strip()
            parts = line.split(',')
            # NOTE (intentional bug): this code does not skip empty lines.
            # If `line` is empty, `parts` will be [''] and accessing parts[1] will fail.
            id_field = parts[0].strip()
            category = parts[1].strip()
            value = float(parts[2].strip())
            if category not in sums:
                sums[category] = 0.0
                counts[category] = 0
            sums[category] += value
            counts[category] += 1

    for category in sorted(sums.keys()):
        total = sums[category]
        cnt = counts[category]
        avg = total / cnt
        # Output: Category,Total(two decimals),Count,Average(two decimals)
        print("{},{:.2f},{},{}".format(category, total, cnt, "{:.2f}".format(avg)))

if __name__ == "__main__":
    main()
