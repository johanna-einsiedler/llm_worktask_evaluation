# task_evaluation.py
import json
import os
import re
import sys

"""
Automated grading script for the Practical Debugging & Fixing — Basic Level exam.

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

Produces:
    test_results.json in the same directory as this script.

Scoring (total 100):
- Functional correctness: 60 points (proportional to tests passed / total tests)
- Explanation quality: 20 points (per-task, heuristics on presence and content)
- Code hygiene / minimal changes: 10 points (checks modified files scope)
- Completeness & reproducibility: 10 points (runtime outputs, commands logs)

The script is defensive about missing fields and reports clear reasons for deductions.
"""

# ---------- Helper utilities ----------

def safe_load_json(path):
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f), None
    except Exception as e:
        return None, str(e)

def normalize_text(s):
    if s is None:
        return ""
    return re.sub(r'\r\n', '\n', str(s)).strip()

def parse_pytest_summary(output_text):
    """
    Parse pytest-like output to extract numbers: collected, passed, failed.
    Returns dict with keys 'collected','passed','failed' (ints or None).
    """
    out = output_text or ""
    res = {'collected': None, 'passed': None, 'failed': None}

    # Attempt "collected X items"
    m = re.search(r'collected\s+(\d+)\s+items', out)
    if not m:
        m = re.search(r'collected\s+(\d+)', out)
    if m:
        res['collected'] = int(m.group(1))

    # "X passed"
    m = re.search(r'(\d+)\s+passed', out)
    if m:
        res['passed'] = int(m.group(1))

    # "X failed"
    m = re.search(r'(\d+)\s+failed', out)
    if m:
        res['failed'] = int(m.group(1))

    # pytest sometimes prints "5 passed in 0.02s"
    if res['passed'] is None:
        m = re.search(r'(\d+)\s+passed\s+in', out)
        if m:
            res['passed'] = int(m.group(1))

    # If no explicit failed but we have collected and passed, infer failed
    if res['collected'] is not None and res['passed'] is not None and res['failed'] is None:
        res['failed'] = max(0, res['collected'] - res['passed'])

    return res

def explanation_score(explanation):
    """
    Heuristic scoring for explanation quality per-task.
    Returns score 0..5 and reasons.
    """
    reasons = []
    if not explanation or not explanation.strip():
        return 0, "Empty explanation."
    text = explanation.strip()
    char_len = len(text)
    # Sentence count heuristic
    sentences = [s for s in re.split(r'[.!?]+', text) if s.strip()]
    sentence_count = len(sentences)

    score = 0
    # baseline for some content and minimal length
    if char_len >= 40 and sentence_count >= 1:
        score += 2
        reasons.append("Contains sufficient length.")
    elif char_len >= 20:
        score += 1
        reasons.append("Short explanation but non-empty.")

    # Keywords indicating root-cause and specific fix
    keywords = ['root', 'cause', 'fix', 'fixed', 'because', 'replace', 'change', 'add', 'remove', 'skip', 'skip non', 'isinstance', 'strip', 'max(', 'min(']
    if any(k in text.lower() for k in keywords):
        score += 2
        reasons.append("Mentions root cause or fix keywords.")

    # Mentions filenames or functions implies concrete reasoning
    if re.search(r'\b(app/|calculations|utils|main|sum_numbers|max_gap|normalize_names)\b', text, re.I):
        score += 1
        reasons.append("Refers to code elements (files/functions).")

    score = min(5, score)
    reason_text = "; ".join(reasons) if reasons else "No strong indicators."
    return score, reason_text

# ---------- Main grading logic ----------

def grade(candidate, answer_key):
    result = {
        "candidate_name": candidate.get("candidate_name", "<unknown>"),
        "errors": [],
        "scores": {},
        "per_task": [],
        "total_points": 0,
        "max_points": 100,
        "percentage": 0.0,
        "overall_score": 0.0
    }

    # Validate presence of expected sections
    if 'final_test_summary' not in answer_key:
        result['errors'].append("Answer key missing 'final_test_summary'. Cannot grade functional correctness.")
        # We will continue with some checks but functional will be 0.
    # Extract totals from answer key if available
    ak_total_tests = None
    try:
        ak_total_tests = int(answer_key.get('final_test_summary', {}).get('total_tests')) if answer_key.get('final_test_summary') else None
    except Exception:
        ak_total_tests = None

    # Candidate final tests info
    cand_final = candidate.get('final_test_summary', {})
    cand_passed = None
    cand_failed = None
    cand_total = None
    try:
        cand_passed = int(cand_final.get('passed')) if cand_final.get('passed') is not None else None
        cand_failed = int(cand_final.get('failed')) if cand_final.get('failed') is not None else None
        cand_total = int(cand_final.get('total_tests')) if cand_final.get('total_tests') is not None else None
    except Exception:
        pass

    # Parse runtime_output for counts
    parsed = parse_pytest_summary(cand_final.get('runtime_output', ""))
    parsed_passed = parsed.get('passed')
    parsed_failed = parsed.get('failed')
    parsed_collected = parsed.get('collected')

    # Functional correctness (60 points)
    func_max = 60
    functional_reasons = []
    func_score = 0.0
    if cand_passed is None and parsed_passed is None:
        functional_reasons.append("No 'passed' count in candidate final_test_summary and runtime_output didn't parse passed tests.")
        func_score = 0.0
    else:
        # Prefer explicit cand_passed, else parsed_passed
        passed = cand_passed if cand_passed is not None else parsed_passed
        total = cand_total if cand_total is not None else (parsed_collected if parsed_collected is not None else ak_total_tests)
        if total is None or total == 0:
            functional_reasons.append("Total tests count unknown; cannot compute functional score. Awarding 0.")
            func_score = 0.0
        else:
            ratio = max(0.0, min(1.0, float(passed) / float(total)))
            func_score = round(func_max * ratio, 2)
            functional_reasons.append(f"Candidate reported {passed}/{total} tests passed -> functional score {func_score}/{func_max}.")

    result['scores']['functional'] = {
        "score": func_score,
        "max": func_max,
        "details": functional_reasons
    }

    # Explanation quality (20 points) -> per-task heuristics
    expl_max = 20
    expl_score_total = 0.0
    expl_reasons = []
    tasks = candidate.get('tasks', [])
    if not tasks:
        expl_reasons.append("No tasks found in candidate submission; explanation score 0.")
    else:
        # Expect 4 tasks A-D
        per_task_explanations = {}
        for t in tasks:
            tid = t.get('id', '<no-id>')
            explanation = t.get('explanation', '')
            sc, reason = explanation_score(explanation)
            # each task has max 5 points
            per_task_explanations[tid] = {
                "score": sc,
                "max": 5,
                "reason": reason
            }
            expl_score_total += sc
        expl_reasons.append("Per-task explanation scores computed.")
    expl_score_total = round(expl_score_total, 2)
    result['scores']['explanation'] = {
        "score": expl_score_total,
        "max": expl_max,
        "details": expl_reasons,
        "per_task": per_task_explanations if tasks else {}
    }

    # Code hygiene / minimal changes (10 points)
    hygiene_max = 10
    hygiene_reasons = []
    hygiene_score = 0.0
    # Allowed files: must be under app/; tests and run_tests.sh must NOT be modified.
    # Collect all modified filenames from candidate
    cand_modified_filenames = []
    for t in tasks:
        for mf in t.get('modified_files', []):
            fname = mf.get('filename')
            if fname:
                cand_modified_filenames.append(fname)
    cand_modified_set = set(cand_modified_filenames)

    # Collect expected modified files from answer key (union across tasks)
    ak_modified_set = set()
    for t in answer_key.get('tasks', []):
        for mf in t.get('modified_files', []):
            fname = mf.get('filename')
            if fname:
                ak_modified_set.add(fname)

    # Check for forbidden modifications (tests/ or run_tests.sh or files outside app/)
    forbidden = [f for f in cand_modified_set if not f.startswith('app/')]
    if forbidden:
        hygiene_reasons.append(f"Modified files outside 'app/': {sorted(list(forbidden))}. This violates rules.")
        # Partial credit if only minor infractions
        hygiene_score = 2.0
    else:
        # All modifications are inside app/
        # If candidate modified exactly the expected set or a subset, full score.
        if ak_modified_set:
            # Compare content equality if answer key provides contents
            # Build mapping from filename->content in answer key
            ak_file_to_content = {}
            for t in answer_key.get('tasks', []):
                for mf in t.get('modified_files', []):
                    if 'filename' in mf and 'content' in mf:
                        ak_file_to_content[mf['filename']] = normalize_text(mf['content'])
            # Build candidate mapping as well
            cand_file_to_content = {}
            for t in tasks:
                for mf in t.get('modified_files', []):
                    if 'filename' in mf and 'content' in mf:
                        cand_file_to_content[mf['filename']] = normalize_text(mf['content'])
            # Compare for exact matches for expected files
            matched = []
            mismatched = []
            missing = []
            for fname in ak_modified_set:
                akc = ak_file_to_content.get(fname)
                cnd = cand_file_to_content.get(fname)
                if cnd is None:
                    missing.append(fname)
                else:
                    # compare normalized content
                    if akc == cnd:
                        matched.append(fname)
                    else:
                        mismatched.append(fname)
            # Score heuristics
            if matched and not mismatched and not missing:
                hygiene_score = hygiene_max
                hygiene_reasons.append("Modified files match answer key contents exactly for expected files.")
            else:
                # award partial credit
                total_expected = max(1, len(ak_modified_set))
                matched_ratio = len(matched) / total_expected
                hygiene_score = round(hygiene_max * (0.5 + 0.5 * matched_ratio), 2)  # baseline 50% then scale
                if matched:
                    hygiene_reasons.append(f"Matched files: {matched}.")
                if mismatched:
                    hygiene_reasons.append(f"Files present but content differs: {mismatched}.")
                if missing:
                    hygiene_reasons.append(f"Expected files missing from candidate modified_files: {missing}.")
        else:
            # No guidance from answer key about expected modified files -> give reasonable credit if modifications are only in app/
            if cand_modified_set:
                hygiene_score = 8.0
                hygiene_reasons.append("No expected modified files in answer key, but candidate changed only files under app/ (partial credit).")
            else:
                # Candidate changed no files; likely they didn't fix anything -> score 0
                hygiene_score = 0.0
                hygiene_reasons.append("Candidate did not modify any source files under app/; no code-hygiene credit.")
    result['scores']['code_hygiene'] = {
        "score": hygiene_score,
        "max": hygiene_max,
        "details": hygiene_reasons
    }

    # Completeness & reproducibility (10 points)
    comp_max = 10
    comp_reasons = []
    comp_score = 0.0

    # Checks:
    # - runtime_output present and non-empty
    runtime_output = cand_final.get('runtime_output', '')
    if runtime_output and runtime_output.strip():
        comp_reasons.append("runtime_output is present.")
        comp_score += 3.0
    else:
        comp_reasons.append("runtime_output missing or empty.")

    # - parsed runtime_output counts consistent with reported passed/failed values
    parsed_vals = parse_pytest_summary(runtime_output)
    parsed_ok = True
    if parsed_vals.get('passed') is None:
        comp_reasons.append("Could not parse passed count from runtime_output.")
        parsed_ok = False
    if parsed_vals.get('failed') is None:
        # failed may be inferred; allow not strict
        pass

    if parsed_ok:
        # Compare to candidate final_test_summary numbers if present
        mismatches = []
        if cand_passed is not None and parsed_vals.get('passed') is not None and cand_passed != parsed_vals.get('passed'):
            mismatches.append(f"passed value mismatch: summary={cand_passed} vs parsed={parsed_vals.get('passed')}")
        if cand_failed is not None and parsed_vals.get('failed') is not None and cand_failed != parsed_vals.get('failed'):
            mismatches.append(f"failed value mismatch: summary={cand_failed} vs parsed={parsed_vals.get('failed')}")
        if mismatches:
            comp_reasons.append("Inconsistencies between final_test_summary and runtime_output: " + "; ".join(mismatches))
            # still give partial credit
            comp_score += 2.0
        else:
            comp_reasons.append("final_test_summary numbers consistent with runtime_output.")
            comp_score += 3.0
    else:
        comp_score += 0.0

    # - commands_run includes both failing and passing run snippets (heuristic)
    commands = []
    for t in tasks:
        for cmd in t.get('commands_run', []) or []:
            if isinstance(cmd, str):
                commands.append(cmd)
    cmd_text = "\n".join(commands)
    has_failing = bool(re.search(r'Error|ERROR|Traceback|SyntaxError|AttributeError|failed', cmd_text, re.I))
    has_passing = bool(re.search(r'\bpassed\b|PASSED|OK\b', cmd_text, re.I))
    if has_failing and has_passing:
        comp_reasons.append("commands_run contains both failing and passing outputs.")
        comp_score += 4.0
    elif has_passing:
        comp_reasons.append("commands_run contains passing output but no explicit failing output detected.")
        comp_score += 2.0
    elif has_failing:
        comp_reasons.append("commands_run contains failing output but no passing output detected.")
        comp_score += 1.0
    else:
        comp_reasons.append("commands_run did not contain recognizable failing or passing outputs.")
        comp_score += 0.0

    # Cap comp_score to comp_max
    comp_score = min(comp_score, comp_max)
    result['scores']['completeness'] = {
        "score": round(comp_score, 2),
        "max": comp_max,
        "details": comp_reasons
    }

    # Aggregate total
    total_points = round(func_score + expl_score_total + hygiene_score + comp_score, 2)
    percentage = round((total_points / result['max_points']) * 100, 2) if result['max_points'] > 0 else 0.0

    result['total_points'] = total_points
    result['percentage'] = percentage
    result['overall_score'] = percentage

    # Per-task detailed checks and reasons
    per_task_details = []
    # Build mapping of answer key modified files and contents for precise checks
    ak_file_to_content = {}
    for t in answer_key.get('tasks', []):
        for mf in t.get('modified_files', []):
            if 'filename' in mf and 'content' in mf:
                ak_file_to_content[mf['filename']] = normalize_text(mf['content'])

    for t in tasks:
        tid = t.get('id', '<no-id>')
        entry = {
            "id": tid,
            "status_before": t.get('status_before', ''),
            "status_after": t.get('status_after', ''),
            "modified_files_listed": [mf.get('filename') for mf in t.get('modified_files', []) if mf.get('filename')],
            "explanation": t.get('explanation', ''),
            "time_spent_minutes": t.get('time_spent_minutes'),
            "checks": []
        }
        # For each modified file, check if it's allowed and whether it matches expected content
        modified_files = t.get('modified_files', [])
        for mf in modified_files:
            fname = mf.get('filename')
            content = normalize_text(mf.get('content', ''))
            check = {"filename": fname}
            if not fname:
                check['result'] = "missing filename"
            else:
                if not fname.startswith('app/'):
                    check['result'] = "forbidden modification (outside app/)"
                else:
                    # If answer key has content for this filename, compare
                    akc = ak_file_to_content.get(fname)
                    if akc is None:
                        check['result'] = "no reference content in answer key to compare"
                    else:
                        if akc == content:
                            check['result'] = "content matches answer key"
                        else:
                            # compute small diff heuristic: whether candidate content contains expected function names or simple changes
                            if akc.strip().splitlines()[0] in content:
                                check['result'] = "content differs but contains expected header"
                            else:
                                check['result'] = "content differs from answer key"
            entry['checks'].append(check)
        per_task_details.append(entry)

    result['per_task'] = per_task_details

    # Add human-friendly summary of major deductions
    deductions = []
    # If functional score < full, note failing tests
    if func_score < func_max:
        deductions.append(f"Functional correctness incomplete: {func_score}/{func_max}. See runtime_output for failed tests.")
    # If any forbidden file modifications
    if forbidden:
        deductions.append(f"Modified forbidden files: {forbidden}. This reduces code hygiene score.")
    # If candidate did not include commands or runtime output
    if not runtime_output or not runtime_output.strip():
        deductions.append("Missing final runtime_output; reproducibility score reduced.")
    result['deductions'] = deductions

    return result

# ---------- Script entry point ----------

def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)

    cand_path = sys.argv[1]
    ak_path = sys.argv[2]

    cand_json, err = safe_load_json(cand_path)
    if err:
        print(f"Error loading candidate submission JSON from '{cand_path}': {err}")
        # still produce a basic results file with error
        out = {
            "error": f"Failed to load candidate submission: {err}"
        }
        with open("test_results.json", "w", encoding='utf-8') as fout:
            json.dump(out, fout, indent=2)
        sys.exit(1)

    ak_json, err = safe_load_json(ak_path)
    if err:
        print(f"Error loading answer key JSON from '{ak_path}': {err}")
        out = {
            "error": f"Failed to load answer key: {err}"
        }
        with open("test_results.json", "w", encoding='utf-8') as fout:
            json.dump(out, fout, indent=2)
        sys.exit(1)

    # Grade
    results = grade(cand_json, ak_json)

    # Save results
    try:
        with open("test_results.json", "w", encoding='utf-8') as out:
            json.dump(results, out, indent=2, ensure_ascii=False)
        print("Grading complete. Results written to test_results.json")
        # Print overall score line for quick reference
        print(f"Overall score: {results.get('overall_score', 0.0)}% ({results.get('total_points', 0)}/100)")
    except Exception as e:
        print(f"Failed to write test_results.json: {e}")
        sys.exit(1)


if __name__ == "__main__":
    main()
