Below is the requested Python grading script. Save it as task_evaluation.py.

It:
- Accepts two command-line args: candidate_submission.json and answer_key.json
- Loads and validates both files
- Applies the rubric described in the exam materials
- Awards automated task points (with partial credit if task tests failed but the candidate documented the correct root cause)
- Awards explanation & reproducibility points and code hygiene points
- Produces detailed test_results.json in the same directory as the script
- Exposes overall_score variable in the saved JSON (percentage 0–100)

Script:

```python
#!/usr/bin/env python3
"""
task_evaluation.py

Usage:
  python3 task_evaluation.py path/to/test_submission.json path/to/answer_key.json

Produces test_results.json in the same directory as this script.

This grader follows the rubric described in the exam materials:
- Automated task correctness: 70 points total (Task1:15, Task2:20, Task3:20, Task4:15)
- Explanation & reproducibility: 20 points (10 for change_summary/root_cause quality, 10 for how_tested evidence)
- Code hygiene & minimality: 10 points

Partial credit:
- If a task's automated test failed, award up to 50% of that task's automated points if
  the candidate documented the correct root cause (token overlap heuristic) and provided a change_summary.
"""

import json
import os
import sys
import re
from collections import defaultdict

# -------------------------
# Helper utility functions
# -------------------------

def load_json(path):
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f), None
    except Exception as e:
        return None, str(e)

def safe_get_task_entry(candidate_json, task_id):
    try:
        for entry in candidate_json.get('task_results', []):
            if entry.get('task_id') == task_id:
                return entry
    except Exception:
        pass
    return None

def normalize_text(s):
    if s is None:
        return ""
    return re.sub(r'\s+', ' ', str(s)).strip().lower()

def tokenize_for_matching(s):
    # return set of significant tokens (alphanumeric, length>2) lowercased
    s = normalize_text(s)
    tokens = re.findall(r"[a-z0-9]+", s)
    return set([t for t in tokens if len(t) > 2])

def token_overlap_ratio(expected, candidate):
    # If expected is empty, return 0. If candidate has overlap >=0.5 => match
    expected_tokens = tokenize_for_matching(expected)
    candidate_tokens = tokenize_for_matching(candidate)
    if not expected_tokens:
        return 0.0
    inter = expected_tokens.intersection(candidate_tokens)
    return len(inter) / len(expected_tokens)

def contains_task_pass_line(text, task_id):
    if not text:
        return False
    text_l = normalize_text(text)
    # Look for "task{n}: pass" or "task {n}: pass" or exact "taskn: pass"
    patterns = [
        fr"task\s*{task_id}\s*:\s*pass",
        fr"task{task_id}\s*:\s*pass"
    ]
    for pat in patterns:
        if re.search(pat, text_l):
            return True
    return False

def contains_task_any_line(text, task_id):
    if not text:
        return False
    text_l = normalize_text(text)
    pat = fr"task\s*{task_id}\s*:"
    return bool(re.search(pat, text_l))

# -------------------------
# Grading configuration
# -------------------------

TASK_POINTS = {1: 15, 2: 20, 3: 20, 4: 15}  # automated points per task summing to 70
AUTOMATED_MAX = sum(TASK_POINTS.values())  # 70
EXPLAIN_MAX = 20
HYGIENE_MAX = 10
TOTAL_MAX = AUTOMATED_MAX + EXPLAIN_MAX + HYGIENE_MAX  # 100

# -------------------------
# Main grading logic
# -------------------------

def grade(candidate_path, answer_key_path):
    results = {
        "meta": {
            "candidate_path": candidate_path,
            "answer_key_path": answer_key_path
        },
        "per_task": [],
        "explanation_and_repro": {},
        "code_hygiene": {},
        "total_points": 0,
        "max_points": TOTAL_MAX,
        "percentage": 0.0,
        "overall_score": 0.0,
        "deductions": [],
        "notes": []
    }

    # Load files
    candidate_json, err = load_json(candidate_path)
    if err:
        results['notes'].append(f"ERROR loading candidate submission JSON: {err}")
        return results
    answer_json, err = load_json(answer_key_path)
    if err:
        results['notes'].append(f"ERROR loading answer key JSON: {err}")
        return results

    # Basic validation
    if 'task_results' not in candidate_json:
        results['notes'].append("Candidate submission missing 'task_results' field.")
        # continue but grading will likely be zero
    if 'task_results' not in answer_json:
        results['notes'].append("Answer key missing 'task_results' field.")
        return results

    # Build expected files_changed set from answer key
    expected_files_by_task = {}
    expected_files_union = set()
    expected_root_causes = {}
    for task_entry in answer_json.get('task_results', []):
        tid = int(task_entry.get('task_id'))
        expected_files = task_entry.get('files_changed', []) or []
        expected_files_by_task[tid] = set(expected_files)
        expected_files_union.update(expected_files_by_task[tid])
        expected_root_causes[tid] = task_entry.get('root_cause', "")

    automated_score = 0.0
    deductions = []

    # For checking how_tested evidence distribution
    how_tested_evidence_count = 0
    root_and_change_count = 0

    # Aggregate candidate changed files
    candidate_files_union = set()
    per_task_file_checks = {}

    # For each task 1..4
    for tid in (1,2,3,4):
        task_info = {
            "task_id": tid,
            "max_points": TASK_POINTS.get(tid, 0),
            "awarded_points": 0.0,
            "passed": False,
            "partial": False,
            "reasons": [],
            "expected_files": sorted(list(expected_files_by_task.get(tid, set()))),
            "candidate_files": [],
            "file_issues": {}
        }

        cand_entry = safe_get_task_entry(candidate_json, tid)
        if cand_entry is None:
            task_info['reasons'].append("No task entry found in candidate submission.")
            # no root/change/how_tested info -> no partial credit
            results['per_task'].append(task_info)
            per_task_file_checks[tid] = task_info
            continue

        # Collect candidate files_changed
        cand_files = cand_entry.get('files_changed') or []
        task_info['candidate_files'] = sorted(cand_files)
        candidate_files_union.update(cand_files)

        # Determine if automated PASS is claimed in candidate how_tested or final_test_output
        how_tested_field = cand_entry.get('how_tested', "") or ""
        final_output_field = candidate_json.get('final_test_output', "") or ""

        passed_claim = contains_task_pass_line(how_tested_field, tid) or contains_task_pass_line(final_output_field, tid)
        task_info['passed'] = bool(passed_claim)

        # Evaluate automated points
        max_pts = TASK_POINTS.get(tid, 0)
        if passed_claim:
            task_info['awarded_points'] = max_pts
            automated_score += max_pts
            task_info['reasons'].append(f"Automated test PASS claimed for Task{tid}; awarded {max_pts} pts.")
        else:
            # Not passed. Check for partial credit: correct root cause & non-empty change_summary
            change_summary = cand_entry.get('change_summary', "") or ""
            root_cause = cand_entry.get('root_cause', "") or ""
            # check token overlap between expected root cause and candidate root cause
            expected_rc = expected_root_causes.get(tid, "")
            overlap = token_overlap_ratio(expected_rc, root_cause)
            has_change_summary = bool(normalize_text(change_summary))
            has_root = bool(normalize_text(root_cause))
            # also consider if candidate included how_tested lines indicating they ran tests (even if fail)
            ran_tests_evidence = contains_task_any_line(how_tested_field, tid) or contains_task_any_line(final_output_field, tid)

            if has_root and has_change_summary and overlap >= 0.5:
                # award 50% of task points
                partial_pts = max_pts * 0.5
                task_info['awarded_points'] = partial_pts
                task_info['partial'] = True
                automated_score += partial_pts
                task_info['reasons'].append(
                    f"Automated test not passed. Candidate documented root cause matching expected (overlap={overlap:.2f}) and provided change_summary. Awarded 50% ({partial_pts}) as partial credit."
                )
            else:
                # No partial credit
                task_info['awarded_points'] = 0.0
                reason = "Automated test not passed."
                if not has_root or not has_change_summary:
                    reason += " Missing or empty root_cause/change_summary — no partial credit."
                else:
                    reason += f" Root cause documented but insufficient match to expected (overlap={overlap:.2f}) — no partial credit."
                task_info['reasons'].append(reason)

        # Evaluate file checks for task
        expected_files = expected_files_by_task.get(tid, set())
        cand_files_set = set(cand_files)
        extra = sorted(list(cand_files_set - expected_files))
        missing = sorted(list(expected_files - cand_files_set))
        file_issue = {}
        if extra:
            file_issue['extra_files'] = extra
            task_info['reasons'].append(f"Changed extra files: {extra}")
        if missing:
            file_issue['missing_expected_files'] = missing
            task_info['reasons'].append(f"Did not modify expected files: {missing}")
        task_info['file_issues'] = file_issue

        # Count evidence for how_tested
        if how_tested_field and contains_task_any_line(how_tested_field, tid):
            how_tested_evidence_count += 1

        # Count good root/change pairs
        if has_root and has_change_summary:
            root_and_change_count += 1

        results['per_task'].append(task_info)
        per_task_file_checks[tid] = task_info

    # Automated score done
    automated_score = round(automated_score, 2)
    results['automated_score'] = automated_score
    results['automated_max'] = AUTOMATED_MAX

    # Explanation & reproducibility scoring (20 points)
    # - 10 points for clear root_cause & change_summary for each task (global)
    # - 10 points for how_tested evidence (did candidate include exact run lines per task?)

    # Score for root/change: proportion of tasks with both non-empty
    tasks_with_both = root_and_change_count
    explain_root_change_pts = (tasks_with_both / 4.0) * 10.0
    explain_root_change_pts = round(explain_root_change_pts, 2)

    # Score for how_tested evidence: number of tasks where how_tested includes a TaskN: line
    how_tested_pts = (how_tested_evidence_count / 4.0) * 10.0
    how_tested_pts = round(how_tested_pts, 2)

    explanation_score = explain_root_change_pts + how_tested_pts
    explanation_score = round(explanation_score, 2)
    results['explanation_and_repro'] = {
        "awarded": explanation_score,
        "max": EXPLAIN_MAX,
        "details": {
            "root_and_change_full_count": tasks_with_both,
            "root_and_change_points": explain_root_change_pts,
            "how_tested_evidence_count": how_tested_evidence_count,
            "how_tested_points": how_tested_pts
        }
    }

    # Code hygiene & minimality scoring (10 points)
    # Criteria: candidate should only change expected files (across all tasks).
    # Start with full points; deduct 2 points per extra file changed and 2 per missing expected file (cap to 10).
    candidate_files_union = set(candidate_files_union)
    expected_files_union = set(expected_files_union)
    extra_files_all = sorted(list(candidate_files_union - expected_files_union))
    missing_expected_all = sorted(list(expected_files_union - candidate_files_union))

    hygiene_score = HYGIENE_MAX
    hygiene_deductions = []
    # Deduct 2 points per extra file (but not below 0)
    for ef in extra_files_all:
        hygiene_score -= 2
        hygiene_deductions.append(f"Changed unexpected file: {ef} (-2)")
    # Deduct 2 points per missing expected file
    for mf in missing_expected_all:
        hygiene_score -= 2
        hygiene_deductions.append(f"Did not change expected file: {mf} (-2)")
    # Floor at 0
    if hygiene_score < 0:
        hygiene_score = 0
    hygiene_score = round(hygiene_score, 2)

    results['code_hygiene'] = {
        "awarded": hygiene_score,
        "max": HYGIENE_MAX,
        "details": {
            "candidate_files_changed": sorted(list(candidate_files_union)),
            "expected_files": sorted(list(expected_files_union)),
            "extra_files": extra_files_all,
            "missing_expected_files": missing_expected_all,
            "deductions": hygiene_deductions
        }
    }

    # Compose per-task deduction messages
    for tinfo in results['per_task']:
        for r in tinfo.get('reasons', []):
            if 'extra files' in r.lower() or 'did not modify expected' in r.lower() or 'changed extra files' in r.lower():
                # capture as deduction text
                deductions.append({
                    "task_id": tinfo['task_id'],
                    "reason": r
                })

    # Consolidate total points
    total_points = automated_score + explanation_score + hygiene_score
    total_points = round(total_points, 2)
    percentage = round((total_points / TOTAL_MAX) * 100.0, 2) if TOTAL_MAX else 0.0

    results['total_points'] = total_points
    results['percentage'] = percentage
    results['overall_score'] = percentage
    results['deductions'] = deductions

    # Extra validations / notes
    # Check total_time_minutes <= 90
    total_time = candidate_json.get('total_time_minutes')
    if total_time is None:
        results['notes'].append("Candidate did not provide total_time_minutes.")
    else:
        try:
            if int(total_time) > 90:
                results['notes'].append(f"Candidate reported total_time_minutes = {total_time} (exceeds 90). No automatic penalty applied; evaluator should review.")
        except Exception:
            results['notes'].append("Could not interpret total_time_minutes; expected integer.")

    # Compare candidate final_test_output to answer key final_test_output (if present)
    ans_final = normalize_text(answer_json.get('final_test_output', ""))
    cand_final = normalize_text(candidate_json.get('final_test_output', ""))
    if ans_final:
        if ans_final == cand_final:
            results['notes'].append("Candidate final_test_output matches answer key final_test_output.")
        else:
            results['notes'].append("Candidate final_test_output differs from answer key. Evaluator should manually verify test runner outputs.")
    else:
        results['notes'].append("Answer key final_test_output not provided; skipping direct final output comparison.")

    return results

# -------------------------
# CLI entrypoint
# -------------------------
def main():
    if len(sys.argv) != 3:
        print("Usage: python3 task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(2)

    cand_path = sys.argv[1]
    ans_path = sys.argv[2]

    # Grade
    graded = grade(cand_path, ans_path)

    # Save test_results.json in same directory as this script
    script_dir = os.path.dirname(os.path.abspath(__file__))
    out_path = os.path.join(script_dir, 'test_results.json')
    try:
        with open(out_path, 'w', encoding='utf-8') as f:
            json.dump(graded, f, indent=2, ensure_ascii=False)
        print(f"Grading completed. Results written to {out_path}")
        # Also print a brief summary to stdout
        print(f"Total points: {graded.get('total_points')} / {graded.get('max_points')}")
        print(f"Overall score (percentage): {graded.get('overall_score')}%")
    except Exception as e:
        print(f"Failed to write results: {e}")
        sys.exit(3)

if __name__ == '__main__':
    main()
```

Notes and behavior summary:
- The script is defensive: if candidate misses fields it still continues and records notes.
- Partial-credit heuristic uses token overlap (>=50%) between expected root_cause and candidate root_cause and presence of non-empty change_summary.
- Explanation/reproducibility points are awarded proportionally across tasks (counts).
- Code hygiene deducts 2 points per unexpected changed file and 2 points per missing expected file (simple, transparent).
- The produced test_results.json contains a per-task breakdown, explanation & hygiene blocks, total_points, percentage, overall_score, and notes/deductions for manual review.
