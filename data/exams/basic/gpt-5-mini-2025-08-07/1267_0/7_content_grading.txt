# task_evaluation.py
import json
import sys
import os
import re
from pathlib import Path

"""
Automated grading script for the Debug & Fix practical exam (basic).

Usage:
    python task_evaluation.py path/to/test_submission.json path/to/answer_key.json

Outputs:
    Creates test_results.json in the same directory as this script with a detailed
    breakdown of scores, explanations for deductions, and overall_score (0-100).
"""

# Configuration / grading weights
TASK_POINTS = {1: 15, 2: 20, 3: 20, 4: 15}  # total 70
EXPLAIN_POINTS_TOTAL = 20  # 10 for root/change summaries, 10 for how_tested output
ROOT_CHANGE_POINTS = 10
HOW_TESTED_POINTS = 10
HYGIENE_POINTS = 10
MAX_POINTS = sum(TASK_POINTS.values()) + EXPLAIN_POINTS_TOTAL + HYGIENE_POINTS  # 100

# Helper: safe lower/strip
def norm(s):
    if s is None:
        return ""
    return str(s).strip()

# Helper: search for "TaskN: PASS" or "TaskN: FAIL" in text (case-insensitive)
def find_task_status_in_text(text, task_id):
    if not text:
        return None
    # normalize line endings and split
    lines = [ln.strip() for ln in text.splitlines() if ln.strip()]
    pattern_pass = re.compile(rf"Task\s*{task_id}\s*:\s*PASS", re.IGNORECASE)
    pattern_fail = re.compile(rf"Task\s*{task_id}\s*:\s*FAIL", re.IGNORECASE)
    for ln in lines:
        if pattern_pass.search(ln):
            return "PASS"
        if pattern_fail.search(ln):
            return "FAIL"
    return None

# Helper: basic fuzzy match for root cause
def root_cause_matches(expected, candidate):
    """
    Very simple heuristic:
    - Tokenize expected into words longer than 3 chars.
    - Count how many of those tokens appear in candidate (case-insensitive).
    - If >= 40% of expected tokens are present, consider it a match.
    """
    exp = norm(expected).lower()
    cand = norm(candidate).lower()
    if not exp or not cand:
        return False
    tokens = re.findall(r"[a-z0-9]{4,}", exp)
    if not tokens:
        # fallback: check substring presence
        return exp in cand
    matched = 0
    for t in tokens:
        if t in cand:
            matched += 1
    return (matched / len(tokens)) >= 0.4

# Main grading logic
def grade(candidate_path, answer_key_path):
    results = {
        "per_task": {},
        "explanations": [],
        "hygiene": {},
        "summary": {},
    }
    deductions_log = []
    # Load JSON files
    try:
        with open(candidate_path, 'r', encoding='utf-8') as f:
            candidate = json.load(f)
    except Exception as e:
        print(f"ERROR: Could not load candidate submission JSON: {e}")
        sys.exit(1)
    try:
        with open(answer_key_path, 'r', encoding='utf-8') as f:
            answer = json.load(f)
    except Exception as e:
        print(f"ERROR: Could not load answer key JSON: {e}")
        sys.exit(1)

    # Validate top-level structure
    cand_tasks = candidate.get("task_results")
    ans_tasks = answer.get("task_results")
    if not isinstance(cand_tasks, list) or not isinstance(ans_tasks, list):
        print("ERROR: Missing 'task_results' arrays in candidate or answer key.")
        sys.exit(1)

    # Collect expected files changed union
    expected_files = set()
    for t in ans_tasks:
        files = t.get("files_changed") or []
        expected_files.update(files)

    candidate_files = set()
    for t in cand_tasks:
        files = t.get("files_changed") or []
        candidate_files.update(files)

    # Score per task: automated + possible partials
    task_total_awarded = 0.0
    task_details = {}

    # Pre-check final_test_output presence (candidate)
    candidate_final_output = norm(candidate.get("final_test_output", ""))
    # Also allow looking into each task's how_tested field
    for ans_task in ans_tasks:
        try:
            task_id = int(ans_task.get("task_id"))
        except Exception:
            continue
        # Find corresponding candidate task entry
        cand_entry = None
        for ct in cand_tasks:
            if ct.get("task_id") == task_id:
                cand_entry = ct
                break
        if cand_entry is None:
            # create empty placeholder
            cand_entry = {
                "task_id": task_id,
                "files_changed": [],
                "change_summary": "",
                "root_cause": "",
                "how_tested": ""
            }

        # Determine automated PASS/FAIL by scanning how_tested and final_test_output
        how_tested_text = norm(cand_entry.get("how_tested", ""))
        status = find_task_status_in_text(how_tested_text, task_id)
        if status is None:
            # fallback to final_test_output
            status = find_task_status_in_text(candidate_final_output, task_id)

        automated_pass = (status == "PASS")
        automated_fail = (status == "FAIL")
        task_max = TASK_POINTS.get(task_id, 0)
        awarded = 0.0
        partial_awarded = False
        partial_reason = ""

        if automated_pass:
            awarded = float(task_max)
            note = f"Task{task_id}: automated PASS detected."
        else:
            # Automated fail or unknown: see if candidate correctly diagnosed root cause -> partial credit up to 50%
            expected_root = ans_task.get("root_cause", "")
            candidate_root = cand_entry.get("root_cause", "")
            root_match = root_cause_matches(expected_root, candidate_root)
            if root_match:
                awarded = 0.5 * task_max
                partial_awarded = True
                partial_reason = "Root cause described matches expected => granted 50% partial credit for this task."
            else:
                awarded = 0.0
                partial_reason = "No automated PASS and root cause did not match expected; no task points awarded."
            note = f"Task{task_id}: automated status={status}; {partial_reason}"

        # Record task detail
        task_details[task_id] = {
            "task_id": task_id,
            "max_points": task_max,
            "earned_points_automated": round(awarded, 2),
            "automated_pass": automated_pass,
            "automated_status_string": status,
            "partial_awarded": partial_awarded,
            "partial_reason": partial_reason,
            "expected_root_cause": ans_task.get("root_cause", ""),
            "candidate_root_cause": cand_entry.get("root_cause", ""),
            "candidate_change_summary": cand_entry.get("change_summary", ""),
            "candidate_files_changed": cand_entry.get("files_changed", []),
        }
        results["per_task"][str(task_id)] = task_details[task_id]
        task_total_awarded += awarded

    # Explanation & reproducibility scoring (20 points)
    # 10 points for root_cause+change_summary clarity (split equally per task)
    # 10 points for inclusion of how_tested/test runner lines (split equally per task)
    root_change_score = 0.0
    how_tested_score = 0.0
    per_task_root_share = ROOT_CHANGE_POINTS / 4.0  # 2.5
    per_task_how_share = HOW_TESTED_POINTS / 4.0    # 2.5

    for ct in cand_tasks:
        tid = ct.get("task_id")
        if tid not in (1, 2, 3, 4):
            continue
        root = norm(ct.get("root_cause", ""))
        change = norm(ct.get("change_summary", ""))
        # Criteria for awarding root/change share:
        # - root length >= 10 chars considered present
        # - change length >= 20 chars considered present
        # award full per-task share if both present, half if one present
        rc_present = len(root) >= 10
        ch_present = len(change) >= 20
        if rc_present and ch_present:
            root_change_score += per_task_root_share
        elif rc_present or ch_present:
            root_change_score += (per_task_root_share / 2.0)
        # how_tested: check how_tested field includes TaskN PASS/FAIL lines or final_test_output includes it
        howtxt = norm(ct.get("how_tested", ""))
        status = find_task_status_in_text(howtxt, tid)
        if status is None:
            status = find_task_status_in_text(candidate_final_output, tid)
        if status is not None:
            how_tested_score += per_task_how_share
        else:
            # partial credit if how_tested contains something non-empty (commands run) even if not PASS/FAIL
            if howtxt:
                how_tested_score += (per_task_how_share / 2.0)

    # Cap explanation scores
    root_change_score = min(root_change_score, ROOT_CHANGE_POINTS)
    how_tested_score = min(how_tested_score, HOW_TESTED_POINTS)

    explanation_total_awarded = round(root_change_score + how_tested_score, 2)

    # Hygiene & minimality scoring (10 points)
    # Heuristics:
    # - If candidate modified exactly expected files -> full points
    # - Missing expected files => deduct 3 points per missing
    # - Extra modified files => deduct 2 points per extra
    # - Short change_summary (<10 chars) for any task deduct 1 point each (up to max)
    hygiene_awarded = HYGIENE_POINTS
    missing_files = expected_files - candidate_files
    extra_files = candidate_files - expected_files
    hygiene_deductions = 0
    hygiene_msgs = []
    if missing_files:
        ded = 3 * len(missing_files)
        hygiene_deductions += ded
        hygiene_msgs.append(f"Missing expected modified files: {sorted(list(missing_files))} (-{ded})")
    if extra_files:
        ded = 2 * len(extra_files)
        hygiene_deductions += ded
        hygiene_msgs.append(f"Extra modified files: {sorted(list(extra_files))} (-{ded})")
    # short summaries penalty
    short_summaries = 0
    for ct in cand_tasks:
        chs = norm(ct.get("change_summary", ""))
        if chs and len(chs) < 10:
            short_summaries += 1
    if short_summaries:
        ded = 1 * short_summaries
        hygiene_deductions += ded
        hygiene_msgs.append(f"Very short change summaries found for {short_summaries} task(s) (-{ded})")

    hygiene_awarded = max(0, hygiene_awarded - hygiene_deductions)

    # Compile detailed deductions
    if hygiene_msgs:
        deductions_log.extend(hygiene_msgs)

    # Final aggregation
    total_awarded = round(task_total_awarded + explanation_total_awarded + hygiene_awarded, 2)
    percentage = round((total_awarded / MAX_POINTS) * 100.0, 2)

    # Build detailed per-task explanations and flags (augment existing)
    for tid_str, info in results["per_task"].items():
        tid = int(tid_str)
        # Add verdict and messages
        automated = info["automated_pass"]
        earned_auto = info["earned_points_automated"]
        expected_task_files = []
        for at in ans_tasks:
            if at.get("task_id") == tid:
                expected_task_files = at.get("files_changed", [])
                break
        candidate_files_for_task = set(info.get("candidate_files_changed", []) or [])
        expected_files_for_task = set(expected_task_files or [])
        files_match = candidate_files_for_task == expected_files_for_task

        # Root cause match
        root_ok = root_cause_matches(info.get("expected_root_cause", ""), info.get("candidate_root_cause", ""))

        # how_tested present?
        ht_present = bool(norm(info.get("candidate_change_summary", ""))) or bool(norm(info.get("candidate_files_changed", "")))

        info.update({
            "files_match_expected_for_task": files_match,
            "root_cause_match_expected": root_ok,
            "how_tested_present": ht_present,
        })

        # Human readable note
        notes = []
        if automated:
            notes.append("Automated PASS recorded; full automated points awarded for this task.")
        else:
            if info["partial_awarded"]:
                notes.append(info["partial_reason"])
            else:
                notes.append("No automated PASS and no sufficient root-cause match; 0 task points.")
        if not files_match:
            notes.append(f"Files changed for task differ from expected (expected: {sorted(list(expected_files_for_task))}, got: {sorted(list(candidate_files_for_task))}).")
        info["grading_notes"] = " ".join(notes)

    # Prepare final results JSON
    script_dir = os.path.dirname(os.path.abspath(__file__))
    output = {
        "candidate_submission_path": os.path.abspath(candidate_path),
        "answer_key_path": os.path.abspath(answer_key_path),
        "max_points": MAX_POINTS,
        "points_awarded": total_awarded,
        "percentage": percentage,
        "overall_score": percentage,
        "breakdown": {
            "tasks_automated_points_awarded": round(task_total_awarded, 2),
            "explanation_points_awarded": explanation_total_awarded,
            "hygiene_points_awarded": hygiene_awarded,
        },
        "task_details": results["per_task"],
        "explanation_scoring": {
            "root_change_points_awarded": round(root_change_score, 2),
            "how_tested_points_awarded": round(how_tested_score, 2)
        },
        "hygiene": {
            "expected_files_changed": sorted(list(expected_files)),
            "candidate_files_changed": sorted(list(candidate_files)),
            "hygiene_points_awarded": hygiene_awarded,
            "hygiene_deductions_notes": hygiene_msgs
        },
        "deductions_log": deductions_log,
        "final_comments": [
            "Automated task points are primarily determined by presence of 'TaskN: PASS' lines in the candidate's how_tested fields or final_test_output.",
            "Partial credit (50% of task points) can be awarded for correct root-cause diagnosis even if the test still failed.",
            "Explanation points require non-trivial root_cause and change_summary fields; how_tested text or final_test_output must include task pass/fail lines to get full reproducibility points.",
            "Hygiene is judged heuristically by matching files changed to expected files and by brevity of change summaries."
        ]
    }

    # Save test_results.json in same directory as script
    outpath = os.path.join(script_dir, "test_results.json")
    try:
        with open(outpath, 'w', encoding='utf-8') as f:
            json.dump(output, f, indent=2)
        print(f"Grading complete. Results written to {outpath}")
    except Exception as e:
        print(f"ERROR: Could not write results file: {e}")
        sys.exit(1)

    return output

def main():
    if len(sys.argv) != 3:
        print("Usage: python task_evaluation.py path/to/test_submission.json path/to/answer_key.json")
        sys.exit(1)
    cand = sys.argv[1]
    key = sys.argv[2]
    grade(cand, key)

if __name__ == "__main__":
    main()
