#!/usr/bin/env python3
"""
Automated grader for the "Test Planning Practical — Basic" exam.

Usage:
    python task_evaluation.py candidate_submission.json answer_key.json

Outputs:
    test_results.json in the current working directory.

Notes:
- Only standard libraries are used.
- Script implements the grading logic and rubrics described in the evaluator materials.
- The script is defensive about invalid/missing data and records clear deduction reasons.
"""

import json
import sys
import os
from datetime import datetime, timedelta, date

# -------------------------
# Helper utilities
# -------------------------
def load_json(path):
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except Exception as e:
        raise ValueError(f"Failed to load/parse JSON file '{path}': {e}")

def to_date(d):
    # Accept date strings in YYYY-MM-DD or ISO format
    if not d:
        return None
    try:
        # Try YYYY-MM-DD first
        return datetime.strptime(d[:10], "%Y-%m-%d").date()
    except Exception:
        try:
            return datetime.fromisoformat(d).date()
        except Exception:
            raise ValueError(f"Cannot parse date string: {d}")

def offset_to_date(project_start_date, offset):
    # offset is integer days from project_start_date (0 => project_start_date)
    return project_start_date + timedelta(days=int(offset))

def safe_get(dct, key, default=None):
    return dct.get(key, default) if isinstance(dct, dict) else default

def round2(x):
    return round(float(x), 2)

# -------------------------
# Main grading logic
# -------------------------
def grade(candidate_path, answer_key_path):
    results = {
        "candidate_file": candidate_path,
        "answer_key_file": answer_key_path,
        "graded_at": datetime.utcnow().isoformat() + "Z",
        "scores": {},
        "deductions": [],
        "total_points_awarded": 0,
        "total_points_possible": 100,
    }

    # Load files
    try:
        candidate = load_json(candidate_path)
    except Exception as e:
        results["error"] = str(e)
        results["overall_score"] = 0.0
        results["comment"] = "Could not load candidate submission."
        _write_results(results)
        return

    try:
        answer = load_json(answer_key_path)
    except Exception as e:
        results["error"] = str(e)
        results["overall_score"] = 0.0
        results["comment"] = "Could not load answer key."
        _write_results(results)
        return

    # Extract some canonical dates from answer key or fallback to brief defaults
    # Prefer explicit fields in answer key; else use known brief values.
    try:
        project_start_date = to_date(safe_get(candidate, 'project_start_date') or safe_get(answer, 'project_start_date') or "2025-11-03")
    except Exception:
        project_start_date = date(2025,11,3)

    # Default brief values:
    default_code_freeze = to_date("2025-11-21")
    default_release = to_date("2025-11-28")

    code_freeze = to_date(safe_get(answer, 'code_freeze') or safe_get(candidate, 'code_freeze') ) or default_code_freeze
    release_date = to_date(safe_get(answer, 'release_date') or safe_get(candidate, 'release_date') ) or default_release

    # Begin schema validation and basic checks
    required_top_keys = [
        "candidate_name", "exam_start", "project_start_date", "assumptions", "test_strategy",
        "tasks", "schedule_summary", "risk_prioritization", "contingency_plan",
        "metrics_and_reporting", "final_comments"
    ]
    missing_keys = [k for k in required_top_keys if k not in candidate]
    if missing_keys:
        results["deductions"].append({
            "area": "JSON schema",
            "points": 0,
            "reason": f"Missing required top-level keys: {missing_keys}"
        })
        # It's a fatal schema error for automatic grading - award zero and explain
        results["scores"]["schema_validity"] = 0
        results["total_points_awarded"] = 0
        results["overall_score"] = 0.0
        results["comment"] = "Submission missing required keys; automatic fail."
        _write_results(results)
        return

    # Validate tasks structure
    tasks = candidate.get("tasks", [])
    if not isinstance(tasks, list) or len(tasks) == 0:
        results["deductions"].append({
            "area": "tasks",
            "points": 0,
            "reason": "Tasks array missing or empty."
        })
        results["scores"]["tasks_present"] = 0
        results["total_points_awarded"] = 0
        results["overall_score"] = 0.0
        results["comment"] = "No tasks provided; automatic fail."
        _write_results(results)
        return

    # Create mapping of tasks by id
    task_map = {}
    invalid_tasks = []
    for t in tasks:
        tid = t.get("id")
        if not tid:
            invalid_tasks.append("task without id")
            continue
        task_map[tid] = t

    if invalid_tasks:
        results["deductions"].append({
            "area": "tasks",
            "points": 0,
            "reason": f"Invalid tasks: {invalid_tasks}"
        })

    # Precompute per-task end offsets and end dates
    computed = {}
    for tid, t in task_map.items():
        try:
            est_off = int(t.get("earliest_start_offset", 0))
            duration = int(t.get("duration_days", 0))
            if duration < 0:
                duration = 0
            # End offset = start offset + duration_days - 1 (inclusive)
            if duration == 0:
                end_offset = est_off
            else:
                end_offset = est_off + duration - 1
            end_date = offset_to_date(project_start_date, end_offset)
            start_date = offset_to_date(project_start_date, est_off)
            computed[tid] = {
                "start_offset": est_off,
                "duration_days": duration,
                "end_offset": end_offset,
                "start_date": start_date,
                "end_date": end_date
            }
        except Exception as e:
            computed[tid] = {
                "start_offset": 0,
                "duration_days": 0,
                "end_offset": 0,
                "start_date": project_start_date,
                "end_date": project_start_date
            }

    # -------------------------
    # Grading area 1: Schedule & resourcing (40)
    # Sub-criteria:
    #  - tasks coverage (8)
    #  - realistic estimates (8)
    #  - dependencies feasibility (8)
    #  - resource allocation respects headcount (8)
    #  - critical path & dates (8)
    # -------------------------
    schedule_points = 0
    schedule_max = 40
    schedule_feedback = []

    # 1.a - tasks coverage (8): must include env-setup, test-design, test-execution, automation, regression, UAT-support, reporting
    required_types = ["env-setup", "test-design", "test-execution", "automation", "regression", "UAT-support", "reporting"]
    present_types = set([safe_get(t, "type") for t in tasks if safe_get(t, "type")])
    missing_types = [tp for tp in required_types if tp not in present_types]
    coverage_score = 8 * (1 - len(missing_types) / len(required_types))
    coverage_score = round2(max(0, coverage_score))
    schedule_points += coverage_score
    if missing_types:
        schedule_feedback.append(f"Missing task types: {missing_types} (coverage score {coverage_score:.2f}/8)")
    else:
        schedule_feedback.append(f"All required task types present (8/8)")

    # 1.b - realistic estimates (8)
    # Check per-task per-day hours <= 8 for the task owner
    unrealistic_reasons = []
    per_day_owner = {}  # owner -> dict(offset -> hours)
    total_effort = 0.0
    tasks_with_zero_duration = []
    for tid, t in task_map.items():
        est_hours = t.get("estimate_hours", 0) or 0
        duration = int(t.get("duration_days", 0)) or 0
        owner = t.get("owner", "")
        total_effort += float(est_hours)
        if duration <= 0:
            tasks_with_zero_duration.append(tid)
            # Avoid division by zero: treat as 1 day for per-day calculations but note problem
            per_day = float(est_hours)
            per_day_owner.setdefault(owner, {})
            per_day_owner[owner].setdefault(0, 0)
            per_day_owner[owner][0] += per_day
        else:
            per_day = float(est_hours) / float(duration)
            # assign per-day evenly across offsets of the task
            for d in range(int(t.get("earliest_start_offset", 0)), int(t.get("earliest_start_offset", 0)) + duration):
                per_day_owner.setdefault(owner, {})
                per_day_owner[owner].setdefault(d, 0.0)
                per_day_owner[owner][d] += per_day

    # Check per-day per-owner >8 hours (impossible)
    impossible_overload = []
    for owner, days in per_day_owner.items():
        for d, hrs in days.items():
            if hrs > 8.0001:  # allow minor float tolerance
                impossible_overload.append((owner, d, round2(hrs)))
    if impossible_overload:
        # Penalize heavily: realistic score reduced proportional to overload severity
        realistic_score = 2.0  # small credit if others good
        schedule_feedback.append(f"Impossible per-day hours detected (owner/day/hours): {impossible_overload} -> realistic score {realistic_score:.2f}/8")
    else:
        # Also check for tasks that unrealistically compress many hours into very few days (e.g., per-day > 8 already caught).
        # Check for zero-duration tasks as suspicious
        if tasks_with_zero_duration:
            realistic_score = 6.0
            schedule_feedback.append(f"Tasks with zero duration_days: {tasks_with_zero_duration} -> conservative realistic score {realistic_score:.2f}/8")
        else:
            realistic_score = 8.0
            schedule_feedback.append("Estimates per-day and durations are feasible (8/8)")
    realistic_score = round2(realistic_score)
    schedule_points += realistic_score

    # 1.c - dependencies defined and feasible (8)
    dependency_issues = []
    dep_score = 8.0
    for tid, t in task_map.items():
        deps = t.get("dependencies", []) or []
        for dep in deps:
            if dep not in task_map:
                dependency_issues.append(f"{tid} depends on unknown {dep}")
            else:
                dep_end = computed[dep]["end_offset"]
                # Candidate should schedule earliest_start_offset >= dep_end + 1 (i.e., next day) OR at least >= dep_end (if can start same day)
                # We'll require start >= dep_end + 0 (allow same-day handoff) but prefer start >= dep_end + 1.
                start = int(t.get("earliest_start_offset", 0))
                if start < dep_end:
                    dependency_issues.append(f"{tid} starts (offset {start}) before dependency {dep} ends (offset {dep_end})")
    if dependency_issues:
        # Deduct proportionally
        dep_penalty = min(len(dependency_issues) / max(1, len(tasks)), 1.0) * 8.0
        dep_score = round2(max(0.0, 8.0 - dep_penalty))
        schedule_feedback.append(f"Dependency issues: {dependency_issues} -> dependencies score {dep_score:.2f}/8")
    else:
        schedule_feedback.append("Dependencies exist and appear feasible (8/8)")
    schedule_points += dep_score

    # 1.d - resource allocation respects headcount (8)
    # We already checked per-owner per-day <=8 as impossible overloading.
    # Now also check aggregate per-day (Tester A + Tester B) <=16.
    aggregate_by_day = {}
    for owner, days in per_day_owner.items():
        # Only consider Tester A and Tester B for headcount constraint
        if owner not in ("Tester A", "Tester B"):
            continue
        for d, hrs in days.items():
            aggregate_by_day.setdefault(d, 0.0)
            aggregate_by_day[d] += hrs
    aggregate_issues = []
    for d, hrs in aggregate_by_day.items():
        if hrs > 16.0001:
            aggregate_issues.append((d, round2(hrs)))
    if aggregate_issues:
        res_score = 2.0
        schedule_feedback.append(f"Aggregate tester-hours overload on days: {aggregate_issues} -> resource allocation score {res_score:.2f}/8")
    else:
        res_score = 8.0
        schedule_feedback.append("Resource allocation respects two-testers headcount (8/8)")
    res_score = round2(res_score)
    schedule_points += res_score

    # 1.e - critical path & dates (8)
    cp_issues = []
    cp_score = 8.0
    summary = candidate.get("schedule_summary", {})
    candidate_cp = summary.get("critical_path", []) or []
    # Check that critical path ids exist
    missing_cp_ids = [tid for tid in candidate_cp if tid not in task_map]
    if missing_cp_ids:
        cp_issues.append(f"Critical path contains unknown tasks: {missing_cp_ids}")
        cp_score -= 3.0
    # Check that cp forms a plausible dependency chain (each next depends on previous or transitively)
    if candidate_cp:
        for i in range(1, len(candidate_cp)):
            prev = candidate_cp[i-1]
            curr = candidate_cp[i]
            curr_deps = task_map[curr].get("dependencies", []) or []
            # Quick check: either curr depends on prev or there is transitive chain (we check reachable)
            if prev not in curr_deps:
                # Check transitive: BFS search
                visited = set()
                stack = list(curr_deps)
                found = False
                while stack:
                    d = stack.pop()
                    if d == prev:
                        found = True
                        break
                    if d in visited:
                        continue
                    visited.add(d)
                    stack.extend(task_map.get(d, {}).get("dependencies", []) or [])
                if not found:
                    cp_issues.append(f"Critical path order not directly supported: {prev} -> {curr} (no dependency chain)")
    # Check schedule_end <= release_date
    try:
        sched_end = to_date(summary.get("schedule_end"))
        if sched_end is None:
            cp_issues.append("schedule_summary.schedule_end missing or invalid")
            cp_score -= 2.0
        else:
            if sched_end > release_date:
                cp_issues.append(f"Schedule end {sched_end} is after release date {release_date}")
                cp_score -= 2.0
    except Exception:
        cp_issues.append("schedule_summary.schedule_end invalid format")
        cp_score -= 2.0

    # Check that core functional/integration tasks complete before code freeze OR candidate documented trade-offs
    # Identify tasks of type test-execution/regression/automation that are likely code-impacting
    code_impacting = []
    for tid, t in task_map.items():
        if t.get("type") in ("test-execution", "regression", "automation"):
            code_impacting.append(tid)
    # find latest end date among these tasks
    latest_code_end = None
    latest_code_end_tid = None
    for tid in code_impacting:
        ed = computed[tid]["end_date"]
        if latest_code_end is None or ed > latest_code_end:
            latest_code_end = ed
            latest_code_end_tid = tid
    code_freeze_issue = False
    if latest_code_end and latest_code_end > code_freeze:
        # Candidate must explicitly explain trade-off in final_comments or contingency_plan
        cont = candidate.get("contingency_plan", "") or ""
        final_comments = candidate.get("final_comments", "") or ""
        explanation = (("defer" in cont.lower() and "post" in cont.lower()) or
                       ("compress" in cont.lower()) or
                       ("scope" in cont.lower()) or
                       ("freeze" in cont.lower()) or
                       ("trade" in final_comments.lower()))
        if not explanation:
            cp_issues.append(f"Code-impacting tests latest end {latest_code_end} (task {latest_code_end_tid}) is after code-freeze {code_freeze} and no explicit trade-off justification found.")
            cp_score -= 3.0
            code_freeze_issue = True
        else:
            cp_score -= 0.5
            schedule_feedback.append("Code-impacting tasks extend after code-freeze but candidate provided trade-off/contingency.")
    if cp_issues:
        schedule_feedback.append(f"Critical path issues: {cp_issues} -> critical path score {max(0, cp_score):.2f}/8")
    else:
        schedule_feedback.append("Critical path plausible and dates respect release/code-freeze (8/8)")
    cp_score = round2(max(0.0, cp_score))
    schedule_points += cp_score

    # Collate schedule area
    schedule_points = round2(schedule_points)
    results["scores"]["schedule_and_resourcing"] = {
        "points_awarded": schedule_points,
        "points_possible": schedule_max,
        "feedback": schedule_feedback
    }

    # -------------------------
    # Grading area 2: Test strategy (30)
    # Sub-criteria (each 6 points):
    #  - scope/objectives
    #  - entry/exit criteria explicit and measurable
    #  - appropriate test types
    #  - automation approach aligned
    #  - reporting metrics & cadence
    # -------------------------
    strategy_points = 0
    strategy_max = 30
    strategy_feedback = []
    ts = candidate.get("test_strategy", "") or ""

    # 2.a scope/objectives (6): presence of feature names or words "scope"/"objectives"
    scope_ok = False
    if any(x in ts for x in ["F1", "F2", "F3", "F4", "F5", "F6"]) or "scope" in ts.lower() or "objective" in ts.lower():
        scope_ok = True
        strategy_points += 6
        strategy_feedback.append("Scope and objectives present (6/6)")
    else:
        strategy_feedback.append("Scope/objectives not explicit (0/6)")

    # 2.b entry/exit criteria explicit (6)
    entry_ok = False
    if ("entry" in ts.lower() or "entry criteria" in ts.lower()) and ("exit" in ts.lower() or "exit criteria" in ts.lower()):
        entry_ok = True
        strategy_points += 6
        strategy_feedback.append("Entry and exit criteria present (6/6)")
    else:
        # Look for mentions of P0/P1 or 'no more than' which indicate exit criteria
        if any(x in ts for x in ["P0", "P1", "critical test", "no more than", "sign-off", "UAT sign-off"]):
            entry_ok = True
            strategy_points += 4
            strategy_feedback.append("Partial entry/exit criteria detected (4/6)")
        else:
            strategy_feedback.append("Entry/exit criteria missing or not measurable (0/6)")

    # 2.c appropriate test types (6): must mention at least 3 of suggested
    suggested = ["smoke", "functional", "integration", "regression", "performance", "UAT"]
    mentioned = [s for s in suggested if s in ts.lower()]
    if len(mentioned) >= 3:
        strategy_points += 6
        strategy_feedback.append(f"Test types appropriate: mentioned {mentioned} (6/6)")
    elif len(mentioned) == 2:
        strategy_points += 3
        strategy_feedback.append(f"Only two test types mentioned: {mentioned} (3/6)")
    else:
        strategy_feedback.append(f"Insufficient test types mentioned: {mentioned} (0/6)")

    # 2.d automation approach (6): must mention automation, 40% or nightly runs etc.
    auto_ok = False
    if "automation" in ts.lower() and ("40" in ts or "40%" in ts or "night" in ts or "nightly" in ts):
        auto_ok = True
        strategy_points += 6
        strategy_feedback.append("Automation approach described and aligned (6/6)")
    else:
        if "automation" in ts.lower():
            strategy_points += 3
            strategy_feedback.append("Automation mentioned but details missing (3/6)")
        else:
            strategy_feedback.append("Automation approach not described (0/6)")

    # 2.e reporting metrics & cadence (6)
    if any(x in ts.lower() for x in ["daily", "cadence", "report", "defect", "pass/fail", "open defects"]):
        strategy_points += 6
        strategy_feedback.append("Reporting metrics and cadence described (6/6)")
    else:
        strategy_feedback.append("Reporting metrics/cadence missing (0/6)")

    strategy_points = round2(strategy_points)
    results["scores"]["test_strategy"] = {
        "points_awarded": strategy_points,
        "points_possible": strategy_max,
        "feedback": strategy_feedback
    }

    # -------------------------
    # Grading area 3: Risk prioritization & contingency (15)
    # - mapping of features to priority and tests (6)
    # - contingency plan (5)
    # - defect/env contingency (4)
    # -------------------------
    risk_points = 0
    risk_max = 15
    risk_feedback = []
    rp = candidate.get("risk_prioritization", []) or []
    rp_map = {entry.get("feature_id"): entry for entry in rp if isinstance(entry, dict) and entry.get("feature_id")}
    # 3.a mapping features (6): must include F1..F6 all present with priorities 1-5 and non-empty tests_to_run
    missing_features = [f"F{i}" for i in range(1,7) if f"F{i}" not in rp_map]
    mapping_score = 6.0
    if missing_features:
        mapping_score = round2(6.0 * (1 - len(missing_features) / 6.0))
        risk_feedback.append(f"Missing risk_prioritization entries for: {missing_features} -> mapping score {mapping_score:.2f}/6")
    else:
        # verify test_to_run non-empty and priority 1-5
        invalid_entries = []
        for fid, ent in rp_map.items():
            tp = ent.get("testing_priority")
            tests = ent.get("tests_to_run", []) or []
            if not (isinstance(tp, int) and 1 <= tp <= 5):
                invalid_entries.append(f"{fid} invalid priority {tp}")
            if not tests:
                invalid_entries.append(f"{fid} empty tests_to_run")
        if invalid_entries:
            mapping_score = 3.0
            risk_feedback.append(f"Invalid mapping entries: {invalid_entries} -> mapping score {mapping_score:.2f}/6")
        else:
            mapping_score = 6.0
            risk_feedback.append("Feature mapping complete and valid (6/6)")
    risk_points += mapping_score

    # 3.b contingency plan (5): must include concrete actions for 1-3 days and >3 days
    cont = (candidate.get("contingency_plan") or "").lower()
    cont_score = 0.0
    reasons_cont = []
    if cont:
        if any(x in cont for x in ["1-3", "1 to 3", "1–3", "1 - 3", "1 day", "1-3 days", "1–3 days", "if slip 1"]):
            cont_score += 2.5
        if any(x in cont for x in [">3", "more than 3", "greater than 3", " >3 days", "if slip >3", "if slip > 3"]):
            cont_score += 2.5
    else:
        # Try to detect phrases implying partial plans
        if "defer" in cont or "compress" in cont or "reduce" in cont:
            cont_score = 2.5
    cont_score = round2(cont_score)
    if cont_score == 5.0:
        risk_feedback.append("Contingency plan includes actions for short and extended slips (5/5)")
    elif cont_score > 0:
        risk_feedback.append(f"Contingency plan partially covers slip scenarios (score {cont_score}/5)")
    else:
        risk_feedback.append("Contingency plan missing or not specific about slip durations (0/5)")
    risk_points += cont_score

    # 3.c defect cycles & environment contingency (4)
    env_and_def_score = 0.0
    cont_lower = cont
    if "env" in cont_lower or "environment" in cont_lower or "provision" in cont_lower or "integration" in cont_lower:
        env_and_def_score += 2.0
    # defect cycle coverage: mention 2 business days or developer SLA or retest
    if any(x in cont_lower for x in ["2 business", "2-business", "2 business days", "dev fix", "fix sla", "2 days", "retest"]):
        env_and_def_score += 2.0
    if env_and_def_score == 0:
        risk_feedback.append("Contingency does not address environment or developer defect cycles (0/4)")
    else:
        risk_feedback.append(f"Contingency addresses defect/env issues (score {env_and_def_score}/4)")
    risk_points += env_and_def_score

    risk_points = round2(risk_points)
    results["scores"]["risk_and_contingency"] = {
        "points_awarded": risk_points,
        "points_possible": risk_max,
        "feedback": risk_feedback
    }

    # -------------------------
    # Grading area 4: Assumptions, clarity, and JSON format (10)
    # - assumptions count and clarity (3)
    # - JSON validity & schema adherence (7)
    # -------------------------
    assumptions_points = 0
    assumptions_max = 10
    assumptions_feedback = []
    assumptions = candidate.get("assumptions", []) or []
    if not isinstance(assumptions, list):
        assumptions_feedback.append("Assumptions not an array")
        assumptions_points += 0
    else:
        n = len(assumptions)
        if 4 <= n <= 8:
            assumptions_points += 3
            assumptions_feedback.append(f"{n} assumptions provided (3/3)")
        elif 1 <= n < 4:
            assumptions_points += 1
            assumptions_feedback.append(f"{n} assumptions only (1/3)")
        else:
            assumptions_points += 2
            assumptions_feedback.append(f"{n} assumptions provided (>8 considered okay but extra assumptions not penalized) (2/3)")

    # JSON validity & schema adherence
    schema_score = 0.0
    schema_reasons = []
    # Required top-level keys already checked earlier; now verify schedule_summary totals
    try:
        schedule_sum = candidate.get("schedule_summary", {})
        total_eff_candidate = float(schedule_sum.get("total_effort_hours", 0) or 0)
        # compute actual sum
        computed_sum = 0.0
        for tid, t in task_map.items():
            computed_sum += float(t.get("estimate_hours", 0) or 0)
        # compare within small tolerance
        if abs(computed_sum - total_eff_candidate) <= max(1.0, 0.05 * computed_sum):
            schema_score += 4.0
            assumptions_feedback.append("schedule_summary.total_effort_hours matches sum of task estimate_hours (4/7)")
        else:
            schema_reasons.append(f"Mismatch total_eff_hours: candidate summary {total_eff_candidate} vs computed sum {computed_sum}")
            schema_score += 1.0
            assumptions_feedback.append(f"schedule_summary.total_effort_hours mismatch (1/7): {schema_reasons[-1]}")
        # Validate critical_path entries exist
        cp = schedule_sum.get("critical_path", []) or []
        missing_cp = [x for x in cp if x not in task_map]
        if missing_cp:
            schema_reasons.append(f"schedule_summary.critical_path references unknown tasks: {missing_cp}")
            # don't award remaining points
        else:
            schema_score += 3.0
            assumptions_feedback.append("schedule_summary.critical_path tasks exist (3/7)")
    except Exception as ex:
        schema_reasons.append(f"Error validating schedule_summary: {ex}")

    # total schema_score out of 7
    schema_score = min(7.0, round2(schema_score))
    assumptions_points += schema_score
    results["scores"]["assumptions_and_format"] = {
        "points_awarded": round2(assumptions_points),
        "points_possible": assumptions_max,
        "feedback": assumptions_feedback + schema_reasons
    }

    # -------------------------
    # Grading area 5: Metrics & reporting (5)
    # -------------------------
    metrics_points = 0.0
    metrics_max = 5.0
    metrics_feedback = []
    metrics = candidate.get("metrics_and_reporting", "") or ""
    metrics_lower = metrics.lower()
    if not metrics_lower:
        metrics_feedback.append("metrics_and_reporting missing or empty")
    else:
        # Look for daily cadence and defect metrics
        if any(x in metrics_lower for x in ["daily", "daily cadence", "daily summary", "morning"]):
            metrics_points += 2.0
        if any(x in metrics_lower for x in ["defect", "open defects", "severity", "p0", "p1", "p2", "pass rate", "pass/fail"]):
            metrics_points += 2.0
        if any(x in metrics_lower for x in ["go/no-go", "go no go", "go/no go", "exit criteria", "exit"]):
            metrics_points += 1.0
    metrics_points = round2(min(metrics_max, metrics_points))
    if metrics_points == metrics_max:
        metrics_feedback.append("Metrics and reporting clear and actionable (5/5)")
    else:
        metrics_feedback.append(f"Metrics/reporting partially missing (score {metrics_points}/5)")
    results["scores"]["metrics_and_reporting"] = {
        "points_awarded": metrics_points,
        "points_possible": metrics_max,
        "feedback": metrics_feedback
    }

    # -------------------------
    # Aggregate totals
    # -------------------------
    total_awarded = 0.0
    for k, v in results["scores"].items():
        pts = v.get("points_awarded", 0) if isinstance(v, dict) else v
        total_awarded += float(pts)
    total_awarded = round2(total_awarded)
    results["total_points_awarded"] = total_awarded
    results["total_points_possible"] = 100.0
    percentage = round2((total_awarded / 100.0) * 100.0)
    results["percentage"] = percentage
    results["overall_score"] = percentage

    # -------------------------
    # Mandatory element checks (automatic fail if any missing)
    # Mandatory elements per brief:
    #  - include env-setup task with correct earliest_start_offset respecting integration provisioning
    #  - schedule critical functional/integration work to finish before code-freeze OR explain trade-offs
    #  - respect two-testers headcount (no impossible parallel full-day tasks)
    #  - provide measurable entry/exit criteria
    # -------------------------
    mandatory_issues = []
    # M1: env-setup task existence and plausible offsets
    env_tasks = [t for t in tasks if t.get("type") == "env-setup"]
    if not env_tasks:
        mandatory_issues.append("No env-setup task present (required).")
    else:
        # Find at least one env-setup with earliest_start_offset <=3 and duration_days >=2
        ok_env = False
        for t in env_tasks:
            if int(t.get("earliest_start_offset", 999)) <= 3 and int(t.get("duration_days", 0)) >= 2:
                ok_env = True
        if not ok_env:
            mandatory_issues.append("env-setup task does not respect integration provisioning lead time (should start <= offset 3 and take >=2 days).")

    # M2: code-freeze compliance
    # We already computed latest_code_end and code_freeze_issue
    if code_freeze_issue:
        mandatory_issues.append("Code-impacting tests extend past code-freeze without explicit trade-off justification in contingency_plan or final_comments.")

    # M3: two-testers headcount respected (no impossible parallel full-day tasks)
    # Check if any owner per-day >8 (we flagged impossible_overload earlier)
    if impossible_overload:
        mandatory_issues.append(f"Per-owner per-day hours exceed 8 hours (impossible): {impossible_overload}")

    # M4: measurable entry/exit criteria
    # Check test_strategy contains exit-related measurable phrases or 'P0' etc.
    ts_lower = ts.lower()
    measurable_exit = False
    if ("exit" in ts_lower or "exit criteria" in ts_lower or "p0" in ts_lower or "p1" in ts_lower or "no more than" in ts_lower or "sign-off" in ts_lower or "u at sign-off" in ts_lower):
        measurable_exit = True
    # Also check candidate's schedule_summary maybe includes criteria? but we focus on test_strategy
    if not measurable_exit:
        mandatory_issues.append("Measurable entry/exit criteria not found in test_strategy (mandatory).")

    if mandatory_issues:
        results["mandatory_fail_reasons"] = mandatory_issues
        # Per spec: "Failure to include any mandatory element results in automatic fail regardless of numeric score."
        results["comment"] = "Automatic fail: missing mandatory elements."
        results["overall_score"] = 0.0
        # Keep detailed breakdown but mark pass=false
        results["pass"] = False
    else:
        # Evaluate pass threshold: >=80 points pass
        results["pass"] = True if percentage >= 80.0 else False
        results["comment"] = "Passed" if results["pass"] else "Failed (score below threshold)"

    # Add some summary notes useful to candidate
    results["grading_notes"] = {
        "project_start_date_used": project_start_date.isoformat(),
        "code_freeze_date": code_freeze.isoformat(),
        "release_date": release_date.isoformat(),
        "computed_task_end_examples": {tid: {
            "start_date": v["start_date"].isoformat(),
            "end_date": v["end_date"].isoformat(),
            "duration_days": v["duration_days"]
        } for tid, v in list(computed.items())[:6]}  # show first few
    }

    # Save detailed results to test_results.json
    _write_results(results)
    return results

def _write_results(results):
    out_path = os.path.join(os.getcwd(), "test_results.json")
    try:
        with open(out_path, "w", encoding="utf-8") as f:
            json.dump(results, f, indent=2, ensure_ascii=False)
        print(f"Grading complete. Results written to {out_path}")
    except Exception as e:
        print(f"Failed to write results file: {e}")

# -------------------------
# Entry point
# -------------------------
if __name__ == "__main__":
    if len(sys.argv) < 3:
        print("Usage: python task_evaluation.py candidate_submission.json answer_key.json")
        sys.exit(2)
    candidate_file = sys.argv[1]
    answer_key_file = sys.argv[2]

        grade(candidate_file, answer_key_file)
    except Exception as exc:
        # Catch-all to ensure deterministic output even on unexpected errors
        err_out = {
            "error": str(exc),
            "overall_score": 0.0,
            "message": "Grader encountered an unexpected error."
        }
        with open("test_results.json", "w", encoding="utf-8") as f:
            json.dump(err_out, f, indent=2)
        print("Grader failed with an exception; see test_results.json for details.")
        sys.exit(1)

Notes about the grader:
- The script implements the evaluator rubric and mandatory checks described in the exam materials.
- It is tolerant of minor format differences (e.g., whitespace, capitalization) when scanning strings for required phrases.
- The grader is defensive: missing or invalid keys cause clear messages and (for mandatory element failures) an automatic fail as required.
- The output file test_results.json contains a breakdown per rubric area, feedback, the numeric overall_score (percentage 0–100), and pass/fail indication.
