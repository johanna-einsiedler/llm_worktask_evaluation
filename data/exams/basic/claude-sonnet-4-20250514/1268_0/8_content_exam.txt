def add(a, b):

def subtract(a, b):
    return a - b

def multiply(a, b):
    return a * b

def divide(a, b):
    if b == 0:

    return a ** b

def calculate(operation, num1, num2):

### data_processor.py
```python
import csv

def process_customer_data(filename):
    customers = []
    total_sales = 0
    
    with open(filename, 'r') as file:
        reader = csv.DictReader(file)
        for row in reader:
            customer = {
                'id': int(row['customer_id']),
                'name': row['customer_name'],
                'age': int(row['age']),
                'sales': float(row['total_sales']),
                'region': row['region']
            }
            customers.append(customer)
            total_sales += customer['sales']
    
    # Calculate statistics
    avg_age = sum(c['age'] for c in customers) / len(customers)
    avg_sales = total_sales / len(customers)
    
    # Filter high-value customers (sales > 5000)
    high_value = [c for c in customers if c['sales'] > 5000]
    
    # Group by region
    regions = {}
    for customer in customers:
        region = customer['region']
        if region not in regions:
            regions[region] = {'count': 0, 'total_sales': 0}
        regions[region]['count'] += 1
        regions[region]['total_sales'] += customer['sales']
    
    # Generate report
    print(f"Total customers processed: {len(customers)}")
    print(f"Average age: {avg_age:.2f}")
    print(f"Average sales: ${avg_sales:.2f}")
    print(f"High-value customers: {len(high_value)}")
    print("\nRegional breakdown:")
    for region, data in regions.items():
        avg_regional_sales = data['total_sales'] / data['count']
        print(f"{region}: {data['count']} customers, avg sales: ${avg_regional_sales:.2f}")
    
    return {
        'total_customers': len(customers),
        'average_age': avg_age,
        'average_sales': avg_sales,
        'high_value_count': len(high_value),
        'regional_data': regions
    }

if __name__ == "__main__":
    result = process_customer_data('customer_data.csv')
```

### inventory_system.py
```python
import json

class InventorySystem:
    def __init__(self, data_file):
        with open(data_file, 'r') as f:
            self.inventory = json.load(f)
    
    def get_total_value(self):
        total = 0
        for item in self.inventory:
            total += item['quantity'] * item['unit_price']
        return total
    
    def find_low_stock_items(self, threshold=10):
        low_stock = []
        for item in self.inventory:
            if item['quantity'] < threshold:
                low_stock.append(item)
        return low_stock
    
    def update_stock(self, product_id, new_quantity):
        for item in self.inventory:
            if item['product_id'] == product_id:
                item['quantity'] = new_quantity
                return True
        return False
    
    def add_new_product(self, product_id, name, quantity, unit_price, category):
        # Check if product already exists
        for item in self.inventory:
            if item['product_id'] == product_id:
                return False
        
        new_item = {
            'product_id': product_id,
            'product_name': name,
            'quantity': quantity,
            'unit_price': unit_price,
            'category': category
        }
        self.inventory.append(new_item)
        return True
    
    def get_category_summary(self):
        categories = {}
        for item in self.inventory:
            cat = item['category']
            if cat not in categories:
                categories[cat] = {'count': 0, 'total_value': 0, 'total_quantity': 0}
            categories[cat]['count'] += 1
            categories[cat]['total_value'] += item['quantity'] * item['unit_price']
            categories[cat]['total_quantity'] += item['quantity']
        return categories
    
    def generate_report(self):
        print("=== INVENTORY REPORT ===")
        print(f"Total inventory value: ${self.get_total_value():.2f}")
        
        low_stock = self.find_low_stock_items()
        print(f"Low stock items: {len(low_stock)}")
        
        categories = self.get_category_summary()
        print("\nCategory Summary:")
        for cat, data in categories.items():
            print(f"{cat}: {data['count']} items, ${data['total_value']:.2f} value")

def main():
    system = InventorySystem('inventory_data.json')
    
    # Test all functions
    print("Testing inventory system...")
    
    # Test 1: Generate initial report
    system.generate_report()
    
    # Test 2: Update stock
    print(f"\nUpdating stock for product P001...")
    success = system.update_stock('P001', 25)
    print(f"Update successful: {success}")
    
    # Test 3: Add new product
    print(f"\nAdding new product P999...")
    success = system.add_new_product('P999', 'Test Product', 50, 15.99, 'Electronics')
    print(f"Add successful: {success}")
    
    # Test 4: Try to add duplicate product
    print(f"\nTrying to add duplicate product P001...")
    success = system.add_new_product('P001', 'Duplicate', 10, 5.00, 'Test')
    print(f"Add successful: {success}")
    
    # Test 5: Final report
    print(f"\n=== FINAL REPORT ===")
    system.generate_report()

if __name__ == "__main__":
    main()
```

## Field Format Requirements

### Boolean Fields
- `execution_successful`: Must be exactly `true` or `false` (lowercase, no quotes)

### String Fields - Severity Classification
- `severity`: Must use exactly one of these values:
  - `"CRITICAL"`
  - `"HIGH"`
  - `"MEDIUM"`
  - `"LOW"`

### String Fields - Deployment Recommendations
- `deployment_recommendation`: Must use exactly one of these values:
  - `"READY"`
  - `"NEEDS_FIXES"`
  - `"REQUIRES_TESTING"`
- `overall_recommendation`: Must use exactly one of these values:
  - `"READY"`
  - `"NEEDS_FIXES"`
  - `"REQUIRES_TESTING"`

### Integer Fields
- All count fields must contain actual integer values (not strings)
- `total_test_cases_run`: Count of test cases executed for calculator
- `failed_test_cases`: Count of test cases that failed for calculator
- `total_records_processed`: Count of customer records processed
- `processing_errors`: Count of errors encountered during data processing
- `functions_tested`: Count of inventory system functions tested
- `functions_with_errors`: Count of functions that had errors
- `total_critical_issues`, `total_high_issues`, `total_medium_issues`, `total_low_issues`: Sum counts across all programs

### String Fields - Issue Documentation
- `issue_description`: Clear, specific description of the problem found
- `test_case`: Exact input or scenario that caused the issue
- `expected_result`: What should have happened according to requirements
- `actual_result`: What actually happened when the test was run

### Array Fields
- `issues_found`: Array of issue objects. If no issues found, use empty array `[]`
- Each issue object must contain all five required fields

## Formatting Standards

### Text Content
- All string values must be enclosed in double quotes
- Issue descriptions should be concise but specific
- Test cases should include exact inputs used
- Results should include actual output values or error messages

### Numerical Precision
- Integer fields: Whole numbers only, no decimal points
- When referencing monetary values in text fields: Use format `$X.XX` (dollar sign, two decimal places)
- When referencing percentages in text fields: Use format `X.XX%` (two decimal places)

### Special Cases
- If a program fails to execute at all, set `execution_successful: false`
- If no issues are found for a program, use `"issues_found": []`
- All numerical count fields must reflect actual testing performed
- Issue arrays can contain multiple issue objects if multiple problems are found

## Validation Requirements
- JSON must be syntactically valid
- All required fields must be present
- No additional fields beyond those specified
- All enum values (severity, recommendations) must match exactly as specified
- File must be named exactly `test_submission.json`
