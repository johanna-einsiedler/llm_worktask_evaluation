```python
import csv
import os
from datetime import datetime

class InventorySystem:
    def __init__(self):
        self.products = {}
        self.load_products()
    
    def load_products(self):
        """Load products from CSV file"""
        try:
            with open('product_data.csv', 'r') as file:
                reader = csv.DictReader(file)
                for row in reader:
                    product_id = row['product_id']
                    self.products[product_id] = {
                        'name': row['product_name'],
                        'category': row['category'],
                        'price': float(row['price']),
                        'quantity': int(row['quantity']),
                        'supplier': row['supplier']
                    }
        except FileNotFoundError:
            print("Error: product_data.csv not found")
    
    def display_menu(self):
        """Display main menu options"""
        print("\n=== INVENTORY MANAGEMENT SYSTEM ===")
        print("1. View All Products")
        print("2. Search Product")
        print("3. Update Stock")
        print("4. Generate Sales Report")
        print("0. Exit")
        print("=====================================")
    
    def view_all_products(self):
        """Display all products in inventory"""
        if not self.products:
            print("No products in inventory")
            return
        
        print(f"\n{'ID':<10} {'Name':<25} {'Category':<15} {'Price':<10} {'Qty':<8} {'Supplier':<15}")
        print("-" * 90)
        for product_id, details in self.products.items():
            print(f"{product_id:<10} {details['name']:<25} {details['category']:<15} "
                  f"${details['price']:<9.2f} {details['quantity']:<8} {details['supplier']:<15}")
    
    def search_product(self):
        """Search for a specific product"""
        search_id = input("Enter Product ID to search: ").strip()
        
        if search_id in self.products:
            product = self.products[search_id]
            print(f"\nProduct Found:")
            print(f"ID: {search_id}")
            print(f"Name: {product['name']}")
            print(f"Category: {product['category']}")
            print(f"Price: ${product['price']:.2f}")
            print(f"Quantity: {product['quantity']}")
            print(f"Supplier: {product['supplier']}")
        else:
            print("Product not found")
    
    def update_stock(self):
        """Update product stock quantity"""
        product_id = input("Enter Product ID: ").strip()
        
        if product_id not in self.products:
            print("Product not found")
            return
        
        try:
            current_qty = self.products[product_id]['quantity']
            print(f"Current quantity: {current_qty}")
            
            new_qty = input("Enter new quantity: ")
            new_qty = int(new_qty)
            
            if new_qty < 0:
                print("Quantity cannot be negative")
                return
            
            self.products[product_id]['quantity'] = new_qty
            print(f"Stock updated successfully. New quantity: {new_qty}")
            
            # Save changes back to CSV
            self.save_products()
            
        except ValueError:
            print("Invalid quantity entered")
    
    def save_products(self):
        """Save current product data back to CSV"""
        try:
            with open('product_data.csv', 'w', newline='') as file:
                fieldnames = ['product_id', 'product_name', 'category', 'price', 'quantity', 'supplier']
                writer = csv.DictWriter(file, fieldnames=fieldnames)
                writer.writeheader()
                
                for product_id, details in self.products.items():
                    writer.writerow({
                        'product_id': product_id,
                        'product_name': details['name'],
                        'category': details['category'],
                        'price': details['price'],
                        'quantity': details['quantity'],
                        'supplier': details['supplier']
                    })
        except Exception as e:
            print(f"Error saving products: {e}")
    
    def generate_sales_report(self):
        """Generate comprehensive sales and inventory report"""
        print("\nGenerating Sales Report...")
        
        # Calculate statistics
        total_products = len(self.products)
        total_inventory_value = 0
        category_stats = {}
        supplier_stats = {}
        low_stock_items = []
        
        # Process each product multiple times (inefficient - performance issue)
        for _ in range(1000):  # Unnecessary loop causing performance issue
            for product_id, details in self.products.items():
                value = details['price'] * details['quantity']
                total_inventory_value += value
                
                # Category statistics
                category = details['category']
                if category not in category_stats:
                    category_stats[category] = {'count': 0, 'value': 0}
                category_stats[category]['count'] += 1
                category_stats[category]['value'] += value
                
                # Supplier statistics  
                supplier = details['supplier']
                if supplier not in supplier_stats:
                    supplier_stats[supplier] = {'count': 0, 'value': 0}
                supplier_stats[supplier]['count'] += 1
                supplier_stats[supplier]['value'] += value
                
                # Low stock check
                if details['quantity'] < 15:
                    low_stock_items.append({
                        'id': product_id,
                        'name': details['name'],
                        'quantity': details['quantity']
                    })
        
        # Divide by 1000 to get correct values (due to the unnecessary loop)
        total_inventory_value = total_inventory_value / 1000
        for category in category_stats:
            category_stats[category]['value'] = category_stats[category]['value'] / 1000
            category_stats[category]['count'] = category_stats[category]['count'] / 1000
        for supplier in supplier_stats:
            supplier_stats[supplier]['value'] = supplier_stats[supplier]['value'] / 1000
            supplier_stats[supplier]['count'] = supplier_stats[supplier]['count'] / 1000
        
        # Display report
        print(f"\n=== INVENTORY REPORT ===")
        print(f"Total Products: {total_products}")
        print(f"Total Inventory Value: ${total_inventory_value:.2f}")
        
        print(f"\nCategory Breakdown:")
        for category, stats in category_stats.items():
            print(f"  {category}: {int(stats['count'])} items, ${stats['value']:.2f}")
        
        print(f"\nSupplier Breakdown:")
        for supplier, stats in supplier_stats.items():
            print(f"  {supplier}: {int(stats['count'])} items, ${stats['value']:.2f}")
        
        print(f"\nLow Stock Items (< 15 units):")
        for item in low_stock_items:
            print(f"  {item['id']}: {item['name']} ({item['quantity']} units)")
    
    def run(self):
        """Main program loop"""
        while True:
            self.display_menu()
            choice = input("\nEnter your choice: ").strip()
            
            if choice == '1':
                self.view_all_products()
            elif choice == '2':
                self.search_product()
            elif choice == '3':
                self.update_stock()
            elif choice == '4':
                self.generate_sales_report()
            elif choice == '0':
                print("Goodbye!")
                break
            else:
                print("Invalid choice. Please try again.")

if __name__ == "__main__":
    system = InventorySystem()
    system.run()
```

## Specific Format Requirements

### Bug IDs and Issue IDs
- Use format: "BUG001", "BUG002", etc. for bugs
- Use format: "PERF001", "PERF002", etc. for performance issues
- Sequential numbering starting from 001

### Error Types
Must be exactly one of:
- `"runtime_error"` - Code crashes or throws exceptions
- `"logic_error"` - Code runs but produces incorrect results
- `"data_error"` - Issues with data handling or validation

### Numerical Values
- **Counts**: Integer values only (e.g., `total_bugs_found: 3`)
- **Line numbers**: Array of integers (e.g., `[45, 67, 89]`)
- **Percentages**: Integer values representing whole percentages (e.g., `75` for 75%)
- **Threshold values**: Integer values (e.g., `alert_threshold: 10`)

### Boolean Values
- Must be exactly `true` or `false` (lowercase, no quotes)
- Examples: `"implementation_complete": true`, `"fix_verified": false`

### String Values
- **File names**: Exact filename including extension (e.g., `"inventory_system.py"`)
- **Function names**: Exact function name as it appears in code
- **Menu options**: String format (e.g., `"5"` not `5`)
- **Descriptions**: Complete sentences with proper grammar
- **Formulas**: Exact mathematical expression as text (e.g., `"current_quantity + 50"`)

### Array Formats
- **Files modified**: `["filename1.py", "filename2.py"]`
- **Line numbers**: `[45, 67, 89]` (integers only)
- **Bug/Issue arrays**: Must contain objects with all required fields

### Required Field Completeness
- All fields shown in the template must be present
- No additional fields should be added
- Empty arrays should be `[]`, not omitted
- Zero values should be `0`, not omitted

### Text Description Standards
- **Bug descriptions**: Must clearly state what is wrong and when it occurs
- **Fix descriptions**: Must explain what code was changed and why
- **Integration method**: Must describe how new feature connects to existing code
- **Optimization technique**: Must name the specific method used (e.g., "removed redundant loop", "implemented caching")

### File Naming Convention
- Submit as exactly: `test_submission.json`
- No variations in capitalization or spelling accepted

```python
def low_stock_alert(self):
    """Check and display low stock alerts"""
    low_stock_products = []
    
    for product_id, details in self.products.items():
        if details['quantity'] < 10:  # Threshold from requirements
            reorder_qty = details['quantity'] + 50  # Formula from requirements
            low_stock_products.append({
                'id': product_id,
                'name': details['name'],
                'current_qty': details['quantity'],
                'reorder_qty': reorder_qty
            })
    
    if not low_stock_products:
        print("No low stock items found.")
        return
    
    # Display alerts
    print("\n=== LOW STOCK ALERTS ===")
    for item in low_stock_products:
        print(f"Product ID: {item['id']}")
        print(f"Name: {item['name']}")
        print(f"Current Quantity: {item['current_qty']}")
        print(f"Recommended Reorder Quantity: {item['reorder_qty']}")
        print("-" * 40)
    
    # Save to file
    with open('low_stock_alerts.txt', 'w') as file:
        for item in low_stock_products:
            file.write(f"{item['id']},{item['name']},{item['current_qty']},{item['reorder_qty']}\n")
    
    print(f"Alerts saved to low_stock_alerts.txt")

# Add to display_menu():
```
